מַעֲשֶׂה בְּעַכְבָּר שֶׁחָזַר אַחֲרֵי אִשָּׁה
	





2.png







עַכְבָּר אֶחָד יַחְסָן גָּדוֹל וְגִבּוֹר מְפֻרְסָם, שֶׁנִּמְלַט מִכַּף חָתוּל עִוֵּר, רָצָה
לָשֵׂאת אִשָּׁה. בָּאוּ אֵלָיו שַׁדְכָנִים מִכָּל־הַחוֹרִים, שַׁדְכָן שַׁדְכָן וְכַלָּתוֹ. הָלַךְ הַבָּחוּר הֶחָתָן
לְבַקֵּר בַּחוֹרִים וְלֹא מָצָא אַף כַּלָּה אַחַת כִּלְבָבוֹ.

כָּאן&nbsp;– כַּלָּה זוּ בְתוּלָה יָפָה הִיא, שֶׁעֵינֶיהָ שְׁחוֹרוֹת וּשְׂפָמָהּ מְגֻדָּל, אַךְ
הַמְּחֻתָּן אֵינוֹ עָשִׁיר.

כָּאן&nbsp;– הַמְּחֻתָּן עָשִׁיר גָּדוֹל הוּא וְחוֹרוֹ מָלֵא נֵרוֹת שֶׁל חֵלֶב, אֲבָל
עַם־הָאָרֶץ הוּא וְאַף דַּף אֶחָד שֶׁל סֵפֶר אֵין לוֹ.

וְכָאן&nbsp;– הַמְּחֻתֶּנֶת אֵינָהּ מְיֻחֶסֶת: מִבֵּיתוֹ שֶׁל סַנְדְּלָר הִיא.

רָאָה הָעַכְבָּר, שֶׁאֵין חוֹר בְּלִי פְּסוּל, וְאָמָר: "אֵין בָּעַכְבָּרִים מָשְׁלִי.
אֵצֵא וְאָשׁוּט בָּאָרֶץ, אוּלַי אֶמְצָא לִי כַּלָּה נָאָה שֶׁלֹּא מִבְּנֵי עַמִּי".


*


הִשְׁכִּים הָעַכְבָּר וְיָצָא בַּבֹּקֶר וְרָאָה אֶת־הַשֶּׁמֶשׁ כְּשֶׁהִיא עוֹלָה מִפְּאַת מִזְרָח
וּמְאִירָה בְּזֹהַר הָאַרְגָּמָן. "כַּלָּה זוֹ נָאָה הִיא לִי"&nbsp;– אָמָר הָעַכְבָּר, וַיִּשָּׂא מְשָׁלוֹ
וַיַּעַן:

"בֹּאִי כַלָּה, בֹּאִי כַלָּה, מְאוֹר עֵינַיִם!

אִתִּי אֶל חוֹרִי תֵּלְכִי תַּחַת הַכִּירַיִם&nbsp;–

שָׁם תֵּדְעִי אשֶׁר, שָׁם תִּרְאִי חַיִּים!"

אָמְרָה לוֹ הַשֶּׁמֶשׁ: "אֲנִי אֵינִי הֲגוּנָה לָךְ, עַכְבָּר נִכְבָּד, שֶׁאַתָּה גִבּוֹר
בֶּן־חוֹרִים, וַאֲנִי&nbsp;– מָה אֲנִי וּמַה־כֹּחִי? עֲנָנָה עוֹבֶרֶת וּמַשְׁחִירָה פָנַי. הָעֲנָנָה&nbsp;– רַק
הִיא נָאָה לָךְ, שֶׁכֹּחָהּ גָּדוֹל מִכֹּחִי: הִיא שׁוֹפֶכֶת מָטָר, מוֹרָה בְרָקִים וּמְחוֹלֶלֶת רְעָמִים; אַךְ
לְךָ נָאָה, אַךְ לְךָ יָאָה כַלָּה כָזוֹ".


*


הָלַךְ הָעַכְבָּר וּמָצָא אֶת־הָעֲנָנָה, כְּשֶׁהִיא שָׁטָה מֵעַל לַנָּהָר, מַבִּיטָה
בִּרְאִי־הַמַּיִם וּמֵיטִיבָה אֶת־מִקְלְעוֹת רֹאשָׁהּ. "גַּם כַּלָּה זוֹ נָאָה הִיא"&nbsp;– אָמַר הָעַכְבָּר, וַיִּשָּׂא
מְשָׁלוֹ וַיַּעַן:

"בֹּאִי כַלָּה, בֹּאִי כַלָּה, עֲנָנָה בַּת שָׁמַיִם!

אִתִּי אֶל חוֹרִי תֵּלְכִי תַּחַת הַכִּירַיִם&nbsp;–

שָׁם תֵּדְעִי אשֶׁר, שָׁם תִּרְאִי חַיִּים!"

אָמְרָה לוֹ הָעֲנָנָה: "אֲנִי אֵינִי הֲגוּנָה לְךָ, עַכְבָּר נִכְבָּד, שֶׁאַתָּה גִבּוֹר
בֶּן־חוֹרִים, וַאֲנִי&nbsp;– מָה אֲנִי וּמַה־כֹּחִי? רוּחַ בָּאָה וְנוֹשְׂאָה אוֹתִי אֶל אֲשֶׁר תַּחְפֹּץ.
הָרוּחַ&nbsp;– רַק הִיא נָאָה לָךְ, שֶׁכֹּחָהּ גָּדוֹל מִכֹּחִי: הִיא מַכָּה גַּלִּים בַּיָּם וְעוֹקֶרֶת עֵצִים
בַּיָּעַר: אַךְ לְךָ נָאָה, אַךְ לְךָ יָאָה כַּלָּה כָזוֹ".


*


הָלַךְ הָעַכְבָּר וּמָצָא אֶת־הָרוּחַ, כְּשֶׁהִיא מִשְׁתַּעֲשַׁעַת בַּהֲדַסִּים וְרוֹקֶדֶת בֵּין
הַשּׁוֹשַׁנִּים. "גַּם כַּלָּה זוֹ נָאָה הִיא"&nbsp;– אָמַר הָעַכְבָּר, וַיִּשָּׂא מְשָׁלוֹ וַיַּעַן:

"בֹּאִי כַלָּה, בֹּאִי כַלָּה, קַלַּת הַכְּנָפַיִם!

אִתִּי אֶל חוֹרִי תֵּלְכִי תַּחַת הַכִּירַיִם&nbsp;–

שָׁם תֵּדְעִי אשֶׁר, שָׁם תִּרְאִי חַיִּים!"

אָמְרָה לוֹ הָרוּחַ: "אֲנִי אֵינִי הֲגוּנָה לָךְ, עַכְבָּר נִכְבָּד, שֶׁאַתָּה גִבּוֹר
בֶּן־חוֹרִים, וַאֲנִי&nbsp;– מָה אֲנִי וּמַה־כֹּחִי? הַחוֹמָה עוֹצֶרֶת בַּעֲדִי. הַחוֹמָה&nbsp;– רַק הִיא
נָאָה לָךְ, שֶׁכֹּחָהּ גָּדוֹל מִכֹּחִי: כֻּלָּהּ אַבְנֵי גָזִית, שְׁעָרֶיהָ בַּרְזֶל וּמִגְדָּלֶיהָ עַנְקֵי אֵל; אַךְ לְךָ
נָאָה, אַךְ לְךָ יָאָה כַּלָּה כָזוֹ".


*


הָלַךְ הָעַכְבָּר וּמָצָא אֶת־הַחוֹמָה, כְּשֶׁהִיא עוֹמֶדֶת בִּמְרוֹם הָהָר; כֻּלָּהּ לְבָנָה
כַשֶּׁלֶג, וּמִתּוֹךְ רָאשֶׁיהָ־מִגְדָּלֶיהָ מִתְנוֹצְצוֹת עֵינַיִם־חַלּוֹנוֹת בְּדִמְדוּמֵי־עָרֶב, וְאַנְשֵׁי־חַיִל
מְזֻיָּנִים סוֹבְבִים עָלֶיהָ וְשׁוֹמְרִים אוֹתָהּ. "גַּם כַּלָּה זוֹ נָאָה הִיא" – אָמַר הָעַכְבָּר, וַיִּשָּׂא
מְשָׁלוֹ וַיַּעַן:

"בֹּאִי כַלָּה, בֹּאִי כַלָּה, חוֹמָה גְבֹהָה בַּשָּׁמַיִם!

אִתִּי אֶל חוֹרִי תֵלְכִי תַּחַת הַכִּירַיִם –

שָׁם תֵּדְעִי אשֶׁר, שָׁם תִּרְאִי חַיִּים!"

אָמְרָה לוֹ הַחוֹמָה: "אֲנִי אֵינִי הֲגוּנָה לָךְ, עַכְבָּר נִכְבָּד, שֶׁאַתָּה גִבּוֹר
בֶּן־חוֹרִים, וַאֲנִי&nbsp;– מָה אֲנִי וּמַה־כֹּחִי? הָעַכְבָּרִים מְחַטְטִים בְּקִירוֹתַי וְחוֹתְרִים
יְסוֹדוֹתָי"…

"אִם כֵּן",&nbsp;– אָמַר הָעַכְבָּר&nbsp;– "הֲרֵי אֵין בָּעוֹלָם כַּלָּה נָאָה וַחֲשׁוּבָה
מִבַּת־עַכְבָּר!"

וּמִיָּד הָלַךְ וְשָׁב אֶל עַמּוֹ וְלָקַח לוֹ בַּת־עַכְבָּר כְּאֹרַח
כָּל־הָעַכְבָּרִים.
