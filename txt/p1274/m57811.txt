


	
	הַמַּצָּה הַחַמָּה
	




בּמַגֶּנְצָא לְפָנִים עִיר וְאֵם בְּיִשְׂרָאֵל, גָר אִישׁ חָסִיד, תָּם וְיָשָׁר, רַבִּי אַמְנוֹן שְׁמוֹ, וְלוֹ בֵּן וּשְׁמוֹ רַבִּי אֱלִיעֶזֶר, כְּשֶׁהִגִּיעָה שַׁעְתּוֹ שֶׁל רַבִּי אַמְנוֹן לְהִסְתַּלֵּק מִן הָעוֹלָם, קָרָא לִבְנוֹ וְאָמַר לוֹ: גוֹזֵרְנִי עָלֶיךָ, בְּנִי, שֶׁלֹּא תַעֲבֹר אֶת הַנָּהָר דָנוּבָה.



נָשַׁק רַבִּי אֱלִיעֶזֶר אֶת יְדֵי אָבִיו, בָּכָה וְאָמַר: מִצְוָתְךָ אֶשְׁמֹר, אֲבִי.

וּלְרַבִּי אַמְנוֹן קָרוֹב בְּרֶגֶנְשְׁבֻּרְגְ שֶׁעַל
הַדָּנוּבָה וּשְׁמוֹ רַבִּי יְהוּדָה הֶחָסִיד, אִישׁ גָדוֹל בְּתוֹרָה
וּבְיִרְאָה וְשֵׁמַע צִדְקוּתוֹ יָצָא לְמֵרָחוֹק. כְּשֶׁשָּׁמַע רַבִּי
אֱלִיעֶזֶר עַל עֹמֶק תּוֹרָתוֹ וְגֹדֶל יִרְאָתוֹ שֶׁל רַבִּי יְהוּדָה
הֶחָסִיד, בִּקֵּשׁ לֵילֵךְ אֶצְלוֹ לִלְמֹד מִפִּיו סִתְרֵי תוֹרָה.
כַּמָּה שָׁנִים הָיָה מִשְׁתּוֹקֵק לֵילֵךְ שָׁמָּה, אֲבָל צַוָּאַת
אָבִיו הָיְתָה מְעַכַּבְתּוֹ. לְבַסּוֹף לֹא יָכֹל לִכְבּשׁ אֶת
תְּשׁוּקָתוֹ, הָלַךְ וְעָבַר אֶת הַנָּהָר דָנוּבָה לָבוֹא אֶל הֶחָסִיד
הַגָּדוֹל.

כְּשֶׁבָּא לְרֶגֶנְשְׁבֻּרְגְ הָלַךְ מִיָּד לְבֵית
רַבִּי יְהוּדָה. אָמַר לוֹ רַבִּי יְהוּדָה: בֶּאֱמֶת לֹא הָיִיתִי
צָרִיךְ לְהַחֲזִיר לְךָ שָׁלוֹם, שֶׁלֹּא קִיַמְתָּ צַוָּאַת אָבִיךָ,
עָלָיו הַשָּׁלוֹם. אֲבָל לִכְבוֹד אַבָּא וּלְפִי שֶׁנַּפְשְׁךָ
חָשְׁקָה בַּתּוֹרָה, הֲרֵינִי אוֹמֵר לְךָ שָׁלוֹם.

נִבְהַל רַבִּי אֱלִיעֶזֶר מִדִּבְרֵי הֶחָסִיד,
שֶׁקָּרָא אֶת מַחְשְׁבוֹת לִבּוֹ, וּמֵרֹב אֵימָה לֹא יָכֹל לְהָשִׁיב
מִלָּה אֶחָת.

יָשַׁב רַבִּי אֱלִיעֶזֶר בְּבֵית רַבִּי יְהוּדָה
הֶחָסִיד שָׁנָה וְעוֹד שָׁנָה וְהָיָה מְהַלֵּךְ אַחֲרָיו וְעָשָׂה
אָזְנוֹ כַּאֲפַרְכֶּסֶת לִלְמֹד מִמֶּנּוּ הַנְהָגוֹת טוֹבוֹת
וְִלִשְׁמֹעַ מִפִּיו סִתְרֵי תוֹרָה, וְלֹא לָמַד וְלֹא שָׁמַע כְּלוּם,
שֶׁהֶחָסִיד הָיָה נוֹהֵג כְּאָדָם פָּשׁוּט וְהָיָה לוֹמֵד אֶת
הַתּוֹרָה הַנִּגְלָה כִּשְׁאָר תַּלְמִידֵי חֲכָמִים.

הִתִחָרֵט רַבִּי אֱלִיעֶזֶר בְּלִבּוֹ, שֶׁהִרְחִיק
נְדֹד מִבֵּיתוֹ וְגַם עַל צַוָּאַת אָבִיו עָבַר וְהֶעֱלָה חֶרֶס
בְּיָדוֹ, שֶׁלֹא לָמַד מִפִּי רַבִּי יְהוּדָה הֶחָסִיד אֲפִלּוּ פֶּרֶק
אֶחָד וַאֲפִלוּ אוֹת אַחַת, וַהֲרֵיהוּ יוֹשֵׁב עִמּוֹ זוֹ הַשָּׁנָה
הַשְּׁלִישִׁית, הַרְחֵק מֵאִשְׁתּוֹ וּמִבְּנוֹ יְחִידוֹ, וּמַה יְהֵא
בְּסוֹפוֹ? הִגִּיעַ עֶרֶב פֶּסַח וְעַל רַבִּי אֱלִיעֶזֶר יָרְדָה
עַצְבוּת גְדוֹלָה. הָיָה מְהַרְהֵר: “רוֹצֶה הָיִיתִי לַעֲרֹךְ אֶת
הַסֵּדֶר עַל שֻׁלְחָנִי עִם בְּנֵּי בֵּיתִי, כַּנָהוּג בְּיִשְׂרָאֵל,
וְלֹא לִהְיוֹת סָמוּךְ עַל שֻׁלְחָן זָר. זֶה לִי הַפֶּסַח
הַשְּׁלִישִׁי, שֶׁאֲנִי רָחוֹק מִבֵּיתִי, וּבְנֵי בֵּיתִי שְׁרוּיִים
בְּצַעַר וְאֵינָם יוֹדְעִים מֶה הָיָה לִי. אֲבָל אָשֵׁם אֲנִי, שֶׁלֹּא
קִיַּמְתִּי צַוָּאַת אָבִי, עָלָיו הַשָּׁלוֹם”.

וְהָיָה מְהַלֵּךְ וּמוֹשֵׁךְ עַל עַצְמוֹ מָרָה
שְׁחוֹרָה, עַד שֶׁפָּרַשׁ לְקֶרֶן־זָוִית וְצָוַח לְנַפְשׁוֹ: “וַי לִי,
לֹא דַי שֶׁאֲנִי רָחוֹק מִקִּנִּי זוֹ הַשָּׁנָה הַשְּׁלִישִׁית, אֶלָּא
שֶׁגַּם לֹא לָמַדְתִּי כְּלוּם, וַאֲפִלּוּ מֵהַמְּעַט שֶׁיָּדַעְתִּי
שָׁכַחְתִּי הַרְבֵּה!”

רָאָה רַבִּי יְהוּדָה הֶחָסִיד אֶת רַבִּי אֱלִיעֶזֶר
וְיָדַע אֶת מַעַרְכֵי לִבּוֹ. אָמַר לוֹ: רוֹאֶה אֲנִי שְֶׁעָצוּב
אַתָּה וְיוֹדֵעַ אֲנִי סִבַּת עִצְּבוֹנְךָ, וְגַם רְצוֹנְךָ לֹא
נֶעְלַם מִמֶּנִּי. רוֹצֶה הָיִיתָ לַעֲרֹךְ הַסֵּדֶר עִם אִשְׁתְּךָ
וְלִשְׁמֹעַ אֶת בִּנְךָ שׁוֹאֵל אַרְבַּע הַקֻּשְׁיוֹת.

נֶאֱנַח רַבִּי אֱלִיעֶזֶר וְאָמַר: אוֹדֶה וְלֹא
אֵבוֹשׁ, רַבִּי, מְאֹד מְאֹד מִשְׁתּוֹקֵק אֲנִי לְכָךְ. אֲבָל
יָדַעְתִּי כִּי תְּפִלַת שָׁוְא הִיא, שֶׂכְּבָר עֶרֶב פֶּסַח הַיוֹם
וְאֵיךְ אַגִּיעַ לְמַגֶּנְצָא?

וּמַה תִּתֵּן לִי אִם אֲבִיאֲךָ וְאַצִּיגְךָ בְּבֵיתְךָ עוֹד לִפְנֵי הִתְקַדֵּשׁ הֶחָג? שָׁאַל הָרַב.

הֻכְפְּלָה עַצְבוּתוֹ שֶׁל רַבִּי אֱלִיעֶזֶר וְאָמַר
בְּקוֹרְטוֹב שֶׁל תַּרְעֹמֶת: רָבִּי, הֲגַם תִּשְׂחַק לְמַר
נַפְשִׁי?

חָלִילָה / עָנָה רַבִּי יְהוּדָה / אַךְ אֱמֶת דִבַּרְתִּי.

נִתְלַהֵב רַבִּי אֱלִיעֶזֶר וְקָרָא: אֶת כָּל הוֹן
בֵּיתִי אֶתֵּן, כִּי כָּלְתָה נַפְשִׁי לִהְיוֹת בְּבֵיתִי בְּלֵיל
הַסֵּדֶר.

חִיֵךְ רַבִּי יְהוּדָה הֶחָסִיד וְאָמַר: הַשָּׁעָה
מְאֻחֶרֶת עַכְשָׁו וְעָלֵינוּ לְמַהֵר לֶאֱפוֹת אֶת הַמַּצָּה
הַשְּׁמוּרָה. אַחֲרֵי אֲפִיַּת הַמַּצּוֹת אֶרְאֶה אִם אוּכַל
לַהֲבִיאֲךָ לְבֵיתְךָ.

תָּמַהּ רַבִּי אֱלִיעֶזֶר וְנָשָׂא קַל־וָחֹמֶר
בְּנַפְשׁוֹ: אִם עַכְשָׁו הַשָּׁעָה מְאֻחֶרֶת, אַחֲרֵי אֲפִיַּת
המַּצּוֹת לֹא כֹּל שֶׁכֵּן.

הָלַךְ רַבִּי יְהוּדָה הֶחָסִיד אֶל בֵּית הָאוֹפֶה
לֶאֱפוֹת אֶת הַמַּצָּה הַשְּׁמוּרָה, וְרַבִּי אֱלִיעֶזֶר הָלַךְ עִמּוֹ
וְהָיָה עוֹזֵר עַל יָדוֹ, מַגִּישׁ לַמְנַקֵּב וְלָאוֹפֶה, וְרַבִּי
יְהוּדָה מִתְהַלֵּךְ בֵּין הָעוֹסְקִים וּמַשְׁגִיחַ עַל הַכַּשְׁרוּת.
כְּשֶׁרָדָה הָאוֹפֶה מִן הַתַּנּוּר אֶת הַמַּצָּה הָאַחֲרוֹנָה, אָמַר
רַבִּי יְהוּדָה לְרַבִּי אֱלִיעֶזְר: טֹל, בְּנִי, מַצָּה זוֹ וְהַנַּח
אוֹתָהּ בְּחֵיקְךָ. כְּשֶׁתַּחֲזֹר הַבַּיְתָה, תִּתְּנֶנָּה
לְאִשְׁתְּךָ בְּעוֹדָהּ חַמָּה.
וְסִיַּע בְּיָדוֹ לְהַנִּיחָהּ בְּחֵיקוֹ.

צִוָּה הֶחָסִיד לְהָבִיא אֶת הַמַּצּוֹת לְבֵיתוֹ, וּלְרַבִּי אֱלִיעֶזֶר אָמַר: בּוֹא נֵצֵא לַשָּׂדֶה.

הָיָה רַבִּי אֱלִיעֶזֶר הוֹלֵךְ לִימִינוֹ שֶׁל רַבִּי
יְהוּדָה הֶחָסִיד שָׁרוּי בְּצַעַר, שֶׁהוּא עוֹמֵד לְהִפָּרֵד מֵרַבּוֹ
וְלֹא לָמַד מִמֶּנּוּ כְּלוּם. וְרַבִּי יְהוּדָה קוֹרֵא הַמַּחֲשָׁבוֹת
נָגַע בִּכְתֵפוֹ וְאָמַר לוֹ: יוֹדֵעַ אֲנִי רְצוֹנְךָ: רוֹצֶה הָיִיתָ
לִלְמֹד מִמֶּנִּי סִתְרֵי תוֹרָה.

הֵן, אָמַר רַבִּי אֱלִיעֶזֶר, לְכָךְ בָּאתִי.

אָמַר הֶחָסִיד: בֶּאֱמֶת אֵינִי צָרִיךְ לְלַמֶּדְךָ,
שֶׁלֹא קִיַמְתָּ צַוָּאַת אָבִיךָ, עָלָיו הַשָּׁלוֹם. אֲבָל מִשּׁוּם
שֶׁקְּרוֹבִי אַתָּה וְיָשַׁבְתָּ עִמִּי שָׁנִים אֲחָדוֹת, אֲלַמֶּדְךָ
מַשֶּׁהוּ.

נָטַל אֶת מַקְלוֹ, כָּתַב בּוֹ בַּחוֹל כַּמָּה שֵׁמוֹת קְדוֹשִׁים וְאָמַר: קְרָא, בְּנִי, מַה כָּתוּב כָּאן.

קָרָא רַבִּי אֱלִיעֶזֶר אֶת הַשֵּׁמוֹת וּמִיָּד
נִפְתַּח לִבּוֹ כְּפִתְחוֹ שֶׁל אוּלָם וְיָדַע תּוֹרָה כֻּלָּה
וְרָזִין שְֶׁלָּהּ, כְּרַבִּי יְהוּדָה הֶחָסִיד.

חָזַר רַבִּי יְהוּדָה וּמָחַק אֶת הַכָּתוּב וְכִסָּהוּ
בַּחוֹל וּמִיָד נִשְׁתַּכַּח הַכֹּל מִלִּבּוֹ שֶׁל רַבִּי
אֱלִיעֶזֶר.

שָׁלֹשׁ פְּעָמִים כָּתַב רַבִּי יְהוּדָה הֶחָסִיד
וְשָׁלֹשׁ פְּעָמִים מָחַק אֶת הַכָּתוּב. נִצְטַעֵר רַבִּי אֱלִיעֶזֶר
שְֶׁשְָּׁכַח בְּכָל פַּעַם אֶת מִשְׁנָתוֹ. וּבַפַּעַם הָרְבִיעִית
נִצְטַוָּה לְלַקֵּק בִּלְשׁוֹנוֹ אֶת הַכָּתוּב בַּחוֹל. עָמַד רַבִּי
אֱלִיעֶזֶר עַל בִּרְכָּיו וְלִקֵּק אֶת הַכָּתוּב וּכְשֶׁבָּלַע אֶת
הָאוֹתִיּוֹת עִם הַחוֹל, נִקְבְּעָה תּוֹרָתוֹ בִּבְנֵי מֵעָיו וְלֹא
אִבֵּד טִפָּה.

לֹא נִשְׁתַּיֵר בֵּין הַחַמָּה הַשּׁוֹקַעַת וְהָאָרֶץ
אֶלָא אַמּוֹת אֲחָדוֹת וְאוֹרָהּ הֶאְדִים כְּאַרְגָמָן. הִנִּיחַ
רַבִּי יְהוּדָה אֶת יָדָיו עַל רֹאשׁוֹ. לָחַשׁ וְאָמַר: לֵךְ
לְשָׁלוֹם!

נִפְטַר רַבִּי אֱלִיעֶזֶר מֵרַבּוֹ וְהָיָה הוֹלֵךְ
וְלִבּוֹ מְרַקֵּד בּוֹ מֵרֹב בִּטָּחוֹן, שֶׁלִּפְנֵי שְׁקִיעַת
הִַחַמָּה יַעַמְדוּ רַגְלָיו בְּמַגֶּנְצָא. עָצַם אֶת עֵינָיו לְרֶגַע
לִרְאוֹת אֶת הַשִּׁנוּי שֶׁבַּדֶּרֶךְ, וּכְשֶׁפְּקָחָן עָמַד בְּעִירוֹ
וְלִפְנֵי בֵּיתוֹ.

נִכְנַס לְבֵיתוֹ וְאָמַר שָׁלוֹם לְאִשְׁתּוֹ וְנָשַׁק
לִבְנוֹ יְחִידוֹ. הוֹצִיא מֵחֵיקוֹ אֶת הַמַּצָּה וּנְתָנָהּ לָאִשָּׁה
וְאָמַר: הֲרֵי לָךְ מַתָּנָה מֵרַבִּי יְהוּדָה הֶחָסִיד.

נָטְלָה הָאִשָּׁה אֶת הַמַּצָּה וְקָרְאָה: הֲרֵי חַמָּה הִיא כִּיצִיאָתָהּ מִן הַתַּנּוּר.




8.png



חִיֵךְ רַבִּי אֱלִיעֶזֶר, לָבַשׁ בִּגְדֵי חָג וְהָלַךְ
עִם בְּנוֹ לְבֵית הַכְּנֶסֶת. רָאוּהוּ חֲשׁוּבֵי הָעֵדָה וְנָתְנוּ לוֹ
שָׁלוֹם וְהֶחֱזִיר לָהֶם שָׁלוֹם. שְׁאֵלוּהוּ הֵיכָן לָן אֶמֶשׁ,
שֶׁאֵין תַּלְמִידֵי־חֲכָמִים מְהַלְּכִים בַּדֶּרֶךְ בְּעַרְבֵי
שַׁבָּתוֹת וּמוֹעֲדִים.

אָמַר לָהֶם: לִפְנֵי שָׁעָה הָיִיתִי
בְּרֶגֶנְשְׁבֻּרְגְ וְעָזַרְתּי עַל יַד רַבִּי יְהוּדָה הֶחָסִיד
לֶאֱפוֹת מַצָּה שְׁמוּרָה לְפֶסַח, וְנָתַן לִי מַצָּה אַחַת מַתָּנָה
לְאִשְׁתִּי. רְצוֹנְכֶם, בּוֹאוֹ לְבֵיתִי וְשַׁאֲלוּ אֶת אִשְׁתִּי,
אִם לֹא הָיְתָה הַמַּצָּה חַמָּה כִּיצִיאָתָה מִן הַתַּנּוּר.

נִעְנְעוּ לוֹ בְּרָאשֵׁיהֶם, שֶׁיָּדְעוּ שֶׁהָיְתָה לוֹ קְפִיצַת הַדֶּרֶךְ.
