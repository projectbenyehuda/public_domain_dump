מֵשִׂיח אִלְּמִים
	




לֹא הַרְחֵק מִן הָעִיר רֶגֶנְשְׁבֻּרְגְ נוֹלַד יֶלֶד
לְאִשָּׁה עִבְרִיה. נַעֲשָׂה הַיֶּלֶד בֶּן שָׁנָה וּבֶן שְׁנָתַיִם
וְלֹא דִבֵּר. כְּשֶׁהָיָה בֶּן שָׁלֹשׁ וַעֲדַיִן לֹא דִבֶּר דִבּוּר
כָּל שֶׁהוּא הִתְחִילָה אִמּוֹ לִשְׁאֹל בְּרוֹפְאִים וּבְיִדְּעוֹנִים
וְלֹא הוֹעִילוּ כְּלוּם. אָמְרָה לָהּ שְׁכֶנְתָּהּ: מַה לָּךְ לָנוּעַ
עַל רוֹפְאֵי אֱלִיל, סְעִי לְרֶגֶנְשְׁבֻּרְגְ אֶל רַבִּי יְהוּדָה
הֶחָסִיד וְסַפְּרִי לוֹ אֶת צָרָתֵךְ, כִּי אֵין כָּמוֹהוּ יוֹעֵץ
פֶּלֶא.

לָקְחָה הָאֵם אֶת יַלְדָהּ וְנָסְעָה עִמּוֹ אֶל רַבִּי
יְהוּדָה הֶחָסִיד וְסִפְּרָה לוֹ אֶת צָרָתָהּ וּבִקְשָׁה מִמֶּנּוּ
עֵצָה מַה לַּעֲשׂוֹת שֶׁיַּלְדָהּ יְדַבֵּר.

אָמַר לָהּ רַבִּי יְהוּדָה: קְחִי אֶת הַיֶּלֶד
וְשָׂאִיהוּ בַּדֶּרֶךְ לְלַנְדְסְהוּט. כְּשֶׁתָּבוֹאִי אֶל הַכְּפָר
גוֹמְלָא תַּעְצְרִי אֶת הָאִישׁ הָרִאשׁוֹן שֶׁיִּזְדַמֵּן לָךְ
וְתֹאמְרִי לוֹ: "יַלְדִי זֶה אִלֵּם, מָה אֶעֱשֶׂה שֶׁיְּדַבֵּר?" וּמַה
שֶׁיֹּאמַר הָאִישׁ תַּעֲשִׂי וְתָבוֹא יְשׁוּעָתֵךְ.

עָשְׂתָה הָאִשָּׁה כְּדִבְרֵי הָרַב הֶחָסִיד,
וּכְשֶׁבָּאָה אֶל הַכְּפָר, פָּגְשָׁה אִישׁ נוֹצְרִי הוֹלֵךְ בְּדֶרֶךְ
הַמֶּלֶךְ. אָמְרָה לוֹ: רְאֵה, אִישׁ טוֹב, יַלְדִי זֶה אִלֵּם, מָה
אֶעֱשֶׂה לוֹ שֶׂיְּדַבֵּר?

נָתַן בָּהּ הַנּוֹצְרִי עַיִן לַגְלְגָנִית וְעָנָה:
הַשְׁלִיכִיהוּ לְתוֹךְ הָאֵשׁ. נִתְכַּעֲסָה הָאֵם הָעֲלוּבָה
וְאָמְרָה: הֱיֵה כַּפָּרָתִי וְכַפָּרַת יַלְדִי! וְהָלְכָה שׁוּב אֵצֶל
רַבִּי יְהוּדָה הֶחָסִיד וְסָחָה לוֹ אֶת כָּל הַמַּעֲשֶׂה. אָמַר לָהּ
רַבִּי יְהוּדָה: בֶּאֱמֶת לֹא הֵיטַבְתְּ לַעֲשׂוֹת, שֶׁהַנּוֹצְרִי
יָעַץ לָךְ יָפֶה. לֹא הָיְתָה כַּוָּנָתוֹ שֶׁתִּשְׂרְפִי אֶת הַיֶּלֶד
אֶלָּא שֶׁתִּתְּנִיהוּ עַל גַבֵּי טַבְלָא וּתְכַסִּיהוּ בַּעֲרֵבָה
וְתַנִּיחִיהוּ לְרֶגַע עַל הַכִּירָה הַמֻּסֶּקֶת. אִלּוּ עָשִׂית
כָּךְ, הָיָה הַיֶּלֶד מַתְחִיל לְדַבֵּר. וְעַכְשָׁו חִזְרִי לְאוֹתוֹ
כְּפָר וְשַׁאֲלִי אֶת הָאִישׁ הָרִאשׁוֹן שֶׁיִּזְדַמֵּן לָךְ
בַּדֶּרֶךְ, מַה תַּעֲשִׂי לְיַלְדֵךְ שֶׁיְּדַבֵּר, כְּכָל מַה
שֶּׁאָמַרְתִּי לָךְ בַּתְּחִלָּה, וַעֲשִׂי כִּדְבָרָיו.

לָקְחָה הָאִשָּׁה אֶת הָאָמָה שֶׁתִּשָּׂא אֶת
הַיֶּלֶד, שֶׁלֹא הָיָה בָּהּ עוֹד כֹּחַ לְשֵׂאתוֹ, וְהָלְכָה שֵׁנִית
אֶל הַכְּפָר הַנִּזְכָּר. פָּגְשָׁה נָכְרִי אַחֵר וְאָמְרָה לוֹ:
אִישִׁי הַטּוֹב, יַלְדִּי זֶה בֶּן שָׁלֹשׁ וַעֲדַיִן אֵינוֹ מְדַבֵּר,
מָה אֶעֱשֶׂה לוֹ שֶׁיִּתְפַּקַּח?

הִסְתַּכֵּל בָּהּ הַָנָּכְרִי כְּבִבְרִיָּה מְשֻׁנָּה
וְאָמַר: הַשְׁלִיכִיהוּ הַמַּיְמָה. וְהֶרְאָה בְּיָדוֹ עַל הַנָּהָר
שֶׁהָיָה עוֹבֵר בְּאוֹתוֹ כְּפָר.

אָמְרָה הָאֵם לָאָמָה הַהוֹלֶכֶת עִמָּהּ: אֲנִי אֶעֱמֹד בְּתוֹךְ הַנָּהָר וְאַתְּ תַּשְׁלִיכִי אֶת הַיֶלֶד עַל יָדִי לְתוֹךְ הַמַּים.
עָשְׂתָה הָאָמָה כְּמִצְוַת אֲדוֹנָתָהּ וְהִשְׁלִיכָה אֶת הַיֶלֶד לְתוֹךְ הַנָּהָר. וּכְשֶׁהָאֵם מָשְׁתָה אוֹתוֹ מִן הַמַּיִם הָיָה הַיֶלֶד צוֹעֵק: "שׁוּ, שׁוּ, שׁוּ,! קַרְרְ לִי!"



מֵאוֹתוֹ רֶגַע הֻתְּרָה לְשׁוֹנוֹ שֶׁל הַיֶּלֶד וְהָיָה מְדַבֵּר כִּשְׁאָר הַיְלָדִים. כְּשֶׁגָּדַל הֵבִיאַתְהוּ אִמּוֹ אֶל רַבִּי יְהוּדָה הֶחָסִיד וְאָמְרָה לוֹ: זֶה בְּנִי שֶׁהָיָה אִלֵּם וְהִשְׁלַכְתִּיו הַמַּיְמָה וְנִתְפַּקַּח. עַכְשָׁו רוֹצָה אֲנִי שֶׁתְּלַמְּדֵהוּ תּוֹרָה.
נִשְׁאַר בְּבֵית רַבִּי יְהוּדָה הֶחָסִיד וְהָיָה לוֹמֵד בְּבֵית־מִדְרָשׁוֹ, עַד שֶׁנַּעֲשָׂה תַּלְמִיד חָכָם מְפֻרְסָם. וְהַפֶּה שֶׁהָיָה אִלֵּם נַעֲשָׂה כַּמַּעְיָן הַמִּתְגַבֵּר.
