הָאֶבֶן הַטּוֹבָה
	




לרַבִּי יִצְחָק אָבִיו שֶׁל רַשִׁ"י זִכְרוֹ לִבְרָכָה
הָיְתָה אֶבֶן טוֹבָה, שֶׁבִּקְשׁוּ לִתֵּן בְּדָמֶיהָ אֶלֶף דִינְרֵי
זָהָב וְלֹא רָצָה לְמָכְרָהּ. וְלַקֵּיסָר שֶׁל אוֹתָהּ מְדִינָה הָיָה
פֶּסֶל עֲבוֹדָה־זָרָה שֶׁשְּׁתֵּי אֲבָנִים יְקָרוֹת קְבוּעוֹת לוֹ
בִּשְׁתֵּי עֵינָיו. אֵרַע שֶׁאֶבֶן אַחַת נָפְלָה מֵעֵינוֹ שֶׁל
הַפֶּסֶל וְאָבְדָה. נוֹדַע לַקֵּיסָר כִּי בִּרְשׁוּתוֹ שֶׁל
הַיְּהוּדִי רַבִּי יִצְחָק מְצוּיָה אֶבֶן רְאוּיָה לְהִקָּבַע
בִּמְקוֹם הַחֲסֵרָה. שָׁלַח אֶל רַבִּי יִצְחָק שֶׁיִּמְכֹּר לוֹ אֶת
הָאֶבֶן וִיקַבֵּל בִּמְחִירָהּ כַּמָּה שֶׁיִּשְׁאַל. הָיָה רַבִּי
יִצְחָק עוֹמֵד וְחוֹכֵךְ בַּדָּבָר.

אָמַר לוֹ הַשָּׁלִיחַ: אִם לֹא תַּעֲשֶׂה בַּקָּשָׁתוֹ
שֶׁל הַקֵּיסָר תַּעֲשֶׂה גְזֵרָתוֹ. מִיָּד הִסְכִּים רַבִּי יִצְחָק.
נָטַל אֶת הָאֶבֶן וּנְתָנָהּ בְּחֵיקוֹ וּבָא בַּסְּפִינָה לְהוֹלִיכָהּ
אֶל הַקֵּיסָר.

הָיָה לִבּוֹ נוֹקְפוֹ שֶׁהוּא הוֹלֵךְ לְהַשְׁלִים
פֶּסֶל שֶׁל עֲבוֹדָה־זָרָה. מֶה עָשָׂה? כְּשֶׁהַסְּפִינָה הָיְתָה
מְפָרֶשֶׂת בְּלֵב יָם נִכְנַס בִּדְבָרִים עִם נָכְרִי אֶחָד שֶׁנָּסַע
בְּאוֹתָהּ סְפִינָה. סִפֶּר לוֹ שֶׁהוּא מוֹלִיךְ אֶבֶן יְקָרָה
בִּשְׁבִיל הַקֵּיסָר.

אָמַר לוֹ הַנָּכְרִי: חָשְׁקָה נַפְשִׁי לִרְאוֹת אֶבֶן זוֹ.

אָמַר לוֹ: הִשָּׁבַע לִי שֶׁאַתָּה נִזְהָר בָּהּ וְאַרְאֶנָּה לְךָ.

נִשְׁבַּע לוֹ.

הוֹצִיא רַבִּי יִצְחָק אֶת הָאֶבֶן מֵחֵיקוֹ
וְהִנִּיחָהּ עַל כַּף יָדוֹ שֶׁל הַנָּכְרִי. הִסְתַּכֵּל בָּהּ
הַנָּכְרִי מִקָּרוֹב וְהָיָה תָמֵהַּ עַל הֲדַר זִיוָהּ שֶׁלֹּא רָאָה
כְּמוֹתוֹ מִיָּמָיו.

וּכְשֶׁבִּקֵּשׁ רַבִּי יִצְחָק לְהַחֲזִירָהּ הִרְעִיד
אֶת יָדוֹ וְהָאֶבֶן נִשְׁמְטָה מִן הַיָּד וְנָפְלָה לִמְצוּלוֹת
יָם.

מִיָּד הֵטִיל עַצְמוֹ עַל קַרְקַע הַסְּפִינָה וְהָיָה
מוֹרֵט שַׂעֲרוֹתָיו וּמַכֶּה אֶת גוּפוֹ בְּקוֹל נְהִי, וְכָךְ הָיָה
מְצַעֵק:

וַי לִי, מֶה עָשִׂיתִי? אִבַּדְתִּי כָּל הוֹנִי, כָּל
עֲמָלִי שֶׁעָמַלְתִּי תַּחַת הַשֶּׁמֶשׁ, וְאִלּוּ הֲבֵאתִיהָ אֶל
הַקֵּיסָר הָיָה נוֹתֵן לִי בְּדָמֶיהָ אֶלֶף אַלְפֵי דִינָרִים, וְעוֹד
הָיָה מַרְצֶה לִי תְּשׁוּאוֹת חֵן וּמְזַכֵּנִי בִּזְכֻיּות טוֹבוֹת,
אוֹתִי וְאֶת זַרְעִי אַחֲרַי עַד סוֹף כָּל הַדּוֹרוֹת. אוֹי לִי
שֶׁאִבַּדְתִּי טוֹבָתִי בְּיָדַיִם, וְעַתָּה מַה תִּקְוָתִי? וְעוֹד
אֲגָרֶה עָלַי חֲמַת הַמֶּלֶך, שְׁבִּקֵּשׁ מִמֶּנִּי אֶת הָאֶבֶן
לְקָבְעָהּ בְּפִסְלוֹ וְאֵין לוֹ אַחֶרֶת כְּמוֹתָהּ, מֶה עָשִׂיתִי,
מֶה עָשִׂיתִי?

וְהָיָה מְטַפֵּחַ רֹאשׁוֹ בַּקַּרְקַע וּפוֹרֵשׂ יָדָיו כְּאִישׁ נוֹאָשׁ.

רָאוּ הָעָם הַנּוֹסְעִים בַּסְּפִינָה אֶת רֹב צַעֲרוֹ, בָּאוּ אֶצְלוֹ וְנִחֲמוּהוּ בְּדִבְרֵי כִּבּוּשִׁים.

וּכְשֶׁהִגִּיעָה הַסְּפִינָה לִנְמֵלָהּ שֶׁל הַבִּירָה
הָלְכוּ אֶל הַקֵּיסָר וְהִתְחַנְנוּ עַל הַיְהוּדִי הָאֻמְלָל שֶׁהֵמַר
לוֹ מַזָּלוֹ. סָבַר הַקֵּיסָר אֶת הַדָּבָר לַאֲמִתּוֹ וְנִצְטַעֵר גַּם
הוּא עַל הָאֲבֵדָה הַגְּדוֹלָה וְאָמַר: חֲבַל עַל דְאָבְדִין. מַה
לַּעֲשׂוֹת לַיָּם שֶׁיָּשִׁיב אֶת גְּזֵלָתוֹ? אוֹי לְמַזָּלוֹ שֶׁל
אוֹתוֹ יְהוּדִי, שֶׁאִלּוּ הֵבִיא לִי אֶת הָאֶבֶן הָיִיתִי מַעֲנִיק
לוֹ כָּךְ וְכָךְ דִינָרִים, וְעַכְשָׁו אֵין בְּיָדוֹ כְּלוּם, וְצִוָּה
לְהַחֲזִירוֹ לְבֵיתוֹ בַּסְּפִינָה חִנָּם אֵין כֶּסֶף, שֶׁאֵין
בְּיָדוֹ לָתֵת שְׂכָרָהּ.

וּכְשֶׁיָצָא רַבִּי יִצְחָק מִן הַסְּפִינָה וְהָלַךְ
בַּדֶּרֶךְ לְבֵיתוֹ פְּגָשׁוֹ אֵלִיָּהוּ הַנָּבִיא זָכוּר לַטּוֹב
וְאָמַר לוֹ: אִבַּדְתָּ הָאֶבֶן הַטּוֹבָה מִדַּעַת כְּדֵי לַעֲשׂות
רְצוֹן קוֹנְךָ, חַיֶּיךָ, כָּעֵת חַיָּה תֵּלֵד לְךָ אִשְׁתְּךָ בֵּן
שֶׁיָּאִיר אֶת הָעוֹלָם יוֹתֵר מִכָּל אֲבָנִים טוֹבוֹת
שֶׁבָּעוֹלָם.

וְלִתְקוּפַת הַשְָּׁנָה נוֹלַד לוֹ בֵּן וְקָרָא שְׁמוֹ שְׁלֹמֹה; הוּא רָבֵּנוּ שְׁלֹמֹה יִצְחָקִי,
שֵׁהֵאִיר עֵינֵינוּ בְּתוֹרָה שֶׁבִּכְתָב וּבְתוֹרָה שֶׁבְּעַל פֶּה,
וְלֹא נִפְחַת מֵאוֹרוֹ עַד הַיּוֹם הַזֶּה.
