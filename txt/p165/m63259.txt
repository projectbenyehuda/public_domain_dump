בֶּעָשׂוֹר לַחֹדֶשׁ הַשְׁבִיעִי
	




תְּנוּעָה רַבָּה מֵאָז בֹּקֶר,

לִפְנוֹת עֶרֶב יָבוֹא צִיר;

זֶה צִיר עֶלְיוֹן בְּהַדְרַת קֹדֶשׁ,

מַטִיל אֵימָה עַל כָּל־יְצִיר.





בִּגְדֵי קֹדֶשׁ לְבָנִים, זַכִּים,

לוֹבְשִׁים הַכֹּל, וּבַצוֹם,

כְּמוֹ מִתְנַעֲרִים מֵעֲפַר חֻמְרָם,

לַעֲמֹד לְפָנָיו כָּל־הַיוֹם.





הַכֹּל בָּאִים אִישׁ לֹא נֶעֱדָר,

הֲמוֹנִים רַבִּים כַּחוֹל;

כְּחֵיל הַצָבָא לְקוֹל הַמַצְבִיא,

הוֹלְכִים עֲרוּכִים בַּכֹּל.





צִיר רָם וְנִשָׂא, שׁוֹפֵט קָשֶׁה,

מְבַקֵשׁ מִשְׁפָּט, מְדַבֵּר רְתֵת;

כָּל־שְׂפַת חָנֵף לֹא יַטֶנוּ,

לְהַטוֹת מִשְׁפָּט, לְהַרְבּוֹת שְׂאֵת.





יוֹם כִּפּוּרִים, זֶה שֵׁם כְּבוֹדוֹ,

אֶחָד מְצֻיָן הוּא וְאֵין עוֹד;

מִשְׁפָּט פִּיהוּ אֱמֶת, צֶדֶק,

מִי יוּכַל לְפָנָיו עֲמֹד.





הַיוֹם שׁוֹבְתִים כָּל־עֲסָקִים,

הַכֹּל שָׂמִים לֵב אֶל עָל;

מְפַשְׁפְּשִׁים בְּמַעֲשֵׂי כָּל־הַשָׁנָה,

קוֹרְאִים עַל כָּל־חֵטְא חֲבָל.





אִם הַחֲלָטַת הַמַחְשָׁבוֹת

תַּשְׁאִיר רשֶׁם עַז בַּלֵב?

אִם הָרָעִים יִהְיוּ טוֹבִים,

וְאָז יָגוּרוּ שֶׂה עִם זְאֵב?
