הַרְהוֹרִים בְּהִתְחַדֵשׁ הַשָׁנָה
	







א.


בְּקֵץ כָּל־שָׁנָה, בְּחַדֵשׁ כָּל־שָׁנָה,

אֲנַחְנוּ מְחַלִים מְיַחֲלִים אֶל אֵל;

כִּי נִוָשַׁע וְזֹאת תְּהִי שְׁנַת רְנָנָה,

וּלְאוֹר בֹּקֶר תִּהְיֶה חֶשְׁכַּת הַלֵיל.





כֹּה תַחֲלֹפְנָה לָנוּ הַשָׁנִים,

וּלְגָלוּתֵנוּ לֹא בָּא עוֹד הַקֵץ;

עוֹד פְּצָעֵינוּ רָעִים נֶאֱמָנִים,

לֹא תַּם הָרוֹמֵס, לֹא אָפֵס הַמֵץ.





עוֹד בְּתוֹכֵנוּ הִתְפַּלְּגוּת הַלְבָבוֹת,

אוֹכֶלֶת כְּרֶקֶב בְּעַצְמוֹת הַלְאֹם;

עוֹד בְּקִרְבֵּנוּ נִרְגָנוּת וּקְרָבוֹת,

הוֹפְכוֹת הַבְּרָכָה לִקְלָלָה יוֹם יוֹם.





מִלְחֶמֶת אַחִים, אֵיבָה מְשַׁכֶּלֶת,

תְּבַלְבֵּל כָּל־רַעֲיוֹן, תְּסַכְסֵךְ כָּל נִיב;

עוֹד לֹא תְחַכְּמֵנוּ מַסָה מַשְׂכֶּלֶת,

לְגָרֵשׁ הַשָׂטָן, לְהַשְׁבִּית הָרִיב.





עַל הַיַהֲדוּת הַתְּמִימָה הַנְקִיָה,

קָמוּ בָּנִים כֶּחָשִׁים כְּלֹא לָהּ הֵם;

בִּלְשׁוֹנָהּ מְדַבְּרִים סָרָה וּרְמִיָה,

צָרִים מִבַּיִת, בּוֹזִים לִיקְהַת אֵם.





הֵם מִתְנַפְּלִים עָלֶיהָ כִּנְחָשִׁים,

לְחַלֵל יִפְעָתָהּ, לָשִׁית לָהּ קֵץ;

מוֹכְרִים הַבְּכוֹרָה בִּנְזִיד עֲדָשִׁים,

מְקַלְעִים דִבְרֵי בֶלַע חַדִים כַּחֵץ.




ב.


שָׁם פַּטְפְּטָנִים מִתְוַכְּחִים בַּשְׁעָרִים,

עַל דְבַר עֲתִיד הַיַהֲדוּת לְדוֹר דוֹר;

אֵלֶּה שׁוֹלְלִים מִמֶנָהּ כָּל־תִּקְוָה

וְאֵלֶּה אוֹמְרִים נְקַוֶה עוֹד לָאוֹר.





יֶשְׁנוֹ מִשְׂתַּכֵּל בֵּין הַמַשְׂכִּילִים,

סוֹפֵג אֶל תּוֹכוֹ כָּל־סִיג וּבְדִיל;

מְצַלְצֵל בְּפִטְפּוּטָיו, מְדַבֵּר תַּהְפּוּכוֹת,

אֶת־כָּל־רוּחוֹ יוֹצִיא הַכְּסִיל.





הֲזֹאת הִיא כָּל־תְּעוּדַת הַהַשְׂכָּלָה,

לְבַלְבֵּל, לְסַכְסֵךְ כָּל־רַעֲיוֹן, כָּל־נִיב;

לַהֲרֹס כָּל־בָּנוּי, לְבַטֵל כָּל־תִּקְוָה,

לְחַלֵל כָּל־קֹדֶשׁ, לְבָעֵר כָּל־כְּתִיב?




ג.


שָׁם עַל הָעִבְרִית קָמוּ מַלְעִיבִים,

לְהַשְׁפִּיל שְׂאֵתָהּ לְהוֹרִידָהּ מִכֵּס;

חֶבֶר זַרְגוֹנִים בַּאֲפֵר בּוֹרְאֵי נִיבִים,

שְׂפַת הַגָלוּת יָרִימוּ עַל נֵס.





לָעִבְרִית, אוֹמְרִים, אֵין תִּקְוַת תְּחִיָה,

אֵין לָהּ עוֹד מִשְׁפַּט קִיוּם כָּעֵת;

סָרַת טַעַם הִיא, בָּלָה עֲנִיָהּ;

מַה־תּוֹעִיל, מַה־תּוּכַל הִיא עוֹד תֵּת!





לִשְׂפַת הַגָלוּת רַבִּים הַקוֹנִים,

בָּהּ עוֹשִׂים הוֹן, קוֹנִים לָהֶם שֵׁם;

מַה לָהֶם עָבָר, עָתִיד, עַם אֵין אוֹנִים,

נַחֲלַת אָבוֹת, יַהֲדוּת, לְאֻמוּת, שְׂפַת אֵם.





מְקוֹמָהּ, אוֹמְרִים, שָׁם בְּאֶרֶץ הָרְפָאִים,

בֵּין שְׂרִידִים בָּלִים אֵין חֵפֶץ בָּם כְּבָר;

שְׂפַת סְפָרִים נוֹשָׁנִים, לֹא עוֹד נִקְרָאִים

וּשְׂפַת הַגָלוּת הִיא יָפָה בִיקָר.





הֵם נוֹתְנִים לָעָם שָׂפָה אַחֶרֶת,

מִי הֵם הַמְדַבְּרִים בְּשֵׁם כָּל־הַלְאֹם?

מְרִימִים הַשִׁפְחָה עַל כֵּס הַגְבֶרֶת,

נְוַת הַבַּיִת לַפִּילֶגֶשׁ הֲדוֹם!





כָּאֵלֶּה עוֹד צוֹרְרִים מִבָּיִת,

רָעִים מַשְׁחִיתִים מְצוֹרְרִים מִלְבָר!

הַכֶּרֶם כֵּן מָלֵא קוֹץ, שָׁמִיר, שָׁיִת,

וּמְחַבְּלִים כָּל־פְּרִי טוֹב, כָּל־בָּר.




ד.


עַד אָנָה יִהְיֶה כַּרְמֵנוּ זָנוּחַ,

מִשְׁלַח שׁוּעָלִים, מִרְמַס כָּל־חֲזִיר;

מָתַי יַעֲבוֹר גַם עָלָיו הָרוּחַ,

יְכַלֶה אֶת־קוֹצָיו, יְנַקֶה הָאֲוִיר?





מִי יַשְׂכִּיל, יְשׁוֹבֵב, יְאַחֵד הַלְבָבוֹת,

יִקְרָא לַשָׂטָן הַמַשְׁחִית דֹם?

יְכַוֵץ הַנִיבִים, יְקַבֵּץ הַמַחֲשָׁבוֹת,

אֶל הָרַעֲיוֹן הָאֶחָד: תְּקוּמַת הַלְאֹם?





מִי יְבַעֵר יְכַלֶה אֶת־כָּל־הַפְּסֹלֶת,

מְלֶאכֶת נְבָלִים מַהְבִּילִים דוֹר דוֹר;

מִי יַפְרִיד יַרְחִיק סוּבִּין מִסֹלֶת,

יַבְדִיל יַפְלִיג בֵּין חשֶׁךְ לָאוֹר?





מַה־תִּקְוָתֵנוּ בַּשָּׁנָה הַחֲדָשָׁה,

מַה־תָּבִיא, מַה־תִּתֵּן לְעַם מֹרָט, לֹא עָז?

מַה־תִּקְוָה לַאֳנִיָה סֹעֲרָה נוֹאָשָׁה?

לְפָנֶיךָ אֵל רֳאִי נִגְלָה הָרָז!




ה.


הַיוֹם נָחֹג בְּהַדְרַת קֹדֶשׁ

יוֹם הוּלֶדֶת כָּל־הַיְצִיר;

כְּבָר זֶה אַלְפֵי אַלְפֵי שָׁנָה,

הַיּוֹם הַזֶה כָּל־לֵב יָעִיר.





כַּמָּה תִּקְוֹת גְדוֹלוֹת, נְעִימוֹת,

קְשׁוּרוֹת בּוֹ לָאִישׁ, לִלְאֹם;

כַּמָּה לְבָבוֹת כּוֹאֲבִים, מְיַחֲלִים,

לִקְרַאת יֶשַׁע בּוֹ בַיוֹם!





כֵּן אֲנַחְנוּ עַם דַל אוֹנִים,

בֵּין פָּרִיצֵי חַיוֹת שֶׂה,

גַּם עֵינֵינוּ כֹּה צוֹפִיוֹת,

לִשְׁנַת יֶשַׁע אֶל יוֹם זֶה.





שָׁנָה שָׁנָה נִתְנַחֵמָה,

בְּתִקְוַת שָׁוְא קִיוּם הַחֲלוֹם;

שֶׁבְּהִתְחַדֵשׁ רֹאשׁ הַשָּׁנָה,

תִּתְחַדֵשׁ תְּחִיַת הַלְאֹם.





שָׁנִים הוֹלְכוֹת, שָׁנִים בָּאוֹת,

וּשְׁנַת יֶשַׁע אֵינָהּ עוֹד;

אַךְ אֲנַחְנוּ לֹא נִתְיָאֵשׁ,

נִתְעוֹדֵד בְּכָל־עֹז וּמְאֹד.





בַּתּוֹחֶלֶת הַמְמֻשָׁכָה,

נִמְצָא נֹחַם, נִמְצָא עֹז;

כִּי בְּרֹאשׁ שָׁנָה חֲדָשָׁה,

תָּחֵל לָנוּ שְׁנַת הַכְּנֹס.





קוֹרוֹתֵינוּ בִּנְדוּדֵינוּ,

נוֹתְנוֹת לָנוּ עֹז וֶאֱיָל;

עַמִים עַזִים סָפוּ תַּמוּ,

וַאֲנַחְנוּ עוֹד עוֹמְדִים עָל.




ו.


הַשָׁנָה הַיְשָׁנָה נוֹטָה לַעֲרֹב,

כָּלְתָה מְרוּצָתָהּ, בָּא לָהּ קֵץ;

אַךְ מַכָּתֵנוּ הַיְשָׁנָה,

טְרִיָה עוֹד מִפִּצְעֵי הַחֵץ.





וְשָׁנָה חֲדָשָׁה צְעִירָה בָּאָה

בְּסֵתֶר פָּנִים תַּחַת אֲפֵר;

הַאִם לָהּ פָּנִים שְׂמֵחִים מְאִירִים,

הֲתִלְבַּשׁ עֶדִי תְּכַהֵן פְּאֵר?





הֲתָאִיר הִיא פָּנֶיהָ אֵלֵינוּ,

תָּבִיא לָנוּ שְׂאֵת וִיקָר?

הַתְרַפֵּא הִיא אֶת־מַכָּתֵנוּ,

תַּהֲפֹּך לָנוּ לְמָתוֹק מָר?





עַל זֹאת הַיוֹם נִתְפַּלֵלָה,

אַךְ לֹא רַק בִּתְּפִלָה לְבָד;

הָבָה, נִתְאַסְפָה נִתְאַחֵדָה,

נָחֵל לַעֲשׂוֹת מַעֲשֵׂה יָד!




ז.


מְבַקְשִׁים אָנוּ אוֹר וָיֶשַׁע,

לִנְחוֹל אֶרֶץ, לִהְיוֹת עָם;

וּמַה־נַעֲשֶׂה לְהִוָשֵׁעַ,

לִהְיוֹת נְכוֹנִים לִשְׁכֹּן שָׁם?





הַכְּבָר שַׁבְנוּ לְיַהֲדוּתֵנוּ,

כִּדְבַר צִיר אֵל בְּדַבְּרוֹ רְתֵת?

הַכְּבָר גַם שְׂפָתֵנוּ בְּפִינוּ,

בְּבוֹא לָשׁוּב לְאַרְצֵנוּ עֵת?





עַד אָן נְפַגֵר בַּעֲצַלְתַּיִם,

הַבְהִתְרַשְׁלוּת לְהַנְחִיל יֵשׁ?

בְּתִקְוָה רַעֲיָה לְיַד עֲשִׂיָה,

נַעֲשֶׂה חַיִל בַּעֲלֵה רֵשׁ!




ח.


זֶה הַיוֹם הוּא יוֹם פְּרִידָה,

פְּרִידָה בֵּין שָׁנָה לְשָׁנָה;

בֵּין הָעָבְרָה וְהָעֲתִידָה,

חֲדָשָׁה תַּחַת יְשָׁנָה.





מַה־תִּהְיֶה לָנוּ הָעֲתִידָה,

שְׁנַת רְעָדָה אוֹ שְׁנַת רְנָנָה?

הֲתָשִׂים הִיא קֵץ לִירִידָה

וְעֵת עֲלִיָה הוּכָנָה?





הֲתָשִׂים הִיא קֵץ לַפְּרִידָה,

לְמִלְחֶמֶת אַחִים שׁוֹנִים?

הֲתָשִׂים הִיא קֵץ לַבְּגִידָה,

לְהֶמְיַת נְבָלִים זַרְגוֹנִים?





הַתְכַלֶה הִיא מִן הַכֶּרֶם,

הַקוֹצִים, הַקִמְשׁוֹנִים?

הֲתַחֲרִיב הִיא אֶת־הַזֶרֶם,

הַמַיִם הַזֵדוֹנִים?





הֲתַטֶה הִיא לֵב מִתְנַכְּרִים

אֶל הַיַהֲדוּת הַשְׁלֵמָה?

הֲתִתֵּן הִיא בְּלֵב מִתְמַכְּרִים

רֶגֶשׁ וְרוּחַ רוֹמֵמָה?





הֲתָסִיר הִיא הַסַנְוֵרִים,

מֵעֵינַי נִתְעִים פּוֹתִים?

לְהַכִּיר הַזְאֵבִים הַנְמֵרִים,

בְּעוֹר שֵׂיוֹת דְמֵיהֶם שׁוֹתִים?





לְהַכִּיר פּוֹשְׁקֵי שְׂפָתָיִם

בִּזְמִירוֹת יוֹשְׁבַי קְרָנוֹת;

כּוֹתְבִים עַל הָאָבְנָיִם

שַׁעֲשׁוּעִים לְמוֹכְרֵי מִגְדָנוֹת?





לְהִבָּדֵל מִן הַתּוֹעִים,

חֶבֶר מְרֵעִים וּמְרֵעוֹת,

כְּמֵיטִיבֵי הָעָם צוֹעִים,

לְהִבָּנוֹת מֵרִיב דֵעוֹת.





הַתְרַפֵּא הִיא אֵת הַפְּצָעִים,

פִּצְעֵי רוּחַ וּגְוִיָה;

פִּצְעֵי אַחִים נִפְשָׁעִים,

מַכָּה טְרִיָה אַכְזְרִיָה?





הַתְזַכֵּנוּ הִיא לִתְחִיָה,

בְּאֶרֶץ לָנוּ מוֹרָשָׁה?

הֲתִהְיֶה גַם לָעִבְרִיָה

תְּחִיָה, תְּקוּמָה חֲדָשָׁה?





הֲתִפְקַח עֵינֵי הַתּוֹעִים,

לְהַכִּיר תְּעוּדַת הַבְּחִירָה,

לְהַכִּיר הַמְלִיצִים הָרוֹעִים

הַמַתְעִים בְּשָׂפָה מְהִירָה?





הֲתִתֵּן בִּלְבַב עַמֵנוּ

רֶגֶשׁ הַכָּרָה בְּהִירָה,

וְיַעֲמֹד בָּנוּ טַעֲמֵנוּ,

וְנִרְאֶה בְאַסְפַּקְלַרְיָה מְאִירָה.





וְלֹא יִרְהֲבוּ עוֹד תַּעֲלוּלִים,

לְשַׁנוֹת, לְחַדֵשׁ חִדוּשִׁים,

לְדַבֵּר עָתָק וַהֲתוּלִים,

וְכַרְמֵנוּ יַעֲשֶׂה בְאוּשִׁים!





וְעוֹד כָּאֵלֶּה שְׁאֵלוֹת

יֵאוּשׁ לְרוּחַ מְיַחֵלָה;

מָתַי תָּסִיר הַמַכְשֵׁלוֹת,

אֵל יִשְׁעֵנוּ? גַם זֹאת שְׁאֵלָה!
