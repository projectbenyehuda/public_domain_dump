החִנָאִים
	



(ע"פ סְוֶן הַדִין)

עַם קְדוּמים הוּא עַם החִנָאִים. עָצוּם וְרַב
בְּמִסְפָּרוֹ מכָּל עַמֵי הַתֵּבֵל. קִיּוּמוֹ הוֹלֵךְ וְנִמְשָׁךְ
מִלִפְנֵי הַרְבּה אֲלָפים שָׁנָה עַד הַיּוֹם. בְּעֵת שֶׁבָּבֶל
וְנִינְוֵה עָמְדוּ בִּמְרוֹם הִתְפַּתְּחוּתָן, הָיְתָה כְבָר
לְחִינָא תַּרְבּוּת גְבֹהָה וּבְמֶשֶׁךְ הַרְבֵּה
אֲלָפים שָנָה לֹא שֻׁנְּתָה בְמַשְהוּ. מִבָּבֶל וְנִינוֵה נִשׁאֲרוּ
הַיּוֹם רַק חָרְבוֹת. אוּלָם חִינָא מִתְקִיֶּמֶת עוֹד בְּכָל
עֻזָהּ.

אַסִּיָה הַמּעֲרָבִית דוֹמָה לִמְקוֹם קְבָרוֹת גָּדוֹל
וּמַצְבוֹת זִכָּרוֹן שׁוֹנִים וְרַבִּים בְּלי מסְפָּר מְספּרוֹת עַל
הַזְמַנִים שֶׁחָלְפוּ. עַמִּים וּשְׁבָטִים נוֹדְדִים שוֹנִים נִלְחֲמוּ
פֹה זֶה עִם זֶה, אֶחָד כָּבַשׁ אֶת הַמֶמְשָׁלָה מִידֵי הַשֵּׁנִי
ויָּחֲרִיבוּ אֶת הָאָרֶץ. אוּלָם הַחינָאִים, הוֹדוֹת לִגְבוּלֵי אַרְצָם הסְּגוּרים מִכָּל צַד וְהוֹדוֹת לְגֹעַל נַפְשָׁם בְּיַחַס לִבְנֵי עַם זָר, נשְׁאֲרוּ כְמוֹ שֶׁהָיוּ. מִנְהָגֵיהֶם הַיְשָנִים שֶׁל עֲבוֹדַת אֱלִילִים וְזֵכֶר נִשְׁמוֹת אֲבוֹתֵיהֶם קְדוֹשִׁים לַדּוֹר הַצָּעִיר כְמוֹ לַדּוֹרוֹת שֶעָבָרוּ.

לִפְנֵי אַלְפַּיִם וּשְׁתֵּי מֵאוֹת שָׁנָה בָּנָה מֶלֶך
חִינָא, אִישׁ נָבוֹן וִרוֹאֶה עֲתִידוֹת, חוֹמָה גְדוֹלָה מִסָּבִיב לָאָרֶץ. בִּנְיָן כָּזֶה עוֹד לֹא הֵקִימוּ יְדֵי אֲנָשִים. אֹרֶךְ הִחוֹמָה, אַלְפַּיִם אַרְבִּע מֵאוֹת וַחֲמִשִׁים
קִילוֹמֶטֶר. שִׁשָׁה עָשָׂר מֶטִרִים גָבְהָהּ וְרָחְבָּהּ – שְׁמֹנָה
מֶטְרִים מִלְּמטָּה וַחֲמִשָׁה מִלְּמַעְלָה. הַחוֹמָה נִבְנְתָה
מֵאֲבָנִים לְבֶנים וְעֵצִים, בָּהּ נִמְצָאִים הַרְבֵּה מִגְדָּלִים
בְּמֶרְחָק יָדוּעַ אֶחָד מֵהַשֵׁנִי וְגם שְׁעָרִים. כַּיּוֹם הֲרוּסָה
הַחוֹמָה בִמְקוֹמוֹת רַבִּים, אוּלָם בִּכְלָלָהּ עוֹמֶדֶת הִיא עוֹד בְּאֵיתָנָהּ.

לְאֵיזוֹ מַטָּרָה נִבְנְתָה הַחוֹמָה הַזֹאת? הֲלֹא
הַחִינָאים הֵם בִּכְלָלָם אוֹהֲבֵי שָׁלוֹם? וַדַּאי שֶׁבָּנוּ אוֹתָהּ לְמעַן הָגֵן עַל הָאָרֶץ מִפְּנֵי הִתְנַפְּלוּת אוֹיְבִים מִבַּחוּץ.

חוֹמוֹת בְּצוּרוֹת מִסָּבִיב לְעָרִים שׁוֹנוֹת יֵשׁ בְּחִינָא אֶלֶף חֲמֵשׁ מֵאוֹת וַחֲמִשִׁים וְשָׁלשׁ. וּמַלְכָּם הַגָּדוֹל מָצָא אָז לְנָכוֹן, כִּי טוֹב יִהְיֶה אִם תִּסְבּוֹב חוֹמָה אַחַת כְּטַבַּעַת אֶת כָּל הָאָרֶץ. הַסַּכָּנָה הֲכִי גְדוֹלָה נִשְׁקְפָה אָז לָאָרֶץ מִצַּד צָפוֹן. שָׁם הָיוּ שִׁבְטֵי תּוּרְקִיסְטַן הַמִּזְרָחִית, שִׁבְטֵי טַטַרִים וּמוֹנְגוֹלִים נוֹדְדִים. כֻּלָּם פְּרָאִים, אַמִיצֵי לֵב וְאוֹהֲבֵי מִלְחָמָה.

בְּעַד כָּל אֵלֶה הַפְּרָאִים הָיְתָה חוֹמַת חִינָא
לְמַעֲצוֹר שֶׁלֹא יָכְלוּ לוֹ. מִסּבָּה זוֹ סָבְלָה אָז אֵירוֹפָּה
הַרְבּה. כּי הַשְּבָטים שֶׁהָיוּ מֻכְרָחִים לָשׁוּב מִדֶּרֶךְ
נְדוּדֵיהֶם הַדְּרוֹמִית, שֶׁהוֹבִילָה לְחִינָא, שָׂמוּ פניהֶם
מַעֲרָבָה. וּבַמאָה הָרְביעִית לִסְפירַת הָאֶזְרָחִים שָׁטְפוּ שִׁבְטֵי הַהוּנִים וְהָאַלַנִים דֶרֶךְ אֲרָצוֹת רַבּוֹת בְּאֵירוֹפָה וְהֶחְרִיבוּ אוֹתָן.
