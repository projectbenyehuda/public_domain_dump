קוּשְׁטָא
	




הָיְתָה מִלְפָנִים בְּאֶרֶץ עִיר גְּדוֹלָה וּשְׁמָהּ "קוּשְׁטָא", לֵאמֹר - אֱמֶת", כִּי
כָל יוֹשְׁבֶיהָ מֵעוֹלָם לֹא שִׁקְּרוּ. וַיִּחְיוּ שָׁם הָאֲנָשִׁים בְּשָׁלוֹם וּבְשַׁלְוָה וַיַּאֲרִיכוּ יָמִים עַל
הָאֲדָמָה, וְלֹא מֵת שָׁם אִישׁ בַּדְמִי יָמָיו וְלֹא בֵן עַל פְּנֵי אָבִיו.

וַיֵּצֵא שֵׁם הָעִיר בְּכָל הָאֲרָצוֹת וַיִּשְׁאֲפוּ אֵלֶיהָ הֲמוֹנִים לָגוּר בְּתוֹכָהּ,
לְמַעַן הִנָּצֵל מִמַר הַמָּוֶת. אַךְ מְעַטִּים הָיוּ, אֲשֶׁר יָכְלוּ לְהִתְאַזְרַח בָּעִיר הַזֹּאת, כִּי לֹא רַבִּים
יַעַמְדוּ בְנִסָּיוֹן לְהִזָהֵר מֵאָבָק שֶׁל שֶׁקֶר וּלְדַבֵּר רַק אֱמֶת תָּמִיד.

וְאִישׁ אֶחָד, אֲשֶׁר אָהַב אֶת הָאֱמֶת בְּכָל לִבּוֹ, וַאֲשֶׁר לֹא הוֹצִיא מִפִּיו
דָּבָר אֲשֶׁר לֹא כֵן, לוּ גַם נִתַּן לוֹ מְלֹא תֵבֵל כֶּסֶף וְזָהָב, בָּא לְקוּשְׁטָא וַיֵּשֶׁב בָּעִיר וַיֵּאָחֶז
בָּהּ.

וְיוֹשְׁבֵי קוּשְׁטָא כִּבְּדוּהוּ עַל בַּר־לִבּוֹ וַיִּתְּנוּ לוֹ אִשָּׁה מִבְּנוֹתֵיהֶם,
וַיְהִי כְאַחַד מִתּוֹשְׁבֵי הָעִיר הַחֲשׁוּבִים. וַיֵּשֶׁב בָּעִיר יָמִים רַבִּים, וַיּוֹלֶד בָּנִים וּבָנוֹת,
וַיְלַמֵד גַּם אוֹתָם, כִּי יֶהֱגוּ רַק אֱמֶת וְלֹא תִהְיֶה בְרוּחָם רְמִיָה. וַיֵּלְכוּ בָנָיו וּבְנוֹתָיו
בְּדֶרֶךְ אֲבִיהֶם וַיִּרְאוּ כֻלָּם חַיִּים מְאֻשָׁרִים וְשׁוֹקְטִים.

וַיְהִי הַיּוֹם וַתִּתְרַחֵץ אֵשֶׁת הָאִישׁ הַזֶּה בַבַּיִת וַתְּחַפֵּף אֶת רֹאשָׁהּ, וְהִנֵּה
שְׁכֶנְתָּהּ דּוֹפֶקֶת עַל הַדֶּלֶת. וַיֵּצֵא הָאִישׁ לִקְרָאתָהּ וַיִּשְׁאָלֶהָ לְחֶפְצָהּ. וַתֹּאמֶר הַשְּׁכֵנָה, כִּי לִשְׁאֹל
מְאֵת אִשְׁתּוֹ כְלִי הֶחָסֵר לָהּ בְּבֵיתָהּ בָּאָה. וַיַּחְשֹׁב הָאִישׁ בְּלִבּוֹ, כִּי לֹא נָאֶה שֶׁתִּכָּנֵס עַתָּה
הַשְּׁכֵנָה הַבַּיְתָה וְתִרְאֶה אֶת אִשְׁתּוֹ, כְּשֶׁהִיא עֲרוּמָה וּמִתְרַחֶצֶת.

וַיַּעֲרֵם לֵאמֹר:

– סִלְחִי נָא, מָרָתִי, אִשְׁתִּי אֵינֶנָה עַתָּה בַבָּיִת; הוֹאִילִי נָא לָבוֹא
בְעוֹד זְמַן מַה וְשָׁאַלְתְּ מֵאִתָּהּ אֶת הַכְּלִי הַצָּרִיךְ לָךְ!

וַתֵּלֶךְ לָהּ הָאִשָּׁה, וְהָאִישׁ שָׂמַח, כִּי עָלְתָה בְיָדוֹ לְהִפָּטֵר מִמֶּנָּה בְדֶרֶךְ
נִמּוּס, וְלֹא הִרְגִישׁ כְּלַל אֵת אֲשֶׁר עוֹלֵל לְנַפְשׁוֹ, בְּהוֹצִיאוֹ דְבַר שֶׁקֶר מִפִּיו!

בַּעֲבוֹר יָמִים אֲחָדִים חָלָה בְנוֹ הַבְּכוֹר וַיָּמָת. וִימֵי אֶבְלוֹ טֶרֶם יִתַּמוּ,
וַיָּמָת לָאִישׁ הַזֶּה גַם בְּנוֹ הַצְעִיר. וַתְּהוֹם כָּל הָעִיר, כִּי מִיּוֹם הִוָּסֵד קוּשְׁטָא בָאָרֶץ, לֹא
קָרָה שָׁם אָסוֹן כָּזֶה, שֶׁיָּמוּתוּ לוֹ לְאָדָם בָּנָיו בְּחַיָּיו.

וַיָּבוֹאוּ זִקְנֵי הָעִיר אֶל הָאִישׁ הַזֶּה וַיְדַבְּרוּ לוֹ כַדְּבָרִים
הָאֵלֶּה:

– תֶּן תּוֹדָה, מַר, מֶה עָשִׂיתָ, כִּי הֵבֵאתָ אֶת מַלְאַךְ הַמָּוֶת
לְעִירֵנוּ?!

וַיַּעַן הָאִישׁ וַיֹּאמַר: לֹא אֶמְצָא בְנַפְשִׁי כָל עָוֹן; אָמְנָם דַּלּוֹתִי מֵהָבִין
אֶת דַּרְכֵי ד', אֲשֶׁר יִסְרַנִּי עַל לֹא דָבָר.

וַיּוֹסִיפוּ זִקְנֵי הָעִיר: אֲבָל מֵעוֹלָם לֹא נוֹעַז הַמָּוֶת לָקַחַת אֶת נֶפֶשׁ
אִישׁ מֵאִתָּנוּ, בְּטֶרֶם יִמָּלֵא צְבָאוֹ עַל הָאָרֶץ – הִזָכֵר נָא, אוּלַי הוֹצֵאתָ מִפִּיךָ דְבַר
שֶׁקֶר?

– בִּי אֲדוֹנַי! לוּ תִתְּנוּ לִי מְלֹא כָל חֲלַל הָעוֹלָם זָהָב וָכֶסֶף, לֹא
אֲטַמֵּא אֶת שְׂפָתַי לְדַבֵּר שֶׁקֶר! טָעַן הָאִישׁ.

– אֲבָל, אוּלַי נִכְשַּׁלְתָּ בִשְׁגָגָה בְאֵיזֶה אָבָק שֶׁל שֶׁקֶר? כִּי לוּלֵי זֹאת
הֲלֹא לֹא שָׁלַט מַלְאַךְ הַמֶּוֶת בְּבָנֶיךָ הַצְּעִירִים!

אָז נִזְכַּר הָאִישׁ אֶת אֲשֶׁר כִּזֵּב לִשְׁכֶנְתּוֹ, וַיְסַפֵּר לְזִקְנֵי הָעִיר: כָּזֹאת
וְכָזֹאת הָיָה מַעֲשֶׂה: חָפְצָה שְׁכֶנְתֵּנוּ לְהִכָּנֵס לְבֵיתֵנוּ לִשְׁאַל אֵיזֶה כְלִי מֵאִשְׁתִּי, אֲשֶׁר הִתְרַחֲצָה
בְאוֹתָהּ שָׁעָה, וָאֹמַר לָהּ, כִּי אֵין אִשְׁתִּי בַבָּיִת!

– הוּא הַדָּבָר, אֲשֶׁר דִּבַּרְנוּ! – הֵשִׁיבוּ הַזְקֵנִים – כִּי לוּלֵי
נִכְשַׁלְתָּ בִדְבַר שֶׁקֶר, לֹא שָׁלַט מַלְאַךְ הַמָּוֶת בְּעִירֵנוּ; וְעַתָּה הוֹאֶל נָא וְלֵךְ מֵאִתָּנוּ, כִּי לֹא
תִשָּׂא עוֹד עִירֵנוּ אֶת שִׁבְתְּךָ עִמָּנוּ פֹּה!

אֲבָל הַאֲמִינוּ נָא לִי אֲדוֹנַי, – הִצְטַדֵּק הָאִישׁ כִּי לֹא בְזָדוֹן כִּחַשְׁתִּי
לָאִשָּׁה הַהִיא, כִּי אִם מִפְּנֵי הַכָּבוֹד, שֶׁהֲרֵי אִשְׁתִּי נִכְלְמָה בְוַדְאַי, לוּ נִכְנְסָה שְׁכֶנְתָּהּ וְרָאֲתָה
אוֹתָהּ עֲרוּמָה!

– אָמְנָם מֵאֲמִינִים אֲנַחְנוּ לָךְ. אֲבָל הַשֶּׁקֶר הוּא דָבָר מְגוּנֶה כָזֶה
הַמְטַמֵּא אֶת נֶפֶשׁ הַמְבַטֵּא אוֹתוֹ לוּ גַם בִּשְׁגָגָה אוֹ בְאוֹנֵס, וְעַל כֵּן אַל נָא תִּתְמַהְמֵהּ לַעֲזוֹב
אֶת הָעִיר וְלֹא תָמִיט עָלֵינוּ עוֹד חֶרְפַּת מָוֶת!

וַיֵּצֵא הָאִישׁ אֶת הָעִיר וַיֵּלֶךְ הָלֹךְ וּבָכֹה עַל אֲשֶׁר בְּדִבּוּר אֶחָד קַל אִבֵּד
לוֹ חַיֵּי אֹשֶׁר וְשַׁלְוָה!
