בֵּין יִשְׂרָאֵל לָעַמִּים
	




יז תְּנָא מִשְּׁמוֹ שֶׁל רַ' מֵאִיר: מִפְּנֵי מָה נִתְּנָה תּוֹרָה לְיִשְׂרָאֵל?
מִפְּנֵי שֶׁהֵם עַזִּים. תְּנָא דְּבֵי רַ' יִשְמָעֵאל: "מִימִינוֹ אֵשׁ דָּת לָמוֹ" (דברים לג, ב)&nbsp;– אָמַר
הַקָּדוֹשׁ בָּרוּךְ הוּא: רְאוּיִים הַלָּלוּ שֶׁתִּנָּתֵן לָהֶם דַּת אֵשׁ. יֵשׁ אוֹמְרִים: דָּתֵיהֶם שֶׁל אֵלּוּ אֵשׁ,
שֶׁאִלְמָלֵא נִתְּנָה תּוֹרָה לְיִשְׂרָאֵל אֵין כָּל אֻמָּה וְלָשׁוֹן יְכוֹלִים לַעֲמֹד בִּפְנֵיהֶם, זֶהוּ שֶׁאָמַר רַ'
שִׁמְעוֹן בֶּן לָקִישׁ: שְׁלוֹשָׁה עַזִּים הֵם: יִשְׂרָאֵל בָּאֻמּוֹת, כֶּלֶב בַּחַיּוֹת, תַּרְנְגוֹל
בָּעוֹפוֹת.

אָמַר רַ' יִצְחָק בַּר רְדִיפָא בְּשֵׁם רַ' אַמִּי: אַתָּה סָבוּר שֶׁהוּא לִגְנַאי,
וְאֵינוֹ אֶלָּא לְשִׁבְחָם: אוֹ יְהוּדִי, אוֹ צָלוּב!

אָמַר רַ' אָבִין: עַד עַכְשָׁו קוֹרִין אֶת יִשְׂרָאֵל בְּחוּצָה לָאָרֶץ: "אֻמָּה שֶׁל
קְשֵׁי עֹרֶף" (ביצה כה ע"ב; שמו"ר מב, ט).






תְּנָא מִשְׁמוֹ&nbsp;– נאמר משמו.

עַזִּים&nbsp;– תקיפים, חצופים ובעלי עוצמה (והתורה תרכך את
אופיים, כאמור להלן).

תְּנָא דְּבֵי וגו'&nbsp;– אמר חכם מבית ר' ישמעאל.

הַלָּלוּ&nbsp;– עם ישראל.

דָּתֵיהֶם&nbsp;– מנהגיהם (של ישראל).

שֶׁאִלְמָלֵא וגו'&nbsp;– לולי ניתנה תורה לישראל, אשר הביאה לריכוכם
ולהתמתנותם, היו מנצחים את כל בני העולם.

שֶׁהוּא לִגְנַאי&nbsp;– שעזותם של ישראל היא גנות להם.

צָלוּב&nbsp;– תלוי (כלומר מוכנים למות ולא לעבור על דתם).

קוֹרִין&nbsp;– קוראים (אומות העולם).





*


יח אָמַר רַ' יוֹחָנָן מִשּׁוּם רַ' אֶלְעָזָר בְּרַ' שִׁמְעוֹן: כָּל מָקוֹם
שֶׁאַתָּה מוֹצֵא דְּבָרָיו שֶׁל רַ' יוֹסֵי הַגְּלִילִי בַּהַגָּדָה&nbsp;– עֲשֵׂה אָזְנֶיךָ כַּאֲפַרְכֶּסֶת: "לֹא מֵרֻבְּכֶם
מִכָּל הָעַמִים חָשַׁק ה' בָּכֶם וַיִּבְחַר בָּכֶם, כִּי אַתֶּם הַמְעַט מִכָּל הָעַמִּים" (דברים ז, ז)&nbsp;–
אָמַר לָהֶם הַקָּדוֹשׁ בָּרוּךּ הוּא לְיִשְׂרָאֵל: חוֹשֵׁקְנִי בָּכֶם, שֶׁאֲפִלּוּ בְּשָׁעָה שֶׁאֲנִי מַשְׁפִּיעַ לָכֶם
גְּדֻלָּה אַתֶּם מְמַעֲטִים עַצְמְכֶם לְפָנַי. נָתַתִּי גְדֻלָּה לְאַבְרָהָם, אָמַר לְפָנַי: "וְאָנֹכִי עָפָר וָאֵפֶר"
(בראשית יח, כז); לְמשֶׁה וְאַהֲרֹן&nbsp;– אָמְרוּ: "וְנַחְנוּ מָה" (שמות טז, ז); לְדָוִד&nbsp;–אָמַר:
"וְאָנֹכִי תוֹלַעַת וְלֹא אִישׁ" (תהלים כב, ז). אֲבָל אֻמּוֹת הָעוֹלָם אֵינָן כֵּן: נָתַתִּי גְּדֻלָּה
לְנִמְרוֹד&nbsp;– אָמַר: "הָבָה נִבְנֶה לָּנוּ עִיר" (בראשית יא, ד); לְפַרְעֹה&nbsp;– אָמַר: "מִי ה' אֲשֶׁר
אֶשְׁמַע בְּקֹלוֹ" (שמות ה, ב); לְסַנְחֵרִיב&nbsp;– אָמַר: "מִי בְּכָל אֱלֹהֵי הָאֲרָצוֹת… כִּי יַצִּיל ה'
אֶת יְרוּשָׁלַיִם מִיָּדִי" (מלכים ב יח, לה); לִנְבוּכַדְנֶצַּר&nbsp;– אָמַר: "אֶעֱלֶה עַל בָּמֳתֵי עָב
אֶדַּמֶּה לְעֶלְיוֹן" (ישעיה יד, יד); לְחִירָם מֶלֶךְ צוֹר&nbsp;– אָמַר: "מוֹשַׁב אֱלֹהִים יָשַׁבְתִּי בְּלֵב
יַמִּים" (יחזקאל כח, ב) (חולין פט ע"א).






הַגָּדָה&nbsp;– דברי אגדה (כגון מה שמובא להלן).

עֲשֵׂה אָזְנֶיךָ כַּאֲפַרְכֶּסֶת&nbsp;– הטה אוזניך לקבל את דבריו (כאותו כלי
הנראה כמשפך שדרכו מגיעים גרגרי החיטה אל הריחיים).

"הַמְעַט"&nbsp;– ומילה זו נדרשת בהמשך מלשון הקטנת ערך והשפלה
עצמית.

"וְנַחְנוּ מָה"&nbsp;– אין אנו שווים דבר.

נִמְרוֹד&nbsp;– מנהיג דור מגדל בבל לפי האגדה.





*


יט "וְהָאֱלֹהִים יְבַקֵּשׁ אֶת נִרְדָּף" (קהלת ג, טו)&nbsp;– אָמַר רַ'
יְהוּדָה בְּרַ' סִימוֹן בְּשֵׁם רַ' יוֹסֵי בְּרַ' נְהוֹרַאי: לְעוֹלָם הַקָּדוֹשׁ בָּרוּךְ הוּא תּוֹבֵעַ דָּמָם שֶׁל
נִרְדָּפִים מִן הָרוֹדְפִים; תֵּדַע לְךָ שֶכָּךְ הוּא, שֶׁכֵּן הֶבֶל נִרְדַּף מִפְּנֵי קַיִן וּבָחַר הַקָּדוֹשׁ בָּרוּךְ
הוּא בְּהֶבֶל, שֶׁנֶּאֱמַר: "וַיִּשַׁע ה' אֶל הֶבֶל וְאֶל מִנְחָתוֹ" (בראשית ד, ד); נֹחַ נִרְדַּף מִפְּנֵי
דּוֹרוֹ וְלֹא בָּחַר הַקָּדוֹשׁ בָּרוּךְ הוּא אֶלָּא בְּנֹחַ, שֶׁנֶּאֱמַר: "כִּי אֹתְךָ רָאִיתִי צַדִּיק לְפָנַי בַּדּוֹר
הַזֶּה" (שם ז, א); אַבְרָהָם נִרְדַּף מִפְּנֵי נִמְרוֹד וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּאַבְרָהָם, שֶׁנֶּאֱמַר:
"אַתָּה הוּא ה' הָאֱלֹהִים אֲשֶׁר בָּחַרְתָּ בְּאַבְרָם" (נחמיה ט, ז); יִצְחָק נִרְדַּף מִפְּנֵי פְּלִשְׁתִּים
וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּיִצְחָק, שֶׁנֶּאֱמַר: "רָאוֹ רָאִינוּ כִּי הָיָה ה' עִמָּךְ" (בראשית כו,
כח); יַעֲקֹב נִרְדַּף מִפְּנֵי עֵשָׂו וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּיַעֲקֹב, שֶׁנֶּאֱמַר: "כִּי יַעֲקֹב בָּחַר
לוֹ יָהּ" (תהלים קלה, ד); יוֹסֵף נִרְדַּף מִפְּנֵי אֶחָיו וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּיוֹסֵף,
שֶׁנֶּאֱמַר: "עֵדוּת בִּיהוֹסֵף שָׂמוֹ" (שם פא, ו); משֶׁה נִרְדַּף מִפְּנֵי פַּרְעֹה וּבָחַר הַקָּדוֹשׁ בָּרוּךְ
הוּא בְּמשֶׁה, שֶׁנֶּאֱמַר: "לוּלֵי משֶׁה בְחִירוֹ" (שם קו, כג); שָׁאוּל נִרְדַּף מִפְּנֵי פְּלִשְׁתִּים
וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּשָׁאוּל, שֶׁנֶּאֱמַר: "הַרְאִיתֶם אֲשֶׁר בָּחַר בּוֹ ה' " (שמואל א י,
כד); דָּוִד נִרְדַּף מִפְּנֵי שָׁאוּל וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּדָוִד, שֶׁנֶּאֱמַר: "וַיִּבְחַר בְּדָוִד
עַבְדּוֹ" (תהלים עח, ע); יִשְׂרָאֵל נִרְדָּפִים מִפְּנֵי הָאֻמּוֹת וּבָחַר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּיִשְׂרָאֵל,
שֶׁנֶּאֱמַר: "וּבְךָ בָּחַר ה' לִהְיוֹת לוֹ לְעַם סְגֻלָּה" (דברים יד, ב) (ויק"ר כז,
ה).






תּוֹבֵעַ דָּמָם וגו'&nbsp;– בוחר בנרדפים ודורש מן הרודפים לתת דין
וחשבון על פגיעתם בהם.

נֹחַ נִרְדַּף וגו'&nbsp;– על יסוד מסורת אגדית שלפיה ביקשו בני דורו
של נח לפגוע בו (תנחומא נח, ה).

נִמְרוֹד&nbsp;מי שהשליך, על פי מסורת האגדה, את אברהם אל כבשן האש
(בר"ר לח, יג).

"עֵדוּת בִּיהוֹסֵף שָׂמוֹ"&nbsp;– הבחירה ביוסף באה לידי ביטוי בהוספת
אות משמו של הקב"ה לתוך השם "יוסף".





*


כ "וַיִּקְרְאוּ שְׁמוֹ עֵשָׂו" (בראשית כה, כה)&nbsp;– הֵא שָׁוְא שֶבָּרָאתִי
בְעוֹלָמִי! אָמַר רַ' יִצְחָק: אַתֶּם קְרָאתֶם לַחֲזִירַתְכֶם שֵׁם, אַף אֲנִי קוֹרֵא לִבְנִי בְּכוֹרִי שֵׁם&nbsp;–
"כֹּה אָמַר ה' בְּנִי בְכוֹרִי יִשְׂרָאֵל" (שמות ד, כב) (בר"ר סג, ח).






הֵא שָׁוְא&nbsp;– הדרשן שם בפי הקב"ה מדרש לשמו של עשיו: הרי (לכם)
שווא והבל.

אַתֶּם&nbsp;– יצחק ורבקה.

חֲזִירַתְכֶם&nbsp;– כינוי גנאי לעשיו (הוא אדום, הוא רומי) על שם
אותה חיה טמאה.

יִשְׂרָאֵל&nbsp;– תחת השם יעקב שנתנו לו הוריו.





*


כא "הֲלוֹא כִבְנֵי כֻשִׁיִּים אַתֶּם לִי בְּנֵי יִשְׂרָאֵל" (עמוס ט,
ז)&nbsp;– וְכִי כּוּשִׁיִּים שְׁמָם, וַהֲלֹא יִשְׂרָאֵל שְׁמָם?! אֶלָּא מָה כּוּשִׁי מְשֻׁנֶּה בְּעוֹרוֹ, אַף יִשְׂרָאֵל
מְשֻׁנִּים בְּמַעֲשֵׂיהֶם מִכָּל אֻמָּה וְלָשׁוֹן (מועד קטן טז ע"ב).






מְשֻׁנֶּה בְעוֹרוֹ&nbsp;– שונה בצבע עורו (משאר בני האדם).






כב אָמַר רַ' לֵוִי: כָּל מַעֲשֵׂיהֶם שֶׁל יִשְׂרָאֵל מְשֻׁנִּים מֵאֻמּוֹת
הָעוֹלָם: בַּחֲרִישָׁתָם וּבִזְרִיעָתָם וּבִקְצִירָתָם וּבְעִמּוּרָם וּבְדִישָׁתָם, בְּגָרְנֵיהֶם וּבְיִקְבֵיהֶם,
בְּתִגְלַחְתָּם וּבְמִנְיָנָם. בַּחֲרִישָׁתָם&nbsp;– "לֹא תַחֲרשׁ בְּשׁוֹר וּבַחֲמֹר יַחְדָּו" (דברים כב, י);
בִּזְרִיעָתָם&nbsp;– "שָׂדְךָ לֹא תִזְרַע כִּלְאָיִם" (ויקרא יט, יט); בִּקְצִירָתָם&nbsp;– "לֹא תְכַלֶּה פְּאַת שָׂדְךָ
לִקְצֹר וְלֶקֶט קְצִירְךָ לֹא תְלַקֵּט" (שם שם, ט); בְּעִמּוּרָם&nbsp;– "וְשָׁכַחְתָּ עֹמֶר בַּשָּׂדֶה לֹא תָשׁוּב
לְקַחְתּוֹ" (דברים כד, יט); בְּדִישָׁתָם&nbsp;– "לֹא תַחְסֹם שׁוֹר בְּדִישׁוֹ" (שם כה, ד); בְּגָרְנֵיהֶם
וּבְיִקְבֵיהֶם&nbsp;– "הַעֲנֵק תַּעֲנִיק לוָ מִצֹּאנְךָ וּמִגָּרְנְךָ וּמִיִּקְבֶךָ" (שם טו, יד); בְּתִגְלַחְתָּם&nbsp;–
"לֹא תַקִּיפוּ פְּאַת רֹאשְׁכֶם וְלֹא תַשְחִית אֵת פְּאַת זְקָנֶךָ" (ויקרא יט, כז); בְּמִנְיָנָם&nbsp;–
שֶׁאֻמּוֹת הָעוֹלָם מוֹנִים לַחַמָּה וְיִשְׂרָאֵל לַלְּבָנָה, שֶׁנֶּאֱמַר: "הַחֹדֶשׁ הַזֶּה לָכֶם" (שמות יב, ב)
(פס"ר טו, ה; במ"ר י, א).




מְשֻׁנִּים&nbsp;– שונים (בגלל המצוות הכרוכות בכל מה שיפורט
להלן).

בְּעִמּוּרָם&nbsp;– ביצירת עומרים (מן התבואה).

בְּמִנְיָנָם&nbsp;– בדרך שהם מונים את הזמן (לוח השמש למול לוח
הלבנה).





*


כג רַ' לֵוִי בְּשֵׁם רַ' יוֹסֵי בְּרַ' אֶלְעַאי אָמַר: דֶּרֶךְ אֶרֶץ הוּא,
שֶׁיְּהֵא הַגָּדוֹל מוֹנֶה לַגָּדוֹל וְהַקָּטָן מוֹנֶה לַקָּטָן. עֵשָׂו, שֶׁהוּא גָדוֹל, מוֹנֶה לַחַמָּה, שֶׁהִיא
גְּדוֹלָה, וְיַעֲקֹב, שֶׁהוּא קָטָן, מוֹנֶה לַלְּבָנָה, שֶׁהִיא קְטַנָּה. אָמַר רַ' נַחֲמָן: וְהוּא סִימָן
טוֹב: עֵשָׂו מוֹנֶה לַחַמָּה&nbsp;– מָה חַמָּה זוֹ שׁוֹלֶטֶת בַּיּוֹם וְאֵינָה שׁוֹלֶטֶת בַּלַּיְלָה, כָּך עֵשָׂו יֵשׁ
לוֹ חֵלֶק בָּעוֹלָם הַזֶּה וְאֵין לוֹ חֵלֶק בָּעוֹלָם הַבָּא; יַעֲקֹב מוֹנֶה לַלְּבָנָה&nbsp;– מָה הַלְּבָנָה הַזּוֹ
שׁוֹלֶטֶת בַּלַּיְלָה וּבַיּוֹם, כָּךְ יַעֲקֹב יֵשׁ לוֹ חֵלֶק בָּעוֹלָם הַזֶּה וּבָעוֹלָם הַבָּא.

וְאָמַר רַב נַחֲמָן: כָּל־זְמַן שֶׁאוֹרוֹ שֶׁל גָּדוֹל מַבְהִיק בָּעוֹלָם אֵין אוֹרוֹ
שֶׁל קָטָן מִתְפַּרְסֵם; שָׁקַע אוֹרוֹ שֶׁל גָּדוֹל&nbsp;– מִתְפַּרְסֵם אוֹרוֹ שֶׁל קָטָן. כָּךְ, כָּל זְמַן שֶׁאוֹרוֹ
שֶׁל עֵשָׂו מַבְהִיק בָּעוֹלָם אֵין אוֹרוֹ שֶׁל יַעֲקֹב מִתְפַּרְסֵם; שָׁקַע אוֹרוֹ שֶׁל עֵשָׂו&nbsp;– מִתְפַּרְסֵם
אוֹרוֹ שֶׁל יַעֲקֹב (בר"ר ו, ג; פס"ר טו, ב; ילק"ש לבראשית, ח).






דֶּרֶךְ אֶרֶץ&nbsp;– כך מקובל וטבעי.

מוֹנֶה לַגָּדוֹל&nbsp;– מחשב את הזמן (וראו מִהּ בקטע הקודם) לפי דבר
המתאים לגודלו.

שׁוֹלֶטֶת&nbsp;– מנהיגה את העולם (בראשית א, טז), נראית.

עֵשָׂו&nbsp;– סמל לרומי ולנצרות.

שׁוֹלֶטֶת בַּלַּיְלָה וּבַיּוֹם&nbsp;– לעיתים נראית הלבנה גם ביום.





*


כד אָמַר רַ' מֵאִיר: שְׁאֵלַנִי רַ' דּוֹסְתַּאי מִן כּוֹכְבָה: מַהוּ
"וּבֶטֶן רְשָׁעִים תֶּחְסָר" (משלי יג, כה)? אָמַרְתִּי לוֹ: גּוֹי אֶחָד הָיָה בְעִירֵנוּ וְעָשָׂה סְעוּדָה
לְכָל בְּנֵי הָעִיר וְזִמְּנַנִי עִמָּהֶם, וְהֵבִיאוּ לְפָנָיו מִכָּל מָה שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעוֹלָמוֹ
בְּשֵׁשֶׁת יְמֵי בְּרֵאשִׁית, וְלֹא הָיָה שֻׁלְחָנוֹ חָסֵר מִכָּל מַטְעַמִּים שֶׁבָּעוֹלָם אֶלָּא אֱגוֹזֵי פֶּרֶךְ בִּלְבַד.
מָה עָשָׂה? נָטַל אֶת הַטַּבְלָה מִלְּפָנֵינוּ, שֶׁהָיְתָה יָפָה כְּשִׁשָּׁה כִּכְּרֵי כֶּסֶף, וּשְׁבָרָהּ. אָמַרְתִּי לוֹ:
מִפְּנֵי מָה עָשִׂיתָ כָּךְ? אָמַר לִי: "אַתֶּם אוֹמְרִים, שֶׁהָעוֹלָם הַזֶּה שֶׁלָּנוּ וְהָעוֹלָם הַבָּא שֶׁלָּכֶם&nbsp;–
אִם אֵין אָנוּ אוֹכְלִים עַכְשָׁו, אֵימָתַי אָנוּ אוֹכְלִים? וְקָרָאתִי עָלָיו: "וּבֶטֶן רְשָׁעִים תֶּחְסָר"
(פס"ר טז, ו; במ"ר כא, כ; מדמ"ש יג, כה).


*

אֱגוֹזֵי פֶּרֶךְ&nbsp;– סוג של אגוזים (הקרויים כך על שם מקום גדילתם
או משום שקליפתם רכה).

נָטַל אֶת הַטַּבְלָה&nbsp;– לקח את השולחן הקטן (שעליו הוגשה
הסעודה).

יָפָה&nbsp;– שווה בערכה.

שֶׁלָּנוּ&nbsp;– של אומות העולם.

עַכְשָׁו&nbsp;– בעולם הזה.

קָרָאתִי עָלָיו&nbsp;– ציטטתי בקשר אליו (את הפסוק).





*


כה כְּשֶׁיִּשְרָאֵל אוֹכְלִים וְשׁוֹתִים מַתְחִילִים בְּדִבְרֵי תוֹרָה וּבְדִבְרֵי
תִּשְׁבָּחוֹת, אֲבָל אֻמּוֹת הָעוֹלָם שֶׁאוֹכְלִים וְשוֹתִים אֵין מַתְחִילִים אֶלָּא בְּדִבְרֵי תִּפְלוּת (מגילה
יב ע"ב).






תִּשְׁבָּחוֹת&nbsp;– שבח והלל לקב"ה.

תִּפְלוּת&nbsp;– בטלה וניבול פה.





*


כו דָּרַשׁ רָבָא: מַהוּ שֶׁנֶּאֱמַר: "לְכָה דוֹדִי נֵצֵא הַשָּׂדֶה" וגו'
(שיר השירים ז, יב)? "לְכָה דוֹדִי"&nbsp;– אָמְרָה כְּנֶסֶת יִשְׂרָאֵל לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא:
רִבּוֹנוֹ שֶׁל עוֹלָם! אַל תְּדִינֵנִי כְּיוֹשְׁבֵי כְּרַכִּים, שֶׁיֵּשׁ בָּהֶם גֵּזֶל וַעֲרָיוֹת וּשְׁבוּעַת שָׁוְא
וּשְׁבוּעַת שֶׁקֶר, אֶלָּא "נֵצֵא הַשָּׂדֶה"&nbsp;– בּוֹא וְאַרְאֲךָ תַּלְמִידֵי חֲכָמִים שֶׁעוֹסְקִים בַּתּוֹרָה מִתּוֹךְ
הַדֹּחַק; "נָלִינָה בַּכְּפָרִים" (שם)&nbsp;– אַל תִּקְרֵי "בַּכְּפָרִים" אֶלָּא "בַּכֹּפְרִים", בּוֹא וְאַרְאֲךָ בְּנֵי
עֵשָׂו, שֶׁהִשְׁפַּעְתָּ לָהֶם טוֹבָה וְהֵם כָּפְרוּ בְּךָ; "נַשְׁכִּימָה לַכְּרָמִים" (שם שם, יג)&nbsp;– אֵלּוּ בָּתֵּי
כְּנֵסִיּוֹת וּבָתֵּי מִדְרָשׁוֹת; "נִרְאֶה אִם פָּרְחָה הַגֶּפֶן" (שם)&nbsp;– אֵלּוּ בַּעֲלֵי מִקְרָא; "פִּתַּח
הַסְּמָדַר" (שם)&nbsp;– אֵלּוּ בַּעֲלֵי מִשְׁנָה; "הֵנֵצוּ הָרִמּוֹנִים" (שם)&nbsp;– אֵלּוּ בַּעֲלֵי תַּלְמוּד; "שָׁם
אֶתֵּן אֶת דֹּדַי לָךְ" (שם)&nbsp;– אַרְאֲךָ כְּבוֹדִי וְגָדְלִי, שֶׁבַח בָּנַי וּבְנוֹתַי (עירובין כא
ע"ב).






דָּרַשׁ רָבָא&nbsp;– והעניק משמעות אלגורית לכל חלקי הפסוק משיר
השירים.

תְּדִינֵנִי&nbsp;– תדון אותי, תתייחס אלי.

כְּרַכִּים&nbsp;– ערים גדולות.

מִתּוֹךְ הַדֹּחַק&nbsp;– כדרכם של יושבי כפרים העמלים קשות כדי
להתפרנס.

אַל תִּקְרֵי&nbsp;– אל תקרא. שיטת מדרש שמציעה קריאה אחרת של מילים
בתנ"ך, בדרך כלל על ידי שינוי ניקודן.

בְּנֵי עֵשָׂו&nbsp;– אומות העולם.

הִשְׁפַּעְתָּ&nbsp;– הענקת בשפע.





*


כז שְׁתֵּי יְרֻשּׁוֹת הִנְחִיל יִצְחָק לִשְׁנֵי בָּנָיו: הִנְחִיל לְיַעֲקֹב
הַקּוֹל, וְכֵן הוּא אוֹמֵר: "הַקֹּל קוֹל יַעֲקֹב" (בראשית כז, כב), וְהִנְחִיל לְעֵשָׂו הַיָּדַיִם,
שֶׁנֶּאֱמַר: "וְהַיָּדַיִם יְדֵי עֵשָׂו" (שם). עֵשָׂו הָיָה מִתְגָּאֶה בִּירֻשָּׁתוֹ, שֶׁנֶּאֱמַר: "וַיֹּאמֶר אֵלָיו
אֱדוֹם לֹא תַעֲבֹר בִּי פֶּן בַּחֶרֶב אֵצֵא לִקְרָאתֶךָ" (במדבר כ, יח), וְיַעֲקֹב מִתְגָּאֶה בִּירֻשָּׁתוֹ,
שֶׁנֶּאֱמַר: "וַנִּצְעַק אֶל ה' אֱלֹהֵי אֲבֹתֵינוּ" (דברים כו, ז); וּלְעָתִיד לָבוֹא שְׁנֵיהֶם נוֹטְלִים
שְׂכָרָם: עֵשָׂו נוֹטֵל שְׂכָרוֹ, שֶׁנֶּאֱמַר: "כִּי רִוְּתָה בַשָּׁמַיִם חַרְבִּי, הִנֵּה עַל אֱדוֹם תֵּרֵד" (ישעיה
לד,ה), וְיַעֲקֹב נוֹטֵל שְׂכָרוֹ, שֶׁנֶּאֱמַר: "עוֹד יִשָּׁמַע בַּמָּקוֹם הַזֶּה… קוֹל שָׂשׂוֹן וְקוֹל
שִׂמְחָה" (ירמיה לג, י–יא) (שמו"ר כא, כ).






הִנְחִיל&nbsp;– נתן לנחלה (כירושה).

עֵשָׂו&nbsp;– הוא אדום, סמלן של רומי והנצרות.

נוֹטְלִים שְׂכָרָם&nbsp;– מקבלים את מה שמגיע להם (כעונש או
כשכר).





*


כח "הַקֹּל קוֹל יַעֲקֹב" (בראשית כז, כב)&nbsp;– אֵין לְךָ תְּפִלָּה
שֶׁמּוֹעֶלֶת שֶׁאֵין בָּהּ מִזַּרְעוֹ שֶׁל יַעֲקֹב; "וְהַיָּדַיִם יְדֵי עֵשָׂו" (שם)&nbsp;– אֵין לְךָ מִלְחָמָה
שֶׁנּוֹצַחַת שֶׁאֵין בָּהּ מִזַּרְעוֹ שֶׁל עֵשָׂו (גיטין נז ע"ב).






מוֹעֶלֶת&nbsp;– מועילה, מתקבלת.

עֵשָׂו&nbsp;– ראו בקטע הקודם.





*


כט הֶן עָם לְבָדָד יִשְׁכֹּן" (במדבר כג, ט)&nbsp;– מַהוּ "הֶן"? כָּל
הָאוֹתִיּוֹת מִזְדַּוְּגוֹת חוּץ מִשְּׁתֵי אוֹתִיּוֹת הַלָּלוּ. כֵּיצַד? א"ט הֲרֵי י', ב"ח הֲרֵי י', ג"ז
הֲרֵי י', ד"ו הֲרֵי י'&nbsp;– נִמְצָא ה' לְעַצְמָּה; וְכֵן הָאוֹת נ' אֵין לָהּ זוּג: י"צ הֲרֵי ק',
כ"פ הֲרֵי ק', ל"ע הֲרֵי ק', מ"ס הֲרֵי ק'&nbsp;– נִמְצָא נ' לְעַצְמָהּ. אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא:
כְּשֵׁם שֶׁשְׁתֵּי אוֹתִיּוֹת הַלָּלוּ אֵינָן יְכוֹלוֹת לְהִזְדַּוֵּג עִם כָּל הָאוֹתִיּוֹת אֶלָּא לְעַצְמָן, כָּך
יִשְרָאֵל אֵינָם יְכוֹלִים לְהִדַּבֵק עִם כָּל הָאֻמּוֹת, אֶלָּא לְעַצְמָם מֻפְרָשִׁים, שֶׁאֲפִלּוּ מֶלֶךְ גּוֹזֵר
עֲלֵיהֶם לְחַלֵּל הַשַּׁבָּת וּלְבַטֵּל אֶת הַמִּילָה אוֹ לַעֲבֹד עֲבוֹדָה זָרָה, הֵם נֶהֱרָגִים וְאֵין מִתְעָרְבִים
בָּהֶם (שמו"ר טו, ז).






מִזְדַּוְּגוֹת&nbsp;– יש להן בן זוג באות אחרת, המשלימה אותו (בערך
המספרי) לעשר או למאה.

אוֹתִיּוֹת הַלָּלוּ&nbsp;– ה"א ונו"ן.

א"ט הֲרֵי י'&nbsp;– אל"ף (1) ועוד טי"ת (9) הן י' (10) וכן
הלאה.

י"צ&nbsp;– עשר ועוד תשעים (וכן הלאה).

אֶלָּא לְעַצְמָן&nbsp;– רק ה"א משלימה ה"א אחרת לעשר (וכן
נו"ן).

לְעַצְמָם מֻפְרָשִׁים&nbsp;– עומדים בנפרד.

מִתְעָרְבִים&nbsp;– מתבוללים (באומות).





*


ל אָמַר לוֹ מִין אֶחָד לְרַ' אֲבִינָא: נֶאֱמַר: "וּמִי כְעַמְּךָ כְּיִשְׂרָאֵל
גּוֹי אֶחָד בָּאָרֶץ" (שמואל ב ז, כג)&nbsp;– מָה גְּדֻלַּתְכֶם? אַף אַתֶּם עִמָּנוּ בִּכְלָל, שֶׁהֲרֵי נֶאֱמַר:
"כָּל הַגּוֹיִם כְּאַיִן נֶגְדּוֹ" (ישעיה מ, יז). אָמַר לוֹ: מִשֶּׁלָּכֶם הֵעִידוּ עָלֵינוּ: "הֵן עָם
לְבָדָד יִשְׁכֹּן וּבַגּוֹיִם לא יִתְחַשָּׁב" (במדבר כג, ט) (סנהדרין לט ע"א–ע"ב).






מִין&nbsp;– נוכרי.

אַף אַתֶּם וגו'&nbsp;– גם אתם כלולים בעיקרון הכללי הזה.

מִשֶׁלָּכֶם&nbsp;– אחד משלכם, בלעם.





*


לא "נְקֻבֵי רֵאשִׁית הַגּוֹיִם" (עמוס ו, א)&nbsp;– שֶׁהֵם בָּאִים מְשְּׁנֵי
רָאשֵׁי הַגּוֹיִם, מִשֵּׁם וּמֵעֵבֶר.

"וּבָאוּ לָהֶם בֵּית יִשְׂרָאֵל" (שם)&nbsp;– אֻמּוֹת הָעוֹלָם בְּשָׁעָה שֶׁהֵן יוֹשְׁבִין
בְּשָׁלוֹם, אוֹכְלִין וְשׁוֹתִין וּמִשְתַּכְּרִין וּמִתְעַסְּקִין בְּדִבְרֵי תִּפְלוּת. מָה הֵן אוֹמְרִין?&nbsp;– מִי
חָכָם כְּבִלְעָם? מִי גִבּוֹר כְּגָלְיָת? מִי עָשִׁיר כְּהָמָן? וְאַחַר כָּךְ בָּאִים יִשְׂרָאֵל וְאוֹמְרִים לָהֶם:
אֲחִיתֹפֶל לֹא הָיָה חָכָם? שְׁלֹמֹה לֹא הָיָה חָכָם? שִׁמְשׁוֹן לֹא הָיָה גִּבּוֹר? דָּוִד לֹא הָיָה גִּבּוֹר?
קֹרַח לֹא הָיָה עָשִׁיר? שְׁלֹמֹה לֹא הָיָה עָשִׁיר? (ויק"ר ה, ג; במ"ר י, ג).






"נְקֻבֵי רֵאשִׁית הַגּוֹיִם"&nbsp;– והפסוק נדרש על עם ישראל שמוצאו משם
ומעבר, מאבותיהן של אומות העולם.

תִּפְלוּת&nbsp;– בטלה ופטפוט.

בָּאִים יִשְׂרָאֵל וגו'&nbsp;– המשך דברי הנביא נדרש כביקורת על עם
ישראל, הנוהגים כדרכם של הגויים, הנוקבים בשמותיהם של גיבוריהם ומתגאים
בהם.





*


לב אָמַר רַ' יִצְחָק: מַעֲשֶׂה בְּקַרְתָּנִית אַחַת, שֶׁהָיְתָה לָהּ שִׁפְחָה
כּוּשִׁית, שֶׁיָּרְדָה לְמַלְאוֹת מִן הָעַיִן הִיא וַחֲבֶרְתָּהּ. אָמְרָה לַחֲבֶרְתָּהּ: חֲבֶרְתִּי, לְמָחָר אֲדוֹנִי
מְגָרֵשׁ אֶת אִשְׁתּוֹ וְנוֹטְלֵנִי לְאִשָּׁה. אָמְרָה לָהּ: לָמָּה?&nbsp;– בִּשְׁבִיל שֶׁרָאָה יָדֶיהָ מְפֻחָמוֹת. אָמְרָה
לָהּ: אִי, שׁוֹטָה שֶׁבָּעוֹלָם, יִשְׁמְעוּ אָזְנַיִךְ מָה שֶׁפִּיךְ מְדַבֵּר: וּמָה אִם אִשְּׁתּוֹ שֶׁהִיא חֲבִיבָה
עָלָיו בְּיוֹתֵר, אַתְּ אוֹמֶרֶת, מִפְּנֵי שֶׁרָאָה יָדֶיהָ מְפֻחָמוֹת שָׁעָה אַחַת רוֹצֶה לְגָרְשָׁהּ; אַתְּ שֶׁכֻּלֵּךְ
מְפֻחֶמֶת וּשְׁחוֹרָה מִמְּעֵי אִמֵּךְ כָּל יָמַיִךְ&nbsp;– עַל אַחַת כַּמָּה וְכַמָּה! כָּךְ לְפִי שֶׁאֻמּוֹת הָעוֹלָם
מוֹנִים לְיִשְׂרָאֵל וְאוֹמְרִים: אֻמָּה זוֹ הֵמִירָה כְּבוֹדָהּ&nbsp;– אוֹמְרִים לָהֶם יִשְׂרָאֵל: וּמָה אִם אָנוּ
לְשָׁעָה כָּךְ נִתְחַיַּבְנוּ, אַתֶּם&nbsp;– עַל אַחַת כַּמָּה וְכַמָּה! (שהש"ר א, ו).






קַרְתָּנִית&nbsp;– בת עיר קטנה.

לְמַלְּאוֹת&nbsp;– לשאוב (מים).

נוֹטְלֵנִי&nbsp;– נושא אותי.

מְפֻחָמוֹת&nbsp;– שחורות ומלוכלכות.

עַל אַחַת כַּמָּה וְכַמָּה&nbsp;– שלא ירצה בך.

מוֹנִים&nbsp;– מצערים, מציקים.

הֵמִירָה כְּבוֹדָהּ&nbsp;– החליפה את אלוהיה באחרים (במעשה העגל [וראו
תהלים קו, כ]).

לְשָׁעָה&nbsp;– לזמן קצר.

נִתְחַיַּבְנוּ&nbsp;– נענשנו (בגלות ובחורבן).





*


לג הַתֶּבֶן וְהַקַּשׁ וְהַמּוֹץ הָיוּ מְרִיבִים זֶה עִם זֶה. זֶה אוֹמֵר:
בִּשְׁבִילִי נִזְרְעָה הַשָּׂדֶה, וְזֶה אוֹמֵר: בִּשְׁבִילִי נִזְרְעָה הַשָּׂדֶה. אָמְרוּ לָהֶם הַחִטִּים: הַמְתִּינוּ
עַד שֶׁתָּבוֹא הַגֹּרֶן וְאָנוּ יוֹדְעִים בִּשְבִיל מִי נִזְרְעָה הַשָּׂדֶה. בָּאָה הַגֹּרֶן, וְיָצָא בַּעַל הַבַּיִת
לִזְרוֹתָהּ. הָלַךְ לוֹ הַמּוֹץ בָּרוּחַ, נָטַל הַתֶּבֶן וְהִשְׁלִיכוֹ לָאָרֶץ, וְנָטַל הַקַּשׁ וּשְׂרָפוֹ; נָטַל אֶת
הַחִטִּים וְעָשָׂה אוֹתָן כְּרִי, וְכָל מִי שֶׁעוֹבֵר מְנַשְּׁקָן. כָּךְ אֻמּוֹת הָעוֹלָם, הַלָּלוּ אוֹמְרִים: אָנוּ
עִקָּר וּבִשְׁבִילֵנוּ נִבְרָא הָעוֹלָם, וְהַלָּלוּ אוֹמְרִים: בִּשְׁבִילֵנוּ נִבְרָא הָעוֹלָם. אָמְרוּ לָהֶם
יִשְׂרָאֵל: הַמְתִּינוּ עַד שֶׁיַּגִּיעַ הַיּוֹם וְאָנוּ יוֹדְעִים בִּשְׁבִיל מִי נִבְרָא הָעוֹלָם. זֶהוּ שֶׁנֶּאֱמַר:
"כִּי הִנֵּה הַיּוֹם בָּא בֹּעֵר כַּתַּנּוּר" (מלאכי ג, יט), וַעֲלֵיהֶם הוּא אוֹמֵר: "תִּזְרֵם וְרוּחַ
תִּשָּׂאֵם, וּסְעָרָה תָּפִיץ אוֹתָם" (ישעיה מא, טז); אֲבָל יִשְׂרָאֵל נֶאֱמַר עֲלֵיהֶם: "וְאַתָּה תָּגִיל
בַּה' בִּקְדוֹשׁ יִשְׂרָאֵל תִּתְהַלָּל" (שם). (בר"ר פג, ה; שהש"ר ז, ג).






בִּשְׁבִילִי&nbsp;– למעני.

תָּבוֹא הַגּרֶן&nbsp;– יגיע זמן איסוף התבואה וזרייתה בגורן.

נָטַל&nbsp;– לקח.

כְּרִי&nbsp;– ערימה.

מְנַשְּׁקָן&nbsp;– שנאמר: "נשקו בר" (תהלים ב, יב).

עֲלֵיהֶם&nbsp;– על אומות העולם.





*


לד דָּרַשׁ רַ' חֲנִינָא בַּר פַּפָּא, וְיֵשׁ אוֹמְרִים רַ' שִׂמְלַאי: לְעָתִיד
לָבוֹא מֵבִיא הַקָּדוֹשׁ בָּרוּךְ הוּא סֵפֶר תּוֹרָה וּמַנִּיחוֹ בְּחֵיקוֹ וְאוֹמֵר: כָּל מִי שֶׁעָסַק בָּהּ יָבוֹא
וְיִטֹּל שְׂכָרוֹ. מִיָּד מִתְקַבְּצִים וּבָאִים הָאֻמּוֹת בְּעִרְבּוּבְיָה, שֶׁנֶּאֱמַר: "כָּל הַגּוֹיִם נִקְבְּצוּ יַחְדָּו
וְיֵאָסְפוּ לְאֻמִּים מִי בָּהֶם יַגִּיד זֹאת וְרִאשֹׁנוֹת יַשְׁמִיעֻנוּ יִתְּנוּ עֵדֵיהֶם וְיִצְדָּקוּ וְיִשְׁמְעוּ
וְיֹאמְרוּ אֱמֶת" (ישעיה מג, ט). אָמַר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: אַל תִּכָּנְסוּ לְפָנַי
בְּעִרְבּוּבְיָה, אֶלָּא תִּכָּנֵס כָּל אֻמָּהּ וְאֻמָּה וְסוֹפְרֶיהָ. מִיָּד נִכְנְסָה לְפָנָיו מַלְכוּת רוֹמִי תְּחִלָּה.
אוֹמֵר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: בַּמֶּה עֲסַקְתֶּם? אוֹמְרִים לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם, הַרְבֵּה
שְׁוָקִים תִּקַּנּוּ, הַרְבֵּה מֶרְחֲצָאוֹת עָשִׂינוּ, הַרְבֵּה כֶּסֶף וְזָהָב הִרְבִּינוּ, וְכֻלָּם לֹא עָשִׂינוּ אֶלָּא
בִּשְׁבִיל יִשְׂרָאֵל, כְּדֵי שֶׁיִּתְעַסְּקוּ בַתּוֹרָה. אוֹמֵר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: שׁוֹטִים שֶׁבָּעוֹלָם,
כָּל מָה שֶׁעֲשִׂיתֶם&nbsp;– לְצֹרֶךְ עַצְמְכֶם עֲשִׂיתֶם: תִּקַּנְתֶּם שְׁוָקִים&nbsp;– לְהוֹשִׁיב בָּהֶם זוֹנוֹת,
מֶרְחֲצָאוֹת&nbsp;– לְעַדֵּן בָּהֶם עַצְמְכֶם; כֶּסֶף וְזָהָב&nbsp;– שֶׁלִּי הוּא, שֶׁנֶּאֱמַר: "לִי הַכֶּסֶף וְלִי הַזָּהָב,
נְאֻם ה' צְבָאוֹת" (חגי ב, ח). כְּלוּם יֵשׁ בָּכֶם מַגִּיד "זֹאת"? שֶׁנֶּאֱמַר: "מִי בָהֶם יַגִּיד
זֹאת"? (ישעיה מג, ט), וְאֵין "זֹאת" אֶלָּא תּוֹרָה, שֶׁנֶּאֱמַר: "וְזֹאת הַתּוֹרָה אֲשֶׁר שָׂם משֶׁה"
(דברים ד, מד)&nbsp;– מִיָּד יוֹצְאִין בְּפַחֵי נֶפֶשׁ.



יָצְאָה מַלְכוּת רוֹמִי וְנִכְנְסָה מַלְכוּת פָּרָס אַחֲרֶיהָ. אוֹמֵר לָהֶם הַקָּדוֹשׁ
בָּרוּךְ הוּא: בַּמֶּה עֲסַקְתֶּם? אוֹמְרִים לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם, הַרְבֵּה גְשָׁרִים גָּשַׁרְנוּ וְהַרְבֵּה
כְּרַכִּים כָּבַשְׁנוּ, הַרְבֵּה מִלְחָמוֹת עָשִׂינוּ, וְכֻלָּם לֹא עָשִׂינוּ אֶלָּא בִּשְבִיל יִשְׂרָאֵל, כְּדֵי
שֶׁיִּתְעַסְּקוּ בַּתּוֹרָה. אוֹמֵר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: כָּל מָה שֶׁעֲשִׂיתֶם&nbsp;– לְצֹרֶךְ עַצְמְכֶם עֲשִׂיתֶם:
תִּקַּנְתֶּם גְּשָׁרִים&nbsp;– לִטֹּל מֵהֶם מֶכֶס, כְּרַכִּים&nbsp;– לַעֲשׂוֹת בָּהֶם אַנְגַּרְיָא; מִלְחָמוֹת&nbsp;– אֲנִי עָשִׂיתִי,
שֶׁנֶּאֱמַר: "ה' אִישׁ מִלְחָמָה" (שמות טו, ג); כְּלוּם יֵשׁ בָּכֶם מַגִּיד "זֹאת"? מִיָּד יוֹצְאִין
מִלְּפָנָיו בְּפַחֵי נֶפֶשׁ. וְכֵן כָּל אֻמָּה וְאֻמָּה.






סוֹפְרֶיהָ&nbsp;– חכמיה.

בִּשְׁבִיל&nbsp;– למען.

לְעַדֵּן&nbsp;– לפנק.

כְּלוּם יֵשׁ וגו'&nbsp;– והפסוק נדרש כשאלה: האם יש ביניכם מי שיודע
את התורה (המכונה "זאת")?

פַּחֵי נֶפֶשׁ&nbsp;– אכזבה.

כְּרַכִּים&nbsp;– ערים גדולות.

אַנְגַּרְיָא&nbsp;– גיוס כפוי של אנשים לעבודת המלכות.





*


אוֹמְרִין הָאֻמּוֹת לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם, כְּלוּם כָּפִיתָ עָלֵינוּ הַר
כְּגִגִּית, כְּדֶרֶךְ שֶׁכָּפִיתָ עַל יִשְׂרָאֵל, וְלֹא קִבַּלְנוּהָ? מִיָּד אוֹמֵר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא:
"וְרִאשֹׁנוֹת יַשְׁמִיעֻנוּ" (ישעיה מג, ט)&nbsp;– שֶׁבַע מִצְווֹת שֶקִּבַּלְתֶּם הֵיכָן קִיַּמְתֶּם? אוֹמְרִין
הָאֻמּוֹת לִפְנֵי הַקָּדוֹשׁ בָּרוךְ הוּא: רִבּוֹנוֹ שֶׁל עוֹלָם, יִשְׂרָאֵל שֶׁקִּבְּלוּהָ הֵיכָן קִיְּמוּהָ? אוֹמֵר
לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: אֲנִי מֵעִיד בְּיִשְׂרָאֵל שֶׁקִּיְּמוּ אֶת הַתּוֹרָה כֻּלָּהּ. אוֹמְרִים לְפָנָיו:
רִבּוֹנוֹ שֶׁל עוֹלָם, כְּלוּם יֵשׁ אָב שֶׁמֵּעִיד עַל בְּנוֹ, שֶׁנֶּאֱמַר: "בְּנִי בְּכֹרִי יִשְׂרָאֵל" (שמות ד,
כב)? אוֹמֵר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: שָׁמַיִם וָאָרֶץ יָעִידוּ בָּהֶם, שֶקִּיְּמוּ אֶת הַתּוֹרָה כֻּלָּהּ.
אוֹמְרִים לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם, אַף שָׁמַיִם וָאָרֶץ נוֹגְעִים בְּעֵדוּתָם, שֶׁנֶּאֱמַר: "אִם לֹא
בְרִיתִי יוֹמָם וָלָיְלָה חֻקּוֹת שָׁמַיִם וָאָרֶץ לֹא שָׂמְתִּי" (ירמיה לג, כה). אוֹמֵר לָהֶם הַקָּדוֹשׁ
בָּרוּךְ הוּא: מִכֶּם יָבוֹאוּ וְיָעִידוּ בָּהֶם בְּיִשְׂרָאֵל, שֶׁקִּיְּמוּ אֶת הַתּוֹרָה כֻּלָּהּ. יָבוֹא נִמְרוֹד
וְיָעִיד בְּאַבְרָהָם, שֶׁלֹּא עָבַד עֲבוֹדָה זָרָה; יָבוֹא לָבָן וְיָעִיד בְּיַעֲקֹב, שֶׁלֹּא נֶחֱשַׁד עַל הַגֵּזֶל;
תָּבוֹא אֵשֶׁת פּוֹטִיפֶרַע וְתָעִיד בְּיוֹסֵף, שֶׁלֹּא נֶחֱשַׁד עַל הָעֲבֵרָה; יָבוֹא נְבוּכַדְנֶצַּר וְיָעִיד
בַּחֲנַנְיָה מִישָׁאֵל וַעֲזַרְיָה, שֶׁלֹּא הִשְׁתַּחֲווּ לַצֶּלֶם; יָבוֹא דַּרְיָוֶשׁ וְיָעִיד בְּדָנִיֵּאל, שֶׁלֹּא בִּטֵּל
אֶת הַתְּפִלָה; יָבוֹא בִּלְדַּד הַשּׁוּחִי וְצוֹפַר הַנַּעֲמָתִי וֶאֱלִיפַז הַתֵּימָנִי וֶאֱלִיהוּא בֶּן בַּרַכְאֵל
הַבּוּזִי וְיָעִידוּ בָּהֶם בְּיִשְׂרָאֵל, שֶׁקִּיְּמוּ אֶת הַתּוֹרָה כֻּלָּהּ, שֶׁנֶּאֱמַר: "יִתְּנוּ עֵדֵיהֶם וְיִצְדָּקוּ"
(ישעיה מג, ט).



אוֹמְרִים לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם, תְּנֶהָ לָנוּ מֵרֹאשׁ וְנַעֲשֶׂנָּה. אוֹמֵר
לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא: שׁוֹטִים שֶׁבָּעוֹלָם, מִי שֶׁטָּרַח בְּעֶרֶב שַׁבָּת אוֹכֵל בְּשַׁבָּת, מִי שֶׁלֹא טָרַח
בְּעֶרֶב שַׁבָּת מֵהֵיכָן יֹאכַל בְּשַׁבָּת? אֶלָּא אַף עַל פִּי כֵן מִצְוָה קַלָּה יֵשׁ לִי, וְסֻכָּה שְׁמָהּ, לְכוּ
וַעֲשׂוּ אוֹתָהּ. מִיָּד כָּל אֶחָד וְאֶחָד נוֹטֵל וְהוֹלֵךְ וְעוֹשֶׂה סֻכָּתוֹ בְּרֹאשׁ גַּגּוֹ. וְהַקָּדוֹשׁ בָּרוּךְ
הוּא מַקְדִּיר עֲלֵיהֶם חַמָּה כְּבִתְקוּפַת תַּמּוּז, וְכָל אֶחָד וְאֶחָד מְבַעֵט בְּסֻכָּתוֹ וְיוֹצֵא, שֶׁנֶּאֱמַר:
"נְנַתְּקָה אֶת מוֹסְרוֹתֵימוֹ וְנַשְׁלִיכָה מִמֶּנּוּ עֲבֹתֵימוֹ" (תהלים ב, ג).



מִיָּד הַקָּדוֹשׁ בָּרוּךְ הוּא יוֹשֵׁב וּמְשַׂחֵק לָהֶם, שֶׁנֶּאֱמַר: "יוֹשֵׁב בַּשָּׁמַיִם
יִשְׂחָק, אֲדֹנָי יִלְעַג לָמוֹ" (שם שם, ד) (עבודה זרה ב ע"א–ג ע"ב; ילק"ש לישעיה,
מג).






כָּפִיתָ עָלֵינוּ וגו'&nbsp;– על יסוד המסורת (שבת פח ע"א), שהקב"ה
תלה מעל עם ישראל את הר סיני, כגיגית הפוכה, ואילץ אותם באיום השמדה לקבל
את התורה.

"וְרִאשֹׁנוֹת יַשְׁמִיעֻנוּ"&nbsp;– נלמד מן המצוות הראשונות (שהוטלו על
בני האדם כולם).

שֶׁבַע מִצְווֹת&nbsp;– הן "שבע מצוות בני נח", שכל בני האדם חייבים
בהן (כגון איסור על שפיכות דמים).

אָב שֶׁמֵעִיד עַל בְּנוֹ&nbsp;– אך קרוב משפחה פסול לעדות.

נוֹגְעִים בְּעֵדוּתָם&nbsp;– יש להם עניין אישי בעדות, כיוון שקיומם
תלוי בקיום התורה (כדרשת הפסוק מספר ירמיה).

נִמְרוֹד&nbsp;– שהשליך את אברהם, לפי האגדה, אל כבשן האש משום
שסירב לעבוד אלוהים אחרים.

לָבָן&nbsp;– ראו בראשית לא, לו–לט.

הָעֲבֵרָה&nbsp;– גילוי עריות.

דָּנִיֵּאל&nbsp;– ראו דניאל ו, יא.

בִּלְדַּד הַשּׁוּחִי וגו'&nbsp;– רעיו של איוב, שיעידו על צדקתו (והתלמוד
מניח שאיוב מבני ישראל היה).

"יִתְּנוּ עֵדֵיהֶם וְיִצְדְקוּ"&nbsp;– והפסוק נדרש כאומר כי אומות העולם
הן שיעידו על צדקת עם ישראל.

תְּנֶּהָ לָנוּ מֵרֹאשׁ&nbsp;– תן לנו כעת את התורה ונתחיל מחדש,
מבראשית.

מִצְוָה קַלָּה&nbsp;– שאינה דורשת הוצאות רבות ולא קשה לבצעה.

נוֹטֵל&nbsp;– לוקח.

מַקְדִּיר עֲלֵיהֶם חַמָּה&nbsp;– מזריח עליהם שמש חמה במיוחד, כבתקופת
שיא הקיץ.

"מוֹסְרוֹתֵימוֹ", "עֲבֹתֵימוֹ"&nbsp;– הן המצוות המגבילות את האדם
כחבלים.

מְשַׂחֵק&nbsp;– לועג, צוחק.
