שהם פלאסר: משירי שיללער
	








עַל מִגְּדַּל הֵיכָלוֹ בֵּין מְרוֹמֵי קָרֶת

נִשְׁקָף מִטוּב לֵב עַל פְּנֵי צוֹר הַגְּבָרֶת

עָמַד פִּלְאֶסֶר וְרֵעוֹ הָאוֹרֵחַ

"כָּל אֵלֶּה יַחַד לִי יַטּוּ אָזְנָיִם

"אָמַר אֶל רֵעֵהוּ מֶלֶךְ מִצְרָיִם

"רְאֵה מַה נָּעִים חֶבְלִי וּמַה פּוֹרֵחַ!"





"אָמְנָם כִּי חֶסֶד אֵל גָּבַר עָלֶיךָ

"כִּי כָל הַגִּבּוֹרִים אֲשֶׁר כְּגִילֶךָ

"תַּחַת עֹז שִׁבְטֶךָ יֶחֶרְדוּ עַתָּה

"אַךְ שָׁאַר עוֹד אֶחָד נָקָם לָקַחַת

"עוֹד לֹא תַרְגִּיעַ עוֹד לֹא תִּמְצָא נַחַת

"כָּל עֵת אֶת אוֹיִבְךָ זֶה לֹא הִכְרַתָּה."





הַמֶּלֶךְ לְדַבֵּר לֹא כִלָּה עֲדֶנָּה

וּמְבַשֵּׂר מִכִּתִּים בָּא דֶּרֶךְ הֵנָּה

וְכֹה אָמַר לָעָרִיץ נוֹפֵל כּוֹרֵעַ

"קַח פָּר וָכֶבֶשׂ מֵחֶלְבֵי הָעֵדֶר

"וּזְבַח לָאֵל תּוֹדָה שַׁלֵּם הַנֶּדֶר

"וּקְרָא יוֹם טוֹב לָעָם יוֹם הִתְרוֹעֵעַ





"שׂנַאֲךָ הַצַּר עַל שְׂדֵה הַמִּלְחֶמֶת

"שָׁם נָפַל שָׁדוּד מֵחֶרֶב נוֹקֶמֶת

"חֵילוֹ פָּנוּ עֹרֶף נָסוּ לַבּשֶׁת

"גַּם כָּל בַּעֲלֵי בְרִיתוֹ שָׁם פָחֲדוּ פָחַד." –

וּמִדֵּי דַבְּרוֹ לִפְנֵי שְׁנֵיהֶם יָחַד

הוֹצִיא קָדְקֹד נוֹרָא מִסֵּפֶל נְחֹשֶׁת





"אוּלָם עוֹד אֵינְךָ שַׁאֲנָן מִכָּל פֶּגַע

"כִּי אַף אִם בְּאַרְצֶךָ לֹא יִקְרַב נֶגַע

אָמַר הַמּוֹלִךְ עַל אֶרֶץ מִצְרָיִם

"הֵן עַל בָּמֳתֵי גַלִּים אֳנִיּוֹתֶיךָ

"הֲלֹא תִירָא הַרְבֵּה לִגְדוּלָתֶךָ

פֶּן יוֹרִידֵן סַעַר בִּמְצוֹלוֹת מָיִם





עוֹד הַמֶּלֶךְ מְדַבֵּר בְּאָזְנֵי הָרֵעַ

וּמֵהַחוֹף קוֹל עַנּוֹת יִשָּׁמֵעַ

קוֹל שָׂשׂוֹן וּתְרוּעָה קוֹל תֹּף וָנֶבֶל

חוֹבְלִים וּמַלָּחִים זִמְרָה יִשָּׂאוּ

כִּי מֵאֶרֶץ מֶרְחָק הָאֳנִיּוֹת בָּאוּ

וּמִכָּל סְגֻלּוֹת חֶמְדָּה נוֹשְׂאוֹת סֵבֶל





נִבְהָל אָמַר הַמֶּלֶךְ הָאוֹרֵחַ

"שִׁמְשְׁךָ כַּיּוֹם שִׁבְעָתַיִם זוֹרֵחַ

"וּבְכָל זֹאת אָחִי אַל תִּבְטַח לָנֶצַח

"הֵן גְּדוּדֵי תַרְשִׁישׁ גִּבּוֹרֵי הֶחָיִל

"בָּאִים עָלֶיךָ לָרוֹב כְּכוֹכְבֵי לָיִל

"וּמְכֵרוֹתֵיהֶם שֹׁד חָמָס וָרֶצַח."





וּבְטֶרֶם אֶת דְבָרָיו כִּלָּה הַמֶּלֶךְ

וְהִנֵּה הָמוֹן חוֹגֵג הָלוֹךְ יֵלֶךְ

כֻּלָּם יָרִיעוּ מִזָּקֵן עַד נָעַר

"בֹּאוּ נִכְרָעָה נוֹדֶה לֵאלֹהֵינוּ

"כִּי הִצִּילָנוּ מִכַּף כָּל אוֹיְבֵנוּ

"כִּי אֳנִיּוֹת תַּרְשִׁישׁ הֵפִיץ הַסָּעַר.





וַיּוֹסֶף מֶלֶךְ מִצְרַיִם לִתְמוֹהַּ

"אָמְנָם כִּי אַתָּה עַתָּה בְּרוּךְ אֱלוֹהַּ

"אַךְ אֶדְאַג לָךְ כִּי אֵל קַנָּא הִנֵּהוּ

"יַעַן עוֹד בַּתֵּבֵל לֹא חָיָה גֶבֶר

"אֲשֶׁר יִמְצָא רַק טוֹב לֹא יִרְאֶה שֶׁבֶר

וּמְאֻשָּׁר לָנֶצַח מִי יִמְצָאֵהוּ?





"אַף אֲנִי כָּמוֹךָ שָׁלֵו הָיִיתִי

"וֵאלֹהִים עוֹזֵר לִי בַּאֲשֶׁר פָּנִיתִי

"אַךְ גַּם עָלֵי עָבַר כּוֹס בָּא הַשֶּׁבֶר

"בֵּן פֹּרָת הָיֹה לִי מִגִּזְעִי נֵצֶר

"זֶה אֲשֶׁר יִמְלוֹךְ תַּחְתָּי יוֹרֵשׁ עֶצֶר

"וּבְעוֹדֶנּוּ בְּאִבּוֹ יָרַד לַקֶּבֶר





"לָכֵן הִתְפַּלֵּל אֶל יֹשֵׁב שָׁמָיִם

"וּבִתְחִנָּה תִּפְרוֹשׂ אֵלָיו כַּפָּיִם

"כִּי אֶל הַטּוֹב גַּם הָרָע בְּךָ יְשַׁלֵּחַ

"אָז אָחִי תֵּלֵךְ לָבֶטַח דַּרְכֶּךָ

"פַּחַד פִּתְאֹם לֹא יִקְרַב בְּאֹהָלֶיךָ

"לֹא תִירָא כֹּל וְהָיִיתָ אַךְ שָׂמֵחַ





"וְאִם לֹא יִשְׁעֶה הָאֵל לִתְפִלָּתֶךָ

"הָבֵא אַתָּה הָרָעָה אֶל בֵּיתֶךָ

"הַאֲזִינָה לִדְבָרִי וְאַל תֵּט מִמֶּנּוּ

"בֹּא אֶל אוֹצְרוֹתֶיךָ וּבְחוֹן אֹתָנָה

"אֶת בְּחִיר חֲמוּדוֹתָן קַח מִכֻּלָּנָה

"וּלְכִפּוּרִים הַיָּמָה תִּשְׁלִיכֶנּוּ





וַיַּעַן רֵעֵהוּ בְּשִׁבְרוֹן מָתְנָּיִם

"מִכָּל אוֹצְרוֹתַי מַחֲמַדֵּי עֵינָיִם

"אַךְ זֶה הַשֹּׁהַם לִי סְגֻלַּת תִּפְאָרֶת

"אֹתוֹ לֵאלֹהִים הַחֲרֵם אַחֲרִימֶנּוּ

"כִּדְבָרְךָ אֶעֱשֶׂה לֹא אָסוּר מִמֶּנּוּ"

וְהַשֹּׁהַם צָלַל בַּיָּם כַּעֹפֶרֶת





וּמִמָּחֳרָת בִּפְנוֹת צִלְלֵי הָאֹפֶל

וְדַיָּג הִשְׁכִּים וַיָּבֹא אֶל הָעֹפֶל

וַיַּעֲמֹד לִפְנֵי הֶעָרִיץ וַיֹּאמֶר

"אֶת הַדַּג הַזֶּה כַּיּוֹם בְּחֶרְמִי צַדְתִּי

"עֲלֵי רֹב גָּדְלוֹ מִשְׁתָּאֶה עָמַדְתִּי

"עַל כֵּן תִּתּוֹ לְךָ שָׁי גָּמַרְתִּי אֹמֶר."





וַיְהִי הַטַּבָּח אֶת הַדָּג פּוֹלֵחַ

וַיָּבֹא בְּחָפְזָה לַהֵיכָל שָׂמֵחַ

וַיִּקְרָא: אֲדוֹנִי יִיטַב לִבֶּךָ

"זֶה הַשֹּׁהַם לְךָ הוּא הֵן תַּכִּירֵהוּ

"בִּמְעֵי הַדָּג מְצָאתִיו הָבָה קָחֵהוּ

"אֵין קֶצֶב אֵין סוֹף אֶל גֹּדֶל אָשְׁרֶךָ!"





וַיִּתֵּן הַמֶּלֶךְ קוֹלוֹ בַּעֲצָבֶת

"הֵן לֹא אוּכַל עוֹד עִמְּךָ יַחַד שָׁבֶת

"כִּי קָרוֹב אֵיד לַפֶתַח רוֹבֵץ קֶרֶץ

"כְּבָר צִוָּה אֵל אֶת הָרָעָה עָלֶיךָ

"לָמָּה אָמוּת גַּם אָנִי בִּגְלָלֶךָ

"וַיִּפֶן בְּאָמְרוֹ וַיֵּלֶךְ אֶל אֶרֶץ."
