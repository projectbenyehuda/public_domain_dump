דַּאֲגַת בָּנִים לַאֲבִיהֶם
	




אִישׁ הָיָה בְּאַחַת הֶעָרִים וְלוֹ אַרְבָּעָה בָּנִים. וַיְהִי כִּי זָקֵן הָאִישׁ וַיִּתֵּן
בַּכּוֹס עֵינוֹ, וַיֵּשְׁתְּ וַיִּשְׁכָּר כָּל הַיָּמִים. וַיְדַבְּרוּ בָּנָיו אֵלָיו יוֹם יוֹם&nbsp;– לַהֲשִׁיבוֹ מִדַּרְכּוֹ,
וְלֹא שָׁמַע אֲלֵיהֶם. וַיִּרְאוּ הַבָּנים כִּי כָּבְדָה אֹזֶן אֲבִיהֶם מִשְּׁמוֹעַ אֶת תּוֹכַחְתָּם, וַיִּוָּעֲצוּ
וַיַּשְׁקוּהוּ יַיִן וַיִּישַּׁן וַיֵּרָדֵם, וַיִּשָּׂאוּהוּ בַּלַּיְלָה וַיְבִיאוּהוּ אֶל בֵּית־הַקְּבָרוֹת. וַיֹּאמְרוּ:
יִיקַץ־נָא אָבִינוּ מִיֵּינוֹ וְרָאָה קְבָרִים סְבִיבוֹתָיו, וְנָפָל עָלָיו פַּחַד אֱלֹהִים וְנָדַר
נֶדֶר־נָזִיר לַיְיָ לְבִלְתִּי שְׁתוֹת עוֹד יַיִן וְשֵׁכָר. הַבֹּקֶר הָיָה, וְהַזָּקֵן עוֹדֶנּוּ יָשֵׁן, וּשְׁנָתוֹ
עָרֵבָה עָלָיו מְאֹד. וְהִנֵּה אֹרְחַת יוֹגְבִים בָּאָה, וּגְמַלֵּיהֶם נוֹשְׂאִים כַּדֵּי תִירוֹשׁ וְיַיִן. הֵם
עוֹבְרִים עַל־פְּנֵי הַקְּבָרוֹת וְהִנֵּה קוֹל שָׁאוֹן עוֹלֶה מִן הָעִיר בְּאָזְנָם, וַיִּבָּהֲלוּ וַיֵּחָפֵזוּ
וַיֹּאמְרוּ אִישׁ אֶל אָחִיו: אֵין זֹאת כִּי מִלְחָמָה בָּעִיר, חֶרֶב אִישׁ בְּרֵעֵהוּ, וְעַתָּה נִמָּלְטָה נָּא
עַל נַפְשֵׁנוּ. וַיְמַלְּטוּ אֶת יֵינָם אֶל בֵּין הָעֵצִים אֲשֶׁר עַל הַקְּבָרִים, וַיַּעַלוּ עַל הַגְּמַלִּים
וַיִּרְכְּבוּ וַיָּנֻסּוּ. וְהַשָּׁאוֹן הוֹלֵךְ וְגָדֵל, וַיַּעַל גַּם בְּאָזְנֵי הַזָּקֵן הַיָּשֵׁן וַיִּיקַץ. וַיִּפְקַח אֶת
עֵינָיו, וַיַּרְא וְהִנֵּה כַּדִּים מְלֵאִים יַיִן לְנֶגֶד עֵינָיו. וַיִּגַּשׁ אֲלֵיהֶם וַיֵּשְׁתְּ מֵהֶם לָרְוָיָה,
וַיִּיטַב לִבּוֹ וַיָּשָׁב וַיִּשְׁכַּב וַיֵּרָדַם. וְאֶת הַקְּבָרוֹת לֹא רָאָה, כִּי עֵינָיו הָיוּ אֶל הַכַּדִּים וְאֶל
דַּם הָעֲנָבִים אֲשֶׁר בְּתוֹכָם. וְהַבָּנִים בָּאוּ לִרְאוֹת אֶת שְׁלוֹם אֲבִיהֶם&nbsp;– אִם הֵקִיץ וְאִם סָר
יֵינוֹ מֵעָלָיו, וְהִנֵּה הוּא הוֹזֶה שׁוֹכֵב וּסְבִיבוֹתָיו חָבִיוֹת מְלֵאוֹת עֲמֻקּוֹת וּרְחָבוֹת וַיֹּאמְרוּ
כֻּלָּם: אִם הִקְרָה יְיָ לְךָ גַּם בַּחֲצֵר מָוֶת,&nbsp;– חָלִילָה לָנוּ לְהַזִּיר אוֹתְךָ עוֹד מִמֶּנּוּ. וַיָּקֻמוּ
וַיִּקְחוּ אֶת אֲבִיהֶם וַיִּשָּׂאוּהוּ הַבַּיְתָה. וַיָּשִׂימוּ לָהֶם לְחֹק לַעֲבֹד אִישׁ יוֹמוֹ, וּלְהַשְׁקוֹת בִּשְׂכַר
הַיּוֹם אֶת אֲבִיהֶם יַיִן. וַיֵּשֵׁב הַזָּקֵן בְּתוֹךְ בָּנָיו יָמִים רַבִּים, וַיֵּשְׁתְּ שָׁתֹה וּמָצֹה עַד בּוֹא
יוֹמוֹ.
