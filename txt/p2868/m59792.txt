


	
	הַכְּרוּב וְהַכְּרוּבִית
	




עַל שְׂפַת נַחַל הַשּׁוֹטֵף סָמוּךְ לְעֵדֶן רִוּוּ אֶת נַפְשָׁם כְּרוּב־הַקּוֹלָס
וְהַכְּרוּבִית וַיִּגְדְּלוּ וַיָּנוּבוּ שְׁנֵיהֶם יַחְדָּו, כִּתְאוֹמֵי אָח וְאָחוֹת. יוֹם יוֹם, כִּזְרֹחַ הַשֶּׁמֶשׁ,
יִפְתְּחוּ הַכְּרוּב וְהַכְּרוּבִית אֶת פִּרְחֵיהֶם מִצֶּבַע הַלִּימוֹן וְיַעֲנִיקוּ סִיג וָצוּף לְכָל מְבַקְרֵיהֶם,
וּבַעֲרֹב הַיּוֹם יִסְגְּרוּ אֶת פִּרְחֵיהֶם הָעֲנֻגִּים וְיִתְרַחֲצוּ בְּטַל הַלָּיְלָה. כֹּה הָיוּ פוֹרְחִים יוֹם
יוֹם כְּחָדְשַׁיִם יָמִים.

וַיְהִי הַיּוֹם וַתֵּלֶךְ חַוָּה, אֵם כָּל חַי, לַשָּׂדֶה לִלְקֹט עֲשָׂבִים וְשָׁרָשִׁים
וַתִּשְׁמַע בְּדַרְכָּהּ הֲמֻלָּה רַבָּה וַתֵּפֶן כֹּה וָכֹה וַתֵּרֶא דְבוֹרִים מִתְעוֹפְפוֹת וְהֵן טְעוּנוֹת סִיג
וָצוּף, וַתִּתְאַו חַוָּה לְהַכִּיר וְלִרְאוֹת אֶת הַצְּמָחִים הַמַּעֲנִיקִים צוּף לַדְּבוֹרִים וַתִּשְׁאַל אֶת אַחַת
הַדְּבוֹרִים עַל הַדֶּרֶךְ וְהַמָּקוֹם אֲשֶׁר מִשָּׁם הֵן בָּאוֹת. וַתַּעַן הַדְּבוֹרָה וַתּאמַר: “בְּמֶרְחַק פַּרְסָה
אַחַת מִפֹּה תָּבוֹאִי אֶל כַּר נִרְחָב וְאַחֲרֵי הַכַּר גִּבְעָה גְדוֹלָה וּמַאֲחוֹרֵי הַגִּבְעָה שׁוֹטֵף נַחַל
מָהִיר וְעַל שְׂפַת הַנַּחַל גְּדֵלִים הַצְּמָחִים הַמַּעֲנִיקִים לָנוּ צוּף וָסִיג כְּאַוַּת נַפְשֵׁנוּ.”

וַתֵּלֶךְ חַוָּה וַתַּעֲבֹר אֶת הַכַּר וְאֶת הַגִּבְעָה וַתָּבוֹא אֶל הַנָּחַל. וַיְהִי כַאֲשֶׁר
קָרְבָה אֶל שְׂפַת הַנַּחַל וַתַּעֲמֹד כְּנִדְהָמָה מֵרֹב הַהוֹד וְהַתִּפְאֶרֶת אֲשֶׁר לַעֲלֵי הַצְּמָחִים הַפּוֹרִים
וְהָרַעֲנַנִים וּמִזִּיו הַפְּרָחִים הָרַבִּים הַמֻּקָּפִים הָמוֹן שׁוֹקֵק שֶׁל דְּבוֹרִים מִתְעוֹפְפוֹת. וַתִּקְרַב
חַוָּה אֶל אַחַד הַצְמָחִים וַתֹּאמַר: “הַגִּידָה נָא לִי, מַה שִּׁמְךָ וּמִי הֵבִיאֲךָ הֲלֹם?” וַיַּעַן הַצֶּמַח:
“שְׁמִי כְרוּב וַאֲנִי בֶּן טִפּוּחִים לְמַזַּל הַכְּרוּב הַמְמֻנֶּה עַל עַנְנֵי הַחֹרֶף”. וַתִּקְרַב חַוָּה אֶל
צֶמַח אַחֵר וַתִּשְׁאֵל גַּם לִשְׁמוֹ וּלְמוֹצָאוֹ הוּא. וַיַּעַן הַצֶּמַח וַיֹּאמַר: “אָנֹכִי שְׁמִי כְרוּבִית
וְהִנְנִי בַת חֲמוּדוֹת לַכְּרוּב הַמְעוֹפֵף מֵעַל לְהָרֵי הַשָּׁלֶג”. וַתֵּפֶן חַוָּה אֶל הַכְּרוּב וְהַכְּרוּבִית
וַתֹּאמַר: “אִם הַדָּבָר יָשָׁר בְּעֵינֵיכֶם, הָקִימוּ לָכֶם נִיר גַּם בְּגִנָּתִי”. וַיִּפְתְּחוּ הַצְּמָחִים אֶת
פִּלְחֵי קְצִיצֵיהֶם שֶׁהִצְהִיבוּ וַיַּעֲנִיקוּ לְחַוָּה זְרָעִים חוּמִים וּזְעִירִים. וַתִּקַּח חַוָּה אֶת
הַזְרָעִים וַתָּשָׁב אֶל גִּנָּתָהּ וַתִּזְרָעֵם בָּעֲרוּגָה אֲשֶׁר הֵכִינָה לְמַעֲנָם. וַיָּצִיצוּ הַזְּרָעִים וַיִּשְׂאוּ
עָלִים יָפִים וַתִּשְׁמֹר עֲלֵיהֶם חַוָּה כְּעַל בָּבַת עֵינָהּ. וַיִּגְדְּלוּ הַכְּרוּב וְהַכְּרוּבִית בַּגִּנָּה
וַיִּתְרַחֲבוּ שִׁבְעָתַיִם מַאֲשֶׁר עַל שְׂפַת הַנָּחַל, כִּי טִפְּלָה בָהֶם חַוָּה בְאַהֲבָה. יוֹם יוֹם הִיא מַשְׁקָה
אוֹתָם, מְעַדֶּרֶת וְגַם מְנַכֶּשֶׁת מִסָּבִיב לְשָׁרְשֵׁיהֶם.

אוּלָם כְּכֹל אֲשֶׁר גָּדְלוּ הַצְּמָחִים יוֹתֵר כֵּן נֶהֱפַךְ לִבּוֹ שֶׁל הַכְּרוּב לְרָעָה
עַל הַכְּרוּבִית, כִּי רָאָה הַכְּרוּב אֶת הַכְּרוּבִית הוֹלֶכֶת וּגְדַלָה, הוֹלֶכֶת וּמַרְחִיבָה אֶת עָלֶיהָ,
הוֹלֶכֶת וּמִתְפַּשֶׁטֶת לְמֵרָחוֹק, וַיִּירָא לְנַפְשׁוֹ פֶּן תָּקוּם הַכְּרוּבִית יוֹם אֶחָד וְתִדְחַק אֶת רַגְלוֹ.
וּמֵרֹב כַּעַס וָפַחַד הָיָה הַכְּרוּב הוֹלֵךְ וּמְקַפֵּל אֶת עָלָיו, מִתְכַּוֵּץ וּמִתְכַּנֵס לְתוֹכוֹ.

פַּעַם בָּאָה חַוָּה לְהַשְׁקוֹת וּלְזַבֵּל אֶת הַכְּרוּבִית וַיִּחַר אַפּוֹ שֶׁל הַכְּרוּב
וַתִּבְעַר בּוֹ אֵשׁ הַקִּנְאָה וּמֵרֹב כַּעַס קִפֵּל אֶת כָּל עָלָיו וַיִּצְטַנֵּף כַּכַּדּוּר. רָאֲתָה הַכְּרוּבִית אֶת
הַדָּבָר וַתִּלְעַג לַחֲבֵרָהּ, אֲשֶׁר הַקִּנְאָה אוֹכֶלֶת אוֹתוֹ, וַתִּפְרַע אֶת רֹאשָׁהּ וַתִּצְחַק צְחוֹק אַדִּיר
וַתֶּחֱשׂף אֶת שִׁנֶּיהָ הַלְּבָנוֹת כַּשָּׁלֶג. וְחַוָּה הִרְגִּישָׁה בַדָּבָר, לָקְחָה עָלִים מִסְפָּר וַתְּכַס בָּהֶם אֶת
פְּנֵי הַכְּרוּבִית, כְּדֵי שֶׁלֹא יִתְגַּלֶּה צְחוֹקָהּ לְאוֹר הַשָּׁמֶשׁ; וְאֶל הַכְּרוּב נִגְּשָׁה
וַתְּנַחֲמֵהוּ.

וּמֵאָז וְעַד הַיּוֹם הַזֶּה צוֹחֶקֶת הַכְּרוּבִית וְהָאָדָם מְכַסֶה אֶת פָּנֶיהָ
בְּעָלִים, וְהַכְּרוּב מְקַפֵּל אֶת עָלָיו, מִצְטַנֵּף וּמִתְכַּנֵּס לְתוֹכוֹ, וְהָאָדָם נִגָּשׁ אֵלָיו בַּלָּאט,
תּוֹהֶה עַל קַנְקַנּוֹ, מַקִּישׁ עָלָיו בְּאֶצְבָּעוֹ, כְּדֵי לָדַעַת אֵת אֲשֶׁר עִם לְבָבוֹ.
