


	
	פֶּרַח נַעֲמָן (הַכַּלָּנִית)
	




בִּימֵי מַלְכֵי יִשְׂרָאֵל יָשְׁבָה בַגִּלְגָּל אִשָּׁה אַחַת מִנְשֵׁי בְנֵי הַנְּבִיאִים.
וְהָאִשָּׁה אַלְמָנָה מֵאִישָׁהּ וְלָהּ בַּת יְחִידָה וִיפַת תֹּאַר, לֹא הָיְתָה כָמוֹהָ נַעֲרָה נֶחְמָדָה וְיַעֲלַת
חֵן בְּכָל הָאָרֶץ. וְהָאֵם וְהַבַּת יָשְׁבוּ בְאֹהֶל דַּל וַתִּתְכַּלְכֵּלְנָה מִיגִיעַ כַּפֵּיהֶן; הָאֵם עֶשְׂתָה
מְלָאכָה בַבַּיִת וְהַבַּת עָבְדָה יוֹם יוֹם בַּגַן הַקָּטֹן אֲשֶׁר עַל יַד בֵּיתָן. הַנַּעֲרָה שָׁמְרָה מְאֹד עַל
הַגַּן הַזֶּה וַתִּשְׁקֹד לְהָסִיר מִמֶּנּוּ שָׁמִיר וָשָׁיִת, כִּי אָהֹב אָהֲבָה אֶת נַחֲלַת אָבִיהָ.

פַּעַם בְּפַעַם הָיְתָה מְבִיאָה מֵהַשָּׂדֶה פִּרְחֵי בַר יָפִים לִנְטֹעַ אוֹתָם בַגִּנָּה.

וַתִּגְדַּל הַנַּעֲרָה וַתִּיף וַתְּהִי כְפֶרַח הָאָבִיב, חַכְלִילַת לְחָיַיִם וּשְׁחוֹרַת עֵינָיִם.

וַיְהִי הַיּוֹם וְהַנַּעֲרָה מְלַקֶּטֶת בַּשָּׂדֶה וּגְדוּדֵי אֲרָם פָּשְׁטוּ בָאָרֶץ וַיִּמְצְאוּ אֶת הַנַּעֲרָה וַיִּקָּחוּהָ בַשֶּׁבִי.

לֵב הָאֵם נִשְׁבַּר בְּקִרְבָּהּ מֵרֹב גַּעְגּוּעִים לְבִתָּהּ. וּבְצֵאתָהּ אֶל הַגֵּן בַּבֹּקֶר
וּבָעֶרֶב שָׁפְכָה הָאִשָּׁה דִמְעוֹתֶיהָ, וְהַדְּמָעוֹת יָרְדוּ וְנָפְלוּ עַל פְּנֵי הַפְּרָחִים, אֲשֶׁר טִפְּחָה אוֹתָם
הַנַּעֲרָה. וּפִרְחֵי הָאָבִיב אֲשֶׁר הָיוּ אֲדֻמִּים תָּמִיד, בִּהְיוֹת הַנַּעֲרָה בַבַּיִת, הָפְכוּ בַשָּׁנָה הַזֹּאת
כָּחֹל וְלָבָן.

וְהַנַּעֲרָה נָשְׂאָה חֵן וָחֶסֶד בְּעֵינֵי נַעֲמָן שַׂר צְבָא מֶלֶךְ אֲרָם וַיְבִיאֶהָ אֶל
אַרְמוֹנוֹ לָשֶׁבֶת לִפְנֵי אִשְׁתּוֹ. וְנֶפֶשׁ הַגְּבֶרֶת נִקְשְׁרָה בַנַּעֲרָה טוֹבַת הַשֵּׂכֶל וַתְּהִי לָהּ
כְּאָחוֹת.

וַיְהִי הַיּוֹם וַתִּשְׁפֹּךְ אֵשֶׁת נַעֲמָן לִפְנֵי הַנַּעֲרָה אֶת מְרִי שִׂיחָהּ, כִּי נַעֲמָן אִישָׁהּ הוּא מְצֹרָע.

וַתֹּאמֶר הַנַּעֲרָה אֶל גְּבִרְתָּהּ: אַחֲלֵי אֲדוֹנִי לִפְנֵי הַנָּבִיא אֲשֶׁר בְּשֹׁמְרוֹן אָז יֶאֱסוֹף אוֹתוֹ מִצָּרַעְתּוֹ.

וְנַעֲמָן שַׂר צְבָא מֶלֶךְ אֲרָם שָׁמַע בַּעֲצַת הַנַּעֲרָה אֲשֶׁר מֵאֶרֶץ יִשְׂרָאֵל, וּבַיּוֹם אֲשֶׁר יָצָא לַדֶּרֶךְ נִגְּשָׁה אֵלָיו הַנַּעֲרָה וַתִּפֹּל לְפָנָיו עַל בִּרְכֶּיהָ וַתֹּאמַר:

– יַעֲשֶׂה נָא אֲדוֹנִי חֶסֶד עִם שִׁפְחָתוֹ. בְּשׁוּב אֲדוֹנִי בְּשָׁלוֹם אֶל בֵּיתוֹ, וְהֵבִיא לִי לִתְשׁוּרָה פֶּרַח אֶחָד מִפִּרְחֵי אַרְצִי.

וַיְהִי אַחֲרֵי אֲשֶׁר טָבַל נַעֲמָן בַּיַּרְדֵּן שֶׁבַע פְּעָמִים וּבְשָׂרוֹ שָׁב כִּבְשַׂר נַעַר קָטָן כַּאֲשֶׁר דִּבֶּר אֵלָיו הַנָּבִיא אֱלִישָׁע, וַיָּשָׁב אֶל אַרְצוֹ שָׂמֵחַ וְטוֹב לֵב.

וּבְעָבְרוֹ עַל פְּנֵי הַגֵּנִּים הַמְלֵאִים פְּרָחִים זָכַר אֶת אֲשֶׁר בִּקְשָׁה מִמֶּנּוּ הַנַּעֲרָה מֵאֶרֶץ יִשְׂרָאֵל.

הוּא עָמַד עַל יַד גַּן אֶחָד לִקְטָף שָׁם פֶּרַח, וְהִנֵּה אִשָּׁה זְקֵנָה יוֹצֵאת לִקְרָאתוֹ וְהִיא מִתְנַפֶּלֶת לְרַגְלָיו.

– מִי אַתְּ וּמַה שְּׁאֵלָתֵךְ? – שָׁאֵל שַׂר הַצָּבָא.

– אִשָּׁה קְשַׁת רוּחַ אָנֹכִי – עָנְתָה הַזְּקֵנָה – כִּי בַעְלִי מֵת וּבִתִּי הַיְחִידָה נִלְקְחָה שְׁבִי בִּידֵי צִבְאוֹת מֶלֶךְ אֲרָם.

וְרַחֲמֵי נַעֲמָן נִכְמְרוּ עַל הָאִשָּׁה הָאַלְמָנָה וַיֹּאמֶר לָהּ:

– בְּשׁוּבִי אֶל אַרְצִי אֲבַקֵּשׁ אֶת בִּתֵּךְ וַאֲשַׁלְחֶנָּה לְבֵיתָהּ.

– הֵא לְךָ הַפֶּרַח הַזֶּה – קָרְאָה הָאִשָּׁה – וְעַל יָדוֹ תִּמְצָא אֶת בִּתִּי, אֲשֶׁר גַּם הִיא חַכְלִילַת לְחָיַיִם וּשְׁחוֹרַת עֵינַיִם כָּמוֹהוּ.

וְכַאֲשֶׁר שָׁב נַעֲמָן אֶל בֵּיתוֹ וַיְסַפֶּר לִפְנֵי אִשְׁתּוֹ וְהַנַּעֲרָה אֲשֶׁר עִמָּהּ אֶת
דְּבַר פְּגִישָׁתוֹ עִם הַזְּקֵנָה הָעִבְרִיָה, וַתַּכֵּר הַנַּעֲרָה אֶת דִּבְרֵי אִמָּהּ וַתִּשָּׂא קוֹלָהּ
וַתֵּבְךְּ.

וְנַעֲמָן שָׁמַר אֶת הַבְטָחתוֹ וַיְשַׁלַּח אֶת הַנַּעֲרָה אֶל אִמָּהּ. וּבְשׁוּבָהּ אֶל
מוֹלַדְתָּהּ, טִפְּלָה בְאַהֲבָה רַבָּה בַּגַּן אֲשֶׁר לִפְנֵי בֵית אִמָּהּ. וְשִׁבְעָתַיִם גָּדְלָה אַהֲבָתָהּ לְפֶרַח
הָאָבִיב, אֲשֶׁר עַל יָדוֹ בָּאָה לָהּ גְאֻלָּתָהּ.

וְהָאֵם קָרְאָה לְפִרְחֵי־אָבִיב אֵלֶּה פִּרְחֵי־נַעֲמָן (נַעֲמָנִיּוֹת) לְזֵכֶר הַחֶסֶד הַגָּדוֹל אֲשֶׁר עָשָׂה עִמָּהּ נַעֲמָן שַׂר צְבָא מֶלֶךְ אֲרָם.
