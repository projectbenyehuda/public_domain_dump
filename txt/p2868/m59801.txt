


	
	אֵיךְ בָּא זְקַן־הַשָּׂב אֶל בֵּין הַקָּמָה
	




בְּרֵאשִׁית יְמֵי הַיְצִירָה, עוֹד טֶרֶם הִשְׂכִּיל הָאָדָם לְהַדְבִּיר תַּחְתָּיו אֶת
אֵיתָנֵי הַטֶּבַע, הִדְשִׁיאָה הָאֲדָמָה שִׁפְעַת דְּשָׁאִים. עֲשָׂבִים מַזְרִיעִים זֶרַע לְמִינֵהוּ וְשִׂיחִים
וְאִילָנוֹת אֲשֶׁר זַרְעָם בָּם לְמִינֵהוּ עָלוּ וַיְשַׂגְשְׂגוּ כֻלָּם יַחְדָּו, זֶה עַל יַד זֶה, שְׁלֵוִים
וּשְׁקֵטִים. עִם הָנֵץ הַחַמָּה פִּתְּחוּ עֻלְשֵׁי־הַשָּׂדֶה לִרְוָחָה אֶת סַלְסִלּוֹת פִּרְחֵיהֶם הַכְּחֻלִּים כְּצֶבַע
הָרָקִיעַ לִפְנוֹת בֹּקֶר, וּבִבְלוֹעַ הַשֶּׁמֶשׁ אֶת רְסִיסֵי טַל הַלָּיְלָה, פָּשְׁטוּ הַפִּשְׁתָּנִיּוֹת הַצְּהֻבּוֹת
וְהַוְּרֻדּוֹת אֶת כּוֹתַרְתָּן הַגְּלוּלָה וַיַּעַמְדוּ כָל הַיּוֹם מַזְהִירִים וְצוֹהֲלִים לְאוֹר הַשֶּׁמֶשׁ הַזּוֹרֵעַ
חַיִּים בִּמְלֹא הַיְקוּם. וַיַּרְא הַיּוֹצֵר כִּי טוֹב, וַיּוֹסֶף לְקַשֵּׁט אֶת הָאָרֶץ, וַיְפַזֵּר עַל פָּנֶיהָ
שְׁלַל פְּרָחִים מַרְהִיבֵי־עַיִן בְּיָפְיָם.

וַיְהִי כַאֲשֶׁר אָכַל הָאָדָם מֵעֵץ הַדָּעַת, וַיָּחֶל לַעֲבוֹד אֶת הָאֲדָמָה וְלֶאֱכוֹל
מִפִּרְיָהּ וַיִּלְמַד לְהַבְחִין בֵּין גִּדּוּלֵי הָאֲדָמָה, וַיִּבְחַר לְגַדֵּל לוֹ אֶת הַדָּגָן לְמִינֵהוּ, לִהְיוֹת
לוֹ לְמִחְיָה. וּמֵאָז אָסַר מִלְחָמָה קָשָׁה וּמָרָה עַל צֶאֱצָאֵי הָאֲדָמָה אֲשֶׁר לֹא מָצָא כָל תּוֹעֶלֶת בָּהֶם
וַיְשַׁחֵת אֶת רַחֲמָיו גַּם מִכָּל הַפְּרָחִים הַיָּפִים, אֲשֶׁר עָלוּ בְשָׂדֵהוּ, וַיְבַעֲרֵם כְּכֹל אֲשֶׁר מָצְאָה
יָדוֹ. אָז תִּטּוֹשׁ מִלְחָמָה כְבֵדָה בֵּין צִמְחֵי־הַבַּר וּבֵין הָאָדָם, וְהָעַזִּים אֲשֶׁר בַּצְמָחִים הָאֵלֶּה
עָמְדוּ לְהָגֵן עַל נַפְשָׁם בְּכָל מַאֲמַצֵי כֹחָם. וְהָיָה אִם זָרַע הָאָדָם אֶת דְּגָנוֹ בַשָּׂדֶה וְעָלוּ הַקּוֹץ
וְהַדַּרְדַּר מִן הָאֲדָמָה וַיַּשְׁחִיתוּ וַיְמַעֲטוּ אֶת הַיְבוּל וַיֵּהָפְכוּ לְרוֹעֵץ לָאָדָם הָעוֹבֵד. וּבְבוֹאוֹ
לְבַעֲרָם, הִתְחַדְּקוּ בִבְשָׂרוֹ וַיִּפְצָעוּהוּ בְעָקְצֵיהֶם וּבְשִׂכּוֹתֵיהֶם. וַיִּקְרָא הָאָדָם לָהֶם בְּשֵׁם
“עֲשָׂבִים רָעִים” מֵאָז וְעַד הַיּוֹם הַזֶּה. וַיִּתְחַכֵּם לְהַתְקִין לוֹ כֵלִים מִכֵּלִים שׁוֹנִים לְמַעַן
הַשְׁמִידָם כָּלִיל מֵעַל פְּנֵי הָאֲדָמָה.

וַיְהִי כִּרְאוֹת הָ“עֲשָׂבִים הָרָעִים”, כִּי קָשְׁתָה וְגַם גָּבְרָה עֲלֵיהֶם יַד
הָאָדָם וַתַּעַל שַׁוְעָתָם הַשָּׁמַיְמָה לֵאמוֹר: “רְאֵה, אֱלֹהִים, וְהַבִּיטָה מָה עוֹלֵל לָנוּ בֶּן־הָאָדָם!”
וַיַּקְשֵׁב שַׂר הַטֶּבַע לְקוֹל שַׁוְעָם וַיְנַחֲמֵם: “עוֹד כָּל יְמֵי הָאָרֶץ לֹא תִשְׁבּוֹת הַמִּלְחָמָה בֵּינֵיכֶם
וּבֵין הָאָדָם. הוּא יִשְׁלוֹט בָּכֶם וְאַתֶּם תִּשְׁלְטוּ בוֹ!” וּמַלְאָכִים שֻׁלְּחוּ גְדוּדִים גְּדוּדִים
מִשָׁמַיִם אָרְצָה, וַיִּפְשְׁטוּ בְכָל הַשָּׂדוֹת לְהַצִּיל אֶת צִמְחֵי־הַבַּר מִידֵי אָדָם וּמִצְוָתָם הָיְתָה חֲזָקָה
עַל כָּל צֶמַח לֵאמוֹר: “עֲלֵה וּגְדָל!”.

וְאַחַד הַמַּלְאָכִים יָרַד בַּלָאט אֶל שְׂדֵה הַדָּגָן, וַיִּתְחַפֵּשׂ וַיִּלְבַּשׁ עָלִים
כֶּעָלִים אֲשֶׁר לַחִטָּה, וַיַּכְתֵּר אֶת רֹאשׁוֹ בְּפֶרַח סָגֹל, גָּדוֹל, אֲשֶׁר דְּמוּת כּוֹכָב לוֹ. וּבְבוֹא
הָאָדָם לִפְנוֹת עֶרֶב אֶל בֵּין הַקָּמָה לִשְׁלוֹחַ אֶת יָדוֹ בְּעִשְׂבֵי־הַבַּר נֶעְלַם הַפֶּרַח מֵעֵינָיו, כִּי
סָגוֹר נִסְגַּר כֻּלּוֹ וְהַצֶּמַח נִדְמָה בְעֵינֵי הָאָדָם לְחִטָּה. כָּכָה יַעֲשֶׂה הַצֶּמַח הַמִּתְחַפֵּשׂ יוֹמָם,
בַּשָּׁעוֹת אֲשֶׁר אַחֲרֵי הַצָּהֳרַיִם עַד הַבְשִׁילוֹ אֶת זְרָעָיו בְּתוֹךְ פִּרְחוֹ הַיְפָה, אָז יַצְמִיחַ בְּראשׁ
כָּל זֶרַע צִיצִית לְבָנָה, וְכָל רוּחַ קַלָּה וּפְזִיזָה בָּאָה וְאוֹחֶזֶת שִׂיחַ הַזְּרָעִים בְּצִיצִיּוֹת רֹאשָׁם
וְנוֹשְׂאָתַם עַד לְמֶרְחַקִים, וְהֵם נְפוֹצִים בָּאָרֶץ. וְרֵאשִׁית־כֹּל – בִּשְׂדוֹת הַדָּגָן.
