


	
	הַצָּבָר
	




הָיֹה הָיָה יוֹם בָּהִיר. עֲצֵי הָאָרֶץ יָשְׁנוּ שְׁנַת הַצָּהֳרָיִם. דּוּמִיָּה מִסָבִיב,
לֹא זָע אַף עָלֶה. אַךְ הִנֵּה עָבְרָה רוּחַ קַלָּה לְהָעִיר אֶת הַיְשֵׁנִים. וּבַהֲקִיצָם הִרְגִישׁוּ הָעֵצִים
בְּמִין אִילָן מְשֻׁנֶּה, עֵץ יָרוֹק הַצּוֹמֵחַ מִן הָאֲדָמָה עַל יָדָם. הֵם הִבִּיטוּ אֶל צוּרָתוֹ הַמְשֻׁנָּה
וְאֶל מַעֲמָדוֹ הַגֵּא. עֲנָפָיו הָיוּ שְׁטוּחִים וּרְחָבִים, וּכְנֶגְדָּם נִרְאוּ מְגֻחָכִים מְאֹד עָלָיו
הַצָּרִים, שֶׁהָיוּ דוֹמִים יוֹתֵר לְקוֹצִים.

בֵּין הָעֵצִים עָבַר רִשְׁרוּשׁ קַל. נִשְׁמְעוּ שִׂיחוֹת חֲשָׁאִיּוֹת עַל אֹדוֹת הָאוֹרֵחַ לֹא־הַקָּרוּא.

בֵּין כֹה וָכֹה עָלָה הָאוֹרֵחַ וְעָמַד אֵיתָן וּמוּצָק, וַיְהִי כְאוֹמֵר לָשֵׂאת נְאוּם.

כָּל הָעֵצִים הֶחֱרִישׁוּ, הִטּוּ אֹזֶן קַשֶּׁבֶת, וּבְפִיּוֹת פְּעוּרִים חִכּוּ לִשְׁמוֹעַ חֲדָשׁוֹת וּנְצוּרוֹת.

וְהָאוֹרֵחַ פָּתָח:

“שָׁלוֹם לָכֶם, צִמְחֵי אֶרֶץ־יִשְׂרָאֵל! מֵאֶרֶץ רְחוֹקָה אֲנִי בָא, מֵאֶרֶץ
מֶכְסִיקוֹ. שְׁמִי: צָבָר. הִנֵּה עָבַרְתִּי דֶרֶךְ רְחוֹקָה וְקָשָׁה. שַׁטְתִּי עַל פְּנֵי הָאוֹקֵיָנוּס. גַּם
בְּאֶרֶץ מוֹלַדְתִּי זוֹרַחַת הַשֶּׁמֶשׁ וּמְחַמֶּמֶת, וְהַשָּׁמַיִם שָׁם בְּהִירִים כִּשְׁמֵי אֶרֶץ־יִשְׂרָאֵל, אַךְ נִכְסוֹף
נִכְסַפְתִּי אֶל אַרְצְכֶם הַבְּרוּכָה. הַלֹא תוֹאִילוּ, אַחַי, לְקַבְּלֵנִי אֶל חֶבְרַתְכֶם”.

וּבְדַבְּרוֹ הִתְבּוֹנֵן וְהִבִּיט כֹּה וָכֹה, מִי מִן הָעֵצִים דּוֹמֶה אֵלָיו בִּתְכוּנוֹתָיו, לְמַעַן יִתְחַבֵּר אֵלָיו.

שָׁמְעוּ הָעֵצִים אֶת הַדְּבָרִים וְתָמָהוּ: מָה רָאָה אִילָן זֶה שֶׁטָּרַח וְעָבַר דֶּרֶךְ כֹּה רְחוֹקָה לָבוֹא אֲלֵיהֶם?

רִאשׁוֹנָה הִתְעוֹרְרָה הַתְּאֵנָה, כִּי נָתְנָה עֵינֶיהָ בְּפֵרוֹתָיו שֶׁל הַצָּבָר
הַמְלֵאִים גַּרְעִינִים וַתִּמְצָא אוֹתוֹ רָאוּי לְהִתְחַבֵּר אֵלָיו, כִּי עַל כֵּן גַּם לָהּ פֵּרוֹת בַּעֲלֵי
גַרְעִינִים רַבִּים.

וְהִיא פָּתְחָה וְאָמָרָה:

“בָּרוּךְ בּוֹאֲךָ, צָבָר! בִּזְרוֹעוֹת פְּתוּחוֹת אֲקַבְּלֶךָ”. אַךְ עוֹד לֹא הִסְפִּיקָה
לִפְתֹּח אֶת זְרוֹעוֹתֶיהָ, וְהִנֵּה הִרְגִּישָׁה בְּמַבְרְשׁוֹת הַקּוֹצִים הַדַּקִּים, שֶׁכִּסּוּ אֶת פְּרִי הַצָּבָר.
וּכְרֶגַע נִרְתְּעָה לְאָחוֹר, לְבַל תִּשָּׁחֵת הַקְּלִפָּה הָרַכָּה שֶׁל פִּרְיָהּ.

נִשְׁאַר הַצָּבָר עוֹמֵד תּוֹהֶה וּמַשְׁמִים רָגַע, אַךְ עַד מְהֵרָה הִתְעוֹדֵד, הִשִּׁיר
מִגִּזְעוֹ אֶחָד מִפֵּרוֹתָיו, הִתְפַּלֵשׁ הַפְּרִי בַחוֹל וְכָל הַקּוֹצִים נָפְלוּ מֵעָלָיו. הִרְגִּישׁ בַּדָּבָר הֵלֶךְ
עָנִי, הֵרִים אֶת הַפֶּרִי, קלְּפוֹ וְהִתְחִיל אוֹכְלוֹ לְתֵאָבוֹן.

רָאָה זֹאת הָרִמּוֹן, נַעֲנָה וְאָמָר:

“שְׁמַע נָא אָחִי הַצָּבָר, אָנֹכִי אֶהְיֶה לְךְ לְחָבֵר”.

אַךְ בִּרְאוֹתוֹ אֶת גַּרְעִינֵי הַצָּבָר הַקָּשִׁים מֻשְׁלָכִים לָאָרֶץ, חָזַר מִדְּבָרָיו
וְקָרָא: “צַר לִי עָלֶיךָ, הַצָּבָר, לֹא אוּכַל לָתֵת יָדִי לָךְ, כִּי גַרְעִינַי הֵם עֲסִיסִיִּים וְרַכִּים,
לֹא כֵן גַּרְעִינֶיךָ אָתָּה”.

אָמְרָה הַגֶּפֶן לְנַסּוֹת, אוּלַי תִּמְצָא מִשְׁעָן בְּטַפְּסָה בִּזְמוֹרוֹתֶיהָ עַל הַצָּבָר,
לְמַעַן תּוּכַל לְהִתְחַבֵּר אֵלָיו. וְאוּלָם כְּשֶׁהִתְחִילָה לְהַעֲלוֹת אֶת זְמוֹרוֹתֶיהָ, נִפְצְעוּ עָלֶיהָ
וְאַשְׁכְּלוֹתֶיהָ בַּעֲלֵי הַצָּבָר הַדּוֹקְרִים.

רָאוּ זאת שְׁאָר הָעֵצִים וְשָׂטוּ מֵעָלָיו.

נָשָׂא הַצָּבָר אֶת עֵינָיו אֶל הָעֵצִים בַּעֲלֵי הַקּוֹצִים הַדּוֹקְרִים, וְהִבִּיט אֲלֵיהֶם כְּמִתְחַנֵּן. אוּלָם אֵלֶּה עָנוּ:

“עוֹלָה אַתָּה עָלֵינוּ בִּתְכוּנוֹתֶיךָ, בְּגִזְעֲךָ הַמָּלֵא מִיץ וּבְפֵרוֹתֶיךָ
הַמְתוּקִים, וִירֵאִים אָנוּ פֶּן תִּתְנַשֵׂא לִמְלוֹךְ עָלֵינוּ. עַל כֵּן לֹא נוּכַל לְקַבֶּלְךָ אֶל
חֶבְרָתֵנוּ”.

נִשְׁאַר הַצָּבָר עוֹמֵד בְּצִדֵּי הַגַּנִּים. פָּחֹד פָּחֲדוּ הָעֵצִים פֶּן יִפְרוֹץ הַצָּבָר לְתוֹכָם וַיְשַׁלְחוּ אֶת הַסִּיכִית לִשְׁמוֹר עָלָיו.

מֵאָז מְטַפֶּסֶת הַסִּיכִית עַל הַצָּבָר וּמַפְרִיעַתּוּ מֵהִתְפַּשֵּׁט כְּחֶפְצוֹ.

כְּשֶׁרָאָה הָאָדָם אֶת הַצָּבָר בְּכָךְ, הִתְחִיל מִשְׁתַּמֵשׁ בּוֹ וְעוֹשֶׂה מִמֶּנּוּ סְיָג לַגַּנִּים.
