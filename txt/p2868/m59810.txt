יְדִיד־הַחוֹלוֹת
	




בַּשָּׁעָה שֶׁאָמַר אֱלֹהִים: "יִקָּווּ הַמַּיִם מִתַּחַת הַשָּׁמַיִם אֶל מָקוֹם אֶחָד וְתֵרָאֶה
הַיָּבָּשָׁה", נֶחֱלַק עוֹלַם הַדּוֹמֵם: הַמְּלָחִים הַשּׁוֹנוֹת הִתְעָרְבוּ בְתוֹךְ הַמַּיִם וַיִּמַּסּוּ בָהֶם;
הַסְּלָעִים, הַצְּדָפִים וּצְרוֹרוֹת הֶעָפָר נֶעֶרְמוּ תִלִּים תִּלִּים וַיִּהְיוּ לְהָרִים וּגְבָעוֹת; וְרַק הַחוֹל
נִשְׁאַר לְבַדּוֹ וְלֹא יָדַע אָנָה יִפְנֶה. הַיָּם אָמַר: לֹא לִי הוּא, בַּאֲשֶׁר הַחוֹל שׁוֹקֵעַ לְמַטָּה אֶל
הָאֲדָמָה. הַיַּבָּשָׁה אָמְרָה: לֹא לִי הוּא, כִּי כַיָּם הוּא מַכֶּה גַלִּים בִּשְׁעַת סְעָרָה.

וְהָיָה הַחוֹל הוֹלֵךְ מִן הַיַּבָּשָׁה אֶל הַיָּם וּמִן הַיָּם אֶל הַיַּבָּשָׁה וְלֹא מָצָא
מָנוֹחַ. מֵרֹב נְדוּדִים הִתְפָּרְדוּ גַרְגְּרָיו וְיַד הָרוּחוֹת הָיְתָה בוֹ לְרָעָה, הֵם דָּחֲפוּ אוֹתוֹ
וְכִסּוּ עַל יָדוֹ כָּל שְׂדוֹת תְּנוּבָה וְגַנֵּי חֶמֶד, וְהָיָה הַחוֹל לִקְלָלָה בְּפִי בְנֵי־הָאָדָם בְּכָל
מְקוֹם בּוֹאוֹ.

וְהַחוֹל הִצְטָעֵר צַעַר רַב עַל הַנֵּזֶק שֶׁהוּא גוֹרֵם לִבְנֵי הָאָדָם.

רָאָה הַקָּדוֹשׁ בָּרוּךְ־הוּא בְּצַעֲרוֹ שֶׁל הַחוֹל וְהִזְמִין לוֹ יְדִיד – אֶת יְדִיד־הַחוֹלוֹת.

וְהָיָה בְהִתְחוֹלֵל הַסַּעַר וְגַרְגְּרֵי הַחוֹל נִזְרָקִים וּמְטַפְּחִים עַל פְּנֵי הַכֹּל,
יִתְקַפֵּל יְדִיד הַחוֹלוֹת, מְכַסֶּה אֶת פָּנָיו וְסוֹגֵר אֶת פִּיפִיּוֹת הַנְּשִׁימָה שֶׁבֵּין גִּידָיו; וְרַק
אֶת גַּבּוֹ הַקָּשֶׁה וְהֶחָסוּם מַפְנֶה הוּא לְצַד הָרוּחַ וּמְקַבֵּל בְּחִבָּה אֶת גַּרְגְּרֵי הַחוֹל
הַנִּדָּפִים.

וּבִשְׁקוֹט הָרוּחַ וּבְרֶדֶת גִּשְׁמֵי בְרָכָה פּוֹתֵחַ הַיְדִיד אֶת עָלָיו וְשׁוֹאֵף רוּחַ
לִרְוָחָה, מְקַבֵּץ בִּנְטִישׁוֹתָיו וּבְשָׁרָשָׁיו אֶת גַּרְגְּרֵי הַחוֹל, מְקָרֵב אוֹתָם אִישׁ לְרֵעֵהוּ וְנוֹתֵן
לָהֶם עֹז לַעֲמוֹד בִּפְנֵי כָל סוּפָה.

גַּם הָאָדָם הִכִּיר בְּמַעֲשָׂיו הַטּוֹבִים שֶׁל יְדִיד־הַחוֹלוֹת וְהוּא עוֹזֵר לוֹ
לְהִתְפַּשֵּׁט וּבְכָל מְקוֹם מִשְׁכַּן הָאָדָם נֶהֱפָכִים הַחוֹלוֹת עַל יְדֵי כָךְ לְיַעֲרוֹת עַד; וְהַשֵּׁם "יְדִיד
הַחוֹלוֹת" חָבִיב וְרָצוּי בְּפִי כָל הַיּוֹשְׁבִים עַל חֻפֵּי הַיַּמִּים.
