


	
	פֶּרַח־הַטַּבָּעוֹת
	




לְעֵת בּוֹא הַשֶּׁמֶשׁ, בְּאֶחָד מִימֵי אֲדָר, שָׁב אֲבִי אַמְנוֹן לְבֵיתוֹ, הָעוֹמֵד בְּפִנָּה חֲרִישִׁית, מֵעֵבֶר לִשְׂדוֹת תְּבוּאָה, חָבוּי בְּצֵל אֹרֶן וּבְרוֹשׁ.

בַּבֹּקֶר, בְּצֵאתוֹ מִבֵּיתוֹ, קָטַף אַמְנוֹן בְּנוֹ צְרוֹר מִצִּמְחֵי־נוֹנִית קְטַנִּים,
בַּעַל פִּרְחֵי־תְכֵלֶת נֶחְמָדִים, וְהִגִּישׁוֹ תְשׁוּרָה לְאָבִיו. בַּכּוֹתָרוֹת הָעֶלְיוֹנוֹת שֶׁל הַפְּרָחִים,
הַדּוֹמוֹת לְקַעֲרִיּוֹת תְּכֵלֶת, נִשְׁמְרוּ עוֹד אֶגְלֵי הַגָּשֶׁם. הָאָב הִסְבִּיר לוֹ, כִּי אֵלֶּה הֵם פִּרְחֵי
הַנּוֹנִית, אַף הֶרְאָה לוֹ מִתּוֹךְ שִׂיחָה עַל הַשְּׂעָרוֹת בְּפֶתַח הַצִּנּוֹר שֶׁל הַכּוֹתָרֶת, הַמִּתְלַכְּדוֹת
יַחְדָּו וְעוֹצְרוֹת בְּעַד טִפֵּי הַגֶּשֶׁם מֵחֲדֹר לָתוֹךְ הַפֶּרַח פְּנִימָה שֶׁלֹא לְהַרְטִיב אֶת הָאַבְקָנִים,
הַשּׁוֹכְנִים לִפְנַי וְלִפְנִים שֶׁל הַפָּרַח. וְעוֹד בֵּאֵר לוֹ אָבִיו, כִּי הַזִּיפִים שֶׁעַל הַצֶּמַח הֵם
הַנֶּשֶׁק, שֶׁהַטֶּבַע חָנַן אֶת הַנּוֹנִית, כְּדֵי שֶׁתּוּכַל לְהָגֵן עַל עַצְמָהּ מִפְּנֵי הַחֲרָקִים הַבָּאִים לֶאֱכֹל
אֶת גִּבְעוֹלֶיהָ וְאֶת עָלֶיהָ.

אַךְ בָּעֶרֶב בְּשׁוּב הָאָב לְבֵיתוֹ, יָצָא אַמְנוֹן לִקְרָאתוֹ וּפָנָיו עֲצוּבִים.

– מַה לְּךָ, בְּנִי?

– אַבָּא – עָנָה הַיָּלֶד – הִנֵּה קָטַפְתִּי פְרָחִים כְּתֻמִּים וּצְהֻבִּים לְהַגִּישָׁם לָךְ. וְאִמָּא שָׂמָה אוֹתָם בַּמָּיִם, וּבְהַבִּיטִי אֲלֵיהֶם וְהִנֵּה נָבֵלוּ.

הָאָב בִּקֵשׁ לְהָבִיא לְפָנָיו אֶת הַפְּרָחִים, וּבְהִתְבּוֹנְנוֹ אֲלֵיהֶם נִחֵם אֶת
בְּנוֹ, כִּי לֹא נָבְלוּ וְרַק נִסְגְרוּ לַלָּיְלָה, וּמִמָּחֳרָת עִם זְרִיחַת הַשֶּׁמֶשׁ יָשׁוּבוּ הַפְּרָחִים
וְיִפְתְּחוּ אֶת סְגוֹר צִיצֵיהֶם.

– הַאִם גַּם הַפְּרָחִים יְשֵׁנִים? – תָּמַהּ אַמְנוֹן.

– כֵּן, בְּנִי, יֵשׁ פְּרָחִים יְשֵׁנִים בַּלָּיְלָה וּמַלְאָכִים טוֹבִים מְסַפְּרִים לָהֶם
חֲלוֹמוֹת עֲרֵבִים. הֲלֹא כֵן אָמְרוּ חֲכָמִים: אֵין כָּל עֵשֶׂב וָעֵשֶׂב, שֶׁאֵין לוֹ מַזָּל בָּרָקִיעַ,
שֶׁמַכֶּה אוֹתוֹ וְאוֹמֵר לוֹ: גְּדָל!

– הַאָמְנָם גַּם פֶּרַח זֶה יֶשׁ לוֹ מַזָּל בָּרָקִיעַ? – שָׁאַל הַיָּלֶד.

– כֵּן, בְּנִי, וְלֹא עוֹד אֶלָא שֶׁמַלְאָכִים אֲחָדִים, הַמְמֻנִּים עַל הַצְּמָחִים,
הִתְחַבְּרוּ יַחַד לִחְיוֹת בִּידִידוּת וְאַחֲוָה, וְהֵם אָסְפוּ לָהֶם אֶת צִיצֵיהֶם לַעֲשׂוֹת מֵהֶם פֶּרַח אֶחָד
מְשֻׁתָּף. וַדַּאי רוֹצֶה אַתָּה לָדַעַת כֵּיצַד חֻבְּרוּ הַצִּיצִים לְפֶרַח אֶחָד? הָבָה אֲסַפֶּר לְךָ מַה
שֶּׁשָּׁמַעְתִּי מִיִּתּוּשׁ שְׁחַרְחֹר וְקָטָן, שֶׁשָּׂם מִשְׁכָּנוֹ בְּתוֹךְ הַפָּרַח.

הַיֶלֶד הִקְשִׁיב בְּדוּמִיָּה וְהָאָב הוֹסִיף לְסַפֵּר:

– פַּעַם אַחַת, עִם רֶדֶת הַיּוֹם, נָח הַיִּתּוּשׁ בְּתוֹךְ הַפָּרַח, וְהִנֵּה נִגְלָה
לְעֵינָיו מַחֲזֶה נִפְלָא. מַלְאָכִים אֲחָדִים, יְלָדִים וִילָדוֹת, אֲשֶׁר נֹגַהּ לָהֶם וְזֹהַר לָהֶם, בָּאוּ
בִיעָף אֶל הַפֶּרַח וַיַּעֲנִיקוּ לוֹ תְשׁוּרוֹת שׁוֹנוֹת, כִּי אוֹהֲבִים הַמַּלְאָכִים אֶת הַפְּרָחִים.
לַצִּיצִים הַקְּטַנְטַנִּים, אֲשֶׁר בְּמֶרְכַּז הַפֶּרַח, חָלְקוּ הַמַּלְאָכִים־הַיְלָדִים אֲבַק־נִיחוֹחַ, וְלַצִּיצִים
אֲשֶׁר בַּתִּתּוֹרָה נָתְנוּ הַמַּלְאָכִים־הַיְלָדוֹת אָרִיג מָשְׁזָר בִּדְמוּת לְשׁוֹנוֹת זָהָב. הַצִּיצִים שָׂמְחוּ
מְאֹד עַל הַמַּתָּנוֹת הַיָּפוֹת. וְכַאֲשֶׁר בָּא הַלַּיְלָה וְעִמּוֹ הָרוּחַ הַקַּר, קִפְּלוּ צִיצֵי הַתִּתּוֹרָה אֶת
לְשׁוֹנוֹתֵיהֶם לְהָגֵן עַל הַצִּיצִים הַקְּטַנְטַנִּים בַּמֶּרְכָּז, לְבַל יִרְטְבוּ מִן הַטָּל. וּלְאוֹת תּוֹדָה
בְּעַד הַחֶסֶד הַזֶּה, חָלְקוּ הַצִּיצִים הַקְּטַנְטַנִּים מֵאֲבַק נִיחוֹחָם לְצִיצֵי הַלְּשׁוֹנוֹת. וְהַמַּלְאָכִים
שָׂמְחוּ מְאֹד, בִּרְאוֹתָם אֶת הָאַחֲוָה הַשּׂוֹרֶרֶת בֵּין הַצִּיצִים (כִּי אוֹהֲבִים הַמַּלְאָכִים אֶת הַשָּׁלוֹם)
וַיְפָאֲרוּ אֶת הַצִּיצִים בְּטַבָּעוֹת וּנְבָלִים. מֵאָז וְעַד הַיּוֹם מְפִיצִים צִיצֵי הָאַבְקָנִים
אֲבַק־נִיחוֹחַ וְצִיצֵי־הַלְּשׁוֹנוֹת צוֹמְחִים בִּדְמוּת טַבָּעוֹת וּנְבָלִים.
