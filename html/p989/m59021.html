השיר השלשה עשר

אֵל נָטָה שָׁמַיִם לֹא יֶחְדַּל כֹּחַ,

לֹא יִיעָף, לֹא יִיגָע, לֹא יִשְׁאַל נוֹחַ,

חֶפְצוֹ יַשְׁלִים, הוּא יֵיטִיב, הוּא יָרֵעַ.

כִּי הוּא אֵל, דּוֹרֵך עַל בָּמֳתֵי אָרֶץ,

מִיָּדוֹ רֹב שָׁלוֹם, מִיָדוֹ פָּרֶץ,

יַשְׁפִּיל רָמֵי קוּמָה, יָקִים צוֹלֵעַ.


אִם לַעֲשׂוֹת כְּמַעֲשֵׂהוּ אֵל בִּלְתּוֹ אַיִן,

אַף כִּי אֱנוֹשׁ גֵּאֶה, גֶּבֶר רוּם עַיִן,

פִּיו דּוֹבֵר גָּבוֹהַּ, אָפֶס מַעֲשֵׂהוּ,

טוֹבוֹתָיו צִיץ שָׂדֶה לֹא יֵאָמֵנוּ,

מַעְיָן אַכְזָב אֶל עָלָיו יִשָּׁעֵנוּ,

פָּעֳלוֹ בֹקֶר טֶרֶם עֶרֶב יַלְאֵהוּ.


חֲמַת אִישׁ כִּי תִּבְעַר יַךְ מִבְּלִי דַעַת,

אֶל מִשְׁפַּט לִבּוֹ אָזְנוֹ בַּל שֹׁמַעַת,

לַעֲשׂוֹת נָקָם יִרְדּוֹף, לֹא יִרְדּוֹף חֵקֶר,

רֵאשִׁית מַעֲשֵׂהוּ עַז וּבְשֶׁטֶף יַעַשׂ,

אִם יוֹסִיף עוֹד יִלְאֶה אָז יִנְחַל כַּעַשׂ,

בִּרְאוֹתוֹ כֹּחוֹ שָׁוְא וִימִינוֹ שָׁקֶר.


וֵאלֹהִים כִּי יִקּוֹם יִקְרַע שָׁמָיִם,

וּגְבוּרוֹת יַעֲשֶׂה יִשְׁקוֹל בְּמֹאזְנַיִם,

חֲמָתוֹ לֹא תַחְשִׁיךְ עֵינֵי הַצֶּדֶק,

רָאשִׁית מוּסָרָיו קַל, בּוֹ יֹאמַר עוּרוּ,

טֶרֶם יִבְעַר אַפִּי מֵרָעָה סוּרוּ,

אַחֲרִית אַף מַפָּלָה, רֵאשִׁיתוֹ בֶּדֶק.


גַּם נִקְמַת אֵל כֻּלָּהּ כָּבוֹד אֹמֶרֶת,

וּנְתִיבוֹת חֲמָתוֹ עֲדִי תִפְאֶרֶת,

אֶל צָרָיו וּבוֹזָיו יִתֵּן יָדַיִם.

כִּי פַרְעֹה וַעֲבָדָיו הֶהָלְכוּ קֶרִי,

הִנִּחָם עַד מוֹעֵד, עַד הִרְבּוּ מֶרִי,

עַד מִשְׁפָּטָם נָגַע אֶל הַשָּׁמָיִם.


נִפְלְאוֹת אֵל מִנִּי קֶדֶם הֵן שָׁמַעְנוּ,

וּמֵאֲשֶׁר הִשְׂכַּלְנוּ בָמוֹ יָדַעְנוּ,

כִּי טוֹב יְיָ וּלְטוּבוֹ אֵין עֶרֶךְ.

אִם מוֹעֵד נָקָם בּוֹ חִצִּים שֹׁלֵחַ,

בַּמִּסְפָּר בַּמִּשְׁקָל צָרָיו פֹּלֵחַ,

מָה רַבּוּ טוֹבוֹתָיו עִם יִשְׁרֵי דָרֶך.


אִם צִדְקוֹת יְיָ עַמִּים יָבִינוּ,

אָז מַעֲשָׂיו מִקֶּדֶם לִבָּם יַרְנִינוּ,

כָּל לָשׁוֹן תַּנְוֵהוּ, כָּל פֶּה יוֹדֶנּוּ,

עַל הַגְּדוֹלוֹת פָּעַל יוֹדְעִים יִתְמָהוּ.

אַחֲרוֹנוֹת עַל רִאשׁוֹנוֹת מַה גָּבָהוּ,

עַד גָּאַל עַם מִיַּד חָזָק מִמֶּנוּ.


מֵישָׁרִים אָרְחוֹתָיו גַּם בִּימֵי זָעַם,

כִּי אוֹתוֹתָיו מִשְׁפָּט, מוֹפְתָיו טוּב טָעַם,

הֵחֵל הִפְלִיא מַטָּה, כִּלָּה גָּבוֹהַּ,

וּגְדוֹלוֹתָיו עַל יָם הֵמָּה נִפְלָאוּ,

מִכְלוּל רִאשׁוֹנוֹת הֵם וּבָם נִמְצָאוּ,

יַעַן כִּי יִגְדַּל תָּמִיד כֹּחַ אֱלוֹהַּ.


כִּי אוֹתוֹת רִאשׁוֹנוֹת בַּיְאוֹר מִתָּחַת,

מֵימָיו הָפַךְ אֶל דַּם וּדְגָתוֹ שָׁחַת,

וַיִּשְׁרוֹץ מַשְׁחִיתִים קִצְרֵי רַגְלָיִם,

עַל יָם, מַיִם נוֹזְלִים כַּחוֹמָה קָמוּ.

שָׁם הָלְכוּ עַם בֶּטַח עַד לַעֲבוֹר תַּמּוּ,

שָׁם מָחַץ רָאשֵׁי תַּנִּינִים עַל מָיִם.


הָאוֹתוֹת הַשְּׁנִיּוֹת הָיוּ בָּאָרֶץ,

כָּל חַיּוֹת הַשָּׂדֶה פָּרְצוּ בָם פָּרֶץ,

וּמִקְנֶה מִצְרַיִם הִסְגִיר לַדָּבֶר,

עַל יָם, הַתְּהוֹמוֹת בִּדְבַר אֵל קָפָאוּ.

לִהְיוֹת מַסְלוּל אֶל עַם מִבּוֹר יָצָאוּ,

וּלְסוּס פַּרְעֹה וּלְרֹכְבוֹ שׁוֹד וָשֶׁבֶר.


אוֹתוֹת הַשְּׁלִישִׁיּוֹת שָׁם בָּרָקִיעַ,

אֵשׁ בָּרָד גֶּשֶׁם, סַעַר קוֹל מַשְׁמִיעַ,

בַּעֲלֵי כָנָף דָּאוּ עַל כַּנְפֵי רוּחַ,

עַל הַיָּם צִוָּה אֶל רַעַם וָזָרֶם,

בַּחֲצָצִים בָּרָד וָאֵשׁ הִכּוּם חֶרֶם,

רוּחַ קָדִים עַזָּה כַּצִּיר שָׁלּוּחַ.


הָאוֹתוֹת הָאַחֲרוֹנוֹת עוֹד גָּבָהוּ,

חֵיל מִשְׁלַחַת אֵימָה רָאוּ תָּמָהוּ,

וּבְכוֹרֵיהֶם מֵחֶזְיוֹנוֹת נִבְהָלוּ.

עַל הַיָּם, חַרְבְּךָ הַקָּשָׁה הֵנַפְתָּ,

כִּי בְּעַמּוּד עָנָן וָאֵשׁ הִשְׁקַפְתָּ

עַל אוֹיֵב, וּבִמְצוֹלוֹת יָם צָלָלוּ.


מִתְנַשֵּׂא לַכֹּל לָרֹאשׁ מַה גָּדַלְתָּ,

וּמַה גָּדְלוּ מַעֲשֶׂיךָ אָז פָּעַלְתָּ,

מִי אֵל אֲשֶׁר יַעֲשֶׂה כִגְבוּרוֹתֶיךָ,

הָאֵיתָנִים תַּהְפּוֹךְ כִּי יוֹצְרָם אַתָּה,

תַּהֲפוּכוֹתָם יוֹשֶׁר, קַו לָהֶם שַׁתָּה,

בַּעֲבוּר יִפְנוּ כֻּלָּם אֶל חָכְמָתֶךָ.


מֵיתְרֵי כִּנּוֹר אֶצְּבְּעוֹת נַעַר יָנִיעוּ,

מֵעֵרֶב קוֹלוֹת שָׁאוֹן זָר יַשְׁמִיעוּ,

וּבְיַד מֵבִין בַּשִּׁיר נַגֵּן יֵיטִיבוּ.

כֵּן תַּהֲפֲּוּכוֹת אֵלֶּה זָרוֹת וָעֵרֶב,

אַךְ נָבוֹן יַיְשִׁירֵם, טוֹב טַעֲמָם יִרֶב,

וּכְקוֹל כִּנּוֹר נָעִים קוֹלוֹ יַקְשִׁיבוּ.


גַּם כִּרְאוֹת יִשְׂרָאֵל עַם זוּ קָנִיתָ,

אֶת הַיָּד הַגְּדוֹלָה אֲשֶׁר עָשִׂיתָ,

עִם פַּרְעֹה אוֹיְבְךָ גַּם עִם מַחֲנֵהוּ,

שָׁרוּ לְךָ שִׁיר חָדָשׁ, כֻּלָּם קָרָאוּ,

חֲזוּ מִפְעֲלוֹת אֵל! אוֹתוֹ תִּרָאוּ!

כִּי גָאֹה גָאָה אֵל, רֵעוֹ יַעֲנֵהוּ.


עַל הַמְשׁוֹרְרִים בַּיָּם צָלְחָה רוּחֶךָ,

גַּם דִּבֶּר רוּחֲךָ הַטּוֹב בִּמְשִׁיחֶךָ,

וּבְשִׁירָיו עַמִּים, רַבִּים לְךָ יָשִׁירוּ,

וַאֲנָחְנוּ אֵיךְ נָשִׁיר בִּימֵי עָנְיֵנוּ,

בָּאֵר מַה מֵהֶם שָׁמַעְנוּ דַּיֵּנוּ,

כִּי זֹאת פָּעֳלִי, שׁוֹמְעֵי שִׁירִי יַכִּירוּ.


אִם מִבֵּית1 עֲבָדִים שִׁלַּח פַּרְעֹה עַם קֹדֶשׁ,

אִם בִּדְמָעוֹת שָׁלִישׁ הִתְחַנֵּן לֵאמֹר צֵאוּ,

לֹא נָכוֹן לִבּוֹ עִם אֵל, לֹא נִטְהַר מִפֶּשַׁע,

עוֹד טֻמְאָתוֹ בוֹ, עוֹד מָלֵא מִרְמָה וָכָחַשׁ,

כִּי אִישׁ דַּרְכּוֹ כָּאֲפֵלָה אוֹר לֹא יִרְאֶה נֶצַח,

אֵל לוֹ לַיְלָה אוֹר, חֹשֶׁךְ לֹא יַחֲשִׁיךְ מִמֶּנוּ,

רָאָה כָּל זֹאת, כִּי רַק מֵהֲלָךְ דֶּבֶר בָּאֹפֶל,

שִׁלַּח אֶת הָעָם, מִמְּחִתָּה עָלָיו גָּבָרָה,

וּמֵעֲוֹנוֹ הַגָּדוֹל נַפְשׁוּ לֹא נִרְפָּאָה,

גַּם לֹא תֵרָפֵּא עוֹד, כִּי עֵין שִׂכְלָהּ שִׁחֵתָה,

וְיֹאמֶר: "הַשְּׁמֵד אַשְׁמִיד חֶבֶר פֹּשְׁעִים אֵלֶּה,

"הֵם עִם חַטֹּאתֵיהֶם בִּמְצוּלוֹת יָם יֹאבֵדוּ,

"וּלְעַם חוֹסֶה בִּי וּבְאוֹר פָּנַי יֵלֵכוּ,

"אָבִיא אֶל הַר קָדְשִׁי שָׁם תּוֹרוֹת אֲלַמְּדֵמוֹ,

"וּלְאוֹר גּוֹיִם יִהְיוּ, קַצְוֵי אֶרֶץ יָאִירוּ.


"עַל כֵּן עַם זֶה גָּאַל יְיָ מִמִּצְרַיִם,

"בַּעֲבוּר הַנְחֵל לָהֶם אַרְצוֹת גּוֹיִם יִירָשׁוּ,

"לֹא הוֹלִיךְ דֶּרֶךְ יָשָׁר מִפְּלִשְׁתִּים לִכְנָעַן,

"הֵסֵב אוֹתָם אֱלֹהִים בַּמִּדְבָּר בַּדָּרֶךְ,

"לַהֲבִיאָם אֶל יָם סוּף לַנְחוֹתָם הַר הַקֹּדֶּשׁ,

"גַּם בִּלְעֲדֵי2 זֹאת אֶרֶץ הַצְּבִי יִתֵּן לָמוֹ,

"אֵיךְ יִנְחָלוּהָ יוֹם יָצְאָה רַגְלָם מִכֶּבֶל,

"טֶרֶם יַעֲשׂוּ לָהֶם לֵב חָדָשׁ יָנוּב אֹמֶץ,

"טֶרֶם יִסּוֹב שֶׁמֶשׁ גַּלְגַּלוֹ פַּעַם אַחַת,

"הֵם יִרְאוּ צָרֵיהֶם יוֹרְדִים לָחֶם שָׁעַר,

"וּבִרְאוֹתָם מִלְחָמָה מִפַּחַד יִגְוָעוּ,

"אַךְ טוֹב לָנוּ אָז מֵעַתָּה הָעָם יֹאמֵרוּ,

"קוּמוּ נָשׁוּבָה אָחוֹר, נָשׁוּבָה מִצְרָיְמָה,

"גַּם בַּעֲבוּר זֹאת בַּדֶּרֶךְ הַקְּרוֹבָה לִנְסוֹעַ,

"מִצֹּעַן לִכְנָעַן לֹא נָחָם אֱלֹהֵימוֹ,

"עַד כִּי בִּישִׁימוֹן נִפְלָאוֹת רַבּוֹת יַרְאֵמוֹ,

"יַרְאֵם שָׂרִים וָמֶלֶךְ אֶרֶץ תִּבְלָעֵמוֹ,

"מִשְּׁחָקִים יֹאכְלוּ לֶחֶם, מִצּוּר יִשְׁתּוּ מָיִם,

"קוֹל אֱלֹהִים חַיִּים, מִתּוֹךְ הָאֵשׁ יִשְׁמָעוֹ,

"שֵׂכֶל טוֹב יִמְצָאוּ, רוּחוֹ יִתֵּן עָלֵימוֹ,

"וִיעַטְּרֵם כָּבוֹד, אָז יִרְחַב לֵב דַּכְּאֵי רוּחַ,

"עֲמָלָם לֹא יִזְכְּרוּ עוֹד, שִׁפְלוּתָם יִשְׁכָּחוּ,

"עַל יְיָ יִשָּׁעֵנוּ, בּוֹ יַעֲשׂוּ חָיִל.

“וּבְיוֹם מִלְחְמָה עַל סוּס עַל עַם רַב יִשְׂחָקוּ”.


וּפְדוּיֵי יְיָ רַגְלִי שֵׁשׁ מֵאוֹת אֶלֶף,

כֻּלָּם אֲנָשִׁים מִלְּבַד נָשִׁים טַף וָעֶרֶב,

קָהָל רַב זֶה לֹא כָעֵדֶר תּוֹעֶה הָלָכוּ,

לֹא כַּנָּסִּים מִבּוֹר הַנֶּחְפָּזִים לָלֶכֶת,

אָכֵן חֲמוּשִׁים עָלוּ מֵאֶרֶץ מִצְרָיִם.

כַּחֲלוּצֵי צָבָא, כַּגִּבּוֹרִים עוֹשִׂים חָיִל,

וּלְמִשְׁפָּחוֹת מִשְׁפָּחוֹת שִׁבְטֵיהֶם נֶחֱלָקוּ,

נָשִׁיא לַשֵּׁבֶט עַל פִּי יְיָ הֵקִימוּ,

וּמֹשֶׁה אִישׁ הָאֱלֹהִים נָגִיד עַל כֻּלָּמוֹ,

עַל רֹאשָׁם שִׂמְחַת עוֹלָם, פַּחַד לֹא יָדָעוּ,

עַל צָרֵיהֶם מִקֶּדֶם רַחֲמֵיהֶם נִכְמָרוּ,

כִּי אֶל כָּל מָקוֹם מִסָּבִיב אֲשֶׁר הִבִּיטוּ,

רָאוּ מִצְרַיִם מוֹרִידִים מֵתִים לַקָּבֶר,

אֲשֶׁר הִכָּה יְיָ בָּהֶם לֵיל הַנֶּגֶף.

גַּם אֱלֹהֵיהֶם עַל הֶהָרִים נוֹפְלִים אַרְצָה,

כִּי גַּם בָּם עָשָׂה יְיָ כָּלָה נֶחֱרֶצֶת,

בַּעֲבוּר הַכְאֵב לִבָּם מִמִּבְטֶחָם יֵבוֹשׁוּ,

וַיֹּאמְרוּ הָעָם: "דַּם זֶה לֹא שָׁפְכוּ יָדֵינוּ,

"מִיָּדָם זֹאת לָהֶם כִּי לַיְיָ חָטָאוּ,

"עַתָּה יִלְמְדוּ צֶדֶק מֵחֶטְאָם יִנָּחֵמוּ,

"דַּבֵּר גָּבוֹהַּ עַל צוּרֵנוּ בַּל יוֹסִיפוּ,

"וּלְהַגִּיד עוֹד ‘בָּכֶם נִמְשֹׁלָה’ יִכָּלֵמוּ,

"כִּי יוֹדְעִים מִי נָחְנוּ, כִּי יְיָ אִתָּנוּ

“וּלְמֶלֶךְ הַכָּבוֹד כָּבוֹד וָעֹז יִתֵּנוּ”.

כֵּן דִּבְּרוּ הַקָּהָל הָרוֹאִים לָעֵינַיִם,

כֵּן דִּבְּרוּ אִישׁ אֶל רֵעֵהוּ וַיִּסָּעוּ,

וּבְיָד רָמָה יוֹצְאִים, מִדָּבָר לֹא יָרֵאוּ

כִּי עוֹד יֵחַתּוּ, עוֹד יִרְדְּפֵם צָר, לֹא זָכָרוּ.


וַיִּסְעוּ3 מִסֻּכּוֹת, עָבְרוּ מוֹשְׁבוֹת צָרֵימוֹ,

בֶּטַח אֵין מַחֲרִיד, קוֹל אוֹיֵב לֹא שָׁמָעוּ,

וַיָּבוֹאוּ אֵיתָם עִיר עַל הַגְּבוּל עֹמֶדֶת,

בִּקְצֵה הַמִּדְבָּר הִיא, אֵין עוֹד אֶרֶץ נוֹשָׁבֶת,

לִימִין הוֹלֵךְ דֶּרֶך מֵאֵיתָם הַמִּדְבָּרָה,

יַבִּיט לִקְצֵה יָם סוּף קֵץ הַלָּשׁוֹן הָאַחַת.

וּלְשׁוֹן יָם הַשְּׁנִיָה עַל קִצָּהּ עֶצְיוֹן גָּבֶר,

בֵּין הַלְּשׁוֹנוֹת הַשְׁתַּיִם הַר סִינַי חוֹרֵבָה

גַּם מֹשֶׁה גַּם הָעָם כֻּלּוֹ יַחְדָּו הֶאֱמִינוּ,

כִּי עַתָּה מִחוּץ לִגְבוּל מִצְרַיִם יָבֹאוּ,

אֶל אֶרֶץ אֵין יוֹשֵׁב בָּהּ אֱלֹהִים יַנְחֵמוֹ,

מַה שָּׂמַח לִבָּם כִּי מֵאֶרֶץ אוֹיֵב שָׁבוּ,

וּמֵאוֹיֵב אַחֵר פֶּן יִקְרַב אֵיךְ יִירָאוּ,

בִּרְאוֹתָם כִּי עַל רֹאשָׁם עֲטֶרֶת תִּפְאֶרֶת,

כִּי יְיָ הוֹלֵךְ לִפְנֵיהֶם יוֹמָם וָלָיְלָה,

עִם עַמּוּד עָנָן יוֹמָם לִהְיוֹת מַחְסֶה לָמוֹ,

שֶׁמֶשׁ לֹא יַכֵּם, גַּם בּוֹ יֶחֱסָיוּן מִזָּרֶם,

מִסַּעַר יַטִּיל גִּבְעוֹת חוֹל עַל עוֹבְרֵי אֹרַח,

מִשֵּׁן חַיְתוֹ שָׂדֶה וּמֵחֲמַת זוֹחֲלֵי אָרֶץ.

וּבְעַמּוּד אֵשׁ לָיְלָה הָאֵר לָהֶם בַּחשֶׁךְ,

אוֹרוֹ טוֹב כָּאוֹר כּוֹכָבִים מָתוֹק לָעָיִן,

גַּם עַמּוּד הָאֵשׁ כַּעֲנַן יוֹמָם מַחְסֶה לָמוֹ.

תָּמִיד הָעַמּוּדִים לִפְנֵי הָעָם לֹא מָשׁוּ,

לֹא עַמּוּד עָנָן יוֹמָם, לֹא עַמּוּד אֵשׁ לָיְלָה,

אֶל הַמָּקוֹם יִסַּע הָעַמּוּד שָׁם יִסָּעוּ,

כִּי עַל פִּי יְיָ יוֹמָם וָלַיְלָה הָלָכוּ.


אֶל רוֹאֶה4 לַלֵּב יֵצֶר מַחְשָׁבוֹת יוֹדֵעַ,

הוּא רָאָה אֶת לֹא רָאוּ הָרוֹאִים לָעָיִן,

רָאָה כִּי לֵב פַּרְעֹה רַע, עוֹד שׂוֹנֵא דָעַת,

וּמִסְפָּר רַב מִיּוֹעֲצָיו חַטָּאִים כָּמוֹהוּ,

כַּכּוּשִׁי לֹא יֵהָפֵך עוֹרוֹ לֹא יֵהָפְכוּ,

גַּם נִפְלְאוֹת לֵיל הַנֶּגֶף לֹא לָמוֹ הוֹעִילוּ,

יִרְאַת אֵל וָדַעַת אֶל לִבָּם לֹא הֵבִיאוּ,

מִפַּחַד שִׁלְּחוּ אֶת הָעָם כִּי עָלָה מָוֶת,

לֹא מִנֶּפֶשׁ חֲפֵצָה מֵרוּחַ נָכוֹנָה.

לוּלֵי כָשַׁל כֹּחָם, לוּ לַקְשׁוֹת לֵב יָכֹלוּ,

לִסְבּוֹל שׁוֹט נָקָם לָשֵׂאת אֵימָתָה וָפַחַד,

בַּעֲבוּר הַנִּפְלָאוֹת לַעֲצַת אֵל לֹא שָׁמָעוּ,

כִּי עֵינֵי הָרְשָׁעִים מֵאוֹר אֱמֶת תִּכְהֶיןָ,

לֹא זֶה עַם בִּינוֹת הוּא וִירַחְמֶנְהוּ עֹשֵׂהוּ,

בָּא מוֹעֵד לָתֵת אֶל חֵיקָם כַּאֲשֶׁר פָּעֳלוּ,

כִּי בַּעֲבוּר הַרְאוֹתָם כֹּחוֹ עַד כֹּה הֶחֱיָמוֹ,

וּבַעֲבוּר עַל אַפָּם יִתְחַנְּנוּ לֵאמֹר צֵאוּ,

עַתָּה כִּי עֶלְיוֹן גָּאוֹן וָגֹבַהּ הֶרְאָמוֹ,

כִּי עַל חֲמָתָם בֵּית יַעֲקֹב חָפְשִׁי יָצָאוּ,

בָּא עֵת לִמְחוֹץ רָאשֵׁי תַּנִּינִים עַל הַמָּיִם,

בַּעֲוֹן אֲבוֹתָם אֲשֶׁר דָּם נָקִי שָׁפָכוּ,

וִילוּדִים לֹא חָטָאוּ בִיאוֹרָם הִשְׁלִיכוּ,

גַּם בַּעֲוֹנוֹתָם כִּי עַל יְיָ הִגְדִּילוּ,

חֹשֶׁךְ שָׂמוּ לָאוֹר גַּם אוֹר שָׂמוּ לַחשֶׁךְ.


לָכֵן דִּבֶּר יְיָ אֶל מֹשֶׁה וַיֹּאמֶר:

"לֹא עַל כֵּן הַקָּהָל הַזֶּה אֵיתָם הֵבֵאתִי,

"בַּעֲבוּר הֲבִיאֵם בַּדֶּרֶךְ הַזֶּה מִדְבָּרָה,

"כִּי אִם לָתֵת מִכְשׁוֹל לִפְנֵי רִשְׁעֵי אָרֶץ,

"לִפְנֵי אַנְשֵׁי חֶרְמִי הֲכִינוֹתִים לָטֶבַח,

"כִּי פַּרְעֹה וַעֲבָדָיו כָּל פּוֹעֲלֵי אָוֶן אֵלֶּה,

"רַחֵם לֹא אֲרַחְמֵם, כָּאֶפֶס לִי נֶחְשָׁבוּ,

"לִמוּדֵי הָרֵעַ הֵם, אוֹתִי לֹא יָדָעוּ,

"גֹּדֶל מַעֲשַׂי גַּם נִפְלְאוֹתַי לֹא לִמְּדוּם דָּעַת,

"וּלְחַזֵק כַּחְשֵׁיהֶם לוּ תּוֹאֵנָה יִמְצָאוּ,

"גַּם עַתָּה עוֹד הָגוּ דִּבְרֵי סָרָה יוֹסִיפוּ,

"אֲנִי הַיּוֹדֵעַ כָּל לָהֶם אַמְצִיאֶנָּה,

"בַּעֲבוּר פִּשְׁעָם בַּסֵּתֶר הֵם לָאוֹר יוֹצִיאוּ,

"וּכְאוֹר אוֹצִיא צִדְקִי מִשְׁפָּטִי כַּצָּהֳרָיִם,

"כִּי לַשְּׁמוּעָה תָבֹא כִּי יָרְדוּ חַיִּים דּוּמָה,

"הַנִּשְׁאָרִים עַל אַדְמַת מִצְרַיִם יָבִינוּ,

"כִּי לֹא עַל שָׁבֵי פֶשַׁע חֲמָתִי שָׁפָכְתִּי,

"אַךְ עַל מוֹרְדֵי אוֹר, עַל עַם אֱלֹהִים נִאֵצוּ.

"דַּבֵּר אֶל הָעָם אָחוֹר מֵאֵיתָם יָשׁוּבוּ,

"לִפְנֵי פִי הַחִירוֹת בֵּין מִגְדּוֹל וּבֵין יָמָּה,

"נֹכַח בַּעַל צְפוֹן אָהֳלֵי אַפַּדְנָם יִטָּעוּ,

"שָׁם פֶּתַח הַסְּלָעִים מִשָּׁם אֶל יַם יָבֹאוּ,

"עַד הַמָּקוֹם הַהוּא בֵּית יִשְׂרָאֵל יִסָּעוּ.

"פַּרְעֹה וַעֲבָדָיו לוּ חָכְמוּ הֲלֹא יַשְׂכִּילוּ,

"כִּי עַל פִּי שָׁבוּ וּבַעֲצָתִי זֹאת הָיָתָה,

"וּמֵחֲשׁוֹב רַע עַל אֵל אוֹ עַל עַמּוֹ יֵבוֹשׁוּ.

"כִּי אֵיךְ יַחְשְׁבוּ רַע, עָלַי עֻזִּי בָם הוֹדַעְתִּי,

"יַחְשְׁבוּ רַע עַל עַמִּי בִּגְלָלָם זֹאת עָשִׂיתִי,

"אַךְ שִׁיבַת עִמִּי אָחוֹר פּוֹשְׁעִים כִּי יִשְׁמָעוּ,

"גַּם לִבּוֹתָם יָשׁוּב אָחוֹר וִידַבְּרוּ כַחַשׁ,

"לֵאמֹר: שֶׁקֶר דִּבֶּר משֶׁה אַךְ קָשַׁר קֶשֶׁר,

"לֹא אֵל צִוָּהוּ לַנְחוֹת אֶת הָעָם הַמִּדְבָּרָה,

"כִּי אֵיךְ אֶל אֵל יוּכַל כֹּל משֶׁה יִקְרָאֶנּוּ,

תָּשָׁב אָחוֹר יָדוֹ מֵהוֹלִיכֵם בּוֹ בֶטַח,

"כִּי כָעִיר הַבְּצוּרָה סוֹגֶרֶת וּמְסֻגֶּרֶת,

"כֵּן סָגַר הַמִּדְבָּר עַל הַבּוֹרְחִים הָאֵלֶּה,

"נָמוֹג לִבָּם מִבּוֹא בוֹ, מִדְּרוֹךְ בִּילֵיל תֹּהוּ,

"עַל כֵּן אָחוֹר שָׁבוּ בַּקֵּשׁ דֶּרֶך לִבְרוֹחַ,

"וּנְבֻכִים הֵם בָּאָרֶץ מַלֵּט בַּל יָדָעוּ,


"כָּעֵת אִם אַנְשֵׁי חֵיל עֲלֵיהֶם יִפּוֹלוּ,

"מִסְפָּר מִצְעָר תּוֹפְשֵׂי קֶשֶׁת אֲחוּזֵי חָרֶב,

"יַשְׁחִיתוּ הָמוֹן רַב זֶה רַכֵּי לֵב אֵלֶּה,

"עַם לֹא יָדְעוּ מִלְחָמָה עַל סוּס לֹא רָכָבוּ.

"כָּל זֹאת יַגִּיד נָבָל לִבְלִי חֹק יִפְעַר פִּיהוּ,

"עַל רָעוֹתָיו הַחֲתוּמִים בּוֹ פִּיו יַרְשִׁיעֶנּוּ,

"אַךְ אִם תִּרְגַז נַפְשׁוֹ תִּדְמֶה אֶל סִיר נָפוּחַ,

"לִרְדּוֹף אַחֲרֵי הָעָם לִבּוֹ בַּל יִמְלָאֶנּוּ,

"אִישׁ רָאֲתָה עֵינוֹ אֶל אַרְמוֹנוֹ עָלָה מָוֶת,

"שָׁמְעָה אָזְנוֹ בָשְׁתּוֹ מִפִּי בֵּן יוֹרֵשׁ עֶצֶר,

"גַּם הַמַּגֵּפוֹת מֵאָז עַל אַרְצוֹ עָבָרוּ,

"עוֹד מַרְגִּיזִים בִּטְנוֹ, זִכְרָם נַפְשׁוּ יַחֲרִידוּ,

"כִּי כַּתְּלָאוֹת אֲשֶׁר מָצָא לֹא יִסְבּוֹל גֶּבֶר,

"הַהוּא יֹאמַר עַל עַם יְיָ אֶגְבַּר חָיִל,

"טֶרֶם יַעֲשֶׂה עַל פִּי, טֶרֶם אֶתֵּן בּוֹ רוּחַ,

"כָּרוּחַ אֲשֶׁר מֵעַל לִבּוֹ הֲסִירוֹתִי,

"כִּי אִם עַד מוֹת יוֹרֵשׁ כִּסְאוֹ חִזַקְתִּיו אָנִי,

"אַחֲרֵי אֶת בֵּית יִשְׂרָאֵל מֵאַרְצוֹ הוֹצֵאתִי,

"לֹא חִזַּקְתִּיו עוֹד, לֹא עוֹד אֶת לִבּוֹ הִכְבַּדְתִּי,

"אָכֵן עַתָּה כִּי הֵבֵאתִי קָהָל זֶה יָמָּה,

"וּבְחַבְלֵי אַפִּי אֶל מִשְׁבְּרֵי מָוֶת אֶמְשְׁכֶנּוּ,

"אָשׁוּב חַזְּקֵהוּ לַעֲשׂוֹת חֶפְצוֹ אֲנִיחֶנּוּ,

"צָרוֹת קַדְמוֹנִיּוֹת מִלִּבּוֹ אֲשַׁכֵּחַ,

"עֲמָלוֹ לֹא יִזְכּוֹר אֶל אֵיתָנוֹ יַגִּיעַ.

"אָז יִרְדּוֹף אַחֲרֵי עַמִּי, יֹאמַר אוֹרִישֵׁמוֹ,

"כִּי מִרְמָה עָשׂוּ מִתַּחַת יָדִי בָּרָחוּ,

" שָׁם יִפּוֹל שׁוֹדֵד, שָׁם יִפְּלוּ כָּל פּוֹעֲלֵי אָוֶן,

"כִּי בּוֹ וּבְכָל חֵילוֹ הַפַּעַם אִכָּבֵדָה,

“כִּי אֲנִי יְיָ אֶת מִצְרַיִם אוֹדִיעַ”.


כַּאֲשֶׁר אִמְרוֹת עֶלְיוֹן אֶל צִיר נֶאֱמָן נוֹדָעוּ,

הוֹדִיעֵם אֶל מַחֲנֶה אֱלֹהִים עֵדֶר קֹדֶשׁ,

מִשּׁוּב אָחוֹר לִבּוֹתָם בַּל אָחוֹר נָסוֹגוּ,

כַּאֲשֶׁר צִוָּם כֵּן עָשׂוּ, בֵּאלֹהִים בָּטָחוּ,

רָאוּ5 יושְׁבֵי אֵיתָם כִּי שָׁבוּ, וַיִּתְמָהוּ,

וּבְכָל מָקוֹם דַּרְכָה רַגְלָם “מַה זֹּאת” שָׁאָלוּ,

רָץ לִקְרַאת רָץ מַגִּיד לִקְרַאת מַגִּיד מִהֵרוּ,

לֵאמֹר לַמֶּלֶךְ: "כַּאֲשֶׁר בָּא הָעָם אֵיתָיְמָה,

נָמוֹג לִבָּם מֵאֲנוּ לָלֶכֶת הַמִּדְבָּרָה,

"גַּם שָׁבוּ אָחוֹר דֶּרֶך הַנֶּגֶב נָסָעוּ,

“בֹּרֵחַ הוּא, לֹא אֵל מוֹצִיאוֹ מִמִּצְרָיִם”.

כִּשְׂמוֹחַ אַסִּיר בַּרְזֶל, כִּמְשׂוֹשׂ יוֹשֵׁב חשֶׁךְ,

בֶּאֱמוֹר לוֹ צֵא לַחָפְשִׁי נַתֵּק מוֹסְרוֹתֶיךָ,

כֵּן לִשְׁמוּעָה זֹאת, אֶל קוֹלָ הַמְּבַשְׂרִים הָאֵלֶּה,

שָׂמַחַ נָבָל זֶה, עֲדַת פוֹשְׁעִים יַחְדָּו שָׂשׂוּ,

דַלְתוֹת יָגוֹן סָגָרוּ בִּפְנֵי זָדוֹן, נִפְתָּחוּ,

וְיַעֲלוּ מַיִם זֵדוֹנִים מִבּוֹרוֹת רֶשַׁע,

וּמְזִמּוֹתֵיהֶם הָרְצוּצִים חָפְשִׁי שִׁלָּחוּ,


עַל יְיָ גַּם עַל מֹשֶׁה סָרָה דִּבֵּרוּ,

סִכְלוּתָם מַה גָּדְלָה, אִוַּלְתָּם מַה שָּׂנָאָה,

כִּי עַל הָעָם אֲשֶׁר הִתְחַנְּנוּ לֵיל הַנֶּגֶף,

לֵאמֹר: “אַל תִּתְמַהְמָהוּ, מֵאַרְצֵנוּ צֵאוּ”.

אָמְרוּ עַתָּה כִּי מִבֵּית עֲבָדִים בָּרָחוּ.

עַתָּה לֵב פַּרְעֹה גַּם לֵב עֲבָדָיו נֶהְפָּכוּ,

לַחְפּוֹץ בְּעַם שָׂמַח לִבָּם יוֹם כִּי יָצָאוּ.

וַיֹּאמְרו: "מַה זֹּאת עָשִׂינוּ אֵיכָה נוֹאַלְנוּ,

" לָמָּה שִׁלַּחְנוּ אֶת יִשְׂרָאֵל מֵעָבְדֵנוּ,

אִישִׁים כָּמוֹנוּ מִצָּרוֹת רַבּוֹת לֹא מָטְנוּ,

“לָמָּה מִדֶּבֶר בָּאֹפֶל יַהֲלֹךְ כָּשַׁלְנוּ”.

מִשְׁכֵנֵיהֶם בּוֹשׁוּ, בִּפְנֵיהֶם הִתְקוֹטָטוּ,

מָה יַעֲשׂוּ בַּל יָדָעוּ, עֲצָתָם אָבָדָה,

כִּי לַעֲשׂוֹת נָקָם וּלְהָשִׁיב הָעָם בַּחֶרֶב,

גַּם אַבִּירֵי לֵב כָּמוֹהֶם לֹא מָצְאוּ חָיִל,

כִּי זָכְרוּ עוֹד שׁוֹאַת לַיְלָה, עוֹד מַה נִּבְהָלוּ

מִדִּבְרֵי הַמּוּמָתִים, מֵחָזוֹן סִפֵּרוּ,

וּכְרוּחַ עַזָּה תַּעֲצוֹר אֶת מֵרוּץ הַנָּחַל,

כֵּן מֵרוּץ מֵי זָדוֹן שֶׁטֶף אַף וּבְלִיָּעַל.

מֵרוּחַ פַּחַד עָבְרָה בָּם מֵאָז נֶעֱצָרוּ,

עוֹד מִתְקוֹטְטִים בִּפְנֵיהֶם לֵאמֹר אֵיךְ נוֹאַלְנוּ,

מַמְתִּיקִים סוֹד. עוֹד מַה לַעֲשׂוֹת הִתְיָעָצוּ,

חֲמַת יְיָ עַד אֵין מַרְפֵּא בָּם בָּעָרָה,

שָׁלַח דָּבָר וַיַּרְפָּאֵם, מִלְּטָם מִפָּחַד,

שִׁכַּח מִלֵּב פּוֹשְׁעִים זִכְרוֹן קוֹרוֹת מִקֶּדֶם,

רָעוֹת מָצָאוּ כַּחֲלוֹם לַיְלָה נִדְמוּ לָמוֹ,

כַּשְּׁמוּעָה הֻגַד לָהֶם, כַּתְּלָאָה נוֹשֶׁנֶת.

כִּי יוֹצֵר רוּחַ אָדָם בּוֹ שַׁלִּיט בָּרוּחַ,

כַּאֲשֶׁר חָפֵץ אֵל כֵּן יִפְעָל, יָדוֹ תַּטֶּנּוּ,

וַיְחַזֵּק לֵב פַּרְעֹה וַיְהִי כַּכְּפִיר בּוֹטֵחַ,

וַיחַזֵּק לֵב עֲבָדָיו אֹמֶץ רַב מָצָאוּ,

לִרְדוֹף אַחֲרֵי בֵּית יִשְׂרָאֵל עַם תּוֹעֵי דָרֶךְ,

כִּי לִנְקוֹם נִקְמָתָם מֵהֶם יַחְדָּו נוֹעָצוּ,

מֵאֵת הָאֱלֹהִים הָיְתָה זֹאת בַּעֲבוּר אַבְּדֵמוּ,

בַּעֲבוּר יִפְּלוּ בִּמְצוּלוֹת יָם קוּם בַּל יוֹסִיפוּ.


חֶבֶר פּוֹשְׁעִים אֵלֶּה תָּמְהוּ אִישׁ אֶל רֵעֵהוּ,

אֵיךְ רַכֵּי לֵב פֶּתַע פִּתְאוֹם עָצְמָה מָצָאוּ.

כִּי מֵאֵת יְיָ זֹאת פּוֹשְׁעִים לֹא הֵבִינוּ,

וּלְזַעֲקַת הַמּוֹשֵׁל בַּכְּסִילִים הִטּוּ אֹזֶן,

בִּגְרוֹנוֹ כַּקֶּבֶר פָּתוּחַ קוֹרֵא לָמוֹ:

"גִּבּוֹרֵי אַפִּי אֶת חַרְבוֹתֵיכֶם לִטֵּשׁוּ,

"לִבְשׁוּ הַשִּׂרְיוֹנִים מַרְכְּבוֹתֵיכֶם אֳסֹרוּ,

"אִסְפוּ חַיִל וּתְרוּעַת מִלְחָמָה הָרִיעוּ,

"עַל בֵּית יִשְׂרָאֵל, עַל עֲבָדִים בָּגָדוּ בָגֶד,

"עַל מַלְכָּם קָשְׁרוּ קֶשֶׁר וּבְמִרְמָה בָּרָחוּ,

"אֶת כָּל חֲמֻדּוֹת אַרְצֵנוּ אִתָּם לָקָחוּ,

"נִכְבַּדֵי נוֹף הָעִיזוּ, שָׂרֵי צֹעַן קוּמוּ,

"רִדְפוּ אַחֲרֵיהֶם, חֶרֶב עֲלֵיהֶם הָרִיקוּ,

"הִרְגוּ כָּל אִישׁ, טַף עִם נָשִׁים תִּשְׁבּוּ שֶׁבִי,

"הוֹן רַב תִּמְצָאוּ, שָׁלָל בֵּינֵיכֶם חַלֵּקוּ,

"הָשִׁיבוּ אֶת הַמַּלְקוֹחַ, חֵיל בָּלְעוּ יָקִיאוּ,

"גַּם מִקְנֵיהֶם עִם כָּל אֲשֶׁר לָהֶם תָּבֹזוּ,

"אַל תִּרָאוּ, כִּי עִם מִי הַיּוֹם תִּלָּחֵמוּ,

"עִם עֲבָדִים נִכְאֵי רוּחַ לַנָּשִׁים נִמְשָׁלוּ,

"עִם עַם תּוֹעִים בַּדֶּרֶךְ וּנְבוּכִים בָּאָרֶץ,

"עִם נִכְלָמִים הַיּוֹדְעִים כִּי מִרְמָה פָּעָלוּ,

"כַּבֹּשֶׁת גַּנָּב אִם יִמָּצֵא בְּמַחְתֶּרֶת.

"יֵבוֹשׁוּ מִפְּנֵיכֶם, מִמַּרְאֵיכֶם יֵחַתּוּ,

"כִּי מִרְמָה מֵאָז לֹא נִשְׁמָעָה עַם זֶה רִמּוּנוּ,

"אִם נֹאמַר נַחֲרִישׁ חֶרְפָּה נִהְיֶה לִשְׁכֵנֵינוּ,

"עַתָּה הֵחָלְצוּ, אִישׁ אֶת רֵעֵהוּ חַזֵּקוּ,

"נָחֲלוּ לָכֶם כָּבוֹד, אֶת חֶרְפַּתְכֶם הָסִירוּ,

"וַאֲנִי בָרֹאשׁ, כִּי אֵצֵא לִפְנֵיכֶם גַּם אָנִי.


כַּאֲשֶׁר6 כִּלָּה דַּבֵּר וּבְרִשְׁעוֹ לָקַח נָפֶשׁ,

כִּי פוֹעֲלֵי אָוֶן יַחְדָּו לַעֲצָתוֹ שָׁמָעוּ,

אָסַר אֶת רִכְבּוּ, אֶת חַרְבּוּ חָגַר עַל יֶרֶךְ,

מִמִּבְחַר מַרְכְּבוֹתָיו לָקַח שֵׁשׁ מֵאוֹת רֶכֶב,

אֲשֶׁר שָׂם תַּחַת יָדוֹ רִאשׁוֹנָה יִסָּעוּ,

אַחֲרֵיהֶם הַנִּשְׁאָרִים כָּל רֶכֶב מִצְרָיִם,

הִפְקִיד תַּחַת יַד שָׁלִישִׁים גִּבּוֹרֵי חֵיל,

רוּחַ עָרִיץ לֹא שָׁקַט עַד כִּלָּה מַעֲשֵׂהוּ,

וּלְמוֹעֵד מִצְעָר אֲנָשָׁיו הַכֹּל הֵכִינוּ.

שָׂמַח לִבּוֹ לִרְאוֹת כִּי הוּא בַּכֹּל מַצְלִיחַ,

מֵאַחֲרִית דָּבָר לֹא דָּאַג, לֹא יָדַע פַּחַד,

כָּל חֲגוּר בַּמַּחֲנֶה הִתְהַלֵּל כִּמְפַתֵּחַ,

וּבְגוֹבַהּ אַפּוֹ רָדַף רָשָׁע עִם מַחֲנֵהוּ

אַחֲרֵי יִשְׂרָאֵל הָשָּׁבִים מֵאֵיתָם יָמָּה,

מִפַּחַד שַׁאֲנַנִּים וּבְיַד רָמָה יָצָאוּ,

כִּי יִדְלֹק אוֹיֵב אַחֲרֵיהֶם אֵיכָה יַאֲמִינוּ,

אִם עֲנַן יְיָ לִפְנֵיהֶם הוּא נוֹסֵעַ,

מִשְּׂאֵתוֹ יִפּוֹל כָּל לֵב, יִרְפּוּ כָּל יָדַיִם.

אַךְ אוֹי לָרָשָׁע יוֹם נִקְמַת אֵל בּוֹ שַׁלָּטֶת,

אִם תִּתֵּן חַחִים בִּלְחָיָיו וּתְקָרְבֵהוּ,

אֶל שַׁעֲרִי מָוֶת תִּשְׁלַח יָדָהּ תִּסְעָדֵהוּ,

מִכֹּל לֹא יִמּוֹט, לַהֲבֵי אֵשׁ לֹא יַחֲרִידוּהוּ.

כֵּן חִזַּק יְיָ אֶת לֵב פּוֹעֲלֵי רַע אֵלֶּה,

כִּי הֵמָּה עַד לִפְנֵי פִי הַחִירוֹת קָרָבוּ,

אִם שָׁם עַמּוּד הֶעָנָן עֵינֵיהֶם הִבִּיטוּ,

לֹא סָר כֹּחָם מֵעֲלֵיהֶם, בִּזְרוֹעָם בָּטָחוּ,

עַל יְיָ עַל עַם מַרְעִיתוֹ הִתְיַצָּבוּ,

וּלְכַלּוֹת אֶת בֵּית יִשְׂרָאֵל גֵּאִים עָלָזוּ,

אִם אֶל חִזֵּק לֵב בּוֹעֲרִים שָׂמוּם חִדְלֵי פַחַד,

מִמַּה יִירְאוּ? הַאִם מִכְּבוֹד מַעֲשָׂיו יַרְאֵמוֹ,

וּבְהַבְלֵיהֶם גַּם מַעֲשֵׂי נִפְלָאוֹת יַהְבִּילוּ,

וַיִּרְדְּפוּ אַחֲרֵיהֶם, כַּסּוּפָה הִתְגַּלְגָּלוּ,

כָּל סוּס רֶכֶב פַּרְעֹה אֲשֶׁר בָּרֹאשׁ הָלָכוּ.

אַחֲרֵיהֶם חֵילוֹ וּפָרָשָׁיו אַנְשֵׁי חַיִל,

עַד פִּי הַחִירוֹת לִפְנֵי בַּעַל צְפוֹן קָרָבוּ,

אֶת יִשְׂרָאֵל חוֹנִים עַל הַיָּם שָׁם הִשִּׂיגוּ.


כַּאֲשֶׁר הִקְרִיב7 פַּרְעֹה עִם שֵׁשׁ מֵאוֹת רֶכֶב,

רָאוּ כִּי בִמְתֵי מִסְפָּר בָּא מֶלֶךְ מִצְרַיִם,

וַיִּתְמָהוּ, כִּי יַעַן מַה בָּא לֹא יָדָעוּ,

כִּי חָפֵץ הִלָּחֵם בָּם עַד כֹּה לֹא הֶאֱמִינוּ,

אֶל מַחֲנֶה הָאֱלֹהִים מִי זֶה יֶעְרַב לָגֶשֶׁת?

גַּם לַעֲרוֹךְ עַמָּם מִלְחָמָה קָטֹן מַחֲנֵהוּ,

אִישׁ אֶל אָחִיו פָּנוּ, יָמִין וּשְׂמֹאל הִבִּיטוּ,

לִרְאוֹת אִם אַחֲרֵיהֶם עוֹד עַם רַב אַנְשֵׁי לָחֶם.

כַּאֲשֶׁר בֵּית יִשְׂרָאֵל אֶת עֵינֵיהֶם נָשָׂאוּ,

אֶת אֲשֶׁר לֹא עָלָה עַל לִבָּם הִתְבּוֹנָנוּ,

כִּי חֵיל מִצְרַיִם כֻּלּוֹ אַחֲרֵיהֶם נוֹסֵעַ,

וַיֵּדְעוּ כִּי לַקְּרָב יָצָאוּ, וַיִּירָאוּ.

עֲבָדִים מִתְּמוֹל שִׁלְשׁוֹם מִפַּחַד נָמוֹגוּ,

כִּי הִלָּחֵם עִמָּם לִבָּם בַּל יִמְלָאֵמוֹ,

לִקְרַאת חֵיל גִבּוֹרִים מַה יַּעֲשׂוּ רוֹמְסֵי חֹמֶר,

תַּחַת הָרֵק חַרְבָּם לִבָּם אֶל אֵל נָשָׂאוּ,

וּלְעוֹשֵׂה נִפְלָאוֹת בַּצַּר לָהֶם צָעָקוּ,

"אֵל מוֹשִׁיעַ! הַאִם לַשָּׁוְא לַעֲשׂוֹת הִפְלֵאתָ,

“וּלְלֹא יוֹעִיל הוֹדַעְתָּ לַצָּרִים עֻזֶּךָ”

"בִּזְרוֹעֲךָ הַנְּטוּיָה מֵאַרְצָם הוֹצֵאתָנוּ,

"אַךְ הֵמָּה עַל נִפְלְאוֹתֶיךָ רֹאשָׁם יָנִיעוּ,

"עוֹדָם דּוֹרְכִים קַשְׁתָּם כּוֹנְנִים חִצָּם עַל יֶתֶר,

"וּבְחֶרֶב גַּאֲוָתָם עַד מַחֲנֶיךָ יִגָּשׁוּ,

"אִם יִפְּלוּ עָלֵינוּ גַּם שָׂרִיד לֹא יַשְׁאִירוּ,

"כִּי מִי נַחְנוּ לַעֲרוֹךְ מוּל עַזֵּי מֶצַח אֵלֶּה,

"נָא קוּמָה אֱלֹהִים וַהֲדוֹף אֶת מִתְקוֹמְמֶיךָ,

"טֶרֶם יָבֹא כַנָהָר צַר לִשְׁטוֹף מַחֲנֵנוּ,

“וּבְטֶרֶם עַל כֹּחֲךָ הַגָּדוֹל יִתְפָּאֵרוּ”.


כַּאֲשֶׁר רַחֲמֵי8 יְיָ רֶגַע הִתְאַפָּקוּ,

וּפַעֲמֵי צָרִים מִפַּחַד אֵל לֹא נָמוֹטוּ,

כַּאֲשֶׁר עֲדַת קִצְרֵי רוּחַ לִרְאוֹת הוֹחִילוּ,

כִּמְעַט יָצָא לִבָּם, כִּמְעַט תִקְוָה אָבָדוּ,

לֹא חִכּוּ לַעֲצַת אֵל, בִּדְבָרוֹ לֹא בָּטָחוּ,

וּבְמֶרֶר רוּחָם קָשׁוֹת אֶל משֶׁה דִּבֵּרוּ:

"יוֹדֵעַ סוֹד עֶלְיוֹן מַה זֹּאת עָשִׂיתָ לָנוּ,

"הַהִגְדַּלְתָּ לַעֲשׂוֹת בַּעֲבוּר כִּי פֹה נָמוּתָה,

"מָה בֶּצַע כִּי מִבֵּית עֲבָדִים הוֹצֵאתָנוּ,

"בַּעֲבוּר כִּי מַחֲנֵּנוֹ כֻלּוֹ יוּבָל לַטֶּבַח,

"הֲמִבְּלִי עַל אַדְמַת חָם אֵין שָׁם מָקוֹם קֶבֶר,

"הוֹצֵאתָנוּ לָמוּת בַּמִדְבָּר מוּתֵי חֶרֶב?

"מִיּוֹם בּוֹאֲךָ מִצְרַיִם וַתֹּאמֶר אֵלֵינוּ

"אוֹצִיא אֶתְכֶם מִפֹּה לַהֲבִיאֲכֶם הַמִּדְבָּרָה,

"שָׁמַעְנוּ וַנִּירָא מֵאָסוֹן שָׁם יִקְרֵנוּ,

"הֲלֹא זֶה הַדָּבָר אָז אֵלֶיךָ דִּבַּרְנוּ,

חֲדַל מִמֶּנוּ נָא נַעַבְדָה אֶת מִצְרַיִם,

“כִּי מִמּוּתֵנוּ בַמִּדְבָּר עַבְדֵּם טוֹב לָנוּ”.


אַף כִּי נָכוֹן לֵב מֹשֶׁה בֶּאֱלֹהָיו בָּטוּחַ,

כִּי מִצָּרָיו יִקּוֹם, כַּאֲשֶׁר דִּבֵּר כֵּן יַעַשׂ,

עַל צָרַת רַכֵּי לֵב אֵלֶּה רַחֲמָיו נִכְמָרוּ,

עַל כֵּן גַּם אִישׁ הָאֱלֹהִים אֶל צוּרוֹ שִׁוֵּעַ,

לֵאמֹר: "כִּי תַשְׁמִיד אֶת חֵיל צָרֵנוּ יָדַעְתִּי,

"כִּי אַתָּה אָמַרְתָּ אֵלַי אִכָּבְדָה בָּמוֹ,

"אַךְ אָמַרְתִּי טֶרֶם יֵרָאוּ פֹה יִפֹּלוּ,

"עַם רַב סוּס וָרֶכֶב אֶת עַמְּךָ לֹא יַבְהִילוּ,

אָכֵן בִּזְרוֹעַ רָמָה עוֹמְדִים מִמּוּלֵנוּ,

"וּכְפֶשַׂע בֵּין הָעָם כֻּלּוֹ וּבֵין הַמָּוֶת,

"כֻּלָּם זוֹעֲקִים כָּאִשָּׁה תִּזְעַק בַּחֲבָלֶיהָ,

כֻּלָּם קִצְרֵי רוּחַ בַּחַיִּים לֹא יַאֲמִינוּ,

"מֵאַחֲרֵיהֶם רוֹאִים חֵיל פַּרְעֹה סוּס וָרֶכֶב,

"מִלִּפְנֵיהֶם לֵב יָם בּוֹ אֳנִיּוֹת יֵלֵכוּ,

"מִימִינָם אֶרֶץ אוֹיְבֵיהֶם שַׁדְמוֹת מִצְרַיִם,

"וּמִשְּׂמֹאלָם מָעוֹן אֲרָיוֹת, הַרְרֵי טֶרֶף,

בִּמְצוּקָה זֹאת עַמֶּךָ עָיֵף וַחֲדַל כֹּחַ,

"וּמָה אֶעֱשֶׂה לָעָם הַזֶּה לֹא הוֹדַעְתַּנִי,

“אֵל נָא מָה אֹמַר לָהֶם? בַּמָּה אֲנַחְמֶנוּ”.

עַל זַעֲקַת9 אִישׁ אֱמוּנִים רַב חֶסֶד עָנָהוּ,

"מָה תִּצְעַק אֵלַי, כִּי אִתְּכֶם אֲנִי יָדַעְתָּ,

"אִם נִפְלֵאת מִמְךָ עֲצָתִי, מָה הִיא אוֹרֶךָּ,

"בַּעֲבוּר אֶל מָרֵי נֶפֶשׁ אֵלֶּה תַגִּידֶנָּה,

"מַהֵר לֵךְ אֲלֵיהֶם דַּבֵּר עַל לֵב אַחֶיךָ,

"שַׁמְּחֵם מִיגוֹנָם וֶאֱמוֹר לָהֶם אַל תִּירָאוּ,

"לַשָּׁוְא הֵנִיף צַר אֶת יָדוֹ, לֹא תַעֲשֶׂה חָיִל,

"גַּם יֶדְכֶם לֹא תִשְׁלְחוּ בּוֹ, כִּי לֹא תִלָּחֵמוּ,

"הַרְחִיקוּ מִמֶּנּוּ וּמִמְּקוֹמְכֶם תִּסָּעוּ.

"כִּי יֹאמְרוּ אָנָה נֵלֵךְ צָפוֹנָה אוֹ נֶגְבָּה?

"אֱמוֹר קָדִימָה לֵכוּ וּבְתוֹךְ הַיָּם בּוֹאוֹ,

"מִגַּלָּיו אַל תֵּחַתּוּ, יַבָּשָׁה אֶעֱשֶׂנּוּ,

"וּלְעֵת שִׁבְטֵי יִשְׂרָאֵל לִשְׂפַת יָם יַגִּיעוּ,

"אָז צֵא אַתָּה לִפְנֵיהֶם הָרֵם אֶת מַטֶּךָ,

"וּנְטֵה אֶת יָדְךָ עַל הַיָּם כִּי תִבְקָעֵהוּ,

"וּבְשִׂמְחַת לֵב אֶל תּוֹךְ הַיָּם עַמִּי יָבוֹאוּ,

"בַּעֲבוֹתוֹת אַהֲבָה אֶמְשָׁכֵם אַחֲרַי יָרוּצוּ,

"כִּי פִתְאוֹם יֵרָאוּ נִפְלָאוֹת לֹא נִשְׁמָעוּ,

"כִּי יֵהָפֵךְ יָם לַבִּקְעָה צוֹמַחַת עֵשֶׂב.

אַז יִרְאוּ כִּי עַבְדִּי אַתָּה עֻזִי אִתֶּךָ,

"וּזְרוֹעַ תִּפְאַרְתִּי הוֹלַכְתִּי לִימִינֶךָ,

"אַךְ צָרַי בַּל יֶחֱזָיוּן מַה בַּיָּם פָּעַלְתִּי,

"כִּי אַחְשִׁיךְ בַּעֲדָם וַאֲחַזֵּק אֶת לֵב מִצְרַיִם,

"סוּס רֶכֶב וּפָרָשִׁים אַחֲרֵיהֶם יִרְדֹּפוּ,

"גַּם הֵמָּה בַּיָּם זֶה עַל יַבָּשָׁה יִדְרֹכוּ,

"אָז בָּם אִכָּבֵדָה אֶעֱשֶׂה לִי שֵׁם תִּפְאָרֶת,

"כִּי פַרְעֹה עִם חֵילוֹ בֵּין מִשְׁבְּרֵי יָם יֹאבֵדוּ,

"גַּם רִכְבוֹ גַּם פָּרָשָׁיו לֹא יִמָּלֵטוּ,

"וּבְהִכָּבְדִי בָם יֵדְעוּ כָּל יוֹשְׁבֵי מִצְרַיִם,

"כִּי אֵל גִּבּוֹר אָנִי, מֶלֶךְ הַכָּבוֹד אָנִי,

"כִּי בוֹזַי וּמְנַאֲצַי שַׁעֲרֵי מָוֶת יָבוֹאוּ,

"כִּי מִכָּל אוֹתוֹתַי עַל כֵּן מַלְכָּם מִלַּטְתִּי,

"וַעֲדַת גֵּאִים מִסָּבִיב לוֹ עַל כֵּן הֶחֱיֵיתִי,

"בַּעֲבוּר הַצְמֵת אוֹתָם בִּמְהוּמָה בַּל נוֹדַעַת,

"לִהְיוֹת אוֹת עוֹלָם וּמוֹפֵת אֶל שׁוֹכְנֵי אָרֶץ,

“לָדַעַת כִּי אַחֲרִית בּוֹזֵי צֶדֶק נִכְרָתָה”.


לִבְשׂוּרָה10 זֹאת מִפִּי אֵל, מֵחֲדָשׁוֹת כָּאֵלֶּה,

שָׂשׂוֹן גַּם שִׂמְחָה רַחֲבֵי לֵב משֶׁה מִלֵּאוּ,

וַיִּפֶן אֶל הָעָם אֶל קְהַל דַּכְּאֵי רוּחַ,

וַיֹּאמֶר אֲלֵיהֶם "הַסּוּ! לֹא עוֹד תִּירָאוּ,

"הָעָם הָרַב הַזֶּה רֶכֶב וָסוּס תַּבִּיטוּ,

"חֲמַת אֵל הֱבִיאָם פֹּה, לַמָּוֶת הוּבָלוּ.

"יָדוֹ תִּהְיֶה בָּם וּבְיַד אָדָם אַל יִפֹּלוּ,

"אַחַי הִתְיַצָּבוּ, נִכְשָׁלִים אִזְרוּ חָיִל,

"וּרְאוּ בִּישׁוּעַת יְיָ, כִּי עוֹשֶׂה פֶלֶא

"יַעֲשֶׂה הַיּוֹם עִמָּכֶם אֶת לֹא רָאֲתָה עַיִן.

"כִּי עַל מִקְוֵה הַמַּיִם, עַל גַּלֵּי יָם אֵלֶּה,

"עַל יָם זוֹעֵף זֶה, יִקְרָא אֵל וִיהִי יַבֶּשֶׁת.

"כַּבְּהֵמָה בַּבִּקְעָה תֵרֵד תֵּרְדוּ בוֹ בֶּטַח,

"בּוֹ תִּשְׂמְחוּ בַיָי, בּוֹ צָרִים יִשְׁתּוֹלָלוּ,

"כִּי בוֹ יִפּוֹל שׁוֹדֵד, בּוֹ יִפּוֹל כָּל מַחֲנֵהוּ.

"קוּמוּ וּסְעוּ עַמִּי מֵאוֹיֵב אַל תִּפְחָדוּ,

רֶגַע תִּסְעוּ מַפֹּה עַיִן לֹא עוֹד תִּרְאֶנּוּ,

"אַף כֵּן לִרְאוֹתָם עוֹד עַד עוֹלָם לֹא תוֹסִיפוּ,

"כִּי לֹא יִמָּלֵט גַּם אֶחָד מֵחֵיל מִצְרָיִם,

"הַבָּאִים אַחֲרֵיכֶם בַּיָּם יַחְדָּו יָמוּתוּ.

"בֵּית יִשְׂרָאֵל! חֶרֶב מִתַּעֲרָהּ לֹא תִשְּׁלֹפוּ,

"וּתְרוּעַת מִלְחָמָה עַל אוֹיֵב לֹא תָרִיעוּ,

"אֲבָל תַּחֲרִישׁוּן וּבְנִפְלְאוֹת אֵל תִּתְבּוֹנָנוּ,

"כִּי יְיָ יִלָּחֵם לָכֶם אֵין כָּמוֹהוּ,

“אֵל נוֹרָא אֲשֶׁר לוֹ הַיָּם כִּי הוּא עָשָׂהוּ”.


כַּאֲשֶׁר כִּלָּה11 דַּבֵּר, בֵּית יִשְׂרָאֵל נָסָעוּ,

פִּתְאוֹם עָלַץ לִבָּם, יָגוֹן וַאֲנָחָה נָסוּ,

אֶת צָרֵיהֶם לֹא רָאוּ עוֹד. אַיָּם שָׁאָלוּ,

גַּם אוֹיֵב לֹא רָאָה עוֹד עֲבָדָיו מִקֶּדֶם,

כִּי מַאֲפֵל שָׂם אֵל בֵּין עַמּוֹ וּבֵין מִצְרָיִם,

בַּל יִרְאֶה צַר גֵּאוּת עֶלְיוֹן, יֶחֱזֶה מַעֲשֵׂהוּ,

לוּ יִרְאוּם צָרִים, מִפַּחַד אָחוֹר יָשׁוּבוּ.

מַאֲפֵל זֶה סָגַר בַּעֲדָם, מִמְּקוֹמָם לֹא מָשׁוּ,

עַד עֲדַת יִשְׂרָאֵל כֻּלּוֹ תּוֹך הַיָּם בָּאוּ,

וּבְמַחֲנֶה מִצְרַיִם אֵין רוֹאֶה אֵין יוֹדֵעַ,

כִּי לִפְנוֹת עֶרֶב שִׁבְטֵי יָהּ אֶל יָם הָלָכוּ,

וּבְטֶרֶם לַיְלָה הַמַּחֲנֶה בַּיָּם הָיָתָה.

כִּי מִסְפָּר הַגְּבָרִים עִם נָשִׂים טַף וָעֵרֶב,

אִם לִשְׁלשֶׁת אַלְפֵי אֲלָפִים יֵחָשֵׁבוּ,

וּבְכָל שׁוּרָה כִּשְׁנֵים עָשָׂר אֶלֶף יֵלֵכוּ,

מִסְפַּר כָּל שׁוּרוֹתָם כַּחֲמִשִּׁים וּמָאתָיִם,

וּלְמִקְנֵה הַצֹּאן גַּם הַבָּקָר אֲשֶׁר לָמוֹ,

אִם גַּם לָהֶם מוֹעֵד זֶה אֶל לֶכְתָּם יִתֵּנוּ,

טֶרֶם בּוֹא הָאוֹר, טֶרֶם כּוֹכָבִים יָהֵלּוּ,

הַהוֹלְכִים בָּאַחֲרוֹנָה אֶל תּוֹךְ יָם הִגִּיעוּ.

וּלְעֵת עַם קָדְשׁוֹ מִפִּי הַחִירוֹת נָסָעוּ,

לֹא נָתַן אֵל אֶת עַמּוּד הֶעָנָן לִנְסוֹעַ

לִפְנֵי מַחֲנֶה יִשְׂרָאֵל לַנְחוֹתָם הַדָּרֶךְ.

עַתָּה הֶרְאָם עֻזּוֹ, הוֹדוֹ כִּסָּה שָׁמָיִם,

כִּי אֱלֹהֵי יִשְׂרָאֵל מִקָּדְשׁוֹ הוֹפִיעַ

עַל מִשְׁבְּרֵי יָם, הָאָרֶץ מִכְּבוֹדוֹ הֵאִירָה,

וְּכַגִבּוֹר מִלְחָמָה יָצָא לִפְנֵי מַחֲנֵהוּ,

רָאוּ הֲלִיכוֹת אֱלֹהִים עַל רַבִּים מָיִם,

הֲלִיכוֹת אֵלִי הֲלִיכוֹת מַלְכִּי בַּקֹּדֶשׁ,

מַרְאוֹת וּתְמוּנוֹת כָּבוֹד, הָעָם כֻּלּוֹ חָזוּ,

גַּם עַל הָעֲבָדִים וּשְׁפָחוֹת שָׁפַךְ הָרוּחַ,

בַּעֲבוּר יֶחֱזוּ נֹעַם אֵל וּשְׂמָחוֹת יִשְׂבָּעוּ.

מַה שָּׂמַח הָעָם כֻּלּוֹ עֵת רָאֲתָה עֵינֵמוֹ,

אֶת משֶׁה אִישׁ הָאֱלֹהִים כַּמַּלְאָךְ נוֹסֵעַ

לִפְנֵי מַחֲנֶה הַגְּאוּלִים וּבְיָדו מַטֵּהוּ,

גַּם אֵל נִצָּב עָלָיו עִם מַחֲנוֹת רִבּוֹתַיִם,

וּזְרוֹעַ תִּפְאַרְתּוֹ לִימִינוֹ יוֹלִיכֶנָּה.

כִּי כַּאֲשֶׁר הַנּוֹסְעִים לִשְׂפַת הַיָּם קָרָבוּ,

וּמשֶׁה נָטָה עָלָיו יָדוֹ וּמַטֵּהוּ,

פִּתְאוֹם נִגְלָה דֶּרֶךְ חָדָשׁ וּרְחַב יָדָיִם,

לִמְסִלַּת רַגְלֵי חֲסִידָיו וּלְמַעֲבַר קֹדֶשׁ,

לִגְזָרִים גָּזַר אֵל יָם סוּף, הוּא הָאֵל עָשָׂהוּ,

וַהֲמוֹן מַיִם אֲשֶׁר בּוֹ, כַּאֲבָנִים נֶעֱרָמוּ,

וּמִשְּׁנֵי צִדֵּי הַמְּסִלָּה כַּנֵּד עָמָדוּ.

כִּי רֹחַב הַמְּסִלָּה אַמּוֹת שֵׁשׁ עֶשְׂרֵה אָלֶף,

כִּשְׁנֵים עָשָׂר אֶלֶף עוֹבְרִים בַּשּׁוּרָה אַחַת,

אָרְכָּהּ אַרְבָּעִים אֶלֶף מֵעֵבֶר אֶל עֵבֶר,

פִּי הַחִירוֹת מִזֶּה, שָׁם הַמְּסִלָּה הֵחֵלָּה,

אָכֵן תַּכְלִיתָהּ עַד מִדְבַּר אֵיתָם הִגִּיעָה,

הוּא מִדְבַּר שׁוּר כִּי שָׁם מִיַּם סוּף יָצָאוּ.


כִּי כַּאֲשֶׁר נִגְלָה הוֹד אֱלוֹהַּ עַל הַמָּיִם,

מַלְאַךְ הָאֱלֹהִים יוֹם יוֹם לִפְנֵיהֶם נוֹסֵעַ,

עָמַד מֵאַחֲרֵיהֶם לִכְבוֹד נֶאְדָּר בַּקֹּדֶשׁ,

וַעֲנַן יוֹמָם הַהוֹלֵךְ לַנְחוֹתָם הַדָּרֶךְ,

נָסַע גַּם הוּא מִפְּנֵיהֶם לַעֲמוֹד מֵאַחֲרֵימוֹ,

וַיָּבֹא הָעַמּוּד בֵּין הַמַּחֲנוֹת הַשְּׁתַּיִם,

וַיְהִי הֶעָנָן וַיְהִי חֹשֶׁךְ וַיְהִי אֹפֶל.

כָּל אֵלֶּה חָבְרוּ בִּסְבִיבוֹת מַחֲנֶה מִצְרָיִם,

מֵעֵת הַגְּאוּלִים מִפִּי הַחִירוֹת נָסָעוּ,

עַד הַמַּחֲנֶה כֻלָּהּ רַגְלָם בַּיָּם הִצִּיגוּ.

אֵיפֹה הָלְכוּ בֵּית יִשְׂרָאֵל, אָנָה נִמְלָטוּ,

וּמַה פָּעַל אֵל בַּיָּם סוּף, אֵיךְ זֹאת יֵדָעוּ,

אִם אֹפֶל חֶלְקָם הַמַּעֲלִים מֵהֶם כָּל אֵלֶּה?

כַּאֲשֶׁר הִרְחִיק הָעָם בַּיָּם אַמּוֹת אַלְפַּיִם,

וַיָּאֶר אֵל אֶת הַלַּיְלָה מִתּוֹךְ הַחשֶׁךְ,

לָתֵת נֵר אֶל רַגְלֵי צָרָיו בַּעֲבוּר יִסָּעוּ,

וּבְגֹבַהּ אַפָּם אַחֲרֵי יִשְׂרָאֵל בַּיָּם יִרְדֹּפוּ.

אַךְ לֹא סָר הָאֹפֶל, בֵּין הַמַּחֲנוֹת עוֹדֶנוּ,

עַל כֵּן כָּל הַלַּיְלָה אֶת יִשְׂרָאֵל לֹא רָאוּ,

עוֹדֶנּוּ עַל צִדֵּי דַרְכָּם בַּיַּרְכָתָיִם,

בַּל יִרְאוּ כִּי בֵין חוֹמוֹת מַיִם יֵלֵכוּ.

לֹא קָרַב מַחֲנֶה אֶל מַחֲנֶה עַד עֵת הַבֹּקֶר,

אַז רָאוּ מַעֲשֶׂה אֵל, אֶת נִפְלְאוֹתָיו הִבִּיטוּ,

כִּי כָּאִישׁ מִתְהַלֵּךְ בַּגָּן בֵּין פִּרְחֵי רֵיחַ,

כֵּן יִשְׂרָאֵל טוֹבֵי לֵב בַּיָּם הִתְהַלָּכוּ,

כִּי עִם מִצְרַיִם הָאֵיתָנִים יִלָּחֵמוּ,

עַמּוּדֵי שַׁחַק וּמוֹסְדוֹת אֶרֶץ יִרְגָּזוּ,

כֻּלָּם לַעֲשׂוֹת נָקָם בָּאוֹיְבֵי אֵל נִקְרָאוּ,

הֵמָּה רָאוּ וַיֶחֱרָדוּ, מִגְּאוֹן אֵל חַתּוּ.

וּבְנֵי יִשְׂרָאֵל כַּאֲשֶׁר נִפְלְאוֹת אֵל הִבִּיטוּ,

כִּמְאוֹרוֹת רִנְּנוּ, כִּבְנֵי אֵלִים הֵרִיעוּ,

"אַחַי! הַקְשִׁיבוּ לִי! מָה פָּעַל אֵל אַזְכִּירָה,

"כִּי אֶזְכּוֹר מַעֲלְלֵי יָהּ אַתֶּם הָבוּ לוֹ גֹּדֶל,

“בֵּית יַעֲקֹב הוֹדוּ לוֹ, עַמִּים כֻּלָּם הַלְלוּהוּ”.


כִּנְסֹעַ יוֹם זֶה עֲנַן יוֹמָם לִפְנוֹת עֶרֶב,

לַעֲמוֹד מֵאַחֲרֵיהֶם וַיַּאֲפֵל עַל מִצְרָיִם,

וּבְנֵי יִשְׂרָאֵל חֲלוּצִים עַד יָם נָסָעוּ,

וּמשֶׁה הִטָּה עָלָיו יָדוֹ וּמַטֵּהוּ,

כַּלְּבוּשׁ יַחֲלִיפוּ, אֵיתָנִים דַּרְכָּם הֶחֱלִיפוּ,

כִּי עַל מִקְוֵה הַמַּיִם בִּלְשׁוֹן יַם מִצְרָיִם,

הִרְעִים אֵל הַכָּבוֹד, קוֹל נָתַן עַל הַמָּיִם,

וּמֵי הַיָּם פִּתְאוֹם כַּעֲרֵמוֹת הִתְנַשָּׂאוּ,

עַל מַסְלוּל חָדָשׁ תּוֹךְ הַיָּם כַּחוֹמוֹת קָמוּ,

עַל מַסְלוּל לֹא עָלָה מֵעוֹלָם עַל לֵב גָּבֶר,

כִּי הִפְלִיא אֵל עַתָּה כַּאֲשֶׁר הִפְלִיא מִקֶּדֶם,

לַהֲפוֹךְ תֹּהוּ אֲשֶׁר בָּרָא אֶל מִדְרַךְ רֶגֶל.

כֵּן דֶּרֶךְ לִפְדוּיָו יָדָיו בַּיָם יָצָרוּ,

דִּבֵּר: וּתְהוֹמוֹת תַּחַת עָלוּ מָעְלָה מַעְלָה.

כַּאֲשֶׁר עָלוּ אוֹצְרוֹת מֵי הַיָּם וַיִּגְבָּהוּ,

סִתְּמוּ אֶת הַיָּם כִּי אֶת הַמִּקְוֶה גָּזָלוּ,

אָז נָס הַיָּם מִלִּפְנֵי אָדוֹן חוּלִי אָרֶץ,

כִּי הֲפָכוֹ עוֹשֵׂהוּ, יַבָּשָׁה שָׂמָהוּ.

בַּעֲלוֹת הַתְּהוֹמוֹת מֵי הַיָּם פִּתְאוֹם נִבְקָעוּ,

כִּי לֹא מָצְאוּ הַמַּיִם עוֹד לָהֶם מָנוֹחַ,

כִּי בִמְנוּסַת יָם מֵימָיו נֹאדוֹתָם אָבָדוּ,

אִם מִגַּעֲרַת אֵל נָסוּ מִפַּחְדּוֹ שָׁטָפוּ.

גַּם לִשְׂפַת הַיָּם לִמְקוֹם הַמַּחֲנֶה יָרוּצוּ,

וּבִמְרוּצָתָם מִי לֹא יִמְצָא קִצּוֹ בַשֶּׁטֶף.

לוּלֵי בּוֹרֵא נִכְבַּדֵּי מַיִם הִשְׁבִּיחָמוֹ,

וּבְרֶגַע הִבָּקְעָם, חֶצְיָם לִפְאַת צָפוֹנָה,

חֶצְיָם לִפְאַת תֵּימָנָה, מִגַּעֲרַת אֵל קָמוּ,

מִימִין וּמִשְּׂמֹאל הַדֶּרֶךְ כַּנֵּד נִצָּבוּ,

כִּי כַמַעֲשֶׂה שֵׁשֶׁת הַיָּמִים מִנִּי קֶדֶּם,

אֲשֶׁר בָּם הִפְלִיא אֵל לַעֲשׂוֹת אֶת כָּל מַעֲשֵׂהוּ,

רוּחַ אֱלֹהִים הַמְרַחֶפֶת עַל הַמָּיִם,

מֵעֱזוּזוֹ נֶחֱצוּ גַּם נִקְווּ וַתְּהִי יַבֶּשֶׁת,

כִּי כּוֹחוֹ הַגָּדוֹל וּזְרוֹעוֹ אָז פָּעָלוּ,

וּבְשֵׁם רוּחַ אֱלֹהִים הֵמָּה יִקָּרֵאוּ,

כֵּן הוֹלִיךְ אֶת הַיָּם וּמֵימָיו כָּל הַלָּיְלָה,

הַיָּם שָׂם לֶחָרָבָה וּמֵימָיו נִבְקָעוּ,

מִמְּקוֹמָם מָשׁוּ וּמֵאֵיתָנָם נֶעֱדָרוּ,

עַד בֹּקֶר הַיָּם אֵינֶנּוּ כַּבִּקְעָה מַרְאֵהוּ,

וּמֵימָיו נֶחֱצוּ נִקְווּ מֵעַל לוֹ עָמָדוּ,

עַד שׁוּב הַיָּם אֶל אֵיתָנוֹ אֵלָיו לֹא שָׁבוּ,

יַעַן רוּחַ אֱלֹהִים בִּשְׁתֵּיהֶן מָשָׁלָה,

רוּחַ עַזָּה רוּחַ קָדִים אֶת כֹּל תַּכְרִיעַ,

הִיא הַמַּחֲרֶבֶת יָם הִיא אֶת מֵימָיו הוֹבִישָׁה.

כַּאֲשֶׁר בִּדְּבַר יְיָ הַתְּהוֹמוֹת עָלוּ

חָלַק לָהֶם בַּקַּו לֵאמֹר: “עַד פֹּה תָבוֹאוּ”!

לֹא גָּבְרוּ לַעֲלוֹת עוֹד אַךְ עַד לֵב יָם הִגִּיעוּ,

בִּמְקוֹם אֳנִיּוֹת יֵלֵכוּן שָׁמָּה קָפָאוּ,

וַתְּהִי אֶל תַּחְתִּית יָם וּשְׂפָתוֹ קוֹמָה אַחַת.

וּכְיוֹרֵד בַּבִּקְעָה מִשְּׂפָתוֹ בּוֹ יֵרֵדוּ,

כִּי גַּם נִבְכֵי יָם תַּבְנִית חֲדָשָׁה קִבֵּלוּ,

עֲמָקִים וּשְׁפָיִים יָצַר אֵל בָּם מִקֶּדֶם,

לֹא הָיוּ בוֹ עוֹד, כַּגַּן לַמִּישׁוֹר חָלָקוּ,

גַּם הַתְּהוֹמוֹת מִימֵי עוֹלָם אוֹצְרוֹת הַמָּיִם,

וַעֲפָרָם טִיט וָרֶפֶשׁ יִטְבְּעוּ בָּם רַגְלָיִם,

כֶּחָרָבָה בִּימֵי אָבִיב פִּתְאוֹם חָרָבוּ,

לֹא כִישִׁימוֹן וִילֵיל תֹּהוּ מַרְאֵיהֶן עַתָּה,

כִּי אֵל צִוָּה מֵאָז: “תַּדְשֵׁא הָאָרֶץ דֶּשֶׁא”,

הִצְמִיחַ עַל חָרְבוֹתֵיהֶן כָּל יֶרֶק עֵשֶׂב.

וּכְדוֹרֵךְ עַל מִרְעֶה טוֹב עֲלֵיהֶן דָּרָכוּ,

אֹרַח סָלוּל מִמִּכְשׁוֹל, מִכָּל אָבְנֵי נֶגֶף,

עַל כֵּן כָּל בֵּית יִשְׂרָאֵל קוֹמְמִיּוּת הָלָכוּ,

תּוֹךְ הַיָּם עַל יַבָּשָׁה תַאֲוָה לָעֵינַיִם,

מִימִינָם וּשְׂמֹאלָם חוֹמָה לָהֶם הַמָּיִם,

אֲשֶׁר מֵרוּחַ אַף אֵל כַּאֲבָנִים נֶעֱרָמוּ.


וּבְיַעַן12 מֵרְשָׁעִים מָנַע אוֹר סִבְבָם אֹפֶל,

דָּרְכוּ גַם הֵם בַּיָם, אֵיפֹה הוֹלְכִים לֹא יָדָעוּ,

חוֹמוֹת הַמַּיִם מִזֶּה וּמִזֶּה לֹא רָאוּ,

תַּחַת רַגְלָם עַפְרוֹת גַּנִּים עֵשֶׂב צוֹמֵחַ,

הַזֹאת יַחְשׁוֹב לִבָּם כִּי בִּתְהוֹמוֹת נִלְכָּדוּ,

בַּאֲבַדּוֹן לֹא יֵצְאוּ מִשָּׁם רַגְלָם הָטְבָּעָה?

מֵחֲרוֹת אַפָּם כִּי עָז מֵחֲמַת אֵל תִּסְעָדֵמוֹ,

לֹא זָכְרוּ אַחֲרִיתָם מָה רָע, בֶּטַח הָלָכוּ,

כָּל סוּס פַּרְעֹה וּפָרָשָׁיו חַיִל וָרֶכֶב

הָרוֹדְפִים אַחֲרֵי יִשְׂרָאֵל, תּוֹךְ הַיָּם בָּאוּ,

כָּל הַלַּיְלָה עַד בֹּקֶר מֵרְדוֹף לֹא חָדֵלוּ,

וּלְעֵת אַשְׁמוֹרֶת הַבֹּקֶר קַרְנָם נִגְדָּעָה.

כִּי אַז הִשְׁקִיף יְיָ אֶל מַחֲנֶה מִצְרַיִם,

וּבְעַמּוּד אֵשׁ וּבְעַמּוּד עָנָן הִבְהִילָמוֹ,

וַיָּהָם אֶת מַחֲנֵיהֶם, מִי הוּא הוֹדִיעָמוֹ,

רָאוּ כִּי אֵל קִבְּצָם יַחַד כֶּעָמִיר גּוֹרְנָה,

רָאוּ יָמִין וָחַתּוּ וּשְׂמֹאל וְיֶחֱרָדוּ,

תַּחַת רַגְלָם רָגְזוּ הַתְּהוֹמוֹת גַּם רָעָשׁוּ,

“כִּי שָּׁוְא מִבְטַח בּוֹגֵד, אֶרֶץ לוֹ מִתְקוֹמֵמָה”.

זָרְמוּ מַיִם עָבוֹת וּשְׁחָקִים קוֹל נָתָנוּ,

בָּרָד הִמְטִירוּ אַף חֲצָצִים הִתְהַלָּכוּ,

קוֹל רַעַם בַּגַּלְגַּל וּבְרָקִים רָב הֵאִירוּ,

נָעוּ חוֹמוֹת הַמַּיִם מִפְּנֵי אֵל יָחִילוּ,

וַעֲפַר מִדְרַךְ רַגְלָם נִהְיָה זֶפֶת בּוֹעֵרָה,

אוֹפַנֵּי מַרְכְּבוֹתֵיהֶם מֵאֵשׁ לוֹהֵט סָרוּ,

אָז סוּסִים אֶת הָרֶכֶב בִּכְבֵדוּת נִהֵגוּ,

שָׁם פָּחֲדוּ פָּחַד, שָׁם אִוַּלְתָּם עַל לֵב שָׂמוּ,

זָכְרוּ כִּי עַל דַּכְּאֵי רוּחַ יָדָם הִכְבִּידוּ,

עַל כֵּן גַּם יַד יְיָ עֲלֵיהֶם כָּבֵדָה,

וַיֵּדְעוּ כָּל פּוֹעֲלֵי אָוֶן אֶת מִי נִאֵצוּ.

עוֹד לָשׁוּב בַּדֶּרֶךְ אֲשֶׁר בָּאוּ הִתְאַמֵּרוּ,

כִּי אָמַר אוֹיֵב "מִפְּנֵי יִשְׂרָאֵל אָנוּסָה,

“כִּי יְיָ נִלְחָם לָהֶם עַתָּה יָדַעְנוּ”.

אַךְ הֵם לֹא יָדְעוּ כִּי בִשְׁחִיתוֹתָם נִלְכָּדוּ,

וּמִפַּח יָקוּשׁ זֶה רַגְלֵיהֶם לֹא יָשִׁיבוּ,

כִּי בָא מוֹעֵד לַעֲשׂוֹת מִשְׁפָּט, עָלָה הַשָּׁחַר.

כִּסְדוֹם וַעֲמוֹרָה גַּם הֵם מוֹעֵד זֶה נִשְׁפָּטוּ,

וּכְחַטָּאתָם הַגְּדוֹלָה כֵּן חַטַּאת מִצְרַיִם,

בִּסְדוֹם נָסַבּוּ הָעָם עַל בֵּית צַדִּיק לָיְלָה,

לִמְשׁוֹל בָּאֲנָשִׁים חָפְשִׁים לֹא בָּהֶם יִמְשֹׁלוּ,

כַּאֲשֶׁר הֻכּוּ בַּסַּנְוֵרִים מִיַּד נוֹרָאָה,

נָעוּ בִּרְחוֹבוֹת הָעִיר אֶל בֵּיתָם לֹא שָׁבוּ.

עַל קִנְיָנֵיהֶם מִלְּפָנִים לֹא עוֹד מָשָׁלוּ,

כִּי טֶרֶם צֵאת שֶׁמֶשׁ מִמִּטְּרַת אֵשׁ אֻכָּלוּ.

כֵּן רִשְׁעֵי מִצְרַיִם הָרוֹדְפִים כָּל הַלַּיְלָה,

לִמְשׁוֹל בַּאֲנָשִׁים חָפְשִׁים לֹא בָהֶם יִמְשֹׁלוּ,

נָעוּ בִּתְהוֹמוֹת הַיָּם אֶל בֵּיתָם לֹא שָׁבוּ,

עַל חֲמוּדוֹתָם מִלְּפָנִים לֹא עוֹד מָשָׁלוּ,

כִּי טֶרֶם צֵאת שֶׁמֶשׁ בִּמְצוּלוֹת יָם יָרָדָוּ.

כִּי כַּאֲשֶׁר13 דִּמוּ הָרוֹדְפִים לַשּׁוּב מִצְרַיְמָה,

וַיֹאמֶר יְיָ אֶל משֶׁה וַיְצַוֵּהוּ,

"קוּם וּנְטֵה יָדְךָ עַל הַיָּם, כִּי אֲשִׁיבֶנּוּ

"לְהיוֹת כְּבָרִאשׁוֹנָה לִמְקוֹם מִקְוֵה מָיִם,

“גַּם מֵימָיו לִמְקוֹמָם הָרִאשׁוֹן אָז יָשׁוּבוּ”.

כַּאֲשֶׁר נָטָה משֶׁה עַל הַיָּם אֶת יָדֵהוּ,

הַמַּעֲבָר אֲשֶׁר נֶהְפַּך אֶל תֹּהוּ לֹא דָרֶךְ,

חֲדַל בָּרֶגַע מִהְיוֹת תַּחְתָּם, כִּי אֵינֶנּוּ,

כִּי הַתְּהוֹמוֹת הַגְּבוֹהוֹת צָלְלוּ מַטָּה מָטָּה,

יַעַן הַיָּם שָׁב אֶל אֵיתָנוֹ לִפְנוֹת בֹּקֶר,

לִהְיוֹת כַּנֹאד אֶל מֵימָיו כִּהְיוֹתוֹ מִקֶּדֶם,

וּבְרִדְתָּם פִּתְאוֹם רִגְבֵיהֶן פּוֹר הִתְפּוֹרֵרוּ,

וּמִצְרַיִם נָסִים לִקְרַאת רִגְבֵי הַדֶּרֶךְ,

נִבְהָלִים וּדְחוּפִים לָבוֹא אֶל הַיַּבָּשֶׁת,

אָז סוּס עִם רוֹכְבוֹ בֵּין רִגְבֵי עָפָר נִבְלָעוּ,

כִּצְלוֹל עוֹפֶרֶת כֵּן עִם הָרְגָבִים צָלָלוּ,

מִתְנַשְּׂאִים עַל אֵל אֶל תַּחְתִּיּוֹת יִנָּעֵרוּ,

כִּי נִעֵר יְיָ תּוֹךְ הַיָּם אֶת מִצְרַיִם,

וַיַּשְׁלִיכֵם בִּמְצוֹלוֹת לֹא עָמְדָה בָּם רֶגֶל,

גַּם חוֹמוֹת הַמַּיִם מִשְּׁנֵי צִדֵּי הַדָּרֶךְ,

פִּתְאֹם לֹא יָכְלוּ קוּם עוֹד עַרְעָר הִתְעַרְעָרוּ,

בִּפְקוּדַת אֵל אֶל אוֹצְרוֹתָם לִמְקוֹמָם שָׁבוּ,

וַיְכַסּוּ אֶת הַפָּרָשִׁים גַּם אֶת הָרֶכֶב,

גַּם כָּל חֵיל פַּרְעֹה רַגְלֵיהֶם בַּיָם הִצִּיגוּ,

לִרְדּוֹף אַחֲרֵי יִשְׂרָאֵל וּלְחַלֵּל כָּל קֹדֶשׁ,

עַל חַלְלֵיהֶם בִּתְהוֹם רַבָּה כִּסּוּ הַמָּיִם,

לֹא נִשְׁאַר מֵהֶם עַד אֶחָד, כֻּלָּם גָּוָעוּ,

וּמְעַטִּים אֲשֶׁר לֹא הִצִּיגוּ בַּיָם רֶגֶל,

רַכֵּי לֵב וּפְלֵטִים לָצֵאת לַקְּרָב מָאָסוּ,

וּמֵעַצְלָה אֶחֱרוּ פַעֲמֵיהֶם מִבּוֹא יָמָּה,

אֵלֶּה הַנֶּחֱשָׁלִים הָאַחֲרוֹנִים נִמְלָטוּ.

עוֹדָם עוֹמְדִים בִּשְׂפַת הַיָּם כַּעֲלוֹת הַשָּׁחַר,

הֵמָּה רָאוּ אֶת מַעֲשֵׂי יְיָ מָה רַבּוּ,

הֵם שָׁבוּ אֶל אַרְצָם שָׁם נִקְמַת אֵל בִּשֵּׂרוּ,

עֲלִילוֹתָיו סִפֵּרוּ וּכְבוֹדוֹ הִגִּידוּ,

אֶל קוֹל הַשְּׁמוּעָה בָּאָה רָפוּ כָּל יָדָיִם,

כִּי אֵיךְ יַחְשֹׁבוּ כִּי הַמְּבַשְּׂרִים שָׁוְא דִּבֵּרוּ,

אִם מִיְדוּעֵיהֶם הֻגַּד לָמוֹ מֵאָח מֵרֵעַ.

אָז יָדְעוּ כִּי יְיָ אֵל נֶאְדָּר בַּקֹּדֶשׁ,

כִּי קָדוֹשׁ מֹשֶׁה לֵאלֹהָיו, אִישׁ מַגִּיד יוֹשֶׁר,

כִּי עִם קָדוֹשׁ יִשְׂרָאֵל, בּוֹ בָּחַר עוֹשֵׂהוּ,

בּוֹשׁוּ כִּי הֶעֱבִידוּם, כִּי עֲלֵיהֶם הִתְגָּאוּ,

לָצֵאת עוֹד לִקְרָאתָם כָּאוֹיְבִים לֹא יָסָפוּ,

עֵת לֶכְתָּם בַּמִּדְבָּר עִמָּהֶם לֹא נִלְחָמוּ,

עֵת נָחֲלוּ אֶרֶץ, אֶת יוֹשְׁבֶיהָ לֹא עָזָרוּ

וּבְשִׁבְתָּם עַל אַדְמָתָם עִמָּם לֹא הֵרֵעוּ,

רַב מֵחֲמֵשׁ מֵאוֹת שָׁנָה רָעָה לֹא חָרָשׁוּ,

לֹא נִרְאָה חֵיל מִצְרַיִם עַל אַדְמַת הַקֹּדֶשׁ,

עַד חַטַּאת יִשְׂרָאֵל מַלְכוּתָם חִלֵּקָה,

אָז עָלָה עֲלֵיהֶם שִׁישַׁק מֶלֶךְ מִצְרָיִם.


עֵת מִתַּחַת מִצְרַיִם הַמְּסִלָּה נָמוֹטָה,

מַחֲצִיתָהּ אֶל הָעֵבֶר הַשֵּׁנִי נִשְׁאָרָה,

אֲשֶׁר עָלֶיהָ רַגְלֵי הַפְּדוּיִם עָמָדוּ,

עוֹד הֵמָּה הוֹלְכִים בֶּטַח בַּיָם בַּיַּבֶּשֶׁת,

וּמִשְׁנֵי צִדֵּיהֶם חוֹמָה לָהֶם הַמָּיִם,

אֶת קֵץ אוֹיְבֵיהֶם רָאוּ, אַחֲרִית פּוֹעֲלֵי אָוֶן,

כִּי אֶת פִּגְרֵי מִצְרַיִם מֵי הַיָּם גָּרָשׁוּ,

לִשְׁאֵרִית הַמְּסִלָּה שָׂפָה לַיָּם עוֹמֶדֶת,

וַיִּרְאוּ כִּי צָרֵיהֶם פַּסּוּ, דּוֹלְקִים תַּמוּ,

כִּי הוֹשִׁיעָם אֵל יוֹם זֶה מִיַּד מִצְרָיִם,

הֵמָּה רָאוּ כֵּן תָּמָהוּ, גַּם כַּאֲשֶׁר רָאוּ

אֶת הַיָּד הַגְּדוֹלָה הִיא הַיָּד הַנִּלְחֶמֶת,

יוֹם יִקּוֹם אֵל מֵאוֹיְבָיו מַה גָּדְלוּ מַעֲשֶׂיהָּ.

רוּחַ יִרְאַת יְיָ אֶת נַפְשָׁם לָבָשָׁה,

כִּי אֶת יְיָ אֵל נוֹרָא עַל כֹּל יִירָאוּ,

רוּחַ אֱמוּנָה אַף הִיא אֶת נַפְשָׁם עָטָתָה,

מִתַּהֲפּוּכוֹת טָהֵרוּ, בַּייָ הֶאֱמִינוּ,

כִּי גָּדוֹל הוּא עַל כָּל אֵל, דּוֹבֵר וִיקִימֶנּוּ,

וּבְמשֶׁה עַבְדּוֹ כִּי כָל אִמְרֵי פִיו נֶאֱמָנוּ,

הֵיטִיבוּ וּבְעֵינֵי יְיָ חֵן מָצָאוּ,

וַיְבָרְכֵם מִמְּעוֹן קָדְשׁוֹ, מִמְּרוֹמוֹ הוֹפִיעַ

עֲלֵיהֶם רוּחַ חָכְמָה, שֵׂכֶל טוֹב מָצָאוּ,

אָז נִפְתַּח פִּי אִלְמִים, אִמְרֵי שֶׁפֶר דִּבֵּרוּ,

וּלְשׁוֹנוֹת הַיְּלָדִים דַּבֵּר צַחוֹת הֵחֵלּוּ,

וּבְהַדְרַת קֹדֶשׁ שִׁירָה חֲדָשָׁה לוֹ שָׁרוּ,

הִיא הַכְּתוּבָה עַל סֵפֶר תּוֹרַת אֱלֹהֵינוּ,

נִפְלְאוֹת אֵל עַל יָם סוּף סִפַּרְתִּי הִיא מַגֶּדֶת.

אֶפֶס כִּי רַק מִקְצַת מִפְּנִינֶיהָ דָּלִיתִי,

כִּי עָמְקוּ כִּי גָאוּ אִמְרוֹתֶיהָ מִמֶּנִי,

וּמְעַט מֵהַרְבֵּה מָצְאָה יָדִי אַצִּיג הֵנָּה.

עַתָּה אַחַי! צוּף אִמְרֵי נֹעַם מִפִּי קָחוּ,

כִּי אָחֵל בָּאֵר דִּבְרֵי שִׁירַת אֲבוֹתֵינוּ,

הֵם שֵׂכֶל טוֹב הִשְׂכִּילוּ וּנְגִידִים דִּבֵּרוּ,

וַיָּשִׁירוּ אֶל צוּר יִשְׂרָאֵל וַיֹּאמְרוּ:


אָשִׁירָה לַיְיָ14 לֵאלֹהֵי קֶדֶם,

"כִּי גָאֹה גָּאָה לִצְבִי וּלְתִפְאָרֶת,

"כִּי הוּא יַעֲדֶה גָאוֹן, הוּא יַעֲדֶה גֹּבַהּ,

"יוֹם יִגְבָּהּ בַּמִשְׁפָּט לִשְׁבּוֹר עַוְלָתָה,

"לַהֲדוֹךְ גֵּאִים תַּחְתָּם וּלְהַכְרִית רֶשַׁע,

"מָה רָם הוֹדוֹ, מָה מָּךְ עֶרְכָּם, יַרְאֵמוֹ,

"כִּי אִם רָם יְיָ, לִרְאוֹת מַשְׁפִּילִי,

"לָחֹן עַנְוֵי אָרֶץ דַּלִים יוֹשִׁיעַ,

"וַעֲדַת גֵּאִים אִם לַכִּסֵּא גָּבָהוּ,

"תַּחְתִּית יָם יוֹרִידֵם, שָׁם יִמְצְאוּ קָבֶר,

"כִּי לֹא אָדָם הֵם, אֶפֶס לוֹ נֶחְשָׁבוּ,

"יַעַן בָּזוּ צֶדֶק, דַּעַת מָאָסוּ,

"אִם אֵל גֵּאוּת לָבֵשׁ רַבָּה תִּפְאֶרֶת,

"כִּי מִפָּנָיו הָאֵיתָנִים יִמּוֹטוּ,

"מָה נּוֹרָאִים מַעֲשָׂיו וּמַה נִּפְלָאוּ,

"רוֹכְבִים עַל סוּסִים עַל אֶרֶץ רָאִינוּ,

"עַל לֵב יָם מִי רָאָם, שָׁם צִים יֵלֵכוּ,

"כַּאֲשֶׁר אֵל בִּגְאוֹנוֹ יוֹם זֶה הֶרְאָנוּ,

"בַּחַחִים מָשַׁךְ אֶל יָם חֵיל מִצְרָיִם,

"וּבְלֶכְתָּם בּוֹ מַצַּב רַגְלָם אֵינֶנּוּ,

"אָז פִּתְאוֹם כָּאֶבֶן מֵהַר נוֹפֶלֶת,

"רָמָה סוּס עִם רֹכְבוֹ בַּיָם מִתָּחַת.


לֹא גִבּוֹרֵי אֶרֶץ כָּכָה יָעֹזוּ,

וּזְמִירוֹת כָּאֵלֶּה לֹא יָשִׁיר גָּבֶר,

עֹז אֱלֹהִים זִמְרַת יָהּ כֵּן פָּעָלוּ.

"הָבוּ לַיְיָ כָּבוֹד, עוֹז לוֹ הָבוּ,

"עֵת בָּעֲרָה חֲמָתוֹ בִּימִינוֹ חֶסֶד,

"עֵת שִׁנֵּן לִמְשַׂנְּאָיו חֶרֶב נוֹקֶמֶת.

"נָתִיב טוֹב אֶל עַמּוֹ יָדוֹ פִּלֵּסָה,

"וַיְהִי לִי לִישׁוּעָה, לִישׁוּעַת נֶצַח,

"כִּי עַתָּה אָפֵס צַר מִמִּי אֶפְחָדָה,

"זֶה אֵלִי אֵין בִּלְתּוֹ עַל כֵּן אַנְּוֵהוּ,

הוֹדוֹ אֵהָב, בִּנְוָתִי בּוֹ אֶדְבָּקָה.

"זֶה אֱלֹהֵי אָבִי צוּרִי מִקֶּדֶם.

"אֶזְכֹּר מַעַלְלֵי אֵלִי וַאֲרוֹמְמֶנְהוּ.


"גִבּוֹרֵי מִצְרַיִם עַתָּה תֵבוֹשׁוּ,

לִלְחוֹם בָּנוּ, יֶדְכֶם שָׁוְא, סוּסְכֶם שָׁקֶר,

"יְיָ אִישׁ מִלְחָמָה, כִּי מַחֲנֵהוּ

"וּכְלֵי מִלְחַמְתּוֹ שָׁמַיִם וָאָרֶץ,

"יִקְרָא לָהֶם לִצְבוֹא צָבָא יָבוֹאוּ,

"כִּי מַעֲשֵׂה יָדָיו הֵם, הוּא אֵל עָשָׂמוֹ,

"כִּי הוּא אֵל עָשָׂם שֵׁם קָדְשׁוֹ יוֹרֵנוּ,

"יְיָ הוּא שְׁמוֹ בָּאֵלִים אֵין כָּמוֹהוּ.


"רֶכֶב בָּחוּר נִהֵג מֶלֶךְ מִצְרָיִם,

"עִם כָּל רִכְבֵי אַרְצוֹ מַה לּוֹ הוֹעִילוּ,

"וּמִבְחַר שָׁלִישָׁיו עִם כָּל הַחַיִל,

"לַשָּׁוְא לָבְשׁוּ שִׂרְיוֹן, לָרִיק הַחֶרֶב.

"אֲשֶׁר לוֹ הַיַּמִּים, לוֹ הַיַּבֶּשֶׁת,

"יָרָם בִּמְצוּלוֹת יָם וּבוֹ טֻבָּעוּ.


"שִׁמְעוּ כָּל הָעַמִּים אֵיךְ נִהְיָתָה,

"אֵיךְ אֶל תַּחְתִּיּוֹת יָם צָרִים נָפָלוּ,

"אֵל גָּעַר בַּתְּהוֹמוֹת דִּבֵּר אֵלֵימוֹ,

"שׁוּבוּ אֶל הָעֲמָקִים לָכֶם יִעַדְתִּי,

"שָׁמְעוּ נָעוּ נָדוּ לִמְקוֹמָם שָׁבוּ,

"אָז תַּחַת רַגְלֵי צַר פּוֹר הִתְפּוֹרֵרוּ,

"וּבְרֶדֶת רַגְלֵיהֶן חַיִל וָרָכֶב,

"יָרְדוּ עִמָּהֶם בִּמְצוּלוֹת כָּאָבֶן.


"אֶל נֶאְדָּרִי בַּכֹּחַ מַה גָּדַלְתָּ,

"וּבְכֹחֲךָ הַגָּדוֹל אֶת כֹּל צִוִּיתָ,

"מִבְּלִימָה עָשִׂיתָ כֹּל בִּימִינֶךָ,

"הִיא טִפְּחָה שָׁמַיִם, הִיא יָסְדָה אָרֶץ,

"גַּם הַיּוֹם הֶרְאָתָה גּוֹדֶל כֹּחֶךָ,

"כַּאֲשֶׁר לִגְאוּלֶיךָ מַסְלוּל כּוֹנָנָה,

"עַל לֵב יָם בִּמְקוֹם אֳנִיּוֹת יֵלֵכוּ,

"וּבְשִׁמְחַת לֵב עַמְּךָ עָלָיו יַעֲבוֹרוּ,

"וִימִינְךָ יְיָ עַל צַר רוֹמֵמָה,

"בַּהֲכִינָהּ לוֹ מַסְלוּל זֶה לִמְקוֹם קָבֶר,

"רַעֲצָה בּוֹ רֹאשׁ אוֹיֵב בֵּין רִגְבֵי אָרֶץ,

"מַסְלוּל זֶה לַחַיִּים אַף הוּא לַמָּוֶת.


"וּבִגְאוֹנְךָ כִּי רַב רַבּוֹת עָשִׂיתָ,

"אֶת מַחֲנֵה אוֹיְבֵינוּ פִּתְאוֹם הַמּוֹתָ,

"זָרְמוּ עָבוֹת וּשְׁחָקִים קוֹל נָתָנוּ,

"קוֹל רַעַם בַּגַּלְגַּל וּבְרָקִים אוֹרוּ,

"וּמָטָר בָּרָד וַחֲצָצִים הִתְהַלָּכוּ,

"בִּמְהוּמוֹת אֵלֶּה הָרַסְתָּ קָמֶיךָ,

"כִּי מָה הֵם רִבְבוֹתָם אִם עַל אוֹיְבֶיךָ,

"חֲרוֹנְךָ שִׁלַּחְתָּ כַּקַּשׁ יֹאכְלֵמוֹ.


"נִפְלְאוֹתֶיךָ יוֹם זֶה תֵּבֵל תַּרְגֵּזְנָה,

"כִּי עַל הָאֵיתָנִים לַעֲשׂוֹת הִגְדַּלְתָּ,

"לַעֲשׂוֹת מִשְׁפָּטֶיךָ יַחְדָּו הִשְׁלִימוּ,

"רוּחַ קָדִים עַזָּה רוּחַ אַפֶּיךָ,

"הִיא רוּחַ יְיָ עֻזֶּךָ לֹבֶשֶׁת,

"רָאָה הַיָּם וַיָּנוֹס, רָאוּ מָיִם

"יָחִילוּ, נוֹזְלֵיהֶם כַּנֵּד נִצָּבוּ,

"מִצִּדֵי הַמַּסְלוּל עַל יָם רָקַעְתָּ,

"הִנּוֹ רָצוּף כֻּלּוֹ מִתְּהוֹמוֹת תָּחַת,

"בִּדְבָרְךָ אֵל הָיְתָה זֹּאת, כִּי צִוִּיתָ

"עַל תַּחְתִּיּוֹת יָם: אֶל עָל הִנָּשֵׂאוּ!

"כַּאֲשֶׁר נִשָּׂאוּ עַד לֶב יָם קָפָאוּ,

"שָׁם מִקְוֵה הַמַּיִם שָׂמוּ לַדָּרֶךְ,

"וּכְהוֹלֵךְ בַּבִּקְעָה עָלָיו הָלָכוּ,

"עֲמָקָיו גָּבָהוּ וּשְׁפָיָיו שָׁפָלוּ,

"וַיְהִי כֶּחָרָבָה אֵין טִיט בּוֹ, אֵין רֶפֶשׁ,

"וּכְיַבָּשָׁה, כִּי צָמַח בּוֹ כָּל עֵשֶׂב.


"אֵל! לוּ יָדְעוּ צָרֶיךָ מַה פָּעַלְתָּ,

"רַב מִפַּחַד עוֹלֶה עַל הַרְרֵי טֶרֶף.

"הַצֵּג רַגְלָם עַל דֶּרֶךְ זֶה יִפְחָדוּ,

"לָכֵן מִסָּבִיב לָהֶם שַׂמְּתָּ אֹפֶל,

"שׁוּר נִפְלְאוֹתֶיךָ בַּיָּם לֹא יָכוֹלוּ,

"כִּי עוֹבְרִים בֵּין חוֹמוֹת מַיִם לֹא רָאוּ.

"כִּי דַּרְכָּם דֶּרֶך מָוֶּת אֵיךְ יַחְשֹׁבוּ,

"אִם נֶגֶד עֵינֵיהֶם מַסְלוּל חִדַּשְׁתָּ,

"כֻּלּוֹ טוֹב, עַל עַפְרוֹת גַּנִּים יִדְרֹכוּ,

"עַל כֵּן אָמַר אוֹיֵב: עַם זֶה אֶרְדֹּפָה,

"אַשִּׂיג רַכֵּי לֵב אֵלֶּה, אַבְהִילֵמוֹ,

"וּשְׁלָלָם בֵּין הַחַיִל אֲחַלֵּקָה,

"כָּל מִשְׁאֲלוֹת נַפְשִׁי פֹּה אֲמַלֵּאָה,

"הוֹן אָבֹז בַּל כֹּפֶר נַפְשָׁם אֶתֵּנָה,

"אַךְ חַרְבִּי אָרִיק יָדִי תוֹרִישֵׁמוֹ.


"כָּל הַלַּיְלָה עַד אַשְׁמֹרֶת הַבֹּקֶר,

"דִּבְּרוּ כֵן הָעָם הַהוֹלְכִים בַּחשֶׁךְ,

"וּכְבוֹא עֵת מִשְׁפָּטָם בַּעֲלוֹת הַשָּׁחַר,

"הֱסִירוֹתָ יָד וּבְרוּחֲךָ נָשַׁפְתָּ,

"וְיָשָׁב הַיָּם לִהְיוֹת מִקְוֵה מָיִם,

"כִּי תַּחְתִּיּוֹת יָם לִמְּקוֹמָם שָׁבוּ,

וּמֵי הַיָּם נִצְבוּ כַּנֵּד, נָזָלוּ,

"אֶל אוֹצְרוֹתֵיהֶם מִלְּפָנִים שָׁטָפוּ,

"אַז רֶכֶב, פָּרָשִׁים, חַיִל וָמֶלֶךְ,

"בַּמַּיִם אַדִּירִים הַיָּם כִּסָּמוֹ;

"צָרֶיךָ, הֶעֱלוּ שִׂיאָם עַד שָׁמַיִם,

"צָלְלוּ פִּתְאוֹם בִּמְצוּלוֹת כָּעֹפֶרֶת.


"יְיָ! בָּאֵלִים מִי אֵל כָּמוֹךָ,

וּמִי יִדְמֶה לָּךְ בֵּין גִּבּוֹרֵי כֹחַ!

"כֻּלָּם עֲבָדֶיךָ, מַעֲשֶׂה יָדֶיךָ,

"גַּם כֹּחָם גַּם פָּעֳלָם אַתָּה הִגְבַּלְתָּ,

"וּגְבוּלוֹת שַׂמְתָּ לָהֶם, לֹא יַעֲבוֹרוּ,

"לֹא כֵן אַתָּה, לִגְדֻלָּתְךָ אֵין חֵקֶר,

"הוֹדְךָ מֵעַל אֶרֶץ, מֵעַל שָׁמָיִם,

"עַל כֵּן רַק אַתָּה אֶת הוֹדְךָ יָדָעְתָּ,

"הַמְּרוֹמָם מִכָּל לֵב, נִפְלָא מִדָּעַת,

"כָּל אַנְשֵׁי שֵׁם מִמְּךָ לֹא יִפָּלֵאוּ,

"כִּי מִי אֵל כָּמוֹךָ נֶאְדָּר בַּקֹּדֶשׁ,

"נוֹרָאוֹת, לֹא נוֹדָעוּ, לֹא נִבְרָאוּ,

"בִּדְבָרְךָ יִבָּרֵאוּ, אוֹר יֵצֵאוּ,

"עַל כֵּן חַכְמֵי לֵב יָד עַל פֶּה יָשִׂימוּ,

כִּי מִי יוּכַל הַלְלָךְ, יוּכַל רוֹמְמֶךָ,

"עַל כָּל שֶׁבַח וּתְהִלָּה נוֹרָא אַתָּה,

"אֵל עוֹשֶׂה חֲדָשׁוֹת, אֵל עוֹשֶׂה פֶּלֶא.


"מַסְלוּל זֶה, מִתְּהוֹמוֹת יָם רָקַעְתָּ,

"לֹא מָט כָּל עֵת בִּימִינְךָ תִּסְעָדֶנוּ

"עֵת גָּעַרְתָּ בּוֹ וִימִינְךָ נָטִיתָ,

"נִשְׁבַּר פִּתְאוֹם לִרְגָבִים כִּכְלִי חֶרֶשׂ,

"וּבְרִדְתָּם עִם חֵיל צַר אֲשֶׁר עָלֵימוֹ,

סוּס גַּם רוֹכֵב בֵּין הַשְּׁבָרִים נִבְלָעוּ,

"אוֹמְרִים “לָנוּ הָאָרֶץ” בָּהּ קֻבָּרוּ.


"וּלְעֵת הַמַּסְלוּל מִימִינְךָ הִשְׁלַכְתָּ,

"לֹא כֻלּוֹ הִשְׁלַכְתָּ, חֶצְיוֹ הִשְׁאַרְתָּ,

"לִהְיוֹת לִמְסִלַּת עָם אֲשֶׁר אָהַבְתָּ,

"וּמִצִּדֶּיהָ עוֹד חוֹמוֹת הַמָּיִם,

"כִּי שָׁם יִרְאוּ אֶת שִׁלּוּמַת אוֹיְבֶיךָ,

"וּבְשִׁירָה חֲדָשָׁה יוֹדוּ לִשְׁמֶךָ,

"וּבְחַסְדְּךָ נָחִיתָ עַם זוּ גָּאָלְתָּ,

"בִּשְׁאֵרִית הַמַּסְלוּל הַזֶּה לָבֶטַח,

"חֶבֶל נַחֲלָתֶךָ לִנְוֵה קָדְשֶׁךָ,

"כָּרוֹעֶה הַמְּנַהֵל עָלוֹת נֵהַלְתָּ

"אוֹתוֹ, וּבְעָזְּךָ עַל עַם זוּ גָּנַנְתָּ.


"נִפְלְאוֹתֶיךָ יַעֲשׂוּ לָךְ שֵׁם תִּפְאָרֶת,

"יִשְׁמְעוּ עַמִּים מַה פָּעַלְתָּ, יִרְגָּזוּ,

"וּמְלָכִים אַדִּירִים מִמְּךָ יָחִילוּ,

"חִיל יֹאחַז עַם צָפוֹן יוֹשְׁבִים בִּפְלָשֶׁת,

"תֵּימָן, אַלּוּפֵי אֱדוֹם יִבָּהֵלוּ,

"אֵילֵי מוֹאָב מִקֶּדֶם תֹּאחֲזֵם רָעַד,

"יוֹשְׁבִים בִּכְנַעַן עַד חוֹף יָם נָמוֹגוּ.


"תִּפּוֹל עֲלֵיהֹם אֵימָתָה וָפַחַד,

"כִּי מִגְּדוֹלוֹת עָשִׂיתָ בִּזְרוֹעֶךָ,

"יָבִינוּ כִּי אֵין מַצִּיל מִיָּדֶךָ,

וּגְאוֹן עַמְּךָ וּכְבוֹדוֹ כִּי יִשְׁמָעוּ,

"יָד עַל פֶּה יָשִׂימוּ, יִדְמוּ כָּאָבֶן,

"וּבְנֵי עַוְלָה לַעֲשׂוֹת רָע לֹא יוֹסִיפוּ,

"לַעֲשׂוֹת רָע עַם עַמְּךָ הַהוֹלְכִים דָּרֶךְ,

"לֹא יֵצְאוּ גוֹיִם לִקְרָאתָם בֶּחָרֶב,

"עַד יַעֲבוֹר עַמְּךָ יְיָ לְבֶטַח,

"עַד יַעֲבוֹר בִּישִׁימוֹן עַם זוּ קָנִיתָ.


"אַתָּה יְיָ תַּנְחֵם וּתְבִיאֵמוֹ

אֶל הָאָרֶץ לָתֵת לָהֶם נִשְׁבַּעְתָּ,

"שָׁם תִּטָּעֵמוֹ עַל הַר נַחֲלָתֶךָ,

"מָכוֹן אֶל שִׁבְתֶּךָ יְיָ פָּעַלְתָּ,

"בָּהָר הַזֶּה בּוֹ אִוִּיתָ, וַתֹּאמֶר:

"פֹּה אֵשֵׁב לַעֲדֵי עַד, אֶשְׁכּוֹן לָנֶצַחַ,

"מִקְּדָּשׁ יְיָ שָׁם כּוֹנְנוּ יָדֶיךָ,

"מִקְּדָשׁ זֶה בָּאָרֶץ עוֹלָם יָסַדְתָּ,

"אִם בַּעֲוֹנֵינוּ הָרֹס תַּהַרְסֶנּוּ,

"עוֹד תָּקִים חָרְבוֹתָיו, תָּשׁוּב תִּבְנֵהוּ,

"כִּי יָבוֹא מוֹעֵד יִכְלֶה בּוֹ זַעֲמֶךָ,

"מוֹעֲדִים לַתּוּשִׁיּוֹת רַק לָךְ נוֹדָעוּ,

"מֵרֵאשִׁית גָּבְלָה אוֹתָם חָכְמָתֶךָ,

"כִּי עִתּוֹת אֵין מִסְפָּר רָאוּ עֵינֶיךָ,

"וַתַּחְלְקֵם לַמּוֹעֲדִים מָתַי תָּבֹאנָה,

"כִּי מַלְכוּת כָּל עוֹלָמִים מַלְכוּתֶךָ,

"וּבְכָל דּוֹר וָדוֹר הִיא בַּכֹּל מָשָׁלָה,

"עַל כֵּן מַלְכֵּנוּ יוֹם זֶה לָךְ נָשִׁירָה,

“לֵאמֹר: יְיָ יִמְלֹךְ לְעוֹלָם וָעֶד”.


כֵּן הֵיטִיבוּ15 שִׁיר יוֹם מִצָּרֵיהֶם נוֹשָׁעוּ,

עוֹמְדִים עַל דֶּרֶך בָּרָא אָדוֹן חוּלִי אָרֶץ,

רִצְפָּתָהּ הַתְּהוֹמוֹת, חוֹמוֹתֶיהָ הַמַּיִם,

חֶצְיָם שָׁב אֶל אֵיתָנָהּ עִם הֲמוֹן דּוֹרְכֶיהָ,

וּשְׁאֵרִיתָהּ עוֹד מַעֲמַד לַמְּשׁוֹרְרִים שִׁיר קֹדֶשׁ,

כִּי בָא סוּס פַּרְעֹה בַיָּם פָּרָשִׁים וָרֶכֶב,

מֵאָז לֹא הִצִּיג עָלָיו מִכָּל הָחַי רֶגֶל,

וּכְשׁוּב הַיָּם אֶל אֵיתָנוֹ אֶל מִקְוֵה מַיִם,

אַחֲרֵי הֵשִׁיב יְיָ מֵי יָם עַל מִצְרַיִם

עוֹד הָלְכוּ בוֹ עֲדַת יִשְׂרָאֵל בַּיַּבָּשֶׁת.


כַּשֶּׁמֶשׁ16 אוֹר, מֵאוֹרוֹ כּוֹכָבִים יַגִּיהוּ,

כֵּן מֵרוּחַ משֶׁה מַרְאוֹת אֵל בּוֹ הֵאִירוּ,

נִפְרַץ חָזוֹן וְיַעֲבוֹר עַל לֵב הָעָם יָחַד,

לִשְׁמוֹעַ מִפִּיו שִׁיר וּלְהִשְׂכִּיל בּוֹ טוּב טָעַם,

כִּי גָּדוֹל הוֹד מֹשֶׁה מֵהוֹד כָּל שָׂרֵי קֹדֶשׁ,

וּבְכָל הַנְּבִיאִים לוֹ אֵין אָח, כִּי אֵין דּוֹמֵהוּ,

אִם מִרְיָם טוֹבַת שֵׂכֶל וּנְבִיאָה הָיָתָה,

גַּם בַּת הוֹרָיו, אֲחוֹתוֹ בִּנְבוּאָה אֵינֶנָּה,

וּבְהוֹדָהּ אֲחוֹת אַהֲרֹן הִיא, אֵלָיו דָּמָתָה,

כִּי גַּם בּוֹ גַּם בָּהּ אֱלֹהִים דָּבָר הוֹפִיעַ,

וּבְיוֹם הַיְּשׁוּעָה עַל הַנָּשִׁים הִתְרוֹמָמָה.

כַּאֲשֶׁר רָאֲתָה מִפְעֲלוֹת מֹשֶׁה מַה נִּשְׂגָּבוּ,

אֲשֶׁר כַּמַּלְאַךְ אֱלֹהִים הִגְדִּיל מַעֲשֵׂהוּ,

וּבִנְטוֹתוֹ יָדוֹ עַל יָם צָרִים נִבְלָעוּ,

זָכְרָה הַקּוֹרוֹת מֵאָז מֶה עָשְׂתָה מִקֶּדֶם,

יוֹם שָׂמוּ מֹשֶׁה בַּסּוּף מִפַּחַד רוֹצֵחַ,

אַף כִּי מַר לָהּ לֹא פָגוּ עֵינֶיהָ מִבֶּכִי,

אִזְּרָה חַיִל וּמִמָּוֶת נַפְשׁוֹ מִלֵּטָה,

וַתֹּאמֶר: "נָאוָה לִי כִּי לַיְיָ אָשִׁירָה,

"בַּמַּקְהֵלִים אָשִׁירָה לוֹ הַיּוֹם גַּם אָנִי,

"וּכְצֵאת מֹשֶׁה אָחִי לִפְנֵי רָאשֵׁי עַם קֹדֶשׁ,

"וּבְשִׁירֵי כָּבוֹד עֹז אֵל וּתְהִלוֹתָיו יַבִּיעוּ,

"אֵצֵא גַּם אָנִי לִפְנֵי רַעֲיוֹתַי כֻּלָּנָה,

"שִׁירֵי קֹדֶשׁ מִפִּי מָשׁוּי מִסּוּף שָׁמַעְתִּי,

"אֶל אָזְנֵי אֲהוּבוֹת אֵלֶּה בַּתֹּף אַשְׁמִיעַ,

"וּבְלֵב יָם נְשִׁירָה יַחְדָּו עֹז אֱלֹהֵינוּ.

כַּאֲשֶׁר תָּפְשָׂה יָדָהּ בַּתֹּף וּכְצֵאת בַּת שָׁחַר

יָצָאָה, וַתֵּצֶאןָ כָּל הַנָּשִׁים אַחֲרֶיהָ,

בִּמְּחוֹלוֹת וּבְתֻפִּיֹם וּבְשִׂמְחָה קוֹל הֵרִימוּ,

וַתַּעַן לָהֶן מִרְיָם: "רַעֲיוֹתַי הַאֲזֵנָה,

"כַּאֲשֶׁר הַגְּבָרִים כֵּן אַתֶּן לַיְּיָ שִׁירוּ,

“כִּי גָּאֹה גָּאָה, סוּס גַּם רֹכְבוֹ רָמָה יָמָּה”.


מַחֲנֶה17 הַגְּאוּלִים לָצֵאת מִיַּם סוּף אִחֵרוּ,

כִּי שִׂמְחַת אֱלֹהִים וּשְׂשׂוֹן לֵב שָׁם הִשִּׂיגוּ,

וַחֲמוּדוֹת צָר מֵי הַיָּם בַּמְּסִלָּה גָרָפוּ,

בָּזְזוּ לָהֶם בֵּית יִשְׂרָאֵל הוֹן רַב מָצָאוּ,

אַךְ מִפִּי אֵל הוֹדִיעָם משֶׁה כִּי יִסָּעוּ,

כִּי עַל פִּי יְיָ מִצְעָדֵיהֶם כּוֹנָנוּ.

כַּאֲשֶׁר אֶל מַעֲשֵׂיהֶם אֶת הַכֹּל הֵכִינוּ,

וַיְצַו מֹשֶׁה אוֹתָם: "מַלְּאוּ כַדֵּיכֶם מַיִם,

"כִּי אֶל מִדְבַּר צִמָּאוֹן אֵין מַיִם תָּבוֹאוּ,

עַל יָמִים שְׁלשֶׁת אֶת נֹאדוֹתֵיכֶם מַלֵּאוּ,

“מַקְצֶה יָמִים שְׁלשֶׁת פַּלְגִּי מַיִם תִּמְצָאוּ”.

וּבְנָסְעָם מִיַּם סוּף אֶל מִדְבַּר שׁוּר יָצָאוּ,

בַּמִּדְבָּר זֶה לֹא רָאוּ אֲפִיקִים בַּנֶּגֶב,

וּלְיָמִים שְׁלשֶׁת לִמְקוֹם פַּלְגֵּי מַיִם בָּאוּ,

אֲבָל מָרִים הָיוּ, לִשְׁתּוֹתָם בַּל יוּכָלוּ,

שָׁם פָּחֲדוּ פָחַד, אִישׁ אֶל רֵעֵהוּ דִּבֵּרוּ,

כִּי יִכְלוּ מֵימֵינוּ בַּצִּמָּאוֹן נָמוּתָה,

לֹא בָטְחוּ בֵאלֹהִים כְּי יֵרֶא וִימַלְּטֵמוֹ,

כַּאֲשֶׁר מִלְּטָם עַל יָם, מִלְּטָם אֶרֶץ מִצְרַיִם,

אַךְ בִּמְצוּקָתָם עַל רוֹעֶה נֶאֱמָן הֵלִינוּ,

לֵאמֹר: "מַה נִּשְׁתֶּה כִּי יִכְלוּ מַיִם מִכֶּלִי,

"מַה זֶּה הִבְטַחְתָּנוּ כִּי יֶשׁ פֹּה נַחֲלֵי מַיִם,

"מָרִים הֵם בַּל יִשְׁתָּיוּן, מָה יּוֹעִילוּ לָנוּ,

“כִּי נִצְמָא הֲיֵשׁ תִּקְוָה! כֻּלָּנוּ אָבָדְנוּ”.

הָשֵׁב אֶת הָעָם דָּבָר לֹא יָדַע הַפָּעַם,

כִּי יְיָ הֶעֱלִים עֲצָתוֹ גַּם מִמֶּנּוּ,

וּבְמַר נֶפֶשׁ צָעַק עַל יְיָ וַיֹּאמֶר:

"מָה אֶעֱשֶׂה עִם צֹאנֶךָ הַלְּקוּחִים לַמָּוֶת,

"יַעַן מֶה עָשִׂיתָ זֹאת לָעָם זוּ אָהַבְתָּ,

"יַעַן מֶה לֹא הִגַּדְתָּ לִי כִּי כֵן יָעַצְתָּ,

"נִשְׂגָּב מֶנִּי אַתָּה יְיָ זֹאת יָדַעְתָּ,

"נָא אֵל, גַּלֵּה לִי סוֹדֶךָ, דַּרְכְּךָ הוֹרֵנִי.


רָצָה אֱלֹהִים מִשְׁאֲלוֹתָיו וַיַּעֲנֵהוּ:

"מֵאַהֲבָתִי אוֹתָם אֶת זֹאת פָּעַלְתִּי לָמוֹ,

"לָדַעַת מַה בִּלְבָבָם אִם מִשְׁפָּט יַחְשֹׁבוּ,

"בִּמְחִיר צֶדֶק וּמִשְׁפָּט אַהֲבָתִי יִנְחָלוּ,

"כִּי אַשְׁרֵי אִישׁ מָצָא חָכְמָה הוּא יִנְחַל אָרֶץ,

"אַךְ מִי זֶה יִמְצָאֶנָּה, לֹא רַבִּים יֶחְכָּמוּ,

"עַיִן לִרְאוֹת, אֹזֶן לִשְׁמוֹעַ, לֵב לָדַעַת,

"זֶה חֵלֶק בַּר לֵב, כָּבוֹד חֲכָמים יִנְחָלוּ,

"וַאֲנִי חוֹקֵר לֵב מִשְׁפַּט לֵב עַמִּי יָדַעְתִּי,

"לֹא נָכוֹן לִבָּם עִמִּי, רוּחָם לֹא נֶאֱמָנָה,

"עַל כֵּן אֲנַסֵּמוֹ וּבַמּוּסָר אֶצְרְפֵמוֹ,

"וּכְסַם אִם יֵמַר אֶל חֵךְ מַרְפֵּא לִבְשַׂר גָּבֶר,

"כֵּן מוּסָר לַתּוֹעִים אֵלֶּה מַרְפֵּא לָנֶפֶשׁ,

"לָכֵן לִמְקוֹם מַיִם מָרִים אוֹתָם הֵבֵאתִי,

"נִסִּיתִים בַּעֲבוּר מַה בִּלְבָבָם אִם יוֹדִיעוּ,

"כִּי כַּאֲשֶׁר פָּחֲדוּ פָּחַד שׁוּר בַּמֶּה בָּחָרוּ,

"כִּי לֹא בָטַח לִבָּם, אֵלַי לֹא הִתְחַנָּנוּ,

"עַל אִישׁ כָּמוֹךָ, לַעֲשׂוֹת הִפְלֵאתָ, הֵלִינוּ,

"כַּאֲשֶׁר עָשׂוּ עַל יָם כִּי גַם שָׁם בָּךְ דִּבֵּרוּ.

"אַךְ אֶסְלַח לָהֶם וּמִפַּחְדָּם אֶרְפָּאֵמוֹ,

"עַתָּה אֱמוֹר אֶל הָעָם: אֶת פַּחְדָּם יַעֲזוֹבוּ,

"כִּי עָנְיָם רָאִיתִי וּתְלוּנָתָם שָׁמַעְתִּי,

"טֶרֶם יִכְלוּ מַיִם מִכֶּלְיָם יִוָּשֵׁעוּ,

"הִנְּנִי רוֹפֵא אֶת הַמַּיִם הַמָּרִים אֵלֶּה,

"וּבְיָדְךָ אֶעֱשֶׂנּוּ, הַדּוֹבְרִים בָּךְ יֵבוֹשׁוּ,

"כִּי לִרְפּוֹא אֶת מֶרֶר הַמַּיִם אֶשְׁלָחֶךָ,

"וּכְרוֹפֵא הַנּוֹתֵן סַמִּים אֶעֱשֶׂה הַפָּעַם,

"אִם עֵץ אוֹרְךָ תַּשְׁלִיךְ אֶל הַמַּיִם יִמְתָּקוּ

"כִּי מַה נָּקֵל לִי לָתֵת בַּמִּדְבָּר זֶה מָיִם,

"אִם עַל הַתְּהוֹמוֹת לָבוֹא עַד לֵב יָם צִוִּיתִי,

"אֶשְׁרוֹק לָּהֶם וּמַעְיְנוֹתֵיהֶם יִבָּקֵעוּ,

"אַךְ הִכִּיתָ אֶת הַיְּאוֹר וּמֵימָיו בָּאָשׁוּ,

"בַּעֲבוּר הוֹבִישׁ גֵּאִים עַל נוֹזְלֵיהֶם בָּטָחוּ,

"כֵּן מֶרֶר מַיִם אֵלֶּה יָדְךָ תִּרְפָּאֵמוֹ,

"בַּעֲבוּר הוֹדִיעַ אֶת יוֹשְׁבֵי אֶרֶץ מִצְרַיִם,

"כִּי יָדְךָ אֲשֶׁר קָדְקֹד הָרְשָׁעִים מָחָצָה,

"כִּי תִּפְתְּחֶנָּה בַּעֲדַת צַדִּיקִים טוֹב יִשְׂבָּעוּ,

“אֶת הָעֵץ הַמַּר הַזֶּה הַשְׁלֵךְ אֶל הַמַּיִם”.

כַּאֲשֶׁר אֶת הָעֵץ וּמַה יַּעֲשֶׂה בּוֹ הוֹרָהוּ,

הִשְׁלִיכוֹ משֶׁה אֶל הַמַּיִם וַיִּמְתָּקוּ.

שָׁם שָׂם לוֹ אֵל חֹק וּמִשְׁפָּט הָעָם בַּדָּרֶךְ,

חֻקֵּי אֵל וּמִשְׁפָּטָיו עִם עַמּוֹ לִמְּדָהוּ,

כִּי מֵאַהֲבָתוֹ אוֹתָם יִפְקוֹד מוּסָר לָמוֹ,

בַּעֲבוּר אִם מִשְּׁרִירוּת לֵב יִרְאָתוֹ יַעֲזוֹבוּ,

מִפַּחַד אַפּוֹ וַחֲמָתוֹ אֵלָיו יָשׁוּבוּ,

כַּנַּעַר לֹא שָׁמַע גְּעָרָה חָפְשִׁי כַפֶּרֶא,

יָזִיד יוֹם יוֹם כִּי עַל אַהֲבַת יוֹלְדָיו בּוֹטֵחַ,

יָשִׂימוּ הֵם אַהֲבַת אֵל לִכְסוּי חֵטְא וָפֶשַׁע,

יוֹם יוֹם יַרְבּוּ מֶרִי, חַטָּאת עַל חַטָאת יוֹסִיפוּ,

וּמָה אַחֲרִית כָּל אֵלֶּה, אֲבַדּוֹן וָמָוֶת,

וּלְהַצִּיל מִשְּׁאוֹל נַפְשׁוֹתָם נָכוֹן הַשֵּׁבֶט

הַכּוֹת בּוֹ שׁוֹגִים מִדֶּרֶךְ בַּעֲבוּר יֵיטִיבוּ.

לֹא הֶאֱמִין עָם אֲשֶׁר עָבְרוּ יַם בַּיַּבָּשֶׁת

כִּי מִכְשׁוֹל וּתְלָאָה תִּמְצָאֵמוֹ בַדָּרֶךְ,

עַד בּוֹאָם מָרָתָה, כִּי שָׁם צוּרָם נִסָּמוֹ,

"שָׁם גִּלָּה עַל עֲוֹנָם, שָׁם הֵחֵל הַרְאֵמוֹ

מָה רוֹבֵץ עַל פִּתְחָם אִם בַּיָי יִמְרֹדוּ,

כִּי רַק אִם יִשְׁמְעוּ לוֹ, אִם תּוֹרוֹתָיו יִנְצוֹרוּ,

לֹא יִרְאוּ רָע, מִשֵּׁבֶט אַפּוֹ לֹא יִפְחָדוּ,

כִּי כֵן דִּבֶּר אֱלֹהִים אֶל משֶׁה וַיֹּאמֶר:

"אִם שָׁמוֹעַ תִּשְׁמַע לְקוֹל יְיָ אֱלֹהֶיךָ,

"כָּל דָּבָר עַל פִּי נָבִיא אֱמֶת יַשְׁמִיעֶךָ,

"תֶּאֱהַב מֵישָׁרִים לַעֲשׂוֹתָם, כָּל מַעְגְּלֵי ישֶׁר,

"לֶאֱהוֹב אֵל, לֶאֱהוֹב מִשְׁפַּט שָׁלוֹם, לֶאֱהוֹב רֵעַ,

"לֶאֱהוֹב אֱמֶת, לִשְׁמוֹר אֱמוּנָה, לִשְׂנֹא עָוֶל,

"כָּל תּוֹרַת חֶסֶד וּפִקּוּדִים אֵל הוֹרֶךָ,

"תַּאֲזִין מִצְוֹתָיו, תַּקְשִׁיב לַחָכְמָה אָזְנֶךָ,

"לַעֲשׂוֹת אֶת מִצְוֹתָיו, תּוֹרוֹת אֱמֶת יוֹרֶךָ,

"גַּם אֶת חֻקָּיו תִּשְׁמוֹר, עֵת וּמִשְׁפָּט שָׂם,

"לֹא תִגְרַע מֵחֻקָּם, גַּם לֹא תוֹסִיף עָלֵימוֹ,

"אָז לֹא תִירָא מִמִּשְׁפָּטִים רָעִים צִוִּיתִי,

"אַרְבָּעָה הֵמָּה, לֹא תִירָא מֵאַרְבַּעְתָּנָה,

"גַּם מִתּוֹצְאוֹתֵיהֶן אַחַת לֹא תָשִּׂיגֶךָ,

"לֹא תֶּחְסַר כֹּל, חַיּוֹת אֶרֶץ עִמְּךָ תַּשְׁלַמְנָה,

"חֶרֶב לֹא תִרְאֶה, וּמִדֶּבֶר אֲמַלְּטֶךָ.

"אֶת אַרְבָּעָה אֵלֶּה אֲשֶׁר רָאוּ עֵינֶיךָ,

"כִּי שַׂמְתִּים עַל מִצְרַיִם לֹא אָשִׂים עָלֶיךָ

"וּמִי בַּלְעָדַי מִכָּל אֵלֶּה יַצִּילֶךָ,

"לִפְדּוֹת מֵרָעָב, מִשֵּׁן חַיָּה אוֹ מֵחָרֶב,

"רוֹפְאֵי אֱלִיל כֻּלָּם, מַצִּיל מִמָּוֶת אָיִן,

"גַּם מִדֶּבֶר אֵין מַצִּיל, מִמַּחְלָה תִדְמֵהוּ,

"אֶחָד מֵחֳלָיִים רָעִים נֶאֱמָנִים אִם בָּךְ יִדְבָּקוּ,

"כָּל סַמִּים, כָּל צֳרִי בַּגִּלְעָד לֹא יַעֲזוֹרוּ,

"אִם טֶרֶם בּוֹאָן מָה תַּעֲשֶׂה לֹא יִדְבְּקוּךָ,

"יָקוּמוּ נָא רוֹפְאֶךָ, יַחְדָּו יִוָּעֵצוּ,

"הֲלֹא גַּם הֵם יֹאמְרוּ לָךְ: כָּמוֹךָ חֻלִּינוּ,

"כִּי מִחוּצָה לָךְ תָּבֹאנָה עַל כַּנְפֵי רוּחַ,

"וּמִי שַׁלִּיט בָּרָקִיעַ אֲשֶׁר מִמָּעַל?

"רַק בִּדְבַר פִּי יִרְחַק כָּל רָעַ מִבּוֹא אֵלֶיךָ,

"כַּאֲשֶׁר צִוִּיתִי עַל מֵי מָרָה וַיִּמְתָּקוּ,

"טֶרֶם כִּלּוּ מַיִם מִכֶּלְיֶךָ, לֹא צָמֵאתָ.

"אִם תֶּחְטָא לִי וַאֲצַוּ רָעָה לָבוֹא עָלֶיךָ,

אִם תָּשׁוּב אֵלַי וַהֲסִירוֹתִיו מִקִּרְבֶּךָ,

“וּמַה תִּירָא אִם אֲנִי יְיָ רוֹפְאֶךָ”.
