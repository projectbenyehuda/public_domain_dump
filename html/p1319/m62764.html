הַקַּיִץ

הַקַּיִץ הִנֵּה־בָא וְרַב מְאֹד חֵילֵיהוּ

צִיָּה גַּם חוֹם כִּמְרִירֵי־יוֹם מַחֲנֵיהוּ

  סוּסָיו לֶהָבוֹת מֶרְכַּבְתּוֹ גַחֶלֶת

עֵת הַזָּמִיר נָסָה כִּסְּתָה פָּנֶיהָ

מִלַּפִּידֵי עֵינָיו תִּסּוֹב עֵינֶיהָ

  וְתִתֵּן תֵּבֵל כֻּלָּהּ אֵלָיו לְמֶמְשֶׁלֶת:


בְּטֶרֶם תָּפִיץ שֶׁמֶשׁ חִצֵּי שַׁלְהֶבֶת

וּבְעוֹד רוּחַ קָרָה צִנָּה נֹשֶׁבֶת

  אֶעֶזְבָה מִטָּתִי אָקוּם מֵעֶרֶשׂ

אֵלֵךְ לִי אֶל־הֲרָרִי אֲבַקֵּשׁ מָנוֹחַ

שָׁם תַּחַת אַלּוֹן וְלִבְנֶה אַחֲלִיף כֹּחַ

  אֲשׁוֹבֵב נַפְשִׁי טֶרֶם יֵצֵא הַחֶרֶס:


עוֹד סְרוּחִים עַל־עַרְשׂוֹתָם חוֹמְסֵי תֵבֵל

מְאַחֲרֵי בַנֶּשֶׁף עֲלֵי כִנּוֹר וָנֵבֶל

  הוֹזִים שׁוֹכְבִים יִשְׁאֲפוּ יֶהֱמוּ כַכֶּלֶב

הָקִיצוּ שִׁכּוֹרִים פִּקְחוּ עֵינָיִם

הִתְבּוֹנְנוּ רְאוּ מִפְלְאוֹת יוֹצֵר שָׁמָיִם

  מָה אֲמוּלָה לִבַּתְכֶם טָפְשָׁה כַּחֵלֶב:


מַה־לָּכֶם תֶּאֶהֲבוּ שֵׁנָה בַּל תָּקִיצוּ

לָלֶכֶת בְּאֶרֶץ נְשִׁיָּה מַה־תָּאִיצוּ

  אוֹ בַחֲלוֹם חֶזְיוֹן לַיְלָה מַה־תֶּהְבָּלוּ

טְפָחוֹת יָמֵינוּ חַיֵּינוּ אַךְ רֶגַע

הֲלֹא אֶת־כֻּלָּנוּ יִקְרֶה עֵת וּפֶגַע

  וּמַדּוּעַ בַּחַיִּים חֲיוֹת תֶּחְדָּלוּ:


אֵלֶּה הַיְּצוּרִים בְּאֵין שֵׂכֶל בְּאֵין דַּעַת

זִיז שָׂדַי עוֹף הָרִים פִּי הַתּוֹלַעַת

  יְאַלְּפוּךָ חָכְמָה יַחַד כְּסִיל וּבַעַר

הֵמָּה יַשְׁכִּימוּ הַשַּׁחַר יָעִירוּ

קוֹל הַתּוֹר נִשְׁמַע צִפֳּרִים יָשִׁירוּ

  יְשַׂחֲקוּ חַיַּת שָׂדֶה גַּם חַיְּתוֹ יַעַר:


הִנֵּה עֲדַת דְּבוֹרִים יִרְחֲקוּ מִמְּגֻרָמוֹ

נָצוּ וְנָעוּ יֵצְאוּ חֹצֵץ כֻּלָּמוֹ

  עַל־עֲרוּגַת פְּרָחִים יָעוּפוּ וְיִשְׁכֹּנוּ

לֹא יִעֲפוּ בְּפָעֳלָם יִשְׁקְדוּ וְיֶחֱרוֹצוּ

לְהָכִין דּוֹנָג וּדְבַשׁ עָסִיס יִמּוֹצוּ

  וּבְגֶזַע עֵץ יִרְקַב אוֹתָם יִצְפֹּנוּ:


שָׁם עֲלֵי שְׂדֵה תְרוּמוֹת תָּרוּץ הָאַרְנֶבֶת

עַתָּה תַּעֲמֹד וְעַל־כְּרָעַיִם יֹשֶבֶת

  מָגוֹר מִסָּבִיב פֶּן יָבֹא הַקָּרֶץ

וַתִּרְגַּז מִקּוֹל צַעַד עוֹבֵר אֹרַח

בְּמוֹרַד הָהָר כְּרָעֶיהָ לָהּ לָטֹרַח

  וַתִּפֹּל עַל־עָרְפָּהּ כַּדּוּר עַלֵי אָרֶץ:


בַּעֲזוּבַת הַחֹרֶשׁ רָץ קַל רַגְלַיִם

הַצְּבִי בִּגְאוֹן צְבִי עֶדְיוֹ פְּאֵר קַרְנַיִם

  וַיָּחֶל מְאֹד מֵהַמּוֹרִים בַּקֶּשֶׁת

יִתְעֶה בְּסִבְכֵי יַעַר לְרוֹחַב וָאוֹרֶךְ

הָלוֹךְ וְטָפוֹף יֵלֵךְ בִּלְבָבוֹ מוֹרֶךְ

  פֶּן עָרְמָה הֵכִינָה לִפְעָמָיו רֶשֶׁת:


עַתָּה עֲטֶרֶת זָהָב עַל־הָרֵי רַהַב

אֵשׁ יָצְאָה מִשַּׁחַק מִקֶּדֶם לַהַב

  מִמֶּנָּה מֵי פֶלֶג כַּדָּם אָדָמוּ

וְהִנֵּה הַשֶׁמֶשׁ בֵּין הָהָר שׁוֹזֶפֶת

לְהַרְחִיב מְדוּרָתָהּ בִּמְקוֹמָהּ שׁוֹאֶפֶת

  וְעַפְרוֹת תֵּבֵל כְּגֶחָלִים יִתְחַמָּמוּ:


הַכַּרְמֶל יָגִיל נְאוֹת דֶּשֶׁא יִשְׂמָחוּ

וְגֶפֶן פּוֹרִיָּה שְׂרוּקֶיהָ יִפְרָחוּ

  הַתְּאֵנָה תִּגְמֹל נִצָּהּ תּוֹצִיא בֹסֶר

רִמּוֹן גַּם תַּפּוּחַ פִּרְיָם יַבְשִׁילוּ

עֲמָקִים שִׁבֳּלֵיהֶם לְמַלְּאֹת יָחִילוּ

  וַתְּהִי לְיוֹגֵב תִּקְוָה לֹא יִירָא חֹסֶר:


אָכֵן מֵחֲצִיר גַּגּוֹת עַד כֶּרֶם זָיִת

מֵעֲצֵי אֶרֶז עַד אֵזוֹב קִיר הַבָּיִת

  מִגְּבוּרַת הַשֶּׁמֶשׁ כֻּלָּם יָקוּמוּ

לְאוֹרָהּ יָקוּמוּ כְּפִיר פֶּתֶן וָשַׁחַל

וְרִמָּה וְתוֹלֵעָה הַשּׁוֹרְצִים בַּנַּחַל

  כֻּלָּם נָגְהָהּ יִשְׁאֲפוּ שָׂבְעוּ וְיָרוּמוּ:


מִדְבָּר גַּם צִיָּה מְקוֹם תַּנִּין רוֹבֶצֶת

אֶרֶץ תַּלְאוּבוֹת שָׁם אַיָּה שׁוֹרֶצֶת

  כִּזְרוֹחַ הַשֶּׁמֶשׁ סָפוּ בַּלְהוֹתֵימוֹ

צִנּוֹר מַיִם שׁוֹטְפִים בְּמוֹרָד נִגָּרוּ

אַף נַהֲרוֹת אֵיתָן מִזָּהֳרָהּ יִטְהָרוּ

  מֵאוֹר פָּנֶיהָ בָּרְחוּ מְחִתּוֹתֵימוֹ:


וְאוֹרָהּ מַה מָתָקָה מַה־טּוֹב לָעֵינָיִם

מֶנָּהּ תֵּצֵא נוֹגַהּ מְלֵאָה קַרְנָיִם

  מֵאַבְנֵי אֶקְדָח מֵאַבְנֵי תִפְאֶרֶת

עֵין הַחַשְׁמָל שְׁבִיב אִשּׁוֹ הִיא מוֹצֶקֶת

מִמֶּנָּהּ יְנוֹצְצוּ פִּטְדָּה וּבָרֶקֶת

  אַף חָצָץ לְמַרְאָהּ יְלַהֵט כְּאֶבֶן יְקֶרֶת:


מַה־טּוּבָהּ מַה־יָפְיָהּ וּמֶה נָעֵמָה

רַבָּה גְּבוּרָתָהּ כֹּחָהּ מֶה עָצֵמָה

  לְמִפְלְאוֹת פְּעֻלּוֹתֶיהָ הֲלֹא אֵין חֵקֶר

עַל־אֵלֶּה תּוֹעֵי לֵבָב אָבְדוּ דֶּרֶךְ

סָגְדוּ לַשֶּׁמֶשׁ גַּם כָּרְעוּ לָהּ בֶּרֶךְ

  נִתְעוּ בְּהַבְלֵי שָׁוְא שָׁעוּ בְּדִבְרֵי שֶׁקֶר:


יַעַן כֹּחָהּ נִדְמָה הַפְלֵא וָפֶלֶא

לָאֵל שָׂמוּהָ קִצְרֵי דַעַת אֵלֶּה

  אַךְ הִיא בַּכֹּל שַׁלֶּטֶת גָּזְרוּ אֹמֶר

מִשְׁטָרָהּ בָּאָרֶץ גַּם בַּשָּׁמָיִם

גַּם אֹכֶל תּוּכַל תֵּת לֶחֶם וּמָיִם

  וַיִּשַּׁח הָאָדָם קָרוּץ מֵחֹמֶר:


לֹא כֵן עִמָּדִי אֲבָל חֶלְקִי בַּדַּעַת

יֵשׁ אָדוֹן עָלֶיהָ נַפְשִׁי יוֹדַעַת

  הָאוֹמֵר לַחֶרֶס וְיֶחְדַּל מִזְּרוֹחַ

עוֹשֵׂי רְצוֹנוֹ הֵם מַזָּרוֹת כֻּלָּמוֹ

יַעַמְדּוּ וְיַחֲלֹפוּ כִּי הוּא צִוָּמוֹ

  לוֹ הַמֶּמְשָׁלָה הַגְּבוּרָה וְהַכֹּחַ:


וְאַבִּיט בַּמֶּרְחָב וְהִנֵּה נְקֻדָּה אַחַת

חוּג הָאָרֶץ אַף חוּג שֶׁמֶשׁ זֹרַחַת

  מַה־נּוֹרָא מְקוֹמוֹ מִי בָא עַד קִצֵּהוּ

בְּכָל־מֶרְחַקָּיו הַגַּלְגַּלִּים יָרוּצוּ

לְעֻמָּתָם אוֹפַנִּים יְסוֹבְבוּ וְיָדוּצוּ

  כַּחוֹל שִׁמְשׁוֹתָיו מִי יִסְפֹּר גְּדוּדֵיהוּ:


וּמִי אָנֹכִי כְּבַד פֶּה נְלוֹז שְׂפָתָיִם

נֶגֶד צִבְאוֹתָיו אַלְפֵי רִבּוֹתָיִם

  כִּי אֲהַלֵּל יָהּ וּמִשִּׁירִי אֲהוֹדֶנּוּ

אֲבָל בְּרָן־כֹּכְבֵי בֹקֶר וְעֻזּוֹ יְזַמֵּרוּ

גַּם עוֹלְלִים וְיוֹנְקִים תְּהִלָּתוֹ יְסַפֵּרוּ

  לָכֵן לֹא אִירָא אִם נַפְשִׁי תְּהַלְּלֶנּוּ:


* * *


עַתָּה הַשֶּׁמֶשׁ תַּבְעִיר כְּאֵשׁ צָרֶבֶת

עַתָּה מִמַּעַל לְהָרִים הִיא נִצֶּבֶת

  מֶנָּה עֲרָפֶל וַעֲנָנִים יִפָּרֵדוּ

קִיטוֹרֵי רָאשֵׁי הֶהָרִים נָמוֹגוּ

עֲנַנֵּי בֹקֶר גַּם הֵם אָחוֹר נָסוֹגוּ

  יָעוּפוּ יַעֲלוּ בַתֹּהוּ וְיֹאבֵדוּ:


וְהִנֵּה בְּרִשְׁפֵי אֵשׁ עֶבְרָתָהּ תָּפִיחַ

עַל־אָדָם עַל־בְּהֵמָה עַל־עֵשֶׂב וְשִׂיחַ

  כֻּלָּם עֲיֵפִים מִנִּי שָׁרָב יַעֲטֹפוּ

יָגֵעַ עֹבֵר דֶּרֶךְ צְרוֹר עַל־שִׁכְמֵהוּ

וְגַם הַחֲמוֹר רוֹבֵץ תַּחַת מַשָּׂאֵהוּ

  וְצִיצֵי תִפְאֶרֶת יִבּוֹלוּ וְיַחֲלֹפוּ:


הָרוֹעֶה יָשׁוּב לְבֵיתוֹ לְאַט פּוֹשֵׂעַ

מֵעֲבוֹדַת הַבֹּקֶר עָיֵף וְיָגֵעַ

  לְפָנָיו עֶדְרוֹ יִרְעֶה לְרִבְצָם יְנַהֲלֵמוֹ

הַכּוֹס וְהַנֵּץ בְּעַנְפֵי אֵלָה יִשְׁכּוֹנוּ

הֵם יִשְׂתָּרְגוּ עֲלֵיהֶם וְאוֹתָם יִצְפּוֹנוּ

  מַחְסֶה מֵחֹרֶב וּמָגֵן עָלֵימוֹ:


גַּם עוֹף הַבַּיִת לְמִינֵיהֶם יָגוּרוּ

וְתַחַת צֵל סֶלַע כָּבֵד הֵם יָדוּרוּ

  וּבָקְעוּ וְדָגְרוּ יִפְעוּ וְיִצְעָקוּ

גַּם נָמוּ הַכְּלָבִים לִמּוּדֵי צָיִד

הָרֹדְפִים אַחֲרֵי קוֹרֵא חֹטְפֵי עָיִט

  אַךְ צִרְעָה תְּעִירֵם וְיֶחֶרְדּוּ וְיִנְהָקוּ:


וְזוֹחֲלֵי עָפָר הֵמָּה בַּעֲלֵי כְנָפָיִם

כְּחוֹל לָרוֹב אַלְפֵי רִבּוֹ רִבּוֹתָיִם

  חוֹם הַיּוֹם אֶת־קִבְרוֹתֵיהֶם פֹּתֵחַ

מִנִּקְרֵי הַצּוּר מִסְּעִיפֵי הַסֶּלַע

מִכָּל־פִּנָּה מִכָּל־עֵבֶר וָצֶלַע

  מִכָּל־חוֹרֵי עָפָר פִּרְחָח פֹּרֵחַ:


לְמִשְׁפְּחוֹתֵיהֶם יֵצְאוּ לְמִינָם יְעוֹפֵפוּ

אֵלֶּה עַל־מֵימֵי הָאֲגַמִּים יְחוֹפֵפוּ

  יָחֹגּוּ וְיָנוּעוּ עַל־פְּנֵי הַמָּיִם

אֵלֶה אֶל־יַעַר אֶת־דַּרְכָּם יִבְחָרוּ

וְתַחַת עֲלֵי רַעֲנָן יַחְדָּו יִתְגֹּרָרוּ

  שָׁם יִמְצְאוּ עֹנֶג וּמַחְמַד עֵינָיִם:


אֵלֶה בִּנְאוֹת דֶּשֶׁא שָׁם יִמְצְאוּ נַחַת

וּבִמְקוֹם סֵתֶר בֵּין צִיצָה פֹּרַחַת

  יָשִׁיתוּ אֶפְרֹחֵיהֶם וַיִּשְׂבָּעוּ

אֵלֶּה בְּתוֹךְ מַרְבֵּק יָדוּרוּ בַּשֵּׁפֶל

יִתְאווּ לֶחָלָב מִתּוֹךְ הַסֵּפֶל

  וּכְסָבְאָם סְבוּאִים בּוֹ יִפְּלוּ וְיִגְוָעוּ:


הֲמוֹנִים הֲמוֹנִים לֹא שְׁזָפָתָם כָּל־עָיִן

קָטְנוּ מֵאָבָק דַּק נֶחְשְׁבוּ לָאָיִן

  יִדְאוּ בָּרוּחַ פְּנֵי תֵבֵל יְמַלֵּאוּ

בְּעוֹמֶק הָאָרֶץ בְּגֹבַהּ הַשָּׁמָיִם

עַל־כָּל־מַאֲכָל עַל־כָּל־מַשְׁקֶה וּמָיִם

  תִּרְמֹשׂ עַם עָצוּם קְהַל גּוֹיִם יֵצֵאוּ:


וּמֵעֵינֵי בָשָׂר נִסְתָּרִים הֵמָּה

זוּ חָכְמַת אֵל לָדַעַת כִּי עָצֵמָה

  בְּכָל־מַעֲשָׂיו יוֹסִיף דַּעַת יוֹסִיף לֶקַח

לוּ לִרְאֹתָם עֵין אֱנוֹשׁ תְּהִי נִפְקַחַת

בְּכָל־מַטְעַמִּים נַפְשׁוֹ לֹא תִמְצָא נַחַת

  וְזִהֵם בְּמַאֲכַל תַּאֲוָה וּבְיֵין הָרֶקַח:


בַּל יָלִיץ הַלֵּץ בַּל יִתֵּן מִגְרַעַת

וְיוֹכַח לָאֵל וְיַעֲלִים עֵצָה בְּלִי דַעַת

  לֵאמֹר מַה יִסְכֹּן כִּי אֵלֶּה נִבְרָאוּ

הַאֱנוֹשׁ שְׁתֻם הָעַיִן יַבִּיט גָּבוֹהַּ

לָשׂוּם תָּהֳלָה בִּגְדֹלֵי מִפְעֲלוֹת אֱלוֹהַּ

  בְּתַעֲלוּמוֹת חָכְמָה אֲשֶׁר עֵינָיו לֹא רָאוּ:


מִי אֶת־קְצֵה רוֹם וְחֵקֶר תְּהוֹם מִי יֹדֵעַ

מִי תַּכְלִית שַׁדַּי וְקוֹלוֹ מִי שֹׁמֵעַ

  מִי בָא בְּסוֹד אֵל בְּהִוָּסְדוֹ מוֹסְדוֹת תֵּבֵל

עַל־כֹּל פָּעַל יְיָ רְצוֹנוֹ הוֹפִיעַ

לַעֲבֹד עֲבוֹדָתוֹ מַעֲלָלוֹ הוֹדִיעַ

  הוּא יִדַּע מְקוֹמוֹ וְחִלֵּק לוֹ בַּחֶבֶל:


* * *


מֶה קוֹל הֶהָמוֹן הַזֶּה שָׁם בָּאָחוּ

קָטוֹן וְגָדוֹל שָׁם הוּא חֲרָבוֹת יִקָּחוּ

  כְּמוֹ בְאֵשׁ פְּלָדוֹת הוּחַדּוּ וְגַם מוֹרָטוּ

אֵין קוֹל עֲנוֹת גְּבוּרָה וְאֵין אוֹיֵב בַּשַּׁעַר

כִּי יַחְדָּו הֵעִיזוּ זָקֵן וָנַעַר

  וְעָזְבוּ בֵיתָם וּבָאָחוּ פָּשָׁטוּ:


אַךְ תֹּפְשֵׂי מַגָּל הֵם חֶרְמֵשׁ יָנִיפוּ

עֵת קְצִיר חָצִיר הוּא לָכֵן שָׁם יַקִּיפוּ

  נְאוֹת דֶּשֶׁא וְיִשְׂמְחוּ כָּל־אִכָּרֵי חֶמֶד

אֵלֶּה יִקְצְרוּ וְאֵלֶּה יִצְבְּרוּ בְיָדָיִם

וְאֵלֶּה יִקְחוּ בְמַזְלֵג שְׁלוֹשׁ הַשִּׁנָּיִם

  מִן הָעֲרֵמָה עַל־הָעֲגָלָה וָצֶמֶד:


וְהִנֵּה כְּהַר תָּלוּל הָעֲגָלָה נִצֶּבֶת

וְעַל רֹאשָׁהּ עַלְמָה אֲדַמְדַּמָּה יֹשֶׁבֶת

  וּבְמַלְמַד הַבָּקָר יְנַהֵג הָעֶבֶד

וַיְּנַהֵל לְאַט הַפָּרוֹת חִזְקֵי גֶרֶם

וְשָׁם עָמְדוּ לְהַבְלִיגָה בְּמִשְׁעֹל הַכֶּרֶם

  כִּי תָעִיק הָעֲגָלָה מִמַּשָׂא כֶבֶד:


נָא פָרוֹת יְקָרוֹת כֹּחֲכֶן עוֹד תֶּעֱצַרְנָה

הִתְאַמְּצוּ לֶכֶת חֲיָלֵיכֶן תִּגְבַּרְנָה

  עוֹד דֶּרֶךְ מְעַט וְתָבֹאנָה אֶל הָרָפֶת

וְחֵלֶף עֲבוֹדַתְכֶן תֹּאכַלְנָה חֲצִיר יֶרֶק

הִתְחַזְּקוּ־נָא בַּל תַּעַמְדוּ עַל הַפֶּרֶק

  הֲלֹא לְמַעַנְכֶן מַעֲמָסַתְכֶן נֶאֱסָפֶת:


וְשָׁמָּה בָּעֵמֶק מְקוֹם יִזּוֹב הַנַּהַר

בִּבְרֵכָה הָעֲשׂוּיָה כְּאַגַּן הַסַהַר

  כָּל־רוֹעֵי הָעֲדָרִים נֶאֶסְפוּ יָחַד

וַיִקָּבְצוּ כָל־בְּנֵי־צֹאן אַיִּל גַּם תַּיִשׁ

מִיַּד כֶּלֶב רוֹעֶה גִבּוֹר כַּלַּיִשׁ

  וּבְבֹאָם עַל־שְׂפָתָהּ שָׁם פָּחֲדוּ פָּחַד:


וְהָרוֹעִים יִצְעֲקוּ וְהָעֲדָרִים יִשְׁרָקוּ

וְהַכְּלָבִים יֶהֱמוּ וְשִׁנֵּיהֶם יֶחֱרָקוּ

  כִּי יָרְאוּ בְּנֵי צֹאן לָרַחְצָה לָרֶדֶת

עַד אַיִּל אֶחָד בְּקַרְנֵי הַגֻּלְגֹּלֶת

נִתְפַּשׂ בְּתוֹקֶף וְהָשְׁלַךְ אֶל הַשִּׁבֹּלֶת

  אָז כָּל־גְּדוּדֵי הַצֹּאן אַחֲרָיו וְאֵין בֹּדֶדֶת:


כָּכָה פַּעֲמַיִם יֵרְדוּ אֶל־הַפֶּלֶג

עַד מְאֹד יַלְבִּינוּ וְצִמְרָם צַח מִשֶּׁלֶג

  וְאָז יַעֲלוּ מִן הַמַּיִם שָׁם רֻחָצוּ

וְהִנֵּה בֵּין גְּבָעוֹת נָפוֹצוּ וְנִפְרָשׁוּ

וְהַשֶּׁמֶשׁ תִּפַּח בָּהֶם וְיִיבָשׁוּ

  וְאָז שֵׁנִית מֵרוֹעִים יַחְדָּו נִקְבָּצוּ:


וְהֵמָּה בֵין גִּדְרוֹת צֹאן יִסָּגֵרוּ

וְשָׁמָּה יֵשְׁבוּ הַגּוֹזְזִים וִיזַמֵּרוּ

  אִישׁ יַחַד תַּעֲרוֹ וּבְאֶבֶן יִלְטְשֵׁהוּ

וְיַעַקְדוּ אֵלֶּה נְקִיִּים אֹכְלֵי עֵשֶׂב

יִסְחֲבוּ צְעִירֵי הַצֹּאן רָחֵל וָכֶשֶׂב

  וּכְאַכְזָר יֹאחֲזוּ הָאַיִּל בְּקַרְנֵיהוּ:


בַּל תִּירְאִי צֹאנִי בַּל תְּהִי מִתְחֹלֶלֶת

לֹא זֹאת הַחֶרֶב לֹא זֹאת הַמַּאֲכֶלֶת

  דָּמֵךְ לֹא יְבַקֵּשׁ הָעֹמֵד עָלָיִךְ

רַק בִּגְדֵךְ יַפְשִׁיטֵךְ צִמְרֵךְ יְגַלֵּחַ

וְאָז יַנִּיחֵךְ וְאוֹתָךְ לְחָפְשִׁי יְשַׁלֵּחַ

  וְנִמְלַטְתְּ מִיָּד כִּי יְפַתַּח מוֹסְרוֹתָיִךְ:


וְאַחֲרֵי כְּלוֹת הַמְּלָאכָה גּוֹזְזִים יָגִילוּ

וְעֵדֶר הַקְּצוּבוֹת עֲלוֹת הָהָר יַעְפִּילוּ

  לִרְבֹּץ מְקוֹם אֵין צֵל הַשֶּׁמֶשׁ זֹרֵחַ

וּמִשְׁתֶּה וְיוֹם טוֹב הַיּוֹם לְגוֹזְזֵי צֶמֶר

אָכוֹל בְּשַׂר אַבִּירִים וְשָׁתוֹת יֵין חֶמֶר

  יְרַנְּנוּ לְקוֹל עוּגָב יְזַמְּרוּ בְּלֵב שָׂמֵחַ:


וְעוֹשֵׂי מִקְנֶה וְקִנְיָן גַּם הֵם יִשְׂמָחוּ

וּבִמְחִיר כֶּסֶף הַגִּזָּה יִקָּחוּ

  וּבְחוּצוֹת קִרְיָה יַעֲשׂוּ מֶנּוּ מֶכֶר

וְאַחֲרֵי צֶמֶר לָבָן וָדַק יִדְרֹשׁוּ

וְאֶת־הַטּוֹב מִן הָרַע הֵיטֵיב יִפְרֹשׁוּ

  לְמַעַן הַרְבּוֹת בִּמְחִירוֹ וְהַרְבּוֹת שֶׂכֶר:


וּמִמֶּרְחָק יָבֹאוּ עֹבְרֵי אֹרַח

מֵעֵבֶר לַיָּם בִּיגִיעָה וָטֹרַח

  אֹרְגֵי בִגְדֵי צֶמֶר וּגְלוֹמֵי תְכֵלֶת

וּנְטִילֵי כֶסֶף בַּצֶּמֶר יִשְׁקֹלוּ

מִן בֹּקֶר עַד עֶרֶב בְּמִסְחָר יַעֲמֹלוּ

  וְהַרְבֵּה יְגִיעָה בְּמֹאזְנַיִם וּבְמִשְׁקֶלֶת:


כָּכָה תְּבֹרַךְ אֶרֶץ בִּרְכַּת שָׁמָיִם

אֲשֶׁר ישְׁבֶיהָ לֹא יִשְׁנוּ בַּעֲצַלְתָּיִם

  וּלְעַשְׁתְּרוֹת צֹאנָם אֶת־לִבָּם יָשִׂימוּ

תְּהִי סְחַר גּוֹיִם אִיִּים רַבִּים רֹכְלֶיהָ

רָעָב לֹא תִירָא יִמָּלְאוּ אֲסָמֶיהָ

  תִּשְׁכֹּן בֶּטַח כָּל־גּוֹי אִתָּהּ יַשְׁלִימוּ:


* * *


עַד כֹּה וְהִנֵּה בָאָה עֵתּ הַצָּהֳרָיִם

וְהַשֶּׁמֶשׁ אָץ לָבוֹא בַּחֲצִי הַשָּׁמָיִם

  הוֹי גַּם עַל־קָדְקֳדִי חַמָּתָהּ יֹרָדֶת

אָנָּא אֵלֵךְ וְאָנָּא אוּכַל לִבְרוֹחַ

אֶתְעַלֵּף בַּחוּץ וְאֶתְהַלֵּךְ בְּלִי כֹחַ

  בִּרְכִּי כָּשַׁל מֵחוֹם וְרַגְלִי מֹעָדֶת:


אֶל־חֹרֶשׁ מֵצַל הָהּ מִי יוֹבִילֵנִי

אוֹ תַחַת סֻכַּת עֲנָפִים מִי יַנְחֵנִי

  עַד שׁוּב חֲרוֹן הַשֶּׁמֶשׁ עַד יַעֲבֹר זָעַם

הִנְנִי אֲכוּלִי חוֹרֶב וּלְחֻמִי רֶשֶׁף

אֶתְאַוֶּה הַלַּיְלָה חִשְׁקִי הַנֶּשֶׁף

  צִלְלֵי עֶרֶב נָא בַּל תְּאַחֲרוּ הַפָּעַם:


הַרְפִּי מְעַט שֶׁמֶשׁ הָסֵבִּי עֵינָיִךְ

עַד כִּי אֵלֵךְ וְאֶסָּתֵר מִפָּנָיִךְ

  עַד אֶמְצָא מָקוֹם מַחְסֶה מִשַּׁלְהָבֶת

שָׁם בֵּין סִבְכֵי יַעַר אֵצֵא לָשׂוּחַ

בֵּין צֵל צֶאֱלִים אַרְגִּיעַ וְאֶשְׁאַף רוּחַ

  עַד כִּי תִשְׁבְּתִי מֵעִיר אוֹחִיל לָשָׁבֶת:


בְּרוּכִים אַתֶּם בָּרוּךְ צִלְּכֶם עֲצֵי חֹרֶשׁ

אַתְּ אֵלָּה עֲבֻתָּה הַמַּזְקֶנֶת שֹׁרֶשׁ

  אֶרֶז לְבָנוֹן בְּרוֹשׁ וְתִדְהָר גְּדֹלֵי כֶתֶר

יְהִי מְקוֹרְכֶם בָּרוּךְ עֲלֵיכֶם לֹא יִבּוֹלוּ

כִּי הֲיִיתֶם מִפְלָט לִי וְעָלַי תַּחְמֹלוּ

  מֵחַרְבוֹנֵי קַיִץ מַחְסֶה וָסֵתֶר:


מַה־טּוֹב צִלְּכֶם לַהֲלָךְ־נֶפֶשׁ בֹּדֶדֶת

פֹּה אִוְּתָה לְמוֹשָׁב לָהּ אֲבָל פֹּה חֹמֶדֶת

  לְהִתְהַלֵּךְ בֵּין בְּנוֹת שִׁיר בְּהַשְׁקֵט וָנַחַת

רָחוֹק מִשְּׁאֹן קִרְיָה מִיַּלְדֵי פֶשַׁע

מִשּׁוֹט לָשׁוֹן רְמִיָּה מִשִּׂפְתֵי רֶשַׁע

  תִּשְׁגֶּה בִּנְעִימוֹת יָהּ וְלֹא תֵרֵד שַׁחַת:


בְּעוֹד הָלְאָה בַּיַּעַר רַגְלִי הֹלֶכֶת

וְהִנֵּה עֲנָפִים יְסֻכָּכוּ כְּמוֹ מַסֶּכֶת

  שָׁם אֶרֶץ עֵפָתָה וְחֹשֶׁךְ יְסַבֵּנִי

זוּלַת קוֹל עָלֶה אֵינֶנִּי שׁוֹמֵעַ

וְקוֹל דְּמָמָה דַקָּה מִנַּחַל נוֹבֵעַ

  וְאֵלֵךְ לְקוֹלוֹ וְעַל מֵי מְנוּחוֹת יְנַהֲלֵנִי:


וְאֶרְאֶה שָׁם יִרְחֲצוּ הַנְּעָרִים בְּמֵי שָׂחוּ

אֵלֶּה עָרוֹם וְיָחֵף יְרַקְּדוּ וְיִשְׂמָחוּ

  וְאֵלֶה יִפְרְשׂוּ לִשְׂחוֹת זְרוֹעוֹת יָדֵימוֹ

כֻּלָּם בְּרִיאִים וּכְיַלְדֵּי עֲנָק יַשְׁמִינוּ

לֹא נִסְגְּרוּ בְּעִיר אַךְ בִּכְפָרִים יָלִינוּ

  כֻּלָּם יְגֵעִים רַךְ וְעָנוֹג אֵין בֵּינֵימוֹ:


נְדִיבֵי עָם עַל־מֶה בְּנֵיכֶם דַּלֵּי תוֹאַר

הֲלֹא יַעַן כִּי תְפַנְּקוּ אוֹתָם מִנּוֹעַר

  וּבְאַהֲבַתְכֶם אוֹתָם נַפְשָׁם תִּקָּחוּ

הֵם כְּלוּאִים בַּבַּיִת לְאוּטים בַּבֶּגֶד

יִנְעָם לְחִכָּם דְּבַשׁ יִתְעַדְּנוּ מִמֶּגֶד

  לָכֵן שָׁרְשָׁם יִבֹּל מִבְּלִי יִצְמָחוּ:


הֲלֹא תִרְאוּ וְתִקְחוּ מוּסָר מִבֶּן־כֹּפֶר

הִנֵּה הוּא עַז כַּנָּמָר קַל כָּעֹפֶר

  לֹא יֵרַךְ לְבָבוֹ מִסּוֹעָה וָסָעַר

הוּא כְּמוֹ עַמּוּד בַּרְזֶל בְּיוֹם קְרָב וָלַחַץ

הוּא יַךּ קָדְקֹד לוֹחֵם פֶּצַע וּמַחַץ

  וְלֹא יֵבוֹשׁ כִּי יְדַבְּרוּ אוֹיְבִים בַּשָׁעַר:


הִנֵּה עָבֵי שְׁחָקִים יִתְקַבְּצוּ יַחַד

נְשִׂיאִים הָרֵי רֶתֶת רַעַד וָפַחַד

  וְחֶשְׁכַּת מַיִם תַּחְשִׁיךְ אֶת הָרָקִיעַ

גָּפְרִית וָמֶלַח יִתְעַלֵּם עָלֵימוֹ

גַּם רוּחַ זִלְעָפוֹת יִסְעַר בְּקִרְבֵּימוֹ

  אֶרֶץ עֵפָתָה מִקַּדְרוּת תַּצִּיעַ:


וְיִגְבְּרוּ הָעֲנָנִים יַחְדָּו יִתְלַכָּדוּ

אִישׁ בְּאָחִיהוּ יִדְבַּק וְלֹא יִתְפָּרָדוּ

  שׁוֹאֲפֵי אֵדִים הָעֹלִים מִן הָאָרֶץ

עַד כִּי הָרוּחוֹת יַחְדָּו יִתְאַנָּפוּ

וּנְשִׂיאִים זֶה בָזֶה בְּחוֹזֶק יֶהְדָּפוּ

  וְיִבָּקְעוּ כְּנֵבֶל יוֹצֵר יִפְרְצוּ פָרֶץ:


עוֹד אֶרֶץ שֶׁקֶט וְשַׁאֲנָן עַל־פָּנֶיהָ

אַךְ קוֹל דְּמָמָה דַקָּה יַחְלִיף עָלֶיהָ

  הַמְבַשֵּׂר סוּפָה מֵי הַיָּם יִגְרָשׁוּ

כְּקוֹל אֶנְקַת חָלָל רוּחַ תִּתְיַפֵּחַ

יִשְׁרֹק לְשׁוֹאָה וּמְשׁוֹאָה מַר צֹרֵחַ

  יִרְגְּזוּן כָּל־עֲצֵי יַעַר חֲרָדוֹת יִלְבָּשׁוּ:


צִפֳּרִים רֶדֶת לָעֵמֶק יָחִישׁוּ

נָבֹכוּ עֶדְרֵי בָקָר דּוֹם יַחֲרִישׁוּ

  יִשְׂאוּ עֵינֵיהֶם שָׁמַיִם יַבִּיטוּ

נִשְׁכָּחִים מִנִּי רוֹעִים וְלֹא נֶאֱסָפוּ

נִסְתְּרוּ בְסֵתֶר הָהָר וְיָדֵיהֶם רָפוּ

  מִגַּעֲרַת אֱלֹהִים אֶת־נַפְשָׁם הִפְלִיטוּ:


חֶרְדַת אֱלֹהִים אִם עַיִן תֵּימָן תַּחַז

וּמִמִּפְלְשֵׂי עָב בָּרָק יְרוֹצֵץ בְּפַחַז

  דְּמוּת נָחָשׁ עֲקַלָּתוֹן יָפִיץ אוֹרֵהוּ

אַחֲרָיו יִשְׁאַג קוֹל מַר יִנְהַם הָרַעַם

כְּקוֹל רַעַשׁ אוֹפָן כְּמוֹ קוֹל הוֹלֶם פַּעַם

  אֲבָל עוֹד מֵרָחוֹק נִשְׁמַע גְּאוֹן קוֹלֵהוּ:


עַד הָלֹךְ וְקָרֵב הַנְּשִׂאִים יַקְרִיבוּ

עַל־כַּנְפֵיהֶן סְעָרוֹת אוֹתָם יַרְכִּיבוּ

  אָז מַרְאֵה הַבָּזָק יַרְחִיב שַׁלְהֶבֶת

וְקוֹל רַעַם בַּגַּלְגַּל יִתְחַזֵּק בּכֹחַ

וּמֵעַל־רֹאשׁ נַהֲרֵי אֵשׁ בְּאֵין מָנוֹחַ

  רָצוֹא וָשׁוֹב בְּזִקִּים בְּחִצֵּי לַהֶבֶת:


פַּעַם יִפָּתְחוּ פַּעַם יִסָּגֵרוּ

פַּעַם יְמַהֲרוּ לָצֵאת פַּעַם יְאַחֵרוּ

  עַד כְּמוֹ אֵשׁ תָּפְתֶּה יִבְעַר הָרָקִיעַ

וּבְתוֹכָם שַׁאֲגַת הֲמוֹן קוֹלוֹת נִשְׁמַעַת

קוֹל אַחֲרֵי קוֹל יְפוֹצֵץ בְּרוֹגֶז וָכַעַס

  יְרוֹפְפוּ מוֹסְדוֹת תֵּבֵל וְאֶרֶץ תָּרִיעַ:


וּבְתוֹכָם מִטְרַת עוֹז תֵּרֵד הֲמוֹן מָיִם

נִבְקְעוּ עֲנָנִים נִפְתְּחוּ אֲרֻבּוֹת שָׁמָיִם

  בָּרָד וְאַבְנֵי אֶלְגָּבִישׁ תּוֹךְ הַשֶּׁצֶף

בֵּינֵיהֶם הַבָּרָק בְּחֵמָה נִצַּחַת

יִלָּחֵם בַּבָּרָד בְּאֵשׁ מִתְלַקַּחַת

  אֵשׁ וּמַיִם יִתְגָּרוּ בַּחֲרוֹן קֶצֶף:


פֹּה אֵלָה עֲבֻתָּה כַּלַּפִּיד בֹּעֶרֶת

כִּי יָרַד רֶשֶׁף שִׁבֵּר הַצַּמֶּרֶת

  הוּרַד עֶדְיָהּ נֶהֶפְכָה לַגַּחֶלֶת

פֹּה מִקְנֵה צֹאן עַל־מַרְעִיתָם נִגָּפוּ

הָיוּ לְמַאֲכֹלֶת אֵשׁ פִּתְאֹם נִשְׂרָפוּ

  וּבֶן־הַבָּקָר אֵיךְ הָיָה לְמַפֶּלֶת:


פָּחֲדוּ חַטָּאִים חִיל וְרַעַד יֹאחֲזֵמוֹ

מִגַּעֲרַת יְיָ כִּי יַרְעִים עָלֵימוֹ

  זוּ נַחֲלַת רָשָׁע זוּ חֶלְקוֹ מִמָּעַל

יֵחַת מִכֹּל יֵרֶא חֶרֶב נֹקֶמֶת

בְּיוֹם שׁוֹאָה יֵחָבֵא בְּחוֹרֵי תִנְשֶׁמֶת

  עֲוֹן עֲקֵבוֹ יְסֻבֶּנּוּ מֶרְיוֹ וּמָעַל:


לֹא־כֵן יִשְׁרֵי לֶכֶת מֵטִיבֵי צַעַד

הַשְׁקֵט בִּלְבָבָם בְּאֵין יִרְאָה וָרַעַד

  לֹא בַעַל חֵמָה אֵל רַחֲמָן יֶחֱוָיוּ

בְּזֹהַר הָרָקִיעַ וּבְשַׂעַר קֶטֶב

יַבִּיטוּ אֱלוֹהַּ כִּי יָחוֹן וַיֵּיטֶב

  כִּסְלָם בֵּאלֹהִים בָּטְחוּ וַיִּשְׁלָיוּ:


* * *


עַתָּה הָעֲנָנִים לֹא נִרְאוּ לָעָיִן

חָלְפוּ נְשִׂאִים וְרוּחַ וָגֶשֶׁם אָיִן

  בָּהִיר בַּשְּׁחָקִים יִטְהַר הָרָקִיעַ

תָּגִיל הָאֲדָמָה וְצֶאֱצָאֶיהָ חָיוּ

יִרְעֲפוּ נְאוֹת דֶּשֶׁא מִגֶּשֶׁם יִרְוָיוּ

  וְרוּחַ צַח יִנְשֹׁף מְחַיֶּה וּמַשְׂבִּיעַ:


כֵּן חֲלִיפוֹת עֲלֵי אֲדָמָה תַּחַת הַשֶּׁמֶשׁ

מִנּוֹגַהּ מִמָּטָר מֵאוֹר מֵאֶמֶשׁ

  שְׂמָחוֹת אֲנָחוֹת שָׂשׂוֹן וְאָסוֹן יַחֲלֹפוּ

כַּאֲשֶׁר יַזְהִיר הַשַּׁחַק אַחֲרֵי רַעַם

כֵּן יָבֹא חֶסֶד אַחֲרֵי יַעֲבֹר זַעַם

  אַחֲרֵי אֵיד הֵד אַחֲרֵי גִיל חִיל יִרְדֹּפוּ:


הִנֵּה יָפוּחַ הַיּוֹם חוֹם אֵינֶנּוּ

הַשֶּׁמֶשׁ יָרַד סָר כֹּחוֹ מִמֶּנּוּ

  יָשׁוּב לְבֵיתוֹ כְּחָתָן יֵלֵךְ אֶל חֶדֶר

יִשְׂחַק לְתֵבֵל יִלְטֹשׁ עַיִן מִיָּמָּה

יִשְׂמַח לֶאֱחֹז בְּכַנְפֵי הָאָרֶץ שָׁמָּה

  יִפְרֹשׂ בֶּגֶד אַרְגָּמָן יִפְשֹׁט אֶדֶר:


אֲבָל עוֹד פַּעַם נִרְאוּ פָנָיו פְּנֵי לַהַב

טֶרֶם יִטְבַּע בַּיָּם יִצְלֹל בְּמֵי רַהַב

  פַּעַם מִבֵּין עֲרָפֶל פַּעַם מִטֹּהַר

עַד דְּמוּת חֲצִי אוֹפָן נִרְאֶה לָעֵינָיִם

וְאָז כְּמַרְאֵה הַחֶרְמֵשׁ מָלֵא קַרְנָיִם

  עוֹד מְעַט וְאֵינֶנּוּ וְאַחֲרָיו מַרְאֵה זֹהַר:


וְשׁוּלֵי מְעִילוֹ מִזָּהָב פָּז חֻטָּבוּ

עֲלֵי הָרֵי גַבְנוּנִים נָגְהָם יָהָבוּ

  אַחֲרֵי הִפָּרְדוֹ נִרְאֶה הוֹד מַרְאֵהוּ

כֵּן צְדָקָה לְחַיִּים אַחֲרֵי מוֹת בְּעָלֶיהָ

הִיא תָאִיר וְתוֹפִיעַ תָּפִיץ קַרְנֶיהָ

  אַחֲרֵי הִפָּרֵד תּוֹם נִשְׁאַר מַעֲשֵׂהוּ:


* * *


הֵן נָסוּ הַצְּלָלִים אָתָה הָעֶרֶב

הוּא יְדַשֵּׁן כָּל־נֶפֶשׁ וּשְׂמָחוֹת יֶרֶב

  יִצֹּק רוּחַ חֵן עַל־כָּל־חַי וְצוֹמֵחַ

עֲטוּפִים מֵחוֹם אֲמוּלִים מֵעֲבוֹדַת פֶּרֶךְ

עוֹרְקֵי צִיָּה נִתְעוּ בְתֹהוּ לֹא דֶרֶךְ

  לְכֻלָּם מִטּוּבוֹ אֶת־יָדוֹ פּוֹתֵחַ:


רוּחַ קָרָה יִנְשֹׁב מֵהָרֵי בֶשֶׂם

יִפְעַל לָנֶפֶשׁ עֲיֵפָה כִּקְסֹם קֶסֶם

  מִפָּנָיו תָּנוּס תּוּגָה יְגָרֵשׁ עֶצֶב

שֹׂבַע נְעִימוֹת יִשְׁכַּח אֶרֶץ מְגוּרֵהוּ

וְיָשִׂים אָדָם בֵּין כֹּכָבִים קִנֵּהוּ

  יִדְמֶה לִבְנֵי אֵלִים וְלֹא יִזְכֹּר קֶצֶב:


מַה־נָּעַמְתָּ לִי עֶרֶב חֶשְׁכַּת לָיִל

לְהִתְהַלֵּךְ בְּחֶבֶר רֵעִים עִם בְּנֵי חָיִל

  לְהִתְעַלֵּס לְהַמְתִּיק סוֹד עַל־הָרֵי נֶשֶׁף

עַד חֲצוֹת לַיְלָה בִּלְתִּי עָיֵף וְיָגֵעַ

עַד כִּי יֶחְשַׁךְ וְאֵין מַכִּיר אִישׁ אֶת־רֵעַ

  עַד עֵת הִרְגִּיעָה לִילִית בַּעֲלַת כֶּשֶׁף:


אוֹ לַעֲבֹר עַל־פְּנֵי הַנַּחַל בָּאֳנִי שַׁיִט

בְּקַל הַמֵּרוּץ כַּאֲשֶׁר יִדְאֶה הָעַיִט

  וְלָשֵׂאת קוֹל זִמְרָה עֲלֵי כִנּוֹר וָנֵבֶל

כְּאַבְנֵי אֶקְדָּח הַכֹּכָבִים עָלֵינוּ

וְקוֹל הֵד הָרִים יַעֲנֶה אַחֲרֵי קוֹלֵנוּ

  נִשְׂחַק לַעֲמֲל אֶרֶץ נִלְעַג לָאֵבֶל:


וְכַאֲשֶׁר יִנְעַם עֶרֶב יִמְתַּק כַּנֹּפֶת

אַחֲרֵי יוֹם חוֹרֶב יוֹם בֹּעֵר כְּאֵשׁ תֹּפֶת

  כֵּן יִנְעַם לָנֶפֶשׁ עֶרֶב חַיֵּינוּ

אַחֲרֵי עִצְּבוֹן יָדֵינוּ עַל־פַּת לֶחֶם

אַחֲרֵי עֲמָלֵנוּ מֵעֵת צֵאת מֵרֶחֶם

  נָגִיל כִּי יָבֹא שֶׁמֶשׁ עֵת קִצֵּנוּ:


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
