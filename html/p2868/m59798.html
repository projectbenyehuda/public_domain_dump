כָּתֵף וְנָטָף

וַיְהִי בְדוֹר הַמַּבּוּל כַּאֲשֶׁר יָרְדוּ בְנֵי הָאֱלֹהִים מִשָּׁמַיִם אָרְצָה וַיִּשְׁכְּחוּ מַהֵר כִּי מַלְאָכִים הֵם וַיַּשְׁחִיתוּ אֶת דַּרְכָּם וַיַּכְעִיסוּ אֶת ה'. וְרַק שְׁנֵי כְרוּבִים, כָּתֵף וְנָטָף, נִשְׁאֲרוּ תְמִימֵי דֶרֶךְ וַיַּעֲשׂוּ אֶת הַטוֹב וְאֶת הַיָּשָׁר בְּעֵינֵי אֱלֹהִים וְאָדָם. עֵת נֹחַ וּבָנָיו עוֹשִׂים מְלָאכָה בַּשָּׂדֶה, חוֹרְשִׁים וְזוֹרְעִים, הָיָה הַכְּרוּב כָּתֵף שׁוֹאֵב מַיִם מִן הַנְּחָלִים הַקְּרוֹבִים, מְבִיאָם עַל כְּתֵפָיו וּמוֹסֵר אוֹתָם לַכְּרוּב נָטָף לְהַשְׁקוֹת אֶת הָאֲדָמָה, וְנָטָף הוֹלֵךְ וּמְנַטֵּף טִפִּין טִפִּין עַל פְּנֵי כָל הַשָּׂדֶה, וַתְּהִי כָל טִפַּת מַיִם לִבְרָכָה בְּקֶרֶב הָאָרֶץ. וַיִּצְמְחוּ וַיִּגְדְּלוּ שִׁבֳּלֵי הַבָּר, הַיְרָקוֹת וַעֲצֵי הַפְּרִי הַשּׁוֹנִים הַזְּרוּעִים וְהַנְּטוּעִים בִּידֵי נֹחַ וּבָנָיו.

וְאַנְשֵׁי דוֹר הַמַּבּוּל הִשְׁחִיתוּ אֶת דַּרְכָּם מְאֹד וַיּוֹסִיפוּ לַעֲשׂוֹת הָרַע בְּעֵינֵי ה'. וַיַּרְא ה' וַיִּנְאֵץ וַיֹּאמַר: אֶמְחָה אֶת כָּל הַיְקוּם אֲשֶׁר עַל פְּנֵי הָאֲדָמָה, מֵאָדָם עַד בְּהֵמָה עַד רֶמֶשׂ וְעַד עוֹף הַשָּׁמָיִם, וְנִשְׁאֲרוּ אַךְ נֹחַ וּבָנָיו, וּמִכָּל הַחַי לְמִינֵהוּ זָכָר וּנְקֵבָה לְהַחֲיוֹת זֶרַע עַל פְּנֵי הָאָרֶץ.

וַיְהִי כַאֲשֶׁר קָרְבוּ יְמֵי הַמַּבּוּל וַיַּעַזְרוּ שְׁנֵי הַכְּרוּבִים, כָּתֵף וְנָטָף, לְנֹחַ וּלְבָנָיו לְקַבֵּץ אֶת כָּל הַבְּהֵמָה, הַחַיָּה וְהָעוֹף לְמִינֵיהֶם אֶל תּוֹךְ הַתֵּבָה וַיָּבִיאוּ אִתָּם מִכָּל מַאֲכָל אֲשֶׁר יֵאָכֵל וְגַם אֶת זְמוֹרוֹת הַגֶּפֶן לֹא שָׁכָחוּ, כִּי אָמְרוּ פֶּן יִתְעַצֵּב נֹחַ אֶל לִבּוֹ עַל כָּל הָרָעָה אֲשֶׁר בָּאָה וְשִׂמַּח אֶת לִבּוֹ בִּפְרִי הַגָּפֶן. וַיִּשְׁמְרוּ הַכְּרוּבִים עַל זְמוֹרוֹת הַגָּפֶן. וְכַאֲשֶׁר תַּמּוּ יְמֵי הַמַּבּוּל וְנֹחַ וּבָנָיו וְכָל הַחַי אֲשֶׁר אִתּוֹ יָצְאוּ מִן הַתֵּבָה, עָזְרוּ כָתֵף וְנָטָף לְנֹחַ לְטַעַת אֶת זְמוֹרוֹת הַגֶּפֶן אֲשֶׁר הָיוּ אִתָּם בַּתֵּבָה. וַיִּשְׁמְרוּ שְׁנֵי הַכְּרוּבִים עַל הַזְּמוֹרוֹת הַנְּטוּעוֹת וַתַּעֲשֶׂינָה אַשְׁכֹּלוֹת. וַתְּהִי הַגֶּפֶן אֲשֶׁר נָטַע נֹחַ בְּעֶזְרַת כָּתֵף עוֹשָׂה אַשְׁכְּלוֹת עֲנָבִים צְפוּפִים הַמֻּנָּחִים אִישׁ עַל כֶּתֶף רַעֵהוּ, וְהַגֶּפֶן הַנְטוּעָה בְּעֶזְרַת נָטָף עוֹשָׂה אַשְׁכְּלוֹת עֲנָבִים דְּלִילִים וְנוֹטְפִים; וַיִּקְרָא נֹחַ לְמִינֵי הָאַשְׁכֹּלוֹת כָּתֵף וְנָטָף עַל שֵׁם שְׁנֵי הַכְּרוּבִים.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
