הַסְּתָוָנִית וְהַנַּחְלִיאֵלִי

מִדֵי שָׁנָה בְשָׁנָה, בַּחֲלֹף הַקַּיִץ בְּאַרְצוֹת הַצָּפוֹן וּבְבוֹא הַחֹרֶף הָעַז, יִתְאַסְּפוּ הָעוֹפוֹת עֵדוֹת עֵדוֹת עַל מְנָת לַעֲזוֹב אֶת אֶרֶץ מְגוּרֵיהֶם וְלָנוּד לַמֶּרְחַקִּים, וְאִתָּם גַּם אֶפְרוֹחֵיהֶם אֲשֶׁר גָּדְלוּ בַקַּיִץ; וְכֻלָּם מְגַמָּתָם קָדִימָה לְאַרְצוֹת הַחֹם, אֲשֶׁר הָאָבִיב שָׁם יִשְׁכֹּן עוֹלָמִים.

גַּם הַנַּחְלִיאֵלִים מִדֵּי יַרְגִּישׁוּ בְאוֹתוֹת הַחֹרֶף יִתְאַחֲדוּ לְלַהֲקָה, בּוֹחֲרִים בְּצוֹפֶה, מוֹרֵה דֶרֶךְ, וּמִתְרוֹמְמוֹת וְעָפוֹת צִפֳּרֵי הַחֶמֶד וְתַעֲשֶׂינָה אֶת דַּרְכָּן בַּיּוֹם וּתְרַחֵפְנָה גַם בַּלֵּילוֹת לְאוֹר הַיָּרֵחַ. הַדֶּרֶךְ כְּבֵדָה עֲלֵיהֶן, אַךְ בִּגְבוּרָה וּבִתְבוּנָה תִּשֶּׂאנָה אֶת עֲמָלָן עַד בּוֹאָן לִמְחוֹז חֶפְצָן.

וַיְהִי הַיּוֹם וַיָּבוֹאוּ הַנַּחְלִיאֵלִים גַּם לְאֶרֶץ יִשְׂרָאֵל, וַתְּפוּצֶינָה הַצִּפֳּרִים בְּכָל הָאָרֶץ, וַתָּרֹצְנָה עַל פְּנֵי שָׂדוֹת וְכָרִים לְבַקֵּשׁ אַחַר מָזוֹן וְלִטְרָף טֶרֶף אֶת הַיִּתּוּשִׁים וְהַזְּבוּבִים הַשּׁוֹרְצִים סָבִיב. וְהָעוֹנָה הִיא רֵאשִׁית הַסְּתָיו, וְהָאָרֶץ יְבֵשָׁה וַחֲרֵבָה וְטֶרֶם נִרְאוּ הַפְּרָחִים הָעֲנֻגִּים, כִּי הָאֲדָמָה הַצְּמֵאָה מְחַכָּה לַיּוֹרֶה כִּי יַרְטִיבָהּ.

וַיִּתְגַּעְגְּעוּ הַנַּחְלִיאֵלִים מְאֹד אֶל הַפְּרָחִים וְהָעֲשָׂבִים הָרַכִּים, כִּי הַסְכֵּן הִסְכִּינוּ לָהֶם בְּאֶרֶץ מוֹלַדְתָּם הַחֲבִיבָה אֲשֶׁר עֲזָבוּהָ. וַיְנַעַנְעוּ הַנַּחְלִיאֵלִים וַיּוֹרִידוּ וַיָּרִימוּ לִרְגָעִים אֶת זְנָבָם הָאָרֹךְ, וַיָּעִירוּ בְקוֹל צִפְצוּפָם אֶת הַבְּצָלִים וְהַפְּקָעוֹת הַטְּמוּנִים בְּחֵיק הָאֲדָמָה, וַיֹּאמְרוּ לָהֶם: צְ, אִי, צְ, אִי. וּבְצַל הַסְּתָוָנִית, עָטוּף בִּקְרוּמִים חוּמִים וְטָמוּן בְּבֶטֶן הָאֲדָמָה, שָׁמַע הָרִאשׁוֹן אֶת קוֹל הַנַּחְלִיאֵלִים הַמְעוֹרֵר, וַיִּבְקַע אֶת קְרוּם הַקַּרְקַע, וַיּוֹשֶׁט אֶת גִּבְעוֹלוֹ לָאֲוִיר.

וּבִרְאוֹת הַנַּחְלִיאֵלִים אֶת גִּבְעוֹל הַסְּתָוָנִית, קָרְאוּ בְיֶתֶר עָז: צְ, אִי, צְ, אִי.

וְהַסְּתָוָנִית נִמְשְׁכָה אַחַר קוֹל הַנַּחְלִיאֵלִים, וַתְּמַהֵר וַתּוֹצִיא שְׁמוֹנָה פְרָחִים עִם שִׁשָּׁה עֲלֵי כוֹתֶרֶת וְרֻדִּים לְכָל פָּרַח, וּבְתוֹכָם שִׁשָּׁה אַבְקָנִים עִם מַאֲבָקִים צְהֻבִּים הַמְחֻבָּרִים אֶל רֹאשׁ הַזֵּר בְּאֶמְצְעִיתָם וּמִתְנוֹעַעִים הֵנָּה וָהֵנָּה כִּזְנַב הַנַּחְלִיאֵלִים אֲשֶׁר קָרְאוּ לָהּ.

מֵאָז, עִם שׁוּב הַנַּחְלִיאֵלִים מֵאַרְצוֹת הַצָּפוֹן, מְמַהֶרֶת גַּם הַסְּתָוָנִית לְהוֹפִיעַ; וּשְׁנֵיהֶם מְבַשְּׂרִים אֶת קִרְבַת הַגְּשָׁמִים.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
