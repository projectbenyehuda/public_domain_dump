מִמִּשְׁפַּחַת הַוְּרָדִים Rosaceae

עֵץ פְּרִי הַגָּדֵל בַּר בֶּהָרִים. עוֹמֵד בַּשַׁלֶכֶת, מַגִּיעַ לְקוֹמָה שֶׁל 10 מֶטֶר. עָלָיו צָרִים, אִזְמְלִיִּים, מְשֻׁנָּנִים בְּשׁוּלֵיהֶם. פֵּרוֹתָיו מְלֵאִים תָּאֵי-אָבֶן. הוּא מְשַׁמֵּשׁ לְכַנָּה לָאַגָּס הַתַּרְבּוּתִי. בְּשׁוּקֵי צִפּוֹרִי הָיוּ מַרְכִּיבִין קְרֻסְתּוֹמְלִין עַל גַּבֵּי אַגָּס (ירוש', כלאים פ"א דף כ’ז). הַקְּרֻסְתּוֹמֵל [אוֹ: קְרֻסְטְמִיל] הוּא מִין אַגָּס תַּרְבּוּתִי.

אַזוֹב Origanum Maru, L

מִמִּשְׁפַּחַת הַשִּׂפְתָּנִים Labiatae

שִׂיחַ הַנּוֹתֵן רֵיחַ טוֹב. נָפוֹץ בְּאֵזוֹר הֶהָרִים. מֻזְכָּר בַּמִּקְרָא בִּמְקוֹמוֹת אֲחָדִים, כְּצֶמַח שֶׁהָיוּ מַזִּים בּוֹ וּמְטַהֲרִים אֶת הַטָּמֵא.

אֵלָה Pistacia Palaestina, Boiss.

מִמִּשְׁפַּחַת הַבָּטְנֶה Anacardiaceae

הָאֵלָה הִנֶּהָ עֵץ בָּר, נְפוֹצָה בְאֵזוֹר הֶהָרִים. בְּחֶבְרוֹן מְגַדְּלִים אוֹתָהּ בְּאֹפֶן תַּרְבּוּתִי. הָאֵלָה מַגִּיעָה לְקוֹמָה שֶׁל שְׁמוֹנָה מֶטֶר וְיוֹתֵר. הִנֶּהָ עוֹמֶדֶת בַּשַּׁלֶּכֶת. הַפֵּרוֹת זְעִירִים, מְסֻדָּרִים בְּאַשְׁכּוֹלוֹת וְצִבְעָם כְּחַלְחַל-יָרֹק. הִנָּם נֶאֱכָלִים בִּהְיוֹתָם קְלוּיִים וְגַם עוֹצְרִים מֵהֶם שָׁמֶן.

בִּימֵי הַמִּקְרָא הָיְתָה הָאֵלָה עֵץ קָדוֹשׁ שֶׁתַּחַת צֵל צַמַּרְתּוֹ הָעֲבֻתָּה הָיוּ מִתְאַסְּפִים אַנְשֵׁי הַכְּפָר וּמְבִיאִים אֶת קָרְבְּנוֹתֵיהֶם אוֹ דָנִים עַל עִנְיְנֵיהֶם.

תַּחַת עֵץ הָאֵלָה מָצָא אַבְשָׁלוֹם אֶת מוֹתוֹ הַטְּרָגִי: “וְאַבְשָׁלוֹם רָכַב עַל הַפֶּרֶד וַיָּבֹא הַפֶּרֶד תַּחַת שׂוֹבֶךְ הָאֵלָה הַגְּדוֹלָה וַיֶּחֱזַק ראשׁוֹ בָאֵלָה וַיֻּתַּן בֵּין הַשָּׁמַיִם וּבֵין הָאָרֶץ; וְהַפֶּרֶד אֲשֶׁר תַּחְתָּיו עָבָר” (שמואל ב', י"ח, ט').

אֵם הַחִטָּה Triticum dicoccum, Heäk, var. dicoccoides, Körnicke.

מִמִּשְׁפַּחַת הַדְּגָנִים Gramineae

אֵם-הַחִטָּה הִנֶּהָ אֶחָד מִשְּׂרִידֵי הַטֶּבַע הַחֲשׁוּבִים וְהַיְקָרִים שֶׁבְּאַרְצֵנוּ. חוֹבָה הִיא לְהַכִּיר אוֹתָהּ, לִבְלִי תְלֹשׁ אוֹתָהּ וְלִשְׁמוֹר עַל מְקוֹם גִּדּוּלָהּ. הַסִימָן הַמֻּבְהָק הַמַּבְדִּיל בֵּין אֵם-הַחִטָּה וּבְנוֹתֶיהָ הַמִּינִים הַתַּרְבּוּתִיִּים הוּא: הַשֶּׁדְרָה שֶׁל הַשִּׁבֹּלֶת אֵצֶל הָרִאשׁוֹנָה נִפְרֶכֶת יַחַד עִם הַמְּלִילָה; הַמַּלְעָנִים מַגִּיעִים לְגֹדֶל שֶׁל 15–14 ס"מ וְכָל הַקֶּלַח לְגֹבַהּ 100–60 ס"מ.

גְּדֵלָה הִיא אֵם-הַחִטָּה בִּמְקוֹמוֹת בִּלְתִּי מְעֻבָּדִים בֵּין סְלָעִים, שִׂיחִים וְקוֹצִים. יֵשׁ שֶׁהַשִּׁבֹּלֶת כֻּלָּהּ שְׁחוֹרָה, אוֹ שֶׁרַק הַגְּלוּמִים אוֹ חֵלֶק מֵהֶם שָׁחוֹר, וְיֵשׁ שֶׁרַק הַמַּלְעָנִים שְׁחוֹרִים. הַגְּלוּמִים יֵשׁ שֶׁהֵם קַרְחִים וְיֵשׁ שֶׁהִנָּם שְׂעִירִים מְאֹד.

הָרִאשׁוֹן שֶׁמָצָא אֶת אֵם-הַחִטָּה בִּמְקוֹם גִּדּוּלָהּ בִּסְבִיבַת רֹאשׁ-פִּנָּה וְהַחֶרְמוֹן וְקָבַע אוֹתָהּ בְּאֹפֶן מְדֻיָּק הָיָה הַמָּנוֹחַ א. אַהֲרָנְסוֹן, אֶחָד מִבְּנֵי הָאִכָּרִים, מְיַסְדֵי הַמּוֹשָׁבָה “זִכְרוֹן יַעֲקֹב”. נְפוֹצָה גַּם בִּסְבִיבוֹת מְנַחֶמְיָה וּמִגְדָּל.

אֹרֶן (אֹרֶן יְרוּשָׁלַיִם) Pinus Hale-pensis, Mill

מִמִּשְׁפַּחַת הָאֲרָזִים Coniferae

עֵץ יַעַר יָפֶה וְיָרֹק תָּדִיר. מְגַדְּלִים אוֹתוֹ בְּכָל הָאָרֶץ. מֻזְכָּר בַּמִּקְרָא “נָטַע אֹרֶן וְגֶשֶׁם יְגַדֵּל” (ישעיה מ"ד, י"ד). לְפָנִים הָיָה נָפוֹץ מְאֹד בָּאָרֶץ, כִּי אֲבוֹתֵינוּ הִשְׁתַּמְשׁוּ בוֹ גַּם לַמַּעֲרָכָה.

בֵּין פְּתּוּחֵי הַפֵּרוֹת וְהַפְּרָחִים עַל מִכְסֵה אֲרוֹן הַמַּתִים שֶׁמָּצְאוּ בְקִבְרֵי הֵילֵנֵי הַמַּלְכָּה נִמְצָא גַם פִּתּוּחַ שֶׁל שְׁלֹשָׁה אִצְטְרוּבָלִים הַדּוֹמִים לְאִצְטְרוּבְלֵי אֹרֶן יְרוּשָׁלָיִם.

בַּבֹּנֶג בַּעַל לֵב הַזָּהָב Matricaria aurea, L.

מִמִּשְׁפַּחַת הַמֻּרְכָּבִים Compositae

מָצוּי בְּכָל הָאָרֶץ, בַּמִּדְבָּר וְגַם בַּקִירוֹת וּבַחוֹמוֹת. נִבְדָּל הוּא מֵאָחִיו הַבַּבֹּנֶג בַּעַל הַתִּתּוֹרָה הַלְבָנָה

Matricaria chamomilla בְּזֶה, שֶׁחֲסֵרִים לוֹ עַלְעֲלֵי תִתּוֹרָה לְבָנִים; הַפְּרָחִים מִצֶּבַע זָהָב מְסֻדָּרִים עַל מַצְעִית מוּרֶמֶת בְּצוּרַת אִצְטְוָנָה וַחֲלוּלָה מִבִּפְנִים.

הַפְּרָחִים שֶׁל שְׁנֵי הַמִּינִים נוֹתְנִים רֵיחַ נִיחוֹחַ וּמְכִילִים שֶׁמֶן אֶתְרִי מִצֶּבַע כָּחֹל כֵּהֶה הַמְשַׁמֵּשׁ לִרְפוּאָה לְקַדַּחַת, לְמַחֲלַת הָאִסְתּוֹמְכָה, לְשִׁעוּל וְלִשְׁאָר מַחֲלוֹת. כְּצֶמַח-רְפוּאָה יָדוּעַ הָיָה הַבַּבֹּנֶג בַּעַל הַתִּתּוֹרָה הַלְבָנָה עוֹד לְחַרְטֻמֵּי מִצְרָיִם. הַשֵּׁם הַמַּדָּעִי Matricaria מְקוֹרוֹ בִיוָנִית Mater – אֵם, עַל שֵׁם שֶׁהִשְׁתַּמְשׁוּ בוֹ הָאִמָּהוֹת לִרְפוּאָה.

בִּלְבִּיל Pycnonotus Xanthopygus

מִמִּשְׁפַּחַת קִצְרֵי-הָרַגְלָיִם Brachypodidae

הַבִּלְבִּיל הוּא “הַזָּמִיר בֶּן אֶרֶץ יִשְׂרָאֵל”, הַלּוֹקֵחַ לֵב כָּל שׁוֹמְעָיו בְּנֹעַם זְמִירוֹתָיו. נִקְרָא כַךְ עַל צַפְצְפוֹ בְּקוֹל רָם: בִּיל-בִּיל-בִּיל. מָצוּי בְּכָל הָאָרֶץ. מַאֲכָלוֹ הֵם פֵּרוֹת הַתְּאֵנָה, הַשִּׁקְמָה, הַגֶּפֶן, הַתּוּת וְיֶתֶר עֲצֵי פֶרִי.

בְּצַל הַנָּחָשׁ Scilla autumnalis, L.

מִמִּשְׁפַּחַת הַחֲבַצֶּלֶת Liliaceae

בְּצַל הַנָּחָשׁ הִנּוֹ חָסִית (נִרְדָּם בְּמֶשֶׁךְ הַקַיִץ בְּתוֹךְ הָאֲדָמָה), צֶמַח יָפֶה מְאֹד, בַּעַל 20–6 פְּרָחִים תְּכֻלִּים. צוּרָתָם כּוֹכָב וְהִנָּם מְסֻדָּרִים בְּאֶשְׁכּוֹל. אֹרֶךְ הַבָּצָל 20–10 ס"מ וְהִנּוֹ אַרְסִי. פּוֹרֵחַ בְּצַל-הַנָּחָשׁ בְּחָדְשֵׁי תִשְׁרֵי וּמַרְחֶשְׁוָן. הֶעָלִים מְלַבְלְבִים לְאַחַר שֶׁנָּבְלוּ הַפְּרָחִים. נָפוֹץ בְּכָל הָאָרֶץ.

בַּר יוֹרֶה Colchicum Steveni,Kunth

מִמִּשְׁפַּחַת הַחֲבַצָּלֶת Liliaceae

בַּר-יוֹרֶה הִנּוֹ חָסִית (נִרְדָּם בְּמֶשֶׁךְ הַקַּיִץ בְּתוֹךְ הָאֲדָמָה). בְּחֹדֶשׁ מַרְחֶשְׁוָן מְלַבְלְבִים 7–5 עָלִים וְאָז גַּם מוֹפִיעִים הַפְּרָחִים. מִסְפַּר הַפְּרָחִים 10–3. צִבְעָם לְבַנְבַּן אוֹ וָרֹד. הַבָּצָל וְגַם הַזְּרָעִים אַרְסִיִּים.

גֶּפֶן Vitis vinifera, L.

מִמִּשְׁפַּחַת הַגָּפֶן Ampelideae

הַגֶּפֶן נִמְנֵית בֵּין שִׁבְעַת מִינֵי הַצְּמָחִים שֶׁאֶרֶץ יִשְׂרָאֵל נִשְׁתַּבְּחָה בָהֶם “אֶרֶץ חִטָה וּשְׂעוֹרָה וְגֶפֶן” וגו' (דברים ח', ח'). אַחֵר הַמַּבּוּל נָטַע נֹחַ, אִישׁ הָאֲדָמָה, גֶפֶן, סָמוּךְ לְהָרֵי אֲרָרָט. “וַיָּחֶל נֹחַ אִישׁ הָאֲדָמָה וַיִּטַע כָּרֶם” (בראשית ט', כ').

הָעִבְרִים גִּדְלוּ אֶת הַגֶּפֶן בְּשָׁלֹשׁ צוּרוֹת שונות:

א) גֶּפֶן סֹרַחַת – רַגְלִיָּה (בִּלְשׁוֹן הַמִּשְׁנָה);

ב) גֶּפֶן אַדֶּרֶת – עָרִיס (בִּלְשׁוֹן הַמִּשְׁנָה);

ג) זְמוֹרָה – דָּלִיָּה.

בְּשֵׁם גֶּפֶן סֹרַחַת קוֹרְאִים לְאוֹתָהּ הַגֶּפֶן הַמְּתוּחָה עַל הָאֲדָמָה וְהַמִּתְפַּשֶּׁטֶת לָאֹרֶךְ וְלָרֹחַב.

בְּשֵׁם גֶּפֶן אַדֶּרֶת קוֹרְאִים לְאוֹתָהּ הַגֶּפֶן, אֲשֶׁר מוּרָמָה הִיא וְנִשְׁעֶנֶת עַל קִיר, קוֹרוֹת, קָנִים וְעוֹד וְאֶפְשָׁר לְהַבְדִּיל בָּהּ חֶלְקֵי גֶזַע וַעֲנָפִים.

זְמוֹרָה (דָּלִיָּה) נִקְרַאת הַגֶּפֶן שֶׁמַּדְלִים אוֹתָהּ עַל אִילָנֵי סְרָק אוֹ עַל עֲצֵי פֶרִי.

אַשְׁכְּלוֹת הָעֲנָבִים מִסְתָּעֲפִים בְּרָאשֵׁיהֶם לְאַשְׁכֹּלוֹת קְטַנִּים (פְּסִיגִים). חֶלְקָם זֶה נִקְרָא “כָתֵף”. בְּחֶלְקָם הַתַּחְתּוֹן מְחֻבָּרִים הָעֲנָבִים יָשָׁר לַשֶּׁדְרָה. חֵלֶק אֶשְׁכּוֹל זֶה נִקְרָא “נָטָף”.

גֶּפֶן שָׂדֶה (חַנְטָל) Citrullus colo-cynthis, L.

מֻזְכָּר בַּמִּקְרָא: “וַיֵּצֵא אֶחָד אֶל הַשָּׂדֶה לִלְקֹט אוֹרוֹת, וַיִּמְצָא גֶפֶן שָׂדֶה, וַיִּלְקֹט מִמֶּנּוּ פַּקְוּעוֹת שָׂדֶה” (מלכים ב' ד', ל"ט) צֶמַח זֶה נָפוֹץ בַּמִּדְבָּר, בַּחוֹלָה, בָּאֲדָמוֹת הַקַלּוֹת בַּשָּׁרוֹן וּבַשְּׁפֵלָה וְגַם בִּסְבִיבוֹת יָם הַמֶּלַח. נוֹתֵן בְּמֶשֶׁךְ כָּל הַקַיִץ פֵּרוֹת עֲגֻלִּים וּמָרִים הַמְשַׁמְּשִׁים לִרְפוּאָה. גֹּדֶל הַפְּרִי הִנּוֹ 10 ס"מ בַּקֹּטֶב. הַגִּבְעוֹלִים מִתְפַּשְּׁטִים עַל פְּנֵי הָאֲדָמָה, כְּגִבְעוֹלֵי הָאֲבַטִיחַ. הֶעָלִים צְפוּפִים וּמְלֵאֵי מִיץ. שָׁרָשיו אֲרֻכִּים וְאוֹגְרִים בְּתוֹכָם מַיִם בְּעוֹנַת הַגְשָׁמִים, מַה שֶּׁנּוֹתֵן לַצֶּמַח אֶת הַיְכֹלֶת לִגְדּוֹל בַּקַּיִץ הַחַם וְהַיָּבֵשׁ, כִּי הֶעָלִים אֵינָם מוּגֵנִּים בִּפְנֵי אִיּוּד רַב וְנוֹבְלִים מִיָּד אַחֲרֵי הִתָּלְשָׁם.

דְּגָנִיָּה אֶרֶץ יִשְׂרְאֵלִית Centaurea cyanoides, Berggr. et Wahl.

מִמִּשְׁפַּחַת הַמֻּרְכָּבִים Compositae

דּוּדָאִים Mandragora officina-rum, L.

מִמִּשְׁפַּחַת הַיַּבְרוּחִים Solanaceae

צֶמַח זֶה גָּדֵל בְּשָׂדוֹת עֲזוּבִים, עוֹנַת גִּדּוּלוֹ הִיא בִימוֹת הַקֹּר. הִנּוֹ פוֹרֵחַ בְּכִסְלֵו. גִּבְעוֹלוֹ אֵינוֹ מִתְרוֹמֵם מֵעַל פְּנֵי הָאֲדָמָה וְהַפְּרָחִים מִצְטַפְּפִים בְּמִסְפָּר רָב, לְמַטָה בְּסֵתֶר הֶעָלִים הַגְּדוֹלִים וְהַמְקֻמָּטִים, מַה שֶּׁנּוֹתֵן לָהֶם אֶת הָאֶפְשָׁרוּת לָשֵׂאת אֶת הָרוּחוֹת הַקָּרוֹת. הַפֵּרוֹת בְּשֵׁלִים בִּימֵי “קְצִיר חִטִּים”; הִנָּם בְּצוּרַת עֲנָבוֹת אֲדֻמּוֹת כְּעֵין עַגְבָנִיּוֹת קְטַנּוֹת וּמְפִיצִים רֵיחַ חָרִיף. הַפַּלָּחִים אוֹכְלִים אֶת צִיפַת הַפֶּרִי, אוּלָם נִזְהָרִים מִן הַגַּרְעִינִים, כִּי הֵם מַזִיקִים מְאֹד.

שָׁרְשֵׁי הַדּוּדָאִים דּוֹמִים בְּמִקְצָת לְצוּרַת צֶלֶם אָדָם וּמִפְּנֵי כַךְ נוֹצְרוּ אֵצֶל הָעַמִּים אַגָּדוֹת שׁוֹנוֹת. בְּעִקָר יִחֲסוּ לָהֶם בַּיָּמִים הַקַּדְמוֹנִים סְגֻלָּה לְעוֹרֵר אַהֲבָה וְלָתֵת פְּרִי בֶטֶן לַעֲקָרוֹת.

מֻזְכָּר בַּמִּקְרָא: “וַיֵּלֶךְ רָאוּבֵן בִּימֵי קְצִיר חִטִּים, וַיִּמְצָא דוּדָאִים בַּשָּׂדֶה, וַיָּבֵא אֹתָם אֶל לֵאָה אִמּוֹ. וַתֹּאמֶר רָחֵל אֶל לֵאָה: תְּנִי נָא לִי מִדּוּדָאֵי בְנֵךְ” (בראשית ל', י"ד).

דַּפְנָה Laurus nobilis, L.

מִמִּשְׁפַּחַת הַדַּפְנָה Laurineae

עֵץ יַעַר, נוֹתֵן רֵיחַ, מַגִּיעַ לְקוֹמָה שֶׁל 6 מֶטֶר. נִפְגָּשׁ בַּר בָּחֳרָשׁוֹת עַל הַכַּרְמֶל, בִּסְבִיבוֹת צְפַת וּמְקוֹמוֹת אֲחֵרִים. מִפּרוֹתָיו עוֹצְרִים שֶׁמֶן לִרְפוּאָה וְעָלָיו מְשַׁמְשִׁים לְתַבְלִין. הַיְּוָנִים וְהָרוֹמָאִים הַקַּדְמוֹנִים הָיוּ מִשְׁתַּמְּשִׁים בְּעַנְפֵי הַדַּפְנָה לִתְהִלָּה וּלְתִפְאֶרֶת. בְּיִחוּד הָיוּ עוֹנְדִים אֶת הַמְשׁוֹרְרִים הַמְהֻלָּלִים בְּזֵרֵי דַפְנָה.

דַּרְדַּר Centaurea pallescens, Del

מִמִּשְׁפַּחַת הַמֻּרְכָּבִים Compositae

צֶמַח דּוּ-שְׁנָתִי נָפוֹץ בְּכָל הָאָרֶץ. נוֹבֵט מִיָּד אַחַר הַגְּשָׁמִים הָרִאשׁוֹנִים. הֶעָלִים מְסֻדָּרִים דּוּרִים-דּוּרִים, הֵילְכָךְ שְׁמוֹ “דַּרְדַּר”. בַּחֹרֶף, בְּעוֹדָם רַכִּים, נוֹהֲגִים הַפַּלָּחִים לֶאֱכוֹל אֶת עוֹרְקֵיהֶם. לְטַרְפֵי הֶעָלִים טַעַם מָר; וְהַיְהוּדִים הַמִּזְרָחִיִּים מִשְׁתַּמְשִׁים בָּהֶם לְמָרוֹר בַּפָּסַח. בְּסוֹף אֲדָר מוֹפִיעוֹת בְּמֶרְכֵּן דּוּרֵי הֶעָלִים קַשְׂקַשִּׂים קָשׁוֹת וְקוֹצָנִיּוֹת בְּצוּרַת שׁוֹשָׁנֶת. בְּחֹדֶשׁ נִיסָן גָּדֵל מִתּוֹךְ הַשּׁוֹשֶׁנֶת גִּבְעוֹל קוֹצָנִי עִם קַרְקְפוֹת פְּרָחִים כְּתֻמִים.

הֲדַס Myrtus communis, L.

מִמִּשְׁפַּחַת הַהֲדַסִּים Myrtaceae

שִׂיחַ בַּר נוֹתֵן רֵיחַ. מַגִּיעַ לְקוֹמָה שֶׁל מֶטֶר וָחֵצִי. נָפוֹץ לָרֹב בִּסְבִיבוֹת צְפַת. מֻזְכָּר בַּמִּקְרָא: "אֶתֵּן בַּמִּדְבָּר אֶרֶז, שִׁטָּה וַהֲדַס וְעֵץ שָׁמֶן (ישעיה מ"א, י"ט); “צְאוּ הָהָר וְהָבִיאוּ עֲלֵי זַיִת וַעֲלֵי עֵץ שֶׁמֶן וַעֲלַי הֲדַס וַעֲלֵי תְמָרִים וַעֲלֵי עֵץ עָבֹת לַעֲשׂוֹת סֻכּוֹת” (נחמיה ח', ט"ו).

עֲלֵי הַהֲדַס – תְּאוֹמִים, עֲרוּכִים אִישׁ מוּל רֵעֵהוּ; וְיֵשׁ שֶׁשְּׁלֹשָׁה עָלִים מְסֻדָּרִים בְּדוּר אֶחָד אוֹ בְקֵן אֶחָד (בִּלְשׁוֹן הַתַּלְמוּד). סֵדֶר עָלִים זֶה הִנּוֹ מְיֻחָד לַעֲנָפִים צְעִירִים, בְּנֵי שָׁנָה, הַיּוֹצְאִים יָשָׁר מִן הָאֲדָמָה. עֲנָפִים אֵלֶּה נִקְרָאִים בִּלְשׁוֹן הַמִּקְרָא “עַנְפֵי עֵץ עָבֹת”, וְהֵם כְּשֵׁרִים לְלוּלָב. פִּרְחֵי הַהֲדַס לְבָנִים, עֲדִינִים וְנוֹתְנִים רֵיחַ נָעִים. הַפֵּרוֹת הֵם עֲנָבוֹת מִצְבַע כָּחֹל-שָׁחוֹר.

זֵיתָה Scrophularia deserti Del.

מִמִּשְׁפַּחַת Scrophulariaceae

צֶמַח רַבְשְׁנָתִי הַמַּגִּיעַ לְקוֹמָה שֶׁל 50 ס"מ. נִפְגָּשׁ בַּמִּדְבָּר וְגַם בְּחוֹלָה. הַפְּרָחִים זְעִירִים מִצֶבַע אָדֹם-חוּם. עוֹנַת הַפְּרִיחָה בְּיַרְחֵי שְׁבָט-אֲדָר.

זְקַן הַשָּׂב Geropogon glabrum L.

מִמִּשְׁפַּחַת הַמֻּרְכָּבִים Compositae

פּוֹרֵחַ וּמַבְשִׁיל אֶת פֵּרוֹתָיו בְּיַרְחֵי שְׁבָט, אֲדָר, נִיסָן וְאִיָר. עָלָיו מֻרְכָּבִים וְעַלְעֲלֵיהֶם מִשֻׁנָּנִים. עֲנָפָיו קְצָרִים, מִסְתָּעֲפִים מְאֹד וּמְפֻצָלִים בְּרָאשֵׁיהֶם, כְּעֵין מַלְגֵּז, לְשִׂכּוֹת קְצָרוֹת וְקָשׁוֹת. הַתִּפְרַחַת – שִׁבֹּלֶת, פִּרְחֵי הָעֱלִי עֲרוּכִים בָּהּ לְמָעְלָה; וּפִרְחֵי הָאַבְקָנִים וְהָאַנְדְּרוֹגִינוֹסִים לְמָטָּה. הַפְּרָחִים מְחֻסְּרֵי כֹתֶרֶת, אַרְבַּעַת עַלְעֲלֵי הַגָּבִיעַ מְחֻבָּרִים לְמַטָּה כְּעֵין כּוֹס וּמָפְרָדִים לְמָעְלָה. הָאַבְקָנִים בַּעֲלֵי זִירִים אֲרֻכִּים וּמְחֻבָּרִים לַגָּבִיעַ. הַצַּלֶקֶת שֶׁל הָעֱלִי הַיָּחִיד מְפֻצֶּלֶת כְּעֵין מִבְרֶשֶׁת לַהֲמוֹן נִימִין אֲדֻמּוֹת. הַשַּׁחְלָה אַחַת, בַּעֲלַת שְׁתַּיִם שָׁלֹשׁ מְגוּרוֹת. בְּכָל מְגוּרָה זֶרַע אֶחָד. הַפְּרִי כַּדּוּר אָדֹם וּבְראשׁוֹ נִשְׁאָרִים שְׂרִידִים מֵעַלְעֲלֵי הַגָּבִיעַ כְּמוֹ אֵצֶל הַתַּפּוּחַ בֶּן-מִשְׁפַּחְתּוֹ. הָעֲרָבִים מַסִּיקִים בַּסִּירָה אֶת תַּנּוּרֵיהֶם וְכִבְשְׁנוֹתֵיהֶם וְשׁוֹפְתִים עֲלֵיהֶם אֶת סִירָם. מְכִינִים מֵהֶם גַּם מַטְאַטְאִים.

עֵץ פְּרִי בַּעַל צַמֶּרֶת רְחָבָה הַנּוֹתֶנֶת צַל רָב. מֻזְכָּר בַּמִּקְרָא; אֶחָד מִשִּׁבְעַת הַמִּינִים שֶׁנִּשְׁתַּבְּחָה בָהֶם הָאָרֶץ. הַתְּאֵנָה הִנֶּהָ אֶחָד מֵעֲצֵי הַמַּאֲכָל הַקַּדְמוֹנִים שֶׁבַּתַּרְבּוּת הַחַקְלָאִית בָּאָרֶץ. פְּרִי הַתְּאֵנָה מָתוֹק, מֵזִין וּמַבְרִיא. כְּמָזוֹן לָאָדָם יָכוֹל הוּא לְשַׁמֵּשׁ בִּמְקוֹם הַלֶּחֶם 3 ל-1. פְּרִי הַתְּאֵנָה שׁוֹנֶה מִפֵּרוֹת אֲחֵרִים. הוּא אֵינוֹ אֶלָּא תִפְרַחַת פְּרָחִים שָׁמַצָּעִיתָהּ תָּפְחָה בְצוּרָה כַדּוּרִית וְנֶהֶפְכָה לְכִפָּה עֲגֻלָּה כְּעֵין פְּרִי הָאַגָּס, עִם פֶּה קָטֹן בְּרֹאשׁוֹ, מְכֻסֶה בְקַשְׂקַשִּׂים. הַצִיפָה הִיא הַמַּצָּעִית שֶׁל הַתִּפְרָחַת, בִּפְנִים הַפְּרִי עַל הַמַּצָּעִית רְפוּדִים פְּרָחִים זְעִירִים בַּעֲלֵי תְקִיף מְחֻמָּשׁ. לְפִרְחֵי הַזָּכָר 5–1 אַבְקָנִים. לְפִרְחֵי הַנְקֵבָה עֱלִי אָרֹךְ בַּעַל שְׁתֵּי צַלָּקוֹת וְשַׁחֲלָה בַּעֲלַת קֵן אֶחָד אוֹ שְׁנָיִם. לַפְּרָחִים הַסָּרִיסִים עֱלִי קָצָר.

רֹב עֲצֵי הַתְּאֵנָה שֶׁגָּדְלוּ עַל יְדֵי יִחוּרִים הִנָּם עֲצֵי נְקֵבָה הַנּוֹשְׁאִים פִּרְחֵי עֱלִי אוֹ פְרָחִים סָרִיסִים. לַעֲצֵי הַזָּכָר פְּרָחִים אַבְקָנִיִּים וּפְרָחִים סָרִיסִים.

בְּפֵרוֹת עֵץ הַזָּכָר הַנִּקְרָאִים בַּמִּשְׁנָה “שִׁיתִין” מְקַנֵּן יַבְחוּש Blastophaga grossorum, הַמַּפְרֶה אֶת פִּרְחֵי הַנְקֵבָה בְּסִיג שֶׁמֵבִיא אִתּוֹ מִפִּרְחֵי הַזָּכָר.

הָאִכָּרִים הָעִבְרִים הַקָּדְמוֹנִים הָיוּ תוֹלִים פֵּרוֹת מֵעֲצֵי זָכָר בְּמַחֲרֹזוֹת עַל עֲצֵי נְקֵבָה, כְּדֵי לְהָקֵל עַל הַהַפְרָאָה. לִפְעֻלָּה זוֹ הָיוּ קוֹרְאִים “גִּמּוּז”.

בְּמַהֲלַךְ הַזְּמָן הִסְתַּגְּלָה הַתְּאֵנָה הַתַּרְבּוּתִית לְהַבְשִׁיל פֵּרוֹתֶיהָ גַּם מִבִּלְעֲדֵי הָאֲבָקָה שֶׁל הַזָּכָר.

רַבִּים מִזְּנֵי הַתְּאֵנָה הַמְּצוּיִים בָּאָרֶץ מְחוֹנָנִים בִּסְגֻלָּה לְהָבִיא שְׁתֵּי עֲנָווֹת בְּמֶשֶׁךְ הַקָּיִץ. הָעֲנָוָה הָרִאשׁוֹנָה הִיא הַבַּכּוּרָה, מוֹפִיעָה בְּרֵאשִׁית הָאָבִיב וּבְשֵׁלָה בְסִיוָן. תְּקוּפַת הַהַבְשָׁלָה שֶׁל הָעֲנָוָה הַשְׁנִיָה אֲרֻכָּה מְאֹד וְנִמְשֶׁכֶת מִסִּיוָן עַד סוֹף מַרְחֶשְׁוָן. תְּנוּבַת הַתְּאֵנָה רַבָּה, מִמָּאתַיִם עַד שְׁלֹשׁ מֵאוֹת ק"ג פְּרִי לָעֵץ.

הַתְּאֵנִים נֶחֱלָקוֹת לִשְׁנֵי סוּגִים: שְׁחוֹרוֹת וּלְבָנוֹת.

הַפֵּרוֹת נֶאֱכָלִים כְּשֶׁהֵם טְרִיִּים וּמְיֻבָּשִׁים בְּצוּרַת גְרוֹגָרוֹת אוֹ דְבֵלִים. בִּסְבִיבַת יְרוּשָׁלַיִם יֶשְׁנָם כְּ-14 זְנִים, מֵהֶם שְׁנַיִם שֶׁשְׁמוֹתֵיהֶם הָעִבְרִיִּים נִשְׁמְרוּ עַד הַיּוֹם: “בְּנוֹת שֶׁבַע” (שִׂבַּעִי, בְּעַרְבִית) וְ"שֹׁעָרִים" (שׁוּרַטִי, בְּעַרְבִית) – פְּרִי מָתוֹק, בִּלְתִּי נָעִים לַאֲכִילָה (וְנָתַתִּי אוֹתָם כַּתְּאֵנִים הַשֹּׁעָרִים אֲשֶׁר לֹא תֵאָכַלְנָה מֵרֹעַ" (ירמיה כ"ט, י"ז).

עֵץ פְּרִי יָרֹק תָּדִיר, בַּעַל גֶזַע יָשָׁר, קִפֵּחַ וְגָמִישׁ, מַגִּיעַ לְגֹבַהּ שֶׁל עֶשְׂרִים מֶטֶר. מְיַפֶּה אֶת הַנּוֹף. הַצַּמֶּרֶת הַגְּדַלָה בְראשׁ הַגֶּזַע יָפָה מְאֹד, דּוֹמָה לְשִׁמְשִׁיָּה וְנִשְׁקֶפֶת מֵרָחוֹק. הִנֶּהָ בַעֲלַת 30–10 כַּפּוֹת מְפֻצָּלוֹת כְּעֵין נוֹצָה. מִתּוֹךְ

הַצַמֶּרֶת עוֹלֶה הַלּוּלָב וּמִמֶּנּוּ מִתְפַּצְּלוֹת 12–10 כַּפּוֹת חֲדָשׁוֹת בְּכָל שָׁנָה.

מֻזְכָּר בַּמִּקְרָא; אֶחָד מִשִּׁבְעַת הַמִּינִים שֶׁנִּשְׁתַּבְּחָה בָהֶם הָאָרֶץ. מִמֶּנּוּ לֻקַּח הַלּוּלָב בֵּין אַרְבַּעַת הַמִּינִים לְחַג הַסֻכּוֹת. עַל הַמַּטְבְּעוֹת הָעִבְרִיִּים הָעַתִּיקִים הָיְתָה טְבוּעָה צוּרַת תָּמָר כְּסֵמֶל אֶרֶץ יִשְׂרָאֵל. לְפָנִים הָיָה נָפוֹץ הַתָּמָר בְּבִקְעַת הַיַּרְדֵּן, בְּיִחוּד בִּסְבִיבַת יְרִיחוֹ עִיר הַתְּמָרִים. כָּעֵת יֶשְׁנָם חֳרָשׁוֹת קְטַנִּים בִּסְבִיבוֹת עַזָּה בְּדִיר-אַל-בָּלָח, בִּסְבִיבוֹת חֵיפָה וּבִסְבִיבוֹת יְרִיחוֹ.

פִּרְחֵי הַתָּמָר הֵם דּוּ-בֵיתִיִּים עַל שְׁנֵי עֵצִים שׁוֹנִים. הֵם קְטַנִּים, מְסֻדָּרִים בְּתִפְרַחַת “מַכְבֵּד” (סַנְסַן); סְגוּרִים תְּחִלָה בְּתוֹךְ קִבּוּר (נַרְתִּיק). מוֹפִיעִים בָּאָבִיב בְּחָדְשֵׁי נִיסָן – אִיָּר, לְאַחַר שֶׁצֶּבַע הַקִּבּוּרִים הוֹפֵךְ חוּם.

מַפְרִים אֶת הַתְּמָרָה בְּדֶרֶךְ מְלָאכוּתִית, כּוֹרְתִים אֶת הַקִּבּוּרִים מֵעַל הָעֵצִים הַזְּכָרִים, טֶרֶם יִפָּתֵחוּ; מוֹצִיאִים אֶת הַסַּנְסַנִּים עִם פִּרְחֵי הָאַבְקָנִים וּמְיַבְּשִׁים אוֹתָם. בְּחָדְשֵׁי נִיסָן וְאִיָּר, כַּאֲשֶׁר נִפְתָּחִים הַקִּבּוּרִים שֶׁל פִּרְחֵי הָעֱלִי, מַכְנִיסִים לְתוֹכָם אֶת הַסַּנְסַנִּים עִם פִּרְחֵי הָאַבְקָנִים. אַנְשֵׁי יְרִיחוֹ הָיוּ נוֹהֲגִים לְהַפְרוֹת דְּקָלִים בְּעַרְבֵי פְסָחִים וְהַחֲכָמִים לֹא מִחוּ בְיָדָם (פסחים נ"ה).

הַפְּרִי – תָּמָר, מָאֳרָךְ, גְלִילִי מִצֶבַע צָהֹב. יֶשְׁנָם זְנִים בַּעֲלֵי פֵרוֹת חוּמִים וַאֲדֻמִּים. הַפֵּרוֹת רַבִּים בְּסַנְסַן אֶחָד, בִּדְמוּת אֶשְׁכּוֹל גָּדוֹל, בְּשֵׁלִים בְּתִשְׁרֵי – מַרְחֶשְׁוָן. הַתָּמָר מוֹנֶה מֵאוֹת זְנִים. בִּסְבִיבוֹת יְרִיחוֹ גְדֵלִים תְּמָרִים בַּעֲלֵי פַרוֹת חַסְרֵי גַרְעִינִים, וְזָן מְשֻׁבָּח בְּשֵׁם “ג’וֹפִי” (הֱבִיאוּהוּ מִמִּדְבַּר ג’וֹף).

מִמִּשְׁפַּחַת הַוְּרָדִים Rosaceae

עֵץ פְּרִי, עוֹמֵד בַּשַּׁלָּכֶת. מַגִּיעַ לְגֹבַהּ שֶׁל 10 מֶטֶר. אַחַד הָעֵצִים הַחֲשׁוּבִים וְהַיָּפִים בְּגַנֵי הַפֶּרִי. מֻזְכָּר בַּמִּקְרָא: “כְּתַפּוּחַ בֵּין עֲצֵי הַיַּעַר” (שה"ש ב', ג'). בִּתְקוּפַת הַמִּשְׁנָה יָדוּעַ הָיָה הַתַּפּוּחַ. אָכְלוּ אֶת הַפְּרִי חַי וְגַם הֵכִינוּ מִמֶּנּוּ מִין עָסִיס – יֵין תַּפּוּחִים (נדרים נג.). בַּזְּמָן הָאַחֲרוֹן מְגַדְּלִים בָּאָרֶץ בְּהַצְלָחָה זְנִים מְשֻׁבָּחִים שֶׁל עֲצֵי תַפּוּחַ.

עֲלֵי הַתַּפּוּחַ הֵם בַּעֲלֵי פְטוֹטָרוֹת קְצָרוֹת, מְשֻׁנָּנִים בְּשׁוּלֵיהֶם, וּשְׂעִירִים בְּצִדָּם הַתַּחְתּוֹן. הַפְּרָחִים מוֹפִיעִים יַחַד עִם הֶעָלִים. נוֹתְנִים רֵיחַ נָעִים. לָהֶם חֲמִשָּׁה עַלְעֲלֵי גָבִיעַ וַחֲמִשָּׁה עַלְעֲלֵי כֹתֶרֶת מִצֶּבַע וָרֹד. אַבְקָנִים רַבִּים עִם מַאֲבָקִים צְהֻבִּים. חֲמִשָּׁה עֲלָיִים מְחֻבָּרִים בְּחֶלְקָם הַתַּחְתּוֹן. מִמַּצְּעִית הַפֶּרַח מִתְפַּתַּחַת צִיפַת הַפְּרִי הַטְּעִימָה וְהַנּוֹתֶנֶת רֵיחַ.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
