הַתְּאֵנָה בַחֹרֶף

בְּאֶחָד מִימֵי הַחֹרֶף הַבְּהִירִים יָצָא אַמְנוֹן הַקָּטֹן עִם אָבִיו אֶל הַגָּן לְהִתְבּוֹנֵן אֶל הָעֵצִים וְלִקְטֹף פְּרָחִים.

קֹדֶם כֹּל הִרְגִישׁוּ בּפִרְחֵי הַשֶּׁסֶק הַיַּפּוֹנִי, כִּי רֵיחָם נוֹדף לְמֵרָחוֹק. אַמְנוֹן שָׂמַח מְאֹד לִקְרַאת הַפְּרָחִים הַלְּבָנִים וְהָרֵיחָנִיִּים שֶׁל הַשֶּׁסֶק וּלְמַרְאֵה עָלָיו הַחֲדָשִׁים. הוּא אָחַז בַּעֲנַף הָעֵץ וַיַּט אוֹתוֹ וַיִּתְבּוֹנֵן אֶל הַפְּרָחִים הַשְּׂעִירִים, אוּלָם קָטֹף לֹא קָטַף מֵהֶם, כִּי חָמַל עַל הַדְּבוֹרִים, אֲשֶׁר הִתְעוֹפְפוּ וְשָׁקְקוּ מִסָּבִיב לַשֶּׁסֶק, בְּאָגְרָן צוּף מִפְּרָחָיו.

– אֵימָתַי נֹאכַל מִפֵּרוֹתָיו שֶׁל הַשֶּׁסֶק? – שָׁאֵל אַמְנוֹן – מַה טְּעִימִים הֵם!

– בָּאָבִיב, בְּנִי – עָנָה הָאָב – וְעַתָּה בֹּא נָא, יַלְדִּי וְנִקְטָף לָנוּ כַּרְכֻּמִּים יָפִים.

בְּדַרְכָּם עָבְרוּ עַל יַד עֵץ הַתְּאֵנָה, שֶׁעָמַד חֲשׂוּף־עָלִים כֻּלּוֹ.

מַה זֹּאת? – קָרָא אַמְנוֹן מִתּוֹךְ צַעַר, בְּעָמְדוֹ עַל יַד הַתְּאֵנָה תּוֹהֶה וּמַשְׁמִים – אַבָּא, רְאֵה, הִנֵּה הַתְּאֵנָה יָבְשָׁה וָמֵתָה.

– הִיא לֹא יָבֵשָׁה – עָנָה הָאָב. וּבְדַבְּרוֹ הִרְכִּין אֶחָד מֵרָאשֵׁי הָעֲנָפִים וְהֶרְאָה לִבְנוֹ אֶת הַפֶּקַע הַיָּרֹק בְּרֹאשׁ הֶעָנָף, קִלֵּף אֶת הַקַּשְׂקַשִּׂים הָעֶלְיוֹנִים וְהִתְחִיל מוֹצִיא, כְּעֵין קֻפְסָה מִתּוֹךְ קֻפְסָה, עַלְעָלִים קְטַנְטַנִּים, שֶׁכָּל אֶחָד מֵהֶם הָיָה מְכֻסֶּה בִשְׁתֵּי קַשְׂקַשִּׂים.

– הֲרוֹאֶה אַתָּה, אַמְנוֹן – הוֹסִיף הָאָב – הִנֵּה פֹּה בְּתוֹךְ פֶּקַע יָרֹק זֶה הֵכִינָה לָהּ הַתְּאֵנָה עַלְעָלִים, הָעֲתִידִים לְלַבְלֵב עִם בּוֹא הָאָבִיב.

– כֵּן, כֵּן, אַבָּא, רוֹאֶה אָנֹכִי – עָנָה הַבֵּן – אֲבָל מַדּוּעַ זֶה הִשִּׁירָה הַתְּאֵנָה אֶת עָלֶיהָ וְנִשְׁאֲרָה כֻלָּהּ עֲרֻמָּה, וְעַתָּה הִיא צְרִיכָה לִדְאֹג לְעָלִים חֲדָשִׁים?

– יָפֶה שָׁאַלְתָּ, בְּנִי, וְעַתָּה הַסְכֵּת וּשְׁמַע אֵת אֲשֶׁר אֲסַפֶּר לָךְ:

– "לִפְנֵי רַבּוֹת, רַבּוֹת בְּשָׁנִים, הָיְתָה הָאָרֶץ כֻּלָּהּ מְכֻסָּה עֵצִּים סְבוּכִים וּצְפוּפִים וּבֵינֵיהֶם גָּדְלָה גַם הַתְּאֵנָה, וְגַם הִיא, כְּיֶתֶר בְּנֵי סוּגָהּ, הָיְתָה יְרֻקָּה תָמִיד וּפוֹרֶשֶׂת צֵל מִסָּבִיב. וּבְתוֹךְ הָאֲדָמָה חִכּוּ בְקֹצֶר־רוּחַ זַרְעֵי הַפְּרָחִים וְהָעֲשָׂבִים עַד אֲשֶׁר יַעַבְרוּ חַרְבוֹנֵי הַקָּיִץ, לְמַעַן יוּכְלוּ לְהָצִיץ מִן הָאָרֶץ וְלִצְמֹחַ גַּם הֵם.

"בָּא הַחֹרֶף. הַגְּשָׁמִים הִרְווּ אֶת הָאֲדָמָה וַעֲשָׂבִים רַכִּים הִתְחִילוּ עוֹלִים מִן הָאֲדָמָה, אַךְ בִּתְחִלַּת צְמִיחָתָם עָמְדוּ חִוְרִים וְחַסְרֵי־אוֹנִים, כִּי הָעֵצִים הַגְּדוֹלִים וְרַחֲבֵי־הֶעָלִים חָסְמוּ אֶת קַרְנֵי הַשֶּׁמֶשׁ מִלַּחְדֹּר אֲלֵיהֶם.

"הִתְאוֹנְנוּ הָעֲשָׂבִים הָעֲלוּבִים עַל הָעֵצִים, עַל אֲשֶׁר יִגְזְלוּ מֵהֶם אֶת אוֹר הַשָּׁמֶשׁ.

"שָׁמְעָה הַתְּאֵנָה וְרַחֲמֶיהָ נִכְמְרוּ עַל הָעֲשָׂבִים הַזְּעִירִים וַתַּשֵּׁר אֶת עָלֶיהָ, לְמַעַן תּוּכַל הַשֶׁמֶשׁ לְחַמֵּם בְּקַרְנֶיהָ גַּם אֶת הַדְּשָׁאִים.

"רָאוּ זאת עֵצִים רַבִּים, וַיַּשִּׁירוּ אֶת עֲלֵיהֶם גַּם הֵם.

“וּגְמוּל הַמַּעֲשֶׂה הַטּוֹב הַזֶּה, מַעֲשֵׂה הָרַחֲמִים, הָיָה לָהֶם מֵאָז הַחֹרֶף לְעֵת מְנוּחָה. רָאֹה תִרְאֶה, בְּנִי, בִּימוֹת הַחֹרֶף אֶת רֹב הָעֵצִים מַשִּׁירִים עֲלֵיהֶם וְהֵם שׁוֹבְתִים מֵעֲבוֹדָתָם, וּבְבוֹא הָאָבִיב יָשׁוּבוּ וְיִתְעַטְּפוּ בִּלְבוּשׁ עָלִים יְרֻקִּים וְרַעֲנַנִּים לְהַרְהִיב עַיִן וּלְשַׂמֵּחַ לֵב”.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
