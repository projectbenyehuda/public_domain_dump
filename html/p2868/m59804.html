לָמָּה יַעֲשֶׂה הַקּוֹץ שְׁנֵי מִינֵי זְרָעִים

אֲרוּרָה הָאֲדָמָה בַּעֲבוּרֶךָ, בְּעִצָּבוֹן תֹּאכְלֶנָּה

כֹּל יְמֵי חַיֶּיךָ. וְקוֹץ וְדַרְדַּר תַּצְמִיחַ לָךְ.

(בראשית ג').


מִן הַיּוֹם אֲשֶׁר הֵחֵל הָאָדָם לַעֲבוֹד אֶת הָאֲדָמָה לֹא חָדַל מַהִלָּחֵם בַּקּוֹץ. “קוֹצִים כְּסוּחִים בָּאֵשׁ יִצַּתּוּ!” וּמִדֵּי שָׁנָה בְשָׁנָה, בְּבוֹא עוֹנַת־הַזְּרִיעָה, יִכְסַח הָאִכָּר אֶת הַקּוֹצִים הַמְכַסִּים אֶת פְּנֵי שָׂדֵהוּ וְיָדוּן אוֹתָם לִשְׂרֵפָה.

וַיְהִי כִרְאוֹת הַקּוֹץ, כִּי כָלְתָה אֵלָיו הָרָעָה מֵאֵת הָאָדָם וַיִּתְעַצֵּב מְאֹד וַיֹּאמַר: “אֲהָהּ! מַה פִּשְׁעִי וּמַה חַטָּאתִי כִּי יִשְׂטְמֵנִי הָאָדָם וִיאַבְּדֵנִי בְלֹא רַחֲמִים מִן הָעוֹלָם?”

וְהִנֵּה נִשְׁמַע קוֹל זִמְזוּם עַלִּיז. דְּבוֹרָה בָאָה לִפְקוֹד אֶת פִּרְחֵי הָאַרְגָּמָן אֲשֶׁר לַקּוֹץ וְלָמֹץ אֶת עֲסִיסָם הַמָּתוֹק.

– הֲלֹא תַגִּידִי לִי, הַדְּבוֹרָה – שָׁאַל הַקּוֹץ – עַל מַה זֶּה וְלָמָּה זֶה כֹּה יִשְׂטְמֵנִי וְיִרְדְּפֵנִי הָאָדָם? מַדּוּעַ הָיִיתִי לוֹ קְלָלָה כָּל הַיָּמִים?

– הֵרָגַע נָא, יַקִּירִי! – עָנְתָה הַדְּבוֹרָה – אַל תָּשֶׁת לִבְּךָ לְשִׂנְאַת בֶּן־הָאָדָם וּלְקִלְלָתוֹ. זְכָר נָא מַה יָּקַרְתָּ וּמַה נָּעַמְתָּ לְכָל עֲדַת הַדְּבוֹרִים. הֵן שֶׁבַע בַּיּוֹם תְּבֹרַךְ בְּפִינוּ עַל הַדְּבַשׁ הַטּוֹב וְהַמָּתוֹק שֶׁאַתָּה מַעֲנִיק לִנְחִילֵנוּ בִּימֵי הַקַּיִץ הַלּוֹהֵט, עֵת נָבֹל יִבְּלוּ כָל פִּרְחֵי הַגַּן וְהַשָּׂדֶה.

– אַל נָא תֵעָצֵב, מַחֲמַדִּי! – קָרָא הָעַכָּבִישׁ אֲשֶׁר יָשַׁב בְּסֵתֶר רִשְׁתּוֹ הַפְּרוּשָׂה בֵּין עֲלֵי הַקּוֹץ – בָּרוּךְ תִּהְיֶה בְכָל קְהַל הָעַכְבִישִׁים הָאוֹרְגִים אֶת קוּרֵיהֶם וְתוֹלִים אֶת רְשְׁתוֹתֵיהֶם בָּךְ.

וַיִּשְׁמַע הַקּוֹץ אֶת דִּבְרֵי הַדְּבוֹרָה וְהָעַכָּבִישׁ וַיִּנָּחֵם לֵאמוֹר: “אָכֵן יֵשׁ יְצוּרִים בְּתֵבֵל אֲשֶׁר יֹאהֲבוּנִי וְגַם בָּרֵךְ יְבָרְכוּנִי – וְאָנֹכִי לֹא יָדָעְתִּי!” וַיּוֹסֶף הַקּוֹץ לְהִתְקַשֵּׁט בְּפִרְחֵי אַרְגָּמָן וּלְהַפְרִישׁ שְׂרָף רֵיחָנִי מֵעָלָיו וּמִקְּלָחָיו, לְמַעַן הַרְבּוֹת לוֹ רֵעִים נֶאֱמָנִים בֵּין הַיְצוּרִים הַחַיִּים. וּמִדֵּי שָׁמְעוֹ אֶת קִלְלַת הָאָדָם אָמָר: “אִם גַּם שָׂרֹף יִשְׂרְפֵנִי הָאָדָם לֹא אֶכָּחֵד מִן הָאָרֶץ, כִּי זַרְעִי הָרַב אֲשֶׁר אַפִּיל אַרְצָה יָשִׂימוּ לִי שְׁאֵרִית בָּעוֹלָם”.

וַיְהִי הַיּוֹם וְעֶדְרֵי עִזִּים פָּשְׁטוּ בַשָּׂדֶה וַיְקַרְסְמוּ1 אֶת רָאשֵׁי הַקּוֹצִים עַל פִּרְחָם וְעַל זַרְעָם. וַיַּרְא הַקּוֹץ וְהִנֵּה הָעִזִּים מְלַקְּטוֹת גַּם אֶת הַזְּרָעִים אֲשֶׁר נָפְלוּ אָרְצָה וַיֶּחֱרַד חֲרָדָה גְדוֹלָה וַיֹּאמַר: “אִם כֹּה הָעִזִּים עוֹשׂוֹת, אַחֲרִיתִי עֲדֵי אוֹבֵד!”

יָמִים רַבִּים הָיָה הַקּוֹץ שָׁרוּי בְּצַעַר וּבִדְאָגָה לְגוֹרָלוֹ, עַד שֶׁמָצָא עֵצָה טוֹבָה אֵיךְ לְהַצִּיל אֶת זַרְעוֹ מִכִּלָּיוֹן: הַקּוֹץ הֵחֵל לַעֲשׂוֹת צִיצִיּוֹת שֵׂעָר לְמִקְצָת זְרָעָיו, לְמַעַן יוּכְלוּ לְהִנָּשֵׂא עַל כַּנְפֵי הָרוּחַ וּלְהִתְעוֹפֵף הַרְחֵק הַרְחֵק מִשְּׂדֵה הָאִכָּר. וּמֵאָז וְעַד הַיּוֹם יַעֲשֶׂה הַקּוֹץ שְׁנֵי מִינֵי זְרָעִים, מִקְצָתָם בַּעֲלֵי צִיצִיּוֹת אֲשֶׁר יוּכְלוּ לְהַרְחִיק נְדוֹד, וְיִתְרָם חַסְרֵי צִיצִיּוֹת אֲשֶׁר יִפְּלוּ סָבִיב לוֹ – בִּמְקוֹם מוֹלַדְתָּם.


  1. כך במקור. הערת פב"י.  ↩
