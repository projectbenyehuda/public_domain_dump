שׁוֹשַׁנַּת־יְרִיחוֹ

וַיְהִי כַאֲשֶׁר הִגִּיעַ הַיּוֹם אֲשֶׁר בּוֹ יָמוּת משֶׁה, כִּדְבַר ה' אֵלָיו וַיַּעַל משֶׁה מֵעַרְבוֹת מוֹאָב אֶל הַר נְבוֹ ראשׁ הַפִּסְגָּה, אֲשֶׁר עַל פְּנֵי יְרֵחוֹ. וַיַּרְאֵהוּ ה' אֶת כָּל הָאָרֶץ: אֶת הַגִּלְעָד עַד דָּן וְאֵת כָּל נַפְתָּלִי וְאֶת אֶרֶץ אֶפְרַיִם וּמְנַשֶּׁה וְאֵת כָּל אֶרֶץ יְהוּדָה עַד הַיָּם הָאַחֲרוֹן וְאֶת הַנֶּגֶב וְאֶת הַכִּכָּר בִּקְעַת יְרֵחוֹ עִיר הַתְּמָרִים עַד צֹעַר. וּמשֶׁה מַחֲרִישׁ בִּיגוֹנוֹ הַגָּדוֹל, מַבִּיט אֶל אֶרֶץ הַצְּבִי, אֶרֶץ הַחֶמְדָּה, וַתִּפָּעֶם רוּחוֹ, וְכִלְיוֹן נֶפֶשׁ גָּדוֹל מִנְּשׂוֹא תְּקָפָהוּ.

וֵאלֹהִּים רָאָה לִלְבַב משֶׁה, וַיְצַו לְרוּחוֹת הַשְּׁפָיִים לְהָבִיא לְמשֶׁה מֵחֲמוּדוֹת הָאָרֶץ.

וַיָּקוּמוּ הָרוּחוֹת, וַיָּבִיאוּ בְכַנְפֵיהֶם רֵיחוֹת עֲדָנִים מֵרֵיחוֹת הַדַּפְנָה, הָאֵזוֹב, הַקּוֹרָנִית וְכָל יֶתֶר צִמְחֵי הֶהָרִים וּמֵרֵיחוֹת הָרִכְפָּה, הַכֹּפֶר וְהַיַּסְמִין אֲשֶׁר בָּעֲמָקִים וּמֵרֵיחַ הַשִּׂיחַ אֲשֶׁר בָּעֲרָבָה.

וַיָּרַח משֶׁה אֶת רֵיחַ הָאָרֶץ מֵרָחוֹק וַיִּרְחַב לִבּוֹ.

פִּתְאוֹם דְּבַר מָה רִפְרֵף בָּאֲוֵיר יָרַד וְנֶחְבַּט לְרַגְלֵי משֶׁה; וַיָּרֶם מֹשֶׁה אֶת הַדָּבָר, וְהִנֵּה הוּא צֶמַח יָבֵשׁ מְצֻנָּף לְכַדּוּר.

רֶטֶט חָלַף אֶת גֵּווֹ שֶׁל משֶׁה וַיִּפְתַּח אֶת פִּיו וַיֹּאמַר: הַאִם הַצֶּמַח הַזֶּה אוֹת הוּא כִּי כָמוֹהוּ אִיבַשׁ וְאֶחֱלוֹף מִן הָאָרֶץ?

וַיַּעֲנֵהוּ אֱלֹהִּים וַיֹּאמַר: קַח לְךָ מְעַט מַיִם וְזַלֵּף עַל הַצֶּמַח. וַיִּקַּח משֶׁה מְעַט מָיִם, וַיְזַלֵּף עַל הַצֶּמַח וַיַּרְטִיבֵהוּ.

וַיְהִי אַךְ נָגְעוּ הַמַּיִם בַּצֶּמַח, וַיָּקוּמוּ וַיִּזְדַּקְּפוּ וַיִּתְרַחֲבוּ גִבְעוֹלָיו וַיֵּהָפְכוּ לְשׁוֹשַׁנָּה.

וַיֹּאמֶר אֱלֹהִים לְמשֶׁה: אַל תִּירָא, משֶׁה עַבְדִּי! כַּצֶּמַח הַזֶּה תִּחְיֶה רוּחֶךָ, וְזִכְרְךָ יִהְיֶה בָרוּךְ לְעוֹלְמֵי עוֹלָמִים.

כִּשְׁמֹעַ משֶׁה אֶת הַדְּבָרִים הָאֵלֶּה יוֹצְאִים מִפִּי אֱלֹהִים, וַיָּנַח לְבָבוֹ וַיֹּאמַר: עַתָּה נַפְשִׁי תִשְׁכֹּן לָבֶטַח.

וַיָּמָת שָׁם משֶׁה, עֶבֶד ה', בְּאֶרֶץ מוֹאָב, עַל פִּי ה', וַיִּקְבֹּר אוֹתוֹ בַגַּי, בְּאֶרֶץ מוֹאָב, מוּל בֵּית פְּעוֹר; וְלֹא יָדַע אִישׁ אֶת קְבוּרָתוֹ עַד הַיּוֹם הַזֶּה.

מֵאָז וְעַד הַיּוֹם, בִּגְמֹל זְרָעֶיהָ מִצְטַנֶּפֶת שׁוֹשַׁנַּת יְרִיחוֹ כַּכַּדּוּר, נִתֶּקֶת מִשָּׁרְשָׁהּ וְנִשֵּׂאת לְכָל מָקוֹם אֲשֶׁר יִשָּׂא אוֹתָהּ הָרוּחַ. אַךְ גֶּשֶׁם כִּי יֵרֵד, אוֹ אִם תְּזַלֵּף עָלֶיהָ מָיִם, יָקוּמוּ גִבְעוֹלֶיהָ וְיִזְדַּקַּפוּ; וְשׁוֹשַׁנַּת יְרִיחוֹ עוֹמֶדֶת שׁוּב חַיָּה וּמְנַעֶרֶת אֶת זְרָעֶיהָ.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
