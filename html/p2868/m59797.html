לְשׁוֹן־הַצִּפּוֹר

כַּאֲשֶׁר כִּלָּה שְׁלֹמֹה הַמֶּלֶךְ לִבְנוֹת אֶת בֵּית ה' וְאֶת בֵּית הַמֶּלֶךְ נָטַע לו גַן עֵצִים גָּדוֹל וְיָפֶה וַיִּבְנֶה בַגֵּן בְּרֵכוֹת מָיִם וּסְבִיב הַבְּרֵכוֹת שָׁתַל פְּרָחִים רַבִּים וְיָפִים וְגַם יֶרֶק־דֶּשֶׁא לֹא הֶחְסִיר. וַיִּבְנֶה גַּם מְעוֹנוֹת לְכָל מִינֵי חַיּוֹת וְעוֹפוֹת שֶׁהִתְאַסְּפוּ וּבָאוּ אֶל גַּנּוֹ מִכָּל קַצְוֵי תֵבֵל, כִּי הָיָה שְׁלֹמֹה הַמֶּלֶךְ מְכַלְכְּלָם בְּיָד רְחָבָה.

גַּם צִפֳּרֵי־שִׁיר הָיוּ מִתְעוֹפְפוֹת בַּגָּן וְצִפְצוּפֵי חֵן וְשִׁירַת גִיל מְמַלְאִים אֶת הַגֵּן תָּמִיד, כִּי דָבְקָה נֶפֶשׁ הַצִּפֳּרִים בְּגַן הַמֶּלֶךְ, וְהַמֶּלֶךְ יָרַד יוֹם יוֹם אֶל הַגָּן לְהַקְשִׁיב לְשִׁירַת הַצִּפֳּרִים וּלְשׂוֹחֵחַ אִתָּן בִּלְשׁוֹנָן, כִּי יָדַע הַמֶּלֶךְ הֶחָכָם אֶת לְשׁוֹן הַצִּפֳּרִים.

פַּעֲמַיִם בַּשָּׁנָה הָיוּ מִן הַצִּפֳּרִים נוֹדְדוֹת לְמֶרְחַקִּים. בִּתְחִלַת הַקַיִץ עָפוּ לָאֲרָצוֹת הַקָּרוֹת בַּצָּפוֹן הַזְּמִירִים, הַחֲרָגִים וְיֶתֶר צִפֳּרֵי הַשִּׁיר, וּבִתְחִלַּת הַחֹרֶף נָדְדוּ הַחִמְרִיָּה, הַגִּבְּתוֹן וְהַדּוּכִיפַת לָאֲרָצוֹת הַחַמּוֹת. פַּעַם פַּעַם בְּשׁוּב הַצִּפֳּרִים מִמַּסָּעָן הָיוּ מְסַפְּרוֹת לַמֶּלֶךְ מִכָּל הַנַּעֲשֶׂה בָאֲרָצוֹת הָרְחוֹקוֹת מֵעֵבֶר לַיָּם, וְגַם תְּשׁוּרוֹת הָיוּ מְבִיאוֹת אִתָּן – מִי נוֹצָה יָפָה מַרְהִיבָה עָיִן, מִי שִׁירָה חֲדָשָׁה וּמִי זְרָעִים שֶׁל פְּרָחִים יָפִים לְיַפּוֹת בָּהֶם אֶת גַּן הַמֶּלֶךְ, כִּי דָבְקָה נֶפֶשׁ הַצִּפֳּרִים לְאַהֲבָה אֶת הַמֶּלֶךְ הֶחָכָם וְאֶת גַּנּוֹ הַנֶּהֱדָר.

גַּם צִפּוֹר הֶחָרָג זָכְרָה אֶת גַּן הַמֶּלֶךְ וּפַעַם בְּשׁוּבָהּ מִמַּסָּעָהּ דֶּרֶךְ אֶרֶץ יָוָן וְאָסִיָּה הַקְּטַנָּה נִקְרָה בְדַרְכָּהּ זֶרַע מִצֶּמַח אֶחָד וַתָּעָף וַתְּבִיאֵהוּ לִשְׁלֹמֹה. וַיִּזְרְעֵהוּ הַמֶּלֶךְ בְּגַנּוֹ וַיִּשְׁקֵהוּ מִמֵּי הַבְּרֵכוֹת.

וַיְהִי כַאֲשֶׁר קָרְבוּ יְמוֹת הַחַמָּה וַיִּזְכֹּר הַזֶּרַע אֶת הַקַיִץ בְּמוֹלַדְתּוֹ וַיָּצֶץ צִיץ וַיִּתֵּן פְּרָחִים שֶׁצִּבְעָם וְצוּרָתָם כְּצִפּוֹר הֶחָרָג אֲשֶׁר בָּחֲרָה בוֹ מִכָּל הַצְּמָחִים לַהֲבִיאֵהוּ אֶל גַּן שְׁלֹמֹה הַמֶּלֶךְ. וַיֶאֱהַב שְׁלֹמֹה אֶת הַצֶּמַח אֲשֶׁר דְּמוּתוֹ הִזְכִּירָה לוֹ אֶת צִפּוֹר הֶחָרָג יְדִידוֹ, וַיְצַו לַהֲפִיצוֹ בְּכָל הָאָרֶץ וַיִּקְרָא שְׁמוֹ “לְשׁוֹן הַצִּפּוֹר” אוֹ “לְשׁוֹן הֶחָרָג”, כִּי בִלְשׁוֹנָהּ שֶׁל צִפּוֹר הֶחָרָג הוּבָא אֶל גַּנּוֹ.

מֵאָז, כְּמוֹ בָאָה לְיַפּוֹת אֶת הָאָרֶץ, גְּדֵלָה לְשׁוֹן הֶחָרָג בִּמְקוֹמוֹת חֲרֵבִים, בֶּהָרִים וּבָעֲמָקִים, וּבְכָל יְמֵי הַקַּיִץ יִתְנוֹסְסוּ לְתִפְאָרָה פְּרָחֶיהָ הַתְּכֻלִים־סְגֻלִּים וּמַרְאֶה לָהֶם מֵרָחוֹק כְּצִפֳּרִים קְטַנּוֹת הַיּוֹשְׁבוֹת עַל פְּנֵי עֲנַף הָעֵץ.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
