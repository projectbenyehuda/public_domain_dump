צִמְחֵי־הַמִּדְבָּר

וַיְהִי אַחֲרֵי אֲשֶׁר חָטָא אָדָם הָרִאשׁוֹן וַה' קִלֵּל אֶת הָאֲדָמָה בַּעֲבוּרוֹ, וַיְכַס אֵד אֶת פְּנֵי כָל הָאָרֶץ וְתוּגָה עֲמוּמָה וּכְבֵדָה יָרְדָה וַתִּרְבַּץ עַל פְּנֵי כָל הַיְקוּם.

וַיִּלְאוּ הַיְצוּרִים לָשֵׂאת אֶת הַמְצוּקָה הַגְּדוֹלָה הַזֹּאת וַיִּזְעֲקוּ מָרָה אֶל ה': “אֲדוֹן הָעוֹלָם, הֲכָלָה אַתָּה עוֹשֶׂה בְּמַעֲשֵׂי יָדֶיךָ!?”

וַיִּשְׁמַע ה' וַיִּנָּחֶם, וַיַּבְדֵּל בְּלֵב הָאָרֶץ חֵלֶק אֲדָמָה גָּדוֹל וּרְחַב־יָדַיִם וַיַּהַפְכֵהוּ לְשִׁמָּמוֹן, וַיִּקְרָא שְׁמוֹ מִדְבָּר, לֵאמֹר: רַק בַּמָּקוֹם הַזֶּה יָקוּם דְּבַר קִלְלָתִי; וּמֵעַתָּה וְעַד הָעוֹלָם לֹא יִצְמְחוּ בוֹ לֹא גֶפֶן וְלֹא תְאֵנָה וְלֹא כָל עֵץ פְּרִי אַחֵר. וַיּוֹצֵא אֱלֹהִים מֵאוֹצְרוֹתָיו אֶת הַשָּׁרָב וְאֶת הַסּוּפָה, וַיִּתְּנֵם שַׁלִּיטִים בְּתוֹךְ הַמִּדְבָּר, וַתִּשְׁתּוֹלֵל הַסּוּפָה וַתְּנַשֵׂא עַמּוּדֵי חוֹל עַד לַשָּׁמָיִם, וְרוּחַ זִלְעָפוֹת עָטָה עַל הַמִּדְבָּר לְלַהֵט וּלְהוֹבִישׁ כָּל שִׂיחַ וְכָל עֵשֶׂב, וַתִּפֹּל חֶרְדַּת מָוֶת עַל כָּל בַּעֲלֵי־הַחַיִּים אֲשֶׁר בַּמִּדְבָּר, וַיָּנוּסוּ מִפְּנֵי הַשָּׁרָב וְהַסּוּפָה.

אָז רִחֵם ה' אֶת הַיְצוּרִים הַתּוֹעִים וְהַנּוֹדְדִים בַּמִּדְבָּר; וּלְבַל יָמוּתוּ בָרָעָב וּבַצָּמָא בָּרָא לְמַעֲנָם בְּתוֹךְ הַצִּיָּה נְאוֹת־דֶּשֶׁא יָפוֹת וּמַרְהִיבוֹת אֶת עֵין הַהֵלֶךְ בַּמִּדְבָּר, מַעְיְנוֹת מַיִם חַיִּים מְפַכִּים בַּעֲלִיצוּת וַעֲצֵי־תָמָר נָאִים שׁוֹלְחִים אֶת כַּפּוֹתֵיהֶם לִפְנֵיהֶם וְהֵם כְּמַזְמִינִים וּמְבַקְשִׁים אֶת הַנּוֹדֵד הֶעָיֵף לַחְסוֹת בְּצִלָּם הָרַעֲנָן, לְרַוּוֹת אֶת צִמְאוֹנוֹ וְלִשְׁבֹּר אֶת רַעֲבוֹנוֹ בְּפִרְיָם הֶעָרֵב לַחֵךְ.

וַיָּבוֹאוּ הָעַזּוֹת בַּחַיּוֹת לָגוּר בִּנְאוֹת־הַדֶּשֶׁא, וַיָּבוֹאוּ שָׁמָה גַּם הָעַזִּים בַּצְּמָחִים, הֲלֹא הֵם: גֶפֶן־הַשָּׂדֶה, הַמֶּלַח, הָרֻחָמָה, הַקְּצִיעָה וּכְדוֹמֶה; וַתְּהִי עֲלֵיהֶם רוּחַ הַיְשִׁימוֹן וַיְשַׁנּוּ אֶת מַרְאֵיהֶם, אֶת טַעְמָם וְאֶת דֶּרֶךְ גִּדּוּלָם.

גֶפֶן־הַשָּׂדֶה – הִיא הַחַנְטָל, קְרוֹבָתוֹ שֶׁל הָאֲבַטִּיחַ – שִׁנְּתָה אֶת טַעְמָהּ וַתֵּהָפֵךְ לְצֶמַח רַב־שְׁנָתִי וְשָׁרָשֶׁיהָ גְדוֹלִים וּבְשָׂרִיִּים, כִּי עַל כֵּן בְּמֶשֶׁךְ יְמֵי הַגְּשָׁמִים הַמְעַטִּים עָלֶיהָ לֶאֱגֹר בָּהֶם מִיץ לְגִבְעוֹלֶיהָ הַזַּחֲלָנִים, לְעָלֶיהָ הַמְשֻׁסָּעִים וּלְפֵרוֹתֶיהָ הַכַּדְרוּרִים. בִּתְקוּפַת אֲדָר־נִיסָן תִּפְרָח; וּפֵרוֹתֶיהָ, בְּטֶרֶם יַסְפִּיקוּ לְהַבְשִׁיל וְלִסְפֹּג לְתוֹכָם רַעַל, יְשַׁמְּשׁוּ מַאֲכָל לַצְּבָאִים וְלַיְּעֵנִים שׁוֹכְנֵי הַמִּדְבָּר.

גַּם הַמֶּלַח עָמַד בִּפְנֵי חַרְבוֹנֵי הַמִּדְבָּר וַיַּעַשׂ שָׁרָשִׁים אֲרֻכִּים מְאֹד וְלָהֶם כֹּחַ סוֹפֵג גָּדוֹל. עֲלֵי הַשִּׂיחַ מַפְרִישִׁים גְבִישֵׁי מֶלַח מְיֻחָדִים בְּמִינָם הַנְמַסִּים בַּלַּיְלָה מִפְּנֵי הַלַּחוּת שֶׁבַּאֲוֵיר1 הַמִּדְבָּר הַצּוֹנֵן וּמִתְקַשִּׁים בַּיּוֹם וְשׁוֹמְרִים עַל הַצֶּמַח בִּפְנֵי אִיּוּד רָב.

וְהָרֻחָמָה לָמְדָה לִקְלֹט בְּשַׂעֲרוֹתֶיהָ לַחוּת מִן הָאֲוֵיר עַד כִּי תּוּכַל לִגְדֹּל גַּם בְּקַרְקַע מְלוּחָה.

וְכֵן הִסְתַּגְלוּ לָאֲוֵיר הַלּוֹהֵט שֶׁל הַצִּיָּה גַם הָעָלֶמֶת, הַזֵּיתָה וְהָאַדָּר וְכָל יֶתֶר צִמְחֵי הַיְשִׁימוֹן; וְהֵם מְחַיִּים אֶת נֶפֶשׁ הַגְּמַלִים וְהַצֹּאן וְאֶת נֶפֶשׁ הָאָדָם שׁוֹכֵן הַמִּדְבָּר.


  1. כך מנוקד במקור. הערת פב"י.  ↩︎
