בַּבּוֹנֶג בַּעַל לֵב־הַזָּהָב

וַיְהִי בִימֵי פַרְעֹה מֶלֶךְ מִצְרַיִם וַיַּעַבְדוּ בְנֵי יִשְׂרָאֵל בְּחֹמֶר וּבִלְבֵנִים וַיִּבְנוּ עָרֵי מִסְכְּנוֹת לְפַרְעֹה, אֶת פִּיתֹם וְאֶת רַעַמְסֵס. וַיִּתְעַמֵּר פַּרְעֹה בִּבְנֵי יִשְׂרָאֵל וַיְצַו לְכָל עַמּוֹ לֵאמֹר: כָּל הַבֶּן הַיִּלּוֹד – הַיְאוֹרָה תַשְׁלִיכוּהוּ וְכָל הַבַּת תְּחַיּוּן. אַךְ הָאִמּוֹת הָעִבְרִיּוֹת בִּקְשׁוּ עֵצוֹת וְתַחְבּוּלוֹת לְהַסְתִּיר אֶת יַלְדֵיהֶן מֵעֵינֵי שׁוֹטְרֵי פַרְעֹה הָרָעִים, וַתַּחְבֶּאנָה אוֹתָם בְּסֵתֶר הַשִּׂיחִים וְהָעֵצִים הַגְּדֵלִים לְמַכְבִּיר עַל שְׂפַת הַיְאוֹר. וַיִּגְדְּלוּ הַיְלָדִים הָעִבְרִים בְּחֵיק הַטֶבַע, וַיִּתְרוֹעֲעוּ עִם בֶּהֱמַת הַשָּׂדֶה וְחַיַּת הָאָרֶץ, וַיִּלְמְדוּ לָרוּץ בָּרֶגֶל כַּצְבִי וְלִהְיוֹת מְהִירִים כַּצִּפֳּרִים.

וַיִּהְיוּ הַתּוֹאִים הָרוֹעִים בַּדֶּשֶׁא מַעֲנִיקִים לַיְלָדִים מֵחֲלָבָם הַשָּׁמֶן, הַצְּבָאִים וְאַיְלוֹת הַשָּׂדֶה מְכַלְכְּלִים אוֹתָם בְּשָׁרָשִׁים וּגְזָעִים מְלֵאִים עָסִיס, וְהַצִּפֳּרִים מְבִיאוֹת לָהֶם גַּרְגְּרִים כְּחֻלִּים וַאֲדֻמִּים, אַף שָׁרוּ בְאָזְנֵיהֶם שִׁירִים נְעִימִים וַעֲנֻגִּים.

וַיֹּאהֲבוּ הַיְלָדִים אֶת הַצִּפֳּרִים הַמְזַמְּרוֹת וְאָזְנָם קָלְטָה כָל צְלִיל דַּק, כָּל נְעִימָה, וַיִּלְמְדוּ לְהַבְחִין בְּקֶצֶב הַקּוֹלוֹת שֶׁל צִפֳּּרֵי הַשִׁיר.

בִּזְרֹחַ הַשֶׁמֶשׁ בַּבֹּקֶר, בְּקוּם צִפּוֹר הַבִּלְבִּיל וּבַת־זוּגָתוֹ מִשְׁנָתָם לְהַשְׁמִיעַ אֶת שִׁירָתָם בְּקוֹל דַּק וְצָלוּל, כְּקוֹל הֶחָלִיל, הָיוּ גַם הַיְלָדִים עוֹזְרִים עַל יָדָם בַּשִּׁיר; בְּהִשָּׁמַע קוֹל צִפְצוּף הַחִמְרִיָּה, הַקּוֹל הָעֵר וְהַקּוֹרֵא, הָיוּ הַיְלָדִים קָמִים וְיוֹצְאִים לִלְקוֹט יְרָקוֹת וּפַקּוּעוֹת שָׂדֶה; וּבַעֲרוֹב הַיּוֹם, עִם יִלְלַת הַזְּוָעָה שֶׁל הַתַּנִים, הָיוּ הַיְלָדִים עוֹלִים עַל מִשְׁכָּבָם, אֲשֶׁר אִמּוֹתֵיהֶם רִפְּדוּ לָהֶם בֵּין עַנְפֵי הָעֵצִים.

כָּכָה גָדְלוּ הַיְלָדִים בְּרִיאִים וּשְׁלֵמִים בְּחֵיק הַטֶבַע. וְאִם חָלָה פְּעָמִים אֶחָד מֵהֶם, וְיָדְעוּ הַיְלָדִים לְהִשְׁתַּמֵּשׁ בַּצְמָחִים לִרְפוּאָה. וְחָבִיב הָיָה עֲלֵיהֶם פֶּרַח הַבַּבֹּנֶג בַּעַל הַתִּתּוֹרָה הַלְּבָנָה, עַל כִּי אֶת עַלְעָלָיו הָיוּ מוֹצְצִים וּמוֹצְאִים בָּהֶם מָזוֹר מִפְּנֵי הַקַּדָּחַת.

וַיְהִי הַיּוֹם וּפַרְעֹה מֶלֶךְ מִצְרַיִם נָפַל לַמִּשְׁכָּב, וַיִּשְׁאַל בַּעֲצַת חַרְטֻמָּיו. וַיְצַוּוּ הַחַרְטֻמִּים לְהָבִיא מִפִּרְחֵי הַבַּבֹּנֶג הַגְּדֵלִים עַל שְׂפַת הַיְאוֹר, לְהַרְתִּיחָם בַּמָּיִם וּלְהַגִּישׁ לַמֶּלֶךְ הַחוֹלֶה אֶת הַמִּיץ.

וַיֵּצְאוּ שְׁלִיחִים מִטַעַם הַמֶּלֶךְ לֶאֱסוֹף מִן הַפְּרָחִים. וּבְלָקְטָם וְהִנֵּה קָרְבוּ אֶל מְקוֹם מַחֲבוֹא הַיְלָדִים, אַךְ יְדִידֵי הַיְלָדִים, הֵן הַצִּפֳּרִים הָעֵרוֹת וּמַקְשִׁיבוֹת לְכָל רַחַשׁ, הִרְגִּישׁוּ פִתְאֹם בִּשְׁלִיחֵי פַרְעֹה וַתָּעֹפְנָה מֵעָנָף לְעָנָף וַתְּצַפְצֵפְנָה בְקוֹל, כְּאִלּוּ בָאוּ לְהַזְהִירָם מִפְּנֵי הַסַּכָּנָה הַמְרֶחָפֶת.

וּפְרְחֵי הַבַּבֹּנֶג, אֲשֶׁר הוֹשִׁיטוּ עֶזְרָה לַיְלָדִים הַמִּסְכֵּנִים בְּעֵת חָלְיָם, רָעֲדוּ מִפַּחַד הָאָסוֹן הַצָּפוּי לַיְלָדִים, וְהִשִּׁירוּ אֶת עַלְעֲלֵי הַתִּתּוֹרָה הַלְּבָנִים וְרַק לִבָּם, לֵב הַזָּהָב, נִשְׁאֵר לָהֶם.

וְהַיְלָדִים הָעִבְרִים מִהֲרוּ לִשְׁמוֹעַ אֶל אַזְהָרַת הַצִּפֳּרִים וְהַפְּרָחִים וַיְטַפְּסוּ וַיַּעֲלוּ אֶל מַחֲבוֹאָם בֵּין הָעֲנָפִים וַיֵּשְׁבוּ דֹם, וְיַחַד עִם הַצִּפֳּרִים הִקְשִׁיבוּ לְכָל הַנַּעֲשֶׂה מִסָבִיב.

וּשְׁלִיחֵי הַמֶּלֶךְ הִבִּיטוּ כֹּה וָכֹה וַיִּרְאוּ וְהִנֵּה בַמָּקוֹם הַזֶּה אֵין פְּרָחִים בַּעֲלֵי תִתּוֹרָה לְבָנָה, כַּאֲשֶׁר אָמְרוּ לָהֶם הָרוֹפְאִים הַחַרְטֻמִּים, וַיִּפְנוּ וַיֵּלְכוּ לְחַפֵּשׂ בִּמְקוֹמוֹת אֲחֵרִים.

כָּכָה נִמְלְטוּ יַלְדֵי הָעִבְרִים מִשְׁלִיחֵי הַמֶּלֶךְ הָאַכְזָר, אֲשֶׁר בִּקֵּשׁ לְאַבְּדָם.

וַיְהִי כַאֲשֶׁר גָּדְלוּ הַיְלָדִים וְיַחַד עִם אֲבוֹתֵיהֶם יָצְאוּ מִמִּצְרָיִם, וַיִּזְכְּרוּ לְטוֹבָה אֶת פִּרְחֵי הַבַּבֹּנֶג בַּעַל־לֵב־הַזָּהָב, וַיִּקְחוּ מֵהֶם אִתָּם רָב.

מֵאָז נָפוֹץ הַבַּבֹּנֶג בַּעַל לֵב הַזָּהָב בְּלִי עַלְעֲלֵי תִתּוֹרָה בַּמִּדְבָּר וּבְכָל אֶרֶץ יִשְׂרָאֵל, כִּי הֵפִיצוּ אוֹתָם יוֹצְאֵי מִצְרַיִם בְּכָל הַמְּקוֹמוֹת.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
