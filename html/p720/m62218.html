רַבִּי יְהוּדָה הַלֵּוִי

2

על-פי הייגע


א

"תִּדְבַּק לְשׁוֹנִי לְחִכִּי, אַף-חִכִּי יִמַּק בְּלֹעִי

יֻתַּן בֶּעָפָר פִּי, תֵּאָלַמְנָה הַשְּׂפָתָיִם,

תִּשְׁכַּח יְמִינִי, מִקָּנָה תִּשָּׁבֵר אֶזְרֹעִי

אִם-אֶשְׁכָּחֵךְ גַּם-רֶגַע, הַבַּת יְרוּשָׁלָיִם!"


אִמְרוֹת שְׁבוּיֵי צִיּוֹן, אֲמָרוֹת מָרוֹת כַּלַּעֲנָה

מָלְאוּ כָּל-חַדְרֵי בִטְנִי, כָּל-קְרָבַי הַפָּעַם,

קוֹל קִינִים הֶגֶה וָהִי כְּמוֹ אָזְנַי תִּשְׁמַעְנָה

קוֹל הֲמוֹן גְּבָרִים שָׁרִים, עוֹרֲרֵי עֶבְרָה וָזָעַם.


וּבְהִתְעוֹפֵף נַפְשִׁי עַל-כַּנְפֵי הַהִגָּיוֹן

אֶחֱזֶה קְוֻצּוֹת זְקָנֵיהֶם, צִלֲלֵיהֶם יְסֻבּוּנִי;

אָנָּא הַגִּידוּ לִי, אֶקְרָא לְצַלְמֵי-חִזָּיוֹן,

מִי-בָכֶם יְהוּדָה הַלֵּוִי? פָּנָיו הַרְאוּנִי!


אֶפֶס כְּצֵל עוֹבֵר חִישׁ חָלְפוּ צַלְמֵי חֶזְיוֹנִי

אֶל-דְּבָרַי לֹא שָׁעוּ, בְּעֵינֵיהֶם לֹא יָקָרְתִּי

כִּי רוּחוֹת שְׁמֵי-עָל יִבְזוּ נִיב אֱנוֹשׁ כָּמֹנִי,

אַךְ גַּם בִּלְעֲדֵיהֶם אֶת-הַלֵּוִי הִכָּרְתִּי:


תָּוֵי פָנָיו יְעִידוּהוּ, מִצְחוֹ זַךְ כַּשֹּׁהַם

רַעְיוֹנוֹת מַאֲלִיפִים מִמַּעַל לוֹ יִרְחָפוּ,

עֵינָיו נֹצֲצוֹת כִּבְרָקִים, מְפִיקוֹת זִיו וָנֹעַם

בִּדְאָבָה חֲרִישִׁית עָלַי יַחַד נִשְׁקָפוּ;


שִׂפְתוֹתָיו שׁוֹשַׁנִּים, נֹטְפוֹת צְחוֹק מָהוּל בַּתּוּגָה

הֵן נָתְנוּ לִי אֹתוֹתָיו, הֵן הֵנָּה בּוֹ עָנוּ,

מַקְבִּילוֹת וּמְתֹאָרוֹת כְּמוֹ בַשֶּׂרֶד וּמְחוּגָה,

בְּשִׂפְתֵי-חֶמֶד כָּאֵלֶּה רַק מְלִיצֵי-יָהּ הוּחָנוּ.


גַּלְגִּלֵּי מַרְכְּבוֹת הָעֵת הַשְׁקֵט לֹא יוּכָלוּ

הַשָּׁנִים כַּעֲנָנִים תְּעוּפֶינָה כֻּלָּהְנָה:

מִיוֹם לֶדֶת יְהוּדָה אֶל-חֵיק הַנֶּצַח הוּבָלוּ

בְּעֶרְכְּךָ זֶה כְּבָר חֲמֵשׁ עֶשְׂרֵה יוֹבְלוֹת שָׁנָה.


בְּקִרְיַת תּוֹלֵדָה אֲשֶׁר בְּקַסְטִילִיָּה הַהֻלָּלָה

שָׁם רָאָה לְמַבָּרִאשׁוֹנָה קַרְנֵי הַחֶרֶס,

עַל-נְהַר תַּחוֹ, גַּלֵּי זְהָבוֹ יְרֻצוּן כְּאַיָּלָה,

שָׁם שׁוֹרֲרָה לּוֹ יוֹלַדְתּוֹ אֶת-שִׁירֵי הָעֶרֶשׂ:


"בְּאֶרֶץ הַשְּׁפַנִּים

פַּרְדֵּסִים וְגַנִּים,

גַּנּוֹת וּכְרָמִים

וְכָל-מִינֵי בְשָׂמִים,

כְּפָרִים עִם-נְרָדִים

וְכָל-פְּרִי מְגָדִים,

אֶל-חֵךְ יִמְתָּקוּ

וּשְׂפָתַיִם יִשָּׁקוּ;

אַךְ תּוֹרָה וְחֻקִּים

מִנֹּפֶת מְתוּקִים,

אֶת-לִבְּךָ יֵיטִיבוּ

אֶת-נַפְשְׁךָ יָשִׁיבוּ;

בַּר-בִּטְנִי יְהוּדָה!

תּוֹרָה וּתְעוּדָה

רִפְאוּת תְּהִי לְשָׁרֶּךָ

וְשִׁקּוּי לְעַצְמוֹתֶיךָ!

*

"בְּאֶרֶץ הַשְּׁפַנִּים

גְּדוֹלִים וּקְטַנִּים,

לִבָּם מְאֹד חָרָד

אֶל לְשׁוֹן סְפָרָד,

הַאֻמְנָם זוֹ שָׂפָה

צַחָה גַּם-יָפָה,

וּנְעִמוֹת הוֹמִיָּה

שְׂפַת קַסְטִילִיָּה;

אַךְ מֶנָּה יְפֵה-פִיּהָ

הַשָּׂפָה הָעִבְרִיָּה,

אוֹצַר כָּל-תּוּשִׁיָּה

וְרַבַּת עֲלִילִיָּה,

עֲנֵפָה וּפוֹרִיָּה

לָהּ תְּהִלָּה דּוּמִיָּה;

בָּהּ אִמְרֵי-שֶׁפֶר

תִּכְתֹּב בַּסֵּפֶר!


"בְּאֶרֶץ הַשְּׁפַנִּים

חוֹרִים וְחַשְׁמַנִּים,

רַבֵּי אוֹצָרוֹת

וַאֲבָנִים יְקָרוֹת,

אֵלֶיהָ יוּבָא בַצִּים

חֵיל כָּל-הָאִיִּים,

סֹחֲרֶיהָ יַעֲשִׁירוּ

לְכָל-חֹסֶן יַגְבִּירוּ;

אַךְ דְּבַר אֱלוֹהַּ

גָּבוֹהַּ מֵעַל-גָּבוֹהַּ,

וְיָקָר שִׁבְעָתָיִם

מִזְּהַב פַּרְוָיִם;

וּבְכֵן יוֹמָם וָלָיִל

בַּתּוֹרָה עֲשֵׂה חָיִל,

אָז יוֹדוּךָ אַחֶיךָ

וְתָגֵל יוֹלַדְתֶּךָ!


"בְּאֶרֶץ הַשְּׁפַנִּים

נֵשְׁבָה שַׁאֲנַנִּים,

הֲמוֹנִים הֲמוֹנִים

בְּטִירוֹת וְאַרְמוֹנִים,

נִלְבְּשָׁה כִסְגָנִים

שָׁנִי עִם-עֲדָנִים,

טוּב הָאָרֶץ נֹאכֵלָה

וְאֶת-מַלְכָּהּ נְהַלֵּלָה;

אַךְ מִי-זֶה יוֹדֵעַ

וּלְיָדֵנוּ יִתָּקֵעַ,

מַה-יֵּלֶד יוֹם

לִלְאֹם וָאֹם?

הָאָרֶץ הַקְּדוֹשָׁה

הִיא אֶבֶן הָרֹאשָׁה,

וּכְבָבַת עֵינָיִם

נְצֹר זֵכֶר יְרוּשָׁלָיִם!"


בִּלְשׁוֹנָהּ מִלֲלָה הָאֵם וּבְיָדוֹ מִלֵּא אָבִיהוּ

וַיְנַהֲלֵהוּ בְיַד חָרוּצִים וּבְרוּחוֹ הַקָּשָׁה,

רֵאשִׁית לִמּוּדִים הוֹרָהוּ לֹא בְּאִמְרֵי פִיהוּ

כִּי אִם-בְּתוֹרַת אֲדֹנָי, לִקְהִלַּת יַעֲקֹב מוֹרָשָׁה.


אַךְ לֹא בִשְׂפַת הָאָרֶץ, כִּי פְשֹׁטָה הִיא וְעֹרָה,

כִּי אִם בִּשְׂפַת-קֹדֶשׁ, שְׁפַת הַר-הַפְּלָאוֹת,

בְּאוֹתִיּוֹתֶיהָ הַחֲמֻדוֹת הֵחֵל קְרוֹא אֶת-הַתּוֹרָה

אוֹתִיּוֹת בְּנוֹת אֶרֶץ אַשּׁוּר, אוֹתִיּוֹת הַמְרֻבָּעוֹת.


עוֹד בְּשָׁנִים קַדְמֹנִיּוֹת, בִּימֵי יַלְדוּת הֶחָלֶד

מִכְתַּב אֱלֹהִים זֶה בִּיפִי תַבְנִיתוֹ הוֹפִיעַ,

וּבְכֵן גַּם-יְשַׁעֲשַׁע תָּמִיד נֶפֶשׁ עוֹלָל וָיָלֶד

לִקְרַאת מְרֻבְּעֵי-פָנִים אֵלֶּה יָשִׂישׂ אַף-יָרִיעַ.


הִנֵּה כִּי-כֵן דִּבְרֵי תוֹרַת מֹשֶׁה בִּשְׂפַת-עֵבֶר

בְּקוֹל הֲמוֹן שִׁירָה וְזִמְרָה מִפִּי יְהוּדָה נִשְׁמָעוּ;

נְגִינוֹת קֹדֶשׁ הָאֵל יָסְדוּ סוֹפְרִים חֹבְרֵי חֶבֶר

לִמְקֹמוֹתֵיהֶם בְּמוֹשְׁבֹתֵיהֶם “טְעָמִים” נִקְרָאוּ.


חֵן הוּצַק בְּשִׂפְתֵי הַיֶּלֶד, מַה-נָּאֲוָה מִדְבָּרוֹ

גַּם הִפְלִיא לְבַטֵּא כַמִּשְׁפָּט אֶת-אוֹתִיּוֹת הַגָּרוֹן,

וַיְרַנֵּן טַעַם “שַׁלְשֶׁלֶת” בְּכָל-תָּקְפּוֹ וַהֲדָרוֹ

כִּצְלִיל רִנֲנַת הַזָּמִיר עַל-חֲבַצֶּלֶת הַשָּׁרוֹן.


גַּם תַּרְגּוּם אוֹנְקְלוֹס הַגֵּר, פֹּעַל יְדֵי צַדִּיק תָּמִים

אֲשֶׁר הִשְׂכִּיל אֶל-סִתְרֵי תוֹרַת-יָהּ וּמַצְפֻּנֶיהָ,

אַךְ שְׂפַת לֹעֵז זוּ “אֲרַמִּית” כִּנּוּהָ חֲכָמִים

כִּי מִן-אֲרָם תִּהֲלַךְ, מֵהַרֲרֵי-קֶדֶם מוֹצָאֹתֶיהָ.


עַל-מִשְׁפְּחוֹת לְשֹׁנוֹת בְּנֵי-שֵׁם הֲכִי גַם-הִיא נֶחֱשָׁבָה

אָכֵן נְעִימַת נְאֻמֵּי הַנְּבִיאִים מֶנָּה נֶעְדָּרֶת,

יֶתֶר שְׂאֵת וָעֹז לִשְׂפַת-קֹדֶשׁ נַעֲלָה וְנִשְׂגָּבָה

וְהָאֲרַמִּית לְעֻמָּתָהּ כְּשִׁפְחָה לִפְנֵי הַגְּבָרֶת.


תַּרְגּוּם אוֹנְקְלוֹס הַלָּז הָיָה לַחְמוֹ לֶחֶם שְׁמֵנָה

בְּעֹדֶנּוּ בְאִבּוֹ הִטָּה אֵלָיו אֹזֶן קַשָּׁבֶת,

וּשְׂפַת-אֲרָם גַּם הָיְתָה לּוֹ עֶזְרָה וּמַשְׁעֵנָה

עֵת אֶל-שֻׁלְחַן “הַתַּלְמוּד” הִקְרִיבוֹ אָבִיו לָשָׁבֶת.


כִּי הִנֵּה בְּשַׁחַר טַל יַלְדוּתוֹ, בְּעוֹדוֹ נָעַר

בְּמִשְׁנַת רַבִּי יְהוּדָה הַנָּשִׂיא הֶרְאָהוּ בְרָכָה,

אַחַר אֶל-הֵיכַל הַתַּלְמוּד פָּתַח לוֹ שָׁעַר

וַיַּנְחֵהוּ בְאֹרַח תְּבוּנוֹת, אֹרַח “הַהֲלָכָה”.


וְהַתַּלְמוּד מַה-הוּא? הֲלֹא הוּא בֵּית-תַּכְסִיסֵי-מִלְחָמָה

אַךְ לֹא-בַחֲרָבוֹת וּבַחֲנִיתוֹת יַעַרְכוּ בוֹ קְרָב,ֹ

גִּבּוֹרֵי הַתּוֹרָה יִתְנַגְּשׁוּ, יֵאָבְקוּ שָׁמָּה

בְּשׂוֹם שֶׂכֶל וְהַעֲמִיק עֵצָה אִישׁ אִישׁ יָדָיו לוֹ רָב!


בַּמַּעְגָּל יָרוּצוּ עַל-פִּי “מִדּוֹת” שְׁלֹשִׁים וּשְׁתָּיִם

יִתְרֹצְצוּ יַחַד, יִלָּחֲמוּ מִלְחָמָה נִפְלָאָה,

וְגִבּוֹרִים אַנְשֵׁי-שֵׁם, לְנַגֵּחַ לָמוֹ קַרְנָיִם

הֵם חַכְמֵי יְשִׁיבוֹת יַבְנֶה, פֻּמְבְּדִיתָא וּנְהַרדָּעָה.


פֹּה לִמֵּד יָדוֹ לַקְּרָב, אַךְ בְּקֶשֶׁת לֹא-רְמִיָּה

בְּפִלְפּוּלִים לֹא-נִפְתָּלִים יְמִינוֹ בַכֹּחַ נֶאְדָּרִי,

אֲשֶׁר עַל-כֵּן הִפְלִיא עֵצָה גַּם-הִגְדִּיל תּוּשִׁיָּה

בְּסֵפֶר אוֹצַר כָּל-חֶמְדָּה, זֶה סֵפֶר “הַכּוּזָרִי”.


אֶפֶס כַּאֲשֶׁר אוֹר-מִשְׁנֶה יְחַלֵּק הָרָקִיעַ

מִשְּׁנֵי הַמְּאֹרֹת הַגְּדֹלִים בַּשַּׁחַק הוּשָׂמוּ,

הַשֶּׁמֶשׁ בִּגְבוּרַת עֻזָּה תִּרְפָּא גַּם-תַּפְגִּיעַ

וְיָרֵחַ יָקָר הֹלֵךְ קַוָּיו תָּמִיד יִנְעָמוּ;


כָּכָה גַּם בִּרְקִיעַ הַתַּלְמוּד לְאוֹר-מִשְׁנֶה מְעֹנָה

לַהֲלָכָה וְהַגָּדָה יִפָּלֵג שֶׁטֶף אוֹרֵהוּ,

“בֵּית-תַּכְסִיסֵי-מִלְחָמָה” קָרָאתִי לָרִאשֹׁנָה

וְגַם לַשֵּׁנִית שֵׁם אִתִּי, יָדַעְתִּי אֲכַנֵּהוּ:


“גִּנַּת הַבִּיתָן” זֶה הַשֵּׁם פִּ יִקֳּבֶנָּה,

בָּהּ כָּל-מַעֲלוֹת רוּחַ, יִצְרֵי מַחְשְׁבוֹת נֹטְעִים מְחֻכָּמִים

כָּל-מַחֲזֶה נַעֲלֶה וְנִשָּׂא לֹא אָצְלוּ מִמֶּנָּה;

הַאֻמְנָם גַּן-פְּלָאוֹת כָּזֶה כְּבָר הָיָה לְעֹלָמִים:


הִנֵּה כָמוֹהוּ גַם-הוּא בְּאֶרֶץ-בָּבֶל יְסוּדָתוֹ

עַל-בִּקְעַת שִׁנְעָר שָׁם עֹלֲלוּ עֲלִילוֹת גְּדֹלוֹת וְרָמוֹת,

כְּאַחַד שֶׁבַע מִפְלְאוֹת תֵּבֵל הִתְנוֹסֵס בְּגַאֲוָתוֹ

הֹרָה וְהֹגָה מִלֵּב מַלְכָּה אַדִּירָה שְׁמִירָמוֹת.


שְׁמִירָמוֹת הַמַּלְכָּה עוֹדָהּ יַלְדָּה קְטַנָּה וּמִצְעָרָה

צִפֳּרֵי הַשָּׁמַיִם כִּלְכְּלוּהָ אַף-גִּדְּלוּהָ,

וְכַאֲשֶׁר צָלְחָה לִמְלוּכָה עַל-כִּסֵּא עֹז וְתִפְאָרָה

דַּרְכֵי צִפֳּרִים וְאָרְחוֹתָם גַּם-אָז לֹא עֲזָבוּהָ.


רַכָּה וַעֲנֻגָּה לֹא-נִסְּתָה הַצֵּג רַגְלָהּ עַל-אָרֶץ

הִתְהַלֵּךְ עַל-אַדְמַת-עָפָר כְּכֹל יוֹנְקֵי שָׁדָיִם,

מְקוֹם שָׁם תִּרְמֹסְנָה רַגְלֵי כָל-בְּהֵמָה וָשָׁרֶץ,

וַתִּטַּע לָהּ גַּן עֵדֶן, תָּלוּי בֵּין אֶרֶץ וְשָׁמָיִם;


וַיִּיף בְּרֹב גֻּדְלוֹ הַתָּמָר וּבְרוֹשִׁים רַעֲנַנִּים

עַל-בָּמֳתֵי עַמּוּדֵי עֲנָק הַבְּנוּיִם לְתַלְפִּיּוֹת,

רִמֹּנֵי זָהָב וּפִרְחֵי הוֹד בֵּין עֲרֻגוֹת שׁוֹשַׁנִּים

פְּסִילֵי שַׁיִשׁ וּשְׂכִיּוֹת חֶמְדָּה עִם-גֻּלֹּת עִלִּיּוֹת.


בְּבִינַת בּוֹנִים אֳמָנִים כָּל-אֵלֶּה חֻבְּרוּ נִסְמָכוּ

עַל-מַעְבְּרוֹת צְרֻרֹת וּגְשָׁרִים קְטֻרִים כְּמוֹ חֲלָאִים,

כְּזַלְזַלֵּי גֶפֶן שׂרֵק יַעֲלוּ שׁוּר, יַחְדָּו יְסֻבָּכוּ

וְכָל-עוֹף כָּנָף יְקַנֵּנוּ שָׁם בֵּינוֹת לָעֳפָאיִם.


שָׁם צִפֳּרִים אַבִּירוֹת מַזְהִירוֹת בִּשְׁלַל-צֶבַע

מַעֲמִיקוֹת חֵקֶר כָּל-הַיּוֹם, אֵין הֶגֶה בִּלְשׁוֹנָן

וְצִפֳּרִים קְטַנּוֹת עֲדִינוֹת, עָפוֹת מֵהַר אֶל-גֶּבַע

אַךְ מְצַפְצְפוֹת וּמַהְגּוֹת, אַף-לֹא יֵחַר גְּרוֹנָן.


שֳָם נִשְׁמַת כָּל-חַי תִּשְׁאַף רֵיחַ מוֹר וְקִנְּמָן-בֶּשֶׂם

עַל-יְקַר כָּרִים וִירַקְרַק חָרוּץ רוּחַ צַחָה נִשְׁפָּכָה,

לֹא יַשְׁבִּית מִטֳּהָרָהּ, כַּאֲשֶׁר לֹא יוּעַם הַכֶּתֶם,

אֵד עוֹלֶה מִן-הָאֲדָמָה, מִן-עֵמֶק הַבָּכָא.


הַהַגָּדָה גַם-הִיא גִּנַּת בִּיתָן כָּזֹאת הִנֶּהָ:

יַעֲרֵי אַלּוֹנִים וּפַרְדְּסֵי רִמֹּנִים מְעוֹפְפִים בָּרוּחַ

כָּל-יְצוּרֵי דִמְיוֹן-עַז יָלִינוּ בֵין שְׂדֵרֹתֶיהָ;

אֵלֶיהָ יַעֲלוּ הֹגֵי הַתַּלְמוּד וְיִתְהַלְּכוּ לָשׂוּחַ.


אִם-קָצְרָה נֶפֶשׁ הָעֶלֶם בַּעֲמַל הַפִּלְפּוּלִים

אִם בַּהֲלָכָה מְמֻשָּׁכָה בִּינָתוֹ מִלֶּכֶת עָמָדָה,

אָז יְשִׁיבֶנָּה יְרַפְּדֶנָּה בְּתַפּוּחֵי קֹדֶש הִלּוּלִים

אָז יִשְׁאַף רוּחַ נְדִיבָה בְּתוֹכֵכִי הַהַגָּדָה.


הֲכֹל הַיּוֹם יַחֲרֹש הַחֹרֵשׁ וִיפַלַּח אַדְמָתוֹ?

הֲלֹא אִם-חֹם שֶׁמֶשׁ בְּחֵמָה יַךְ עַל-קָדְקֳדוֹ

אָז יַעֲזֹב לִרְגָעִים אֶת-אֵתוֹ וּמַחֲרֵשָׁתוֹ

וּבְצֵל אֵלָה עֲבֻתָּה יִנָּפֵשׁ מִמַּעְבָּדוֹ.

  • * *

הַּהַגָּדוֹת הַנֶּחֱמָדוֹת מִמְּקוֹר נֶאֱמָן נָבָעוּ

בְּשְׁנוֹת-קֶדֶם דּוֹרוֹת עוֹלָמִים חוֹלֲלוּ נוֹלָדוּ,

עַל-מִסְתְּרֵי אֵל אֵלִים וְאֶרְאֶלָּם בְּחִידוֹת יִשְתָּעוּ

רָזֵי עוֹלָם כַּחֲרוּזֵי פְנִינִים יַחַד נִצְמָדוּ.


עַל-נְצוּרוֹת סוֹד הַיְצִירָה וּשְׁכִינַת-אֵל הַכְּבוּדָּה

עַל-אֲבוֹת הַקֹּדֶשׁ, חֹזֵי-יָהּ וּצְבִי מַלְכֵי יְרוּשָׁלָיִם,

עַל-גְּאוֹן מִקְדְּשֵׁי שַׁדַּי וְעַל-שֶׁבֶר גִּבֹּרֵי יְהוּדָה

שֶׁמָּסְרוּ נַפְשָׁם לַהוֹרֵג עַל-קְדֻשַּׁת שֵׁם שָׁמָיִם.


הַדִּמְיוֹן וְהַגּוּזְמָא הָיוּ לִכְלִי-חֵפֶץ בִּידֵיהֶם

חִידוֹת, מְשָׁלִים וַהֲתֻלִּים לִסְגֻלָּתָם בָּחָרוּ,

לְעוֹרֵר יְשֵׁנִים מִתַּרְדֵּמָה, לִפְקֹחַ עֵינֵיהֶם

לְהַטּוֹת לַמּוּסָר אָזְנָם וּלְבַל יִכְסְלוּ וְיִבְעָרוּ;


לָתֵת מַרְפֵּא לִשְׁבוּרֵי רוּחַ, לְחַבֵּשׁ לְעַצְּבוֹתָם

לְחַזֵּק יְדֵי פְלִיטֵי יְשֻׁרוּן הַנֹּדְדִים בַּלְאֻמִּים,

לְהָבִיא תִקְוָה לַנּוֹאָשִׁים וּלְחַזֵּק לִבּוֹתָם

כִּי עוֹד יוֹסִיף יִזְרַח לָמוֹ אוֹר יֶשַׁע וְנִחֻמִּים.


כְּמַעְיָן מִזְעָר יִזַּל, יְזַנֵּק מֵרֹאשׁ נִקְרוֹת צֻרִים

יֵלֵךְ הָלֹךְ וָרָב וְהָיָה לִנְחָלִים נִכְבַּדֵּי-מָיִם,

כֵּן רֵאשִׁית הַהַגָּדָה מִצְעָר, סִפּוּרֶיהָ סְפוּרִים

וְאַחֲרִיתָהּ יִשְׂגֶּא מְאֹד – מִדְרָשִׁים רַחֲבֵי יָדָיִם.

*

אֶל-גִּנַּת הַהַגָּדוֹת אָז יֵרֵד יְהוּדָה הַנַּעַר

יִשְׁאַב יִשְׁאַף רוּחַ צַח וְרִגְשֵׁי לִבּוֹ נֵעֹרוּ,

יִתְעַנֵּג עַל-שְׁתִילָיו, יִשְׁתַּעֲשַׁע לְנֹגַהּ הַסַּהַר

אַף-יִטְעַם מִּיַּעֲרַת הַדְּבָשׁ – וּשְׁתֵּי עֵינָיו אֹרוּ!


חַדְרֵי מַשְׂכִּיּוֹת לִבּוֹ חֶרְדַּת קֹדֶשׁ חָדָרָה

לַהֶבֶת שַׁלְהֶבֶת-יָהּ אֶת-בָּתֵּי נַפְשׁוֹ הֵאִירָה;

אָז יָחוּשׁ כִּי רוּחַ מִמָּרוֹם עָלָיו עָבָרָה,

כִּי נָחָה עָלָיו רוּחַ אֱלֹהִים, רוּחַ הַשִּׁירָה.


יֵשׁ מְשׁוֹרְרִים נַפְשׁוֹתָם רוּחַ רֶגֶשׁ תִּמָּלֶאנָה

אֶפֶס כְּמוֹ גַלֵּי יַם-זֹעֵף מִלֵּיהֶם תַּעֲבֹרְנָה;

זִמְרוֹתֵיהֶם כֻּלָּן בְּלִי-סְדָרִים פְּרָצוֹת תֵּצֶאנָה,

יַעַן מִתֵּת לָרוּחַ מִשְׁקָל יְדֵיהֶם תִּקְצֹרְנָה.


אוּלָם יְהוּדָה רָד עִם-חֻקֵּי הַזִּמְרָה כֻּלָּמוֹ

וְעִם-מִשְׁפְּטֵי הַשִּׁירָה הָיָה נֶאֱמָן כָּל-הַיָּמִים,

וְשָׁקַל בַּפֶּלֶס כָּל-שִׁירָיו וְשַׁכֻּלָּה אֵין בָּמוֹ;

גַּם-זֹאת נַחֲלַת שַׁדַּי הִיא, נַחֲלַת מְשׁוֹרֵר תָּמִים.


כִּי לֹא רַק חָכָם נוֹדָע לְשֵׁם בָּאֲרָצוֹת בָּאִיִּים

לֹא רַק חוֹקֵר אַדִּיר וְנָאוֹר הָיָה זֶה הַגֶּבֶר,

אַךְ גַּם כְּהֵימָן וּכְאָסָף רָאשֵׁי מְשׁוֹרְרֵי הַלְוִיִּם

מְשׁוֹרֵר מָרוֹם וְנִשָּׂא מְאֹד, נֶשֶׁר אֶרֶךְ הָאֵבֶר!


אָמְנָם כֵּן, מְשׁוֹרֵר מַפְלִא לַעֲשׂוֹת מֵאֵין כָּמֹהוּ

כּוֹכַב הַשַּׁחַר לִבְנֵי-יְהוּדָה הָיְתָה שִׁירָתוֹ,

וְכִמְנֹרַת הַמָּאוֹר הֵאִירָה חֶשְׁכַת לֵיל תֹּהוּ,

מְשׁוֹרֵר כַּבִּיר וְנִשְׂגָּב הוּא, אֵין-חֵקֶר לִגְדֻלָּתוֹ.


כְּעַמּוּד הָאֵשׁ אֲשֶׁר-הָלַךְ לִפְנֵי הָעָם בַּלָּיִל

בִּנְסֹעַ יִשְׂרָאֵל בַּמִּדְבָּר, אֶרֶץ נִמְלָחָה,

כֵּן לִפְנֵי אֹרְחַת הַגָּלֻת הַחֵל יִצְעַד בֶּחָיִל

לַנְחוֹתָהּ בַּדֶּרֶךְ הָעוֹלָה קֵדְמָה מִזְרָחָה!


קֹשֶט וָתוֹם שִׁירָתוֹ, זַכָּה צַחָה וּבָרָה

כְּנַפְשׁוֹ הַתְּמִימָה וּנְעִימָה בִּיקַר חֲמֻדוֹתֶיהָ;

עֵת יַד אֱלֹהֵי הָרוּחוֹת נֶפֶשׁ זוּ יָצָרָה

שָׂמַח בִּיצוּר כַּפָּיו וַיָּשֶׂשׂ מָשׂוֹשׂ עָלֶיהָ,


וַיִּשַּׁק לְמוֹ-פִיו אֶת-הַנֶּפֶשׁ הַנֶּאֱהָבֶת,

וְצִלְצַל מַשַּׁק נְשִׁיקַת-יָהּ עַל-פֶּה וָמֶצַח

בְּכָל-שִּׁירוֹת הַמְשׁוֹרֵר אָזְנֵנוּ קַשָּׁבֶת

כִּי בִנְשִׁיקַת אֱלֹהֵי הַשִּׁירָה קָדְשׁוּ לָנֶצַח!


כִּבְכָל-פֹּעַל גֶּבֶר, כֵּן בִּמְלִיצָה, שִׁיר וָשִׂיחַ

לֹא-יַשְׂגִּיב אִישׁ בְּכֹחוֹ בִּלְתִּי אִם-אֲדֹנָי יְחֻנֵּהוּ,

הָאִישׁ חֲנָנוֹ יָהּ – בְּכֹל אֲשֶׁר-יִפְנֶה יַצְלִיחַ

גַּם בְּשִׁירוֹ גַּם בְּשִׂיחוֹ לֹא-יֶחֱטָא יְסַלֵּף דַּרְכֵּהוּ.


לִמְשׁוֹרֵר כָּזֶה, אֲשֶׁר חֵן-שַׁדַּי חוֹפֵף עָלָיו

אִישׁ רוּחַ יִקָּרֵא לוֹ, מוֹפֵת הוּא לִבְנֵי דוֹרוֹ,

בְּמַמְלֶכֶת הָרַעֲיוֹנוֹת בְּכָל-מַעֲלָלָיו וּמִפְעָלָיו

שִׁלְטוֹן בְּלִי-מְצָרִים לוֹ, יֵלֵךְ וְיִסַּע לְאוֹרוֹ.


דִּין וְחֶשְׁבּוֹן יִתֵּן רַק לְיוֹצֵר רוּחוֹ וְנִשְׁמָתוֹ

רַק בִּידֵי מְחוֹלְלוֹ לְשָׁפְטוֹ, לְהַאְשִׁים אוֹ לִסְלֹחַ,

אֶפֶס הָעָם עַל-תְּבוּנַת כַּפָּיו וְעַל-דִּבְרַת שִׁירָתוֹ

הֵן יוּכַל הֲמִיתוֹ, אַךְ לְשָׁפְטוֹ לֹא יַעֲצָר-כֹּחַ!



ב

"עַל-נַהֲרוֹת בָּבֶל, מִשְּׁאוֹן דָּכְיָם צָלְלוּ אָזְנֵינוּ

שָׁם יָשַׁבְנוּ גַּם-שָׁפַכְנוּ דִמְעָה כְּנַחַל-מָיִם,

עַל-עֲרָבִים בְּתוֹכָהּ תָּלִינוּ כִּנֹּרוֹתֵינוּ

עַל-אַלּוֹנֵי בָכוּת תָּלִינוּ עֻגָב וּמְצִלְתָּיִם".


הֲתֵדַע שִׁיר-קְדוּמִים זֶה? הֲבָאתָ עַד-תְּכוּנָתוֹ?

זֶה הַחִלּוֹ בְּקוֹל בִּכְיָה תַּאֲנִיָּה מְעוֹרֶרֶת,

מְעַט מְעַט יִרְעַשׁ יִתְגָּעַשׁ, יָהִים בְּקוֹל יִלְלָתוֹ

כַּהֲמוֹת דּוּד נָפוּחַ עַל-הָאָח הַמְבֹעָרֶת.


זֶה שְׁנוֹת אַלְפַּיִם מֵעַי רֻתְּחוּ וְלֹא-דָמּוּ

לִבִּי יִתְחַמֵּץ בְּקִרְבִּי וּבְחֵיקִי כִּלְיוֹתַי כָּלוּ;

רַבּוֹת צְרָרוּנִי מִנְּעוּרָי, רַבּוֹת בִּי נִלְחָמוּ

חִנָּם הִרְבּוּ פְצָעַי וְכָל-דְּמֵי לִבִּי הָרְעָלוּ.


בִּשְׁחִין רָע הֻכֵּיתִי מִכַּף רַגְלִי עַד קָדְקֳדִי

נַחְלָה מַכָּתִי אֲנוּשָׁה, מַכַּת אִיּוֹב דִּמְיוֹנָהּ,

גַּם חֶרֶשׂ לְהִתְגָּרֵד בּוֹ הֲלֹא אֵין יֵשׁ בְּיָדִי,

רַק הָעֵת כַּכֶּלֶב תָּלֹק זֹב פְּצָעַי בִּלְשׁוֹנָהּ.


רַב-תּוֹדוֹת לָךְ, הַכֶּלֶב, בְּרִיר שְׂפָתֶיךָ תְּחָנֵּנִי

הֵן תָּקֵל מַכְאֹבִי, אַךְ הַתְרַפֵּא מַכַּת צַלְמָוֶת?

הָהּ, רַק הַמָּוֶת לְבַדּוֹ מֵחָלְיִי יַחֲלִימֵנִי,

אֶפֶס הֵן נֵצַח יִשְׂרָאֵל אָנִי, בֵּן אַל-מָוֶת…


הַיָּמִים קַלּוּ מִנִּי-אָרֶג, נֶחְפְּזוּ לָלָכֶת

הַשָּׁנִים תְּנוּסֶינָה כָּעֹפֶר עַל-הָרֵי בָתֶר,

וְיַד עֹשָׂה פֶלֶא תִּתְמֹךְ בָּאֶרֶג וּבְמַסָּכֶת

תֶּאֱרֹג תִּקְוַת חוּט הַמִּקְרִים, תִּרְקֹם בַּסָּתֶר.


רַק קוֹלָהּ יִשָּׁמַע וְעֵין רֹאִי לֹא תְשׁוּרֶנָּה

וּמַה-זֹּאת הָרִקְמָה נִבְעַר כָּל-אֹרֵג מִדָּעַת,

בִּלְתִּי הָאֱלֹהִים לְבַדּוֹ, רַק הוּא יֵדָעֶנָּה

אַךְ לוֹ נִתְכְּנוּ עֲלִילוֹת, לֹא לְבֶן-אָדָם תּוֹלָעַת.


הַיָּמִים קַלּוּ מִנִּי-רָץ, מַה-נֶּחְפְּזוּ לָלָכֶת,

הַשָּׁנִים חִישׁ תַּחֲלֹפְנָה, כִּצְבִי מְדַלֵּג עַל-הֶהָרִים,

וְדִמְעַת עֲשׁוּקִים כְּקֶדֶם אַרְצָה נִתָּכֶת

וּדְמֵי עֲנִיִּים נְקִיִּים עוֹד בְּמוֹרָד נִגָּרִים.


הָרִשְׁעָה תַּעֲרֹץ אָרֶץ, עַל כֹּל זְרֹעָהּ רָמָה

תִּטְרֹף כְּנָמֵר שֹׁקֵד, תֹּאכַל וְתָדֹק בְּשִׁנֶּיהָ;

וּדְמָעוֹת וְדָמִים גַּם-יַחַד תִּבְלַע הָאֲדָמָה

כְּאֶל-מַלְקוֹשׁ תִּפְעַר-פֶּה וּתְכַסֶּה עַל-הֲרוּגֶיהָ…


הָהּ רִשְׁעָה בַּת-תָּפְתֶּה, אֵיךְ אֲכַלְכֵּל מַעֲלָלָיִךְ?

מַה-יִּרְתַּח הַדּוּד! מִכְסֵהוּ נָקַע כְּאֶבֶן הַקָּלַע!

אַשְׁרֵי הַגֶּבֶר יִתְעוֹרֵר עַל-יוֹצְאֵי חֲלָצָיִךְ

“אַשְׁרֵי שֶׁיֹּאחֵז עֹלָלַיִךְ וְנִפֵּץ אֶל-הַסָּלַע!”,


רוּחַ בִּטְנִי הֱצִיקַתְנִי, נִלְאֵיתִי נְשׂא מַתְּלָאָה

רוּחַ עִוְעִים תְּקָפַתְנִי וּמְרֵרָתִי נִשְׁפָּכָה,

אַךְ דִבַּרְתִּי וְרָוַח לִי, סָרָה הָרוּחַ הָרָעָה

יִשְׁקֹט הַדּוּד עַל-גֶּחָלָיו – וַחֲמָתִי שָׁכָכָה.


וְהִנֵּה זֶה הַכְּרוּב, לְרִכְבֵי דִמְיוֹנִי סוּסָתִי,

בְּרָגְזִי עָמַד מַשְׁמִים וַיָּלֶט פָּנָיו בַּאֲגַפָּיו,

הִתְנַעֵר חָלַף רוּחַ וַיַצְהֵל פָּנָיו לִקְרָאתִי

וּכְמוֹ דָבָר לָאַט עִמּוֹ יִקָּחֵנִי בְעַפְעַפָּיו:


"הִנֵּה-נָא שָׁב אַפְּךָ אֲדֹנִי, חָדַל גַּם-כְּאֵבִי

הֲתָשׁוּב אֶל-גֵּיא הַחִזָּיוֹן, מְקוֹם רַגְלִי צָעָדָה,

הַאֶשָּׂאֲךָ שֵׁנִית אֶל-בְּחִירְךָ יְהוּדָה הַלֵּוִי? – "

וָאֶרְכַּב עַל-כְּרוּב וָאָעוּף, וָאֵדֶא סְפָרָדָה.


אָמְנָם כֵּן בְּחִיר הַמְשׁוֹרְרִים הוּא, לוֹ שְׂאֵת וְיֶתֶר

מִכָּל-בְּנֵי גִילוֹ יֶחְכַּם בִּמְלֶאכֶת שִׁיר וּרְנָנָה,

הוּא מֶלֶךְ בָּעוֹלָם הָרַעְיוֹן, עַל-רֹאשׁוֹ הַכֶּתֶר

כִּי חֲנָנוֹ אֵל-חַנּוּן שְׁאָר רוּחַ וְלֵב מַתָּנָה.


וּבְרוּחוֹ הַכַּבִּיר בָּרָא לוֹ לַבְּקָרִים חֲדָשִׁים

שִׁירִים מִשִּׁירִים שׁוֹנִים, מְאֹד נִפְלָאִים לְמִינֵיהֶם

שַׁלְהֶבֶת-יָהּ יֹקֶדֶת בָּם, אֵשׁ קֹדֶשׁ הַקֳּדָשִׁים,

אַף-נַפְשׁוֹ נְשׁוּקַת-אֵל תַּצְהִיר בֵּין שׁוּרוֹתֵיהֶם!


הַאֻמְנָם לֹא נֹפֵל הוּא הָעִבְרִי רַבִּי יְהוּדָה

מִכָּל-מְשׁוֹרֲרֵי זְמַנּוֹ, בְּשֵׁם “טְרֻבַּדּוּרִים” נִקְרָאוּ,

כָּל-תֹּפְשֵׂי מִנִּים וְעֻגָב בְּאֶרֶץ צָרְפַת הַחֲמוּדָה

שֶׁשָּׁם תַּפּוּחֵי זָהָב עֲצֵי-גַנִּים יִשָּׂאוּ.


בַּמְּדִינָה הָעֲדִינָה הַזֹּאת, יְפֵה נוֹף וְחֶבֶל,

בַּעֲלֹץ עֲלִיזֵי רוּחַ וְנַפְשָׁם גָּרְסָה לְתַאֲבָה,

עַל-עֲלָמוֹת שִׁירָה פָּרְטוּ עַל-פִּי כִנּוֹר וָנֵבֶל

לִבָּם רָחַשׁ יְדִידוֹת, חֵן וָחֶסֶד חֵשֶׁק וְאַהֲבָה.


מַּה-יָּפְיָפִית עֵת הַזָּמִיר, נָעַמְתְּ עַד-אֵין-חֵקֶר!

עֵת הֵמִירוּ כְבוֹד אֱלֹהֵי אֱמֶת, צֶדֶק וּמֵישָׁרִים

בְּתַבְנִית אֱלִיל הָאַהֲבָה, מִקְסַם כָּזָב וָשֶׁקֶר

וְלִבְנוֹת-הַשִּׁיר הִשְׁתַּחֲווּ הַשָּׂרִים עַל-הַכְּמָרִים.


תְּפִלּוֹת בֶּן-יִשַׁי עָזְבוּ הַכֹּהֲנִים הַנְּזִירִים

נִזְרֵי הַמְשׁוֹרְרִים חָבְשׁוּ עַל-קָרַחַת רָאשֵׁיהֶם,

עַל-אַהֲבָה בַתַּעֲנוּגִים כֻּלָּם שָׁרוּ בַשִּׁירִים

וְהָאַבִּירִים אָהֲבוּ כֵן וַיַּעְזְרוּ עַל-יְדֵיהֶם.


הֵם הָרַכָּבִים שֹׁקְדִים עַל-טִירוֹת וּבִירָנִיּוֹת

בְּגָאוֹן יִרְכְּבוּ עַל-סוּסִים אַבִּירִים וְמְיוּזָנִים,

כֻּלָּם חָרְזוּ חֲרוּזִים וַיְשׁוֹרְרוּ בְּרָאשֵׁי הוֹמִיּוֹת

לִכְבוֹד נְשֵׁי-חֵן בָּחָרוּ עָשׂוּ תָמִיד כַּוָּנִים.


אֵין שִׁיר וּמִזְמוֹר בִּלְעֲדֵי אֲהוּבָה וְרַעְיָה

רַק הָאַהֲבָה יֹלֶדֶת הַזִּמְרָה, שִׁירִים מְחֹלֶלֶת,

בְּאֹפַנֵּי הַשִּׁירָה הִיא לְבַדָּהּ רוּחַ הַחַיָּה

וְשִׁירָה קְשׁוּרָה בְאִשָּׁה צְעִירָה כְּשַׁלְהֶבֶת בְּגַחֶלֶת!


גַּם לִיהוּדָה הַלֵּוִי הָיְתָה אַיֶּלֶת אֲהָבִים

לָהּ הִקְדִּישׁ כִּנֹּרוֹ וּבְנַפְשָׁהּ נַפְשׁוֹ נִקְשָׁרָה,

אַךְ לֹא כְרַכָּבִים שָׁר עֲגָבִים עַל-אֲשִׁישֵׁי עֲנָבִים

גַּם לֹא כִשְׁאָר יַעֲלוֹת חֵן אֲהוּבָתוֹ הַמְאֻשָּׁרָה:


לֹא כְלוֹרָא כְּלִילַת הַהוֹד, הַנַּעֲרָה הַיְפֵה-פִיָּה

בְּעֵינַיִם מְאַזְּרוֹת זִיקוֹת וְכִידוֹדֵי אֵשׁ בֹּעֶרֶת

הִצִּיתָה לֵב פֶּטְרַרְקָה, נְעִים זְמִרוֹת אִיטַלִּיָּה

בְּשִׁבְתּוֹ בֶטַח בְּהֵיכַל-הַתְּפִלָּה, בְּיוֹם צוֹם וַעֲצֶרֶת;


לֹא כְעַלְמָה בְּרֹאשׁ בַּחוּרִים עֹרְכֵי מִשְׂחָק לְפָנֶיהָ

וְזֹאת הַיָּפָה בַבָּנוֹת, עַל-הָאַרְמוֹן הַגְּבָרֶת

נִזְרֵי עֲלֵי-זַיִת לַמְנַצְּחִים תְּחַלֵּק בְּכַפֶּיהָ

בְּתוֹךְ הָמוֹן חוֹגֵג, לְשֵׁם לִתְהִלָּה וּלְתִפְאָרֶת;


אַף-לֹא כְאִשָּׁהּ מַשְׂכֶּלֶת בְּשִׂיחַ סוֹד מְשֹׁרְרֵי אַהֲבָה

לִשְׁפֹּט עַל-חֻקֵּי אֲהָבִים לָהּ נִתְּנָה הַמִּשְׁמֶרֶת,

בְּדֵי-חֵקֶר וּבְרֹב מְזִמּוֹת, בְּגֹדֶל לֵבָב וּבְגַאֲוָה

עַל-אֹדוֹת רִיב נְשִׁיקַת-פֶּה מִשְׁפַּט חָרוּץ גֹּזֶרֶת…


הָרַעְיָה אֲשֶׁר בָּחַר רַבִּי יְהוּדָה לִסְגֻלָּתוֹ

עֲנִיָּה עֹטְיָה הִיא, אַלְמָנָה שְׁכוּלָה וְגַלְמוּדָה,

הַשִּׁמָּמוֹן כִּסָּה פָנֶיהָ בִּצְעִיף חֲתֻלָּתוֹ

וּשְׁמָהּ מִימֵי עוֹלָם – יְרוּשָׁלַיִם אֲשֶׁר בִּיהוּדָה!


עוֹד בַּאֲבִיב יְמֵי עֲלוּמָיו הִיא לָקְחָה כָּל-לְבָבוֹ

לָהּ צָמְאָה נַפְשׁוֹ כָּמַהּ בְּשָׂרוֹ, קִרְבָתָהּ שָׁאָפוּ,

בָּהּ הָגָה כָל-הַיּוֹם, גַּם בִּתְנוּמוֹת עַל-מִשְׁכָּבוֹ

לִשְׁמָהּ וּלְזִכְרָהּ כָּל-עַצְמוֹתָיו בִּרְעָדָה רָחָפוּ.


מִדֵּי אֶל-תּוֹלֵדָה מֵאֶרֶץ-קֶדֶם סָר אֹרֵחַ

חָרַד אֵלָיו יְהוּדָה, לַקְשִׁיב רַב-קֶשֶׁב מַאֲוַיָּו,

וְעַל-דִּבְרַת אֲהוּבָתוֹ הַהֵלֶךְ כִּי יְשׂוֹחֵחַ

הָמוּ מֵעָיו בְּקִרְבּוֹ, כַּשָּׁנִי יִתְאַדְּמוּ לְחָיָיו.


הִנֵּה שָׁם נִצָּב הֵלֶךְ, אֶחָיו עֹטְרִים סְבִיבֹתָיו

תַּלְתַּלֵּי רֹאשׁוֹ מִמּוּל עָרְפּוֹ כַּכֶּסֶף לָטֹהַר,

זְקָנוֹ כְּלִבְנַת הַשֶּׁלֶג יֹרֵד עַל-פִּי מִדּוֹתָיו

אַךְ קָצֵהוּ הָפַךְ שָׁחוֹר, כְּמוֹ יָשׁוּב לִימֵי נֹעַר…


כִּפְנֵי מַלְאַךְ אֱלֹהִים נֶהְדְּרוּ פְּנֵי זֶה הָאֹרֵחַ

אֲשֶׁר שָׁב מֵחוֹנֵן עֲפַר צִיּוֹן, מֵרְצוֹת אֲבָנֶיהָ

עֵינָיו מְזָרוֹת עֹצֶב, מִקֶּרֶב לֵב יֵאָנֵחַ

וִיקוֹנֵן עַל-יְרוּשָׁלַיִם, יֵבְךְ עַל-מִשְׁבַּתֶּיהָ:


"פַּלָּצוּת בִּעֲתָתְנִי, אַחַי, מָלְאוּ מָתְנַי חַלְחָלָה

עֵת כַּפּוֹת רַגְלַי עַל-אַדְמַת הַקֹּדֶשׁ דָּרָכוּ,

יְרוּשָׁלַיִם לְנִדָּה הָיָתָה, מוֹרְאָה וְנִגְאָלָה

עַל-כָּל מַחֲמַדֶּיהָ כֻּלָּם שִׁקּוּצִים הָשְׁלָכוּ.


"מְקוֹם עוֹד יָאִירוּ עִקְּבוֹת נְבִיאֵי הָאֱמֶת וָצֶדֶק

לֶקַח טוֹב וַאֲמָרוֹת טְהֹרוֹת כְּטַל אוֹרֹת הִזִּילוּ,

שָׁם קוֹץ וְדַרְדַּר יַעֲלוּ, סִירִים סְבֻכִים וָחֶדֶק

הֵד חֲרָבוֹת נִשְׁמָע, בְּנוֹת יַעֲנָה וְאֹחִים יְיֵלִילוּ.


"מְקוֹם אַוֵּיר חַיֵּי נְשָׁמוֹת כִּבְגַן עֵדֶן יָפוּחַ

מִנִּשְׁמַת-אֵל עוֹד יִתֵּן רֵיחַ נֵרְדְּ וְקִנָּמוֹן,

שָׁם צִפְעֹנִים וְתַנִּים בְּאַוָּתָם יִשְׁאֲפוּ רוּחַ

וִירוּשָׁלַיִם כֻּלָּהּ הָיְתָה לְסֵמֶל שַׁמָּה וְשִׁמָּמוֹן!


"קִרְיַת רַבָּתִי עָם אֵיךְ נָשַׁמּוּ חוּצוֹתֶיהָ

רְחֹבֹתֶיהָ יַחַד אֵיךְ נֶהְפְּכוּ לְמִדְבַּר שְׁמָמָה,

כְּגַלִּים עַל-תַּלְמֵי שָׂדָי כֵּן רַבּוּ חָרְבוֹתֶיהָ

מָעוֹן לְכָל-שֶׁרֶץ וָרֶמֶשׂ, שְׂעִירִים יְרַקְּדוּ שָׁמָּה.


"בִּבְקִיעֵי הַחוֹמָה קָאַת גַּם-קִפֹּד יָלִינוּ

שָׁם יִנְווּ זֹחֲלֵי עָפָר, לִילִית תַּרְגִּיעַ,

שׁוּעָלִים מִבַּעַד לַחֲרַכִּים לָצִיץ יָהִינוּ

שָׁם מִשְׁכְּנוֹת מִבְטַחִים לָמוֹ, אֵין מַחֲרִיד מַפְגִּיעַ.


"עַל-מִגְדְּלֵי הַרֲרֵי צִיּוֹן עַד-עָבִים הִגִּיעוּ

כִּפּוֹתֵיהֶם הַמּוּפָזוֹת כִּשְׁמָשׁוֹת הֵאִירוּ,

הֲדַר כְּבוֹד מַלְכֵי יְהוּדָה וּגְדֻלָּתָם הִבִּיעוּ

וְשִׁלְטֵי הַגִּבּוֹרִים עֲלֵיהֶם כַּבָּזָק הִזְהִירוּ;


"עַתָּה פִּתּוּחֵיהֶם כְּשֵׁבֶר נֵבֶל יוֹצְרִים נִשְׁבָּרוּ

עַל-כִּפּוֹתֵיהֶם עָלוּ יְרַק דֶּשֶׁא וַחֲצִיר גַּגּוֹת,

הַחוֹמוֹת שְׁחֹרוֹת כָּעוֹרֵב, עַרְעֵר הִתְעַרְעָרוּ

אֵבֶל עָטְפוּ כֻלָּן וּכְמוֹ בְקוֹל-מַר מְנַהֲגוֹת.


"הַאֻמְנָם תִּבְכֶּינָה הַחֹמוֹת, תֵּרַדְנָה מָיִם

בַּתְּשִׁיעִי לְיֶרַח אָב, יוֹם יַד-הָאוֹיֵב גָּבֵרָה

מוֹעֲדֵי-אֵל וּמִקְדָּשָׁיו בּוֹ נִשְׂרְפוּ פַעֲמָיִם,

בְּעֵינַיִם מְלֵאוֹת דִּמְעָה זֶה-חָזִיתִי וַאֲסַפֵּרָה:


"הִנֵּה בְיוֹם צוֹם זֶה הַמִּגְדָּלִים גַּם-הֵם יִבְכָּיוּן

מִבֵּינוֹת לָאֲבָנִים אֶגְלֵי נִטְפֵי-מַיִם נָזָלוּ,

שְׂרִידֵי עַמּוּדֵי הַהֵיכָל בִּמְרִירוּת יֶהֱמָיוּן

הַנְּתוּצִים וּנְפוּצִים תַּחַת שֹׁאָה יִתְגַּלְגָּלוּ.


"גַּם-פֶּרֶא וְעָרוֹד אַחַר כָּל-יָרוֹק יִדְרֹשׁ:

יָתוּרוּ מִרְעֶה לָמוֹ, דֶּרֶךְ הֵנָּה פְנֵיהֶם,

כִּמְעַט יָבוֹאוּ אֶל-הַחֶלְקָה חִישׁ אוֹתָהּ יִטֹּשׁוּ,

עַל-אַבְנֵי הַמַּפֵּלָה לְבַל-יִתְנַגְּפוּ רַגְלֵיהֶם.


"אַךְ זְעֵיר שָׁם עֲרָבִי, יְלִיד אֶרֶץ מְלֵחָה וַעֲרָבָה

בְּלוֹיֵ מְלָחִים עַל-מַעֲרֻמָּיו, בְּיָדוֹ חִצִּים וָקֶשֶׁת

עִם-גְּמַלּוֹ יַעֲבֹר עַל-הָעִיר הַמְשֻׁלָּחָה וְנֶעֱזָבָה

וִילַחֵךְ עִשְׂבָּהּ וַחֲצִירָהּ בַּעַל תָּאֳמֵי-הַדַּבֶּשֶׁת.


"אֶפֶס שְׁכִינַת-יָהּ גַּם עַל-הֲרִיסֹתֶיהָ תִּשְׁכֹּנָה,

וַחֲסִידִים יִרְאֵי אֱלֹהִים בְּאָזְנֵיהֶם שָׁמָעוּ

בַּת-קוֹל יוֹצֵאת מִן-הֶחָרְבָּה מְנַהֶמֶת כַּיּוֹנָה:

"אוֹי לַבָּנִים בַּעֲוֹנָם קַרְנֵי עֻזָּם גֻּדָּעוּ;


"אוֹי לַבָּנִים בִּגְלַל גִּלּוּלֵיהֶם בָּם מָאָסְתִּי

וַהֲנִחֹתִי חֲמָתִי בָם, הֵבֵאתִי הַקָּרֶץ,

בְּחַטֹּאתֵיהֶם כִּי עָצְמוּ מְאֹד מִקְדָּשַׁי הָרָסְתִּי

וְהִנָּם רְצוּצִים וּלְחוּצִים בְּכָל-כַּנְפוֹת הָאָרֶץ;


"אַךְ אוֹי וָמָר גַּם לְעשְׁקֵיהֶם, אֹכְלֵיהֶם יֱאשָׁמוּ

יוֹם נָקָם בְּלִבִּי, אֶל-חֵיקָם אָשׁוּב כִּפְלָיִם,

וְאֶת-בָּנַי אֲרַחֵם, אֶל-אַרְצָם יָשׁוּבוּ יְנֻחָמוּ

כִּי אֲנַחֵם אֶת-צִיּוֹן וְאֶבְחַר עוֹד בִּירוּשָׁלָיִם!"


כְּהִנָּתֵק פְּתִיל-הַנְּעֹרֶת בַּהֲרִיחוֹ לְשׁוֹן לַהֶבֶת

נִתְּקוּ מוֹרָשֵׁי לִבּוֹ הָרַךְ, רוּחוֹ חֻבָּלָה

מִדֵּי אֲמָרוֹת כָּאֵלֶּה, שְׂפַת אֹרֵחַ דּוֹבֶבֶת –

וַתְּכַל נַפְשׁוֹ לִרְאוֹת אֶת-אֲהוּבָתוֹ הָאֻמְלָלָה.


כְּכֹל אֲשֶׁר הוֹסִיף לִדְרשׁ בַּכְּתוּבִים וְהַגָּדוֹת

וְהִרְבָּה לְהַעֲמִיק חֵקֶר בְּחָכְמָה מִכָּל-בְּנֵי דוֹרוֹ,

כֵּן פָּרְצָה תְשׁוּקָתוֹ לִירוּשָׁלַיִם עֶשֶׂר יָדוֹת

אַךְ בָּהּ שָׂם כָּל-מַעְיָנָיו, הִיא הָיְתָה יִשְׁעוֹ וְאוֹרוֹ.


לִתְשׁוּקַת מְשׁוֹרֵר אֵין קֵץ, אֵין דְּלָתַיִם וּבְרִיחַ!

אֶת-נַפְשׁוֹ יַקְרִיב עַל-מִזְבְּחָהּ, לֹא יִירָא שָׁחַת…

כֵּן רָאִיתִי אֶת-רוּדֶלָּה, מְשׁוֹרֵר מַשְׂכִּיל מַצְלִיחַ

אַבִּיר הַטְּרֻבַּדּוּרִים, בְּהֵיכָלוֹ שָׂבַע נָחַת –


בְּשׁוּב מְרֵעָיו מֵאֶרֶץ קֶדֶּם וּלְהֵיכָלוֹ בָאוּ

עַל-בְּנוֹת חֵן וְשֵׂכֶל טוֹב אִישׁ אֶל-רֵעֵהוּ נִדְבָּרוּ,

אָז עַל-כּוֹס-יַיִן מָלֵא הָרַכָּבִים נִשְׁבָּעוּ

כֻּלָּם יַחַד אִישׁ לֹא-נֶעְדָּר אֹמֶר גָּזָרוּ:


כִּי מֶלִיזַנְדָּה שָׂרָתִי בִּטְרִפּוֹלִיס הַבִּירָה

בִּצְבִי תִפְאֶרֶת וָרֹם עוֹלָה עַל-כָּל-בְּנוֹת חָיִל,

כִּי מִכְלַל-יֹפִי וְהַצְנֵעַ לֶכֶת זֹאת הַגְּבִירָה

וּבֵין נְשֵׁי-חֵן תָּאִיר כַּיָּרֵחַ בֵּין כּוֹכְבֵי-לָיִל…


אָז נֶפֶשׁ הַמְשׁוֹרֵר אַחֲרֵי מֶלִיזַנְדָּה דָּבָקָה

לָהּ הִקְדִּישׁ כָּל-שִׁירָיו, כִּנֹּרוֹ עִם עֻגָבוֹ

רַעְיָה לֹא-רָאָה רוּחַ חַיִּים עָלָיו יָצָקָה

וַתְּהִי מַשְׂאַת רוּחוֹ תָּמִיד, כָּל-חֵשֶׁק לְבָבוֹ.


מִיּוֹם לְיוֹם גָּבְרָה אַהֲבָתוֹ וַתְּהִי כְּאֵשׁ בֹּעֶרֶת

וַתַּעֲרֹג נַפְשׁוֹ לִרְאוֹת יָפָתוֹ עַיִן בְּעָיִן,

בְּעָבְרוֹ אָרְחוֹת יָם הִדְבִּיקַתּוּ מַחֲלָה מַמְאֶרֶת

וּבְבֹאוֹ טְרִפּוֹלִיס כָּל-תִּקְוָתוֹ הָיְתָה כְּאָיִן.


הֵן אָמְנָם בְּעֵינֵי בָשָׂר אֶת-יְדִידוּת נַפְשׁוֹ רָאָה,

אַךְ שִׁירָתוֹ הָאַחֲרוֹנָה טֶרֶם כִּלָּה בְּאָזְנֶיהָ

חֶרֶב מָוֶת לֹא-יַחְמֹל עַד-לִבּוֹ נָגָעָה

שְׁמֻרוֹת עֵינָיו סֻגָּרוּ – וַיִּפֹּל מֵת לְרַגְלֶיהָ.


בִּשְׁנֵי מְשׁוֹרְרִים אֵלֶּה פָּגַע אָסוֹן בְּשֵׁבֶט עֶבְרָתוֹ

גּוֹרָל אֶחָד לִשְׁנֵיהֶם, בְּלֹא-עִתָּם רָאוּ שָׁחַת;

אַךְ זֶה בִּימֵי חָרְפּוֹ יָרַד לִרְאוֹת פְּנֵי אֲהוּבָתוֹ

וְהִיא עוֹדָהּ לֹקַחַת נְפָשׁוֹת וְכַתָּמָר פֹּרָחַת;


וִיהוּדָה בָא בַיָּמִים, כִּמְעַט זָרְקָה בוֹ שֵׂיבָה

לִרְאוֹת רַעְיָתוֹ הַזְּקֵנָה עָבַר מֵי אַפְסָיִם,

וַיִּפֹּל לְרַגְלֵי שׁוּלַמִּית הַשּׁוֹמֵמָה וַחֲרֵבָה

וַיַּנַּח רֹאשׁוֹ הַשָּׁדוּד עַל-בִּרְכֵּי – יְרוּשָׁלָיִם!



ג

אַחֲרֵי הַכּוֹת אַלֶּכְּסַנְדֶּר הַגָּדוֹל בִּזְרוֹעַ-כֹּחַ

אֶת-דָּרְיָוֶשׁ קֹדֹמַנּוֹס, מֶלֶךְ בָּבֶל וּפָרָס,

עַל-יַד אַרְבֶּלָּה הוֹרִיד חֵילוֹ כְּכָרִים לִטְבּוֹחַ

קִרְיַת תעֻזּוֹ שָׂרַף בָּאֵשׁ, מִבְצָרָיו הָרָס;


כָּל-הָאֲרָצוֹת וְכָל-הַגּוֹיִם הָיוּ לוֹ לְשָׁלָל,

הַהֵיכָל, הַרְמוֹן הַנָּשִׁים, גִּנְזַכֵּי מַטְמֹנִים

נֵזֶר וְשַׁרְבִיט, כְּלֵי זָהָב וָכֶסֶף, נְחֹשֶׁת קָלָל

סוּסִים וְשֶׁנְהַבִּים, קַרְנוֹת-שֵׁן הָבְנִים וַאֲדַרְכֹּנִים.


בִּלַּע כַּתַּנִּין כְּבוֹד פָּרָס, חִלֵּל גְּאוֹן כָּל-צֶבִי

נַפְשׁוֹ הִרְחִיב כִּשְׁאוֹל, בִּטְנוֹ שָׂבְעָה לֹא יָדָעָָה,

וַיִּירָא דָרְיָוֶשׁ פֶּן גַּם-אוֹתוֹ יוֹלִיךְ שֵׁבִי

וַיְמַהֵר לְהִמָּלֵט וּלְהִנָּצֵל מִכַּף-רָעָה.


וּבְאֹהֶל אַפַּדְנוֹ יַד הַמְנַצֵּחַ חָתָרָה

וַתִּמְצָא שָׁם אַרְגָּז קָטָן, כֻּלּוֹ זְהַב פַּרְוָיִם,

מְשֻׁבָּץ בִּפְנִינִים וּמְעֻלָּף בְּכָל-אֶבֶן יְקָרָה

נֶחְפָּה בְּפִתּוּחֵי חֹתָם מַחֲמַדֵּי עֵינָיִם.


וּבָאַרְגָּז הַזֶּה הָיוּ אֲרֻזִים וַחֲבֻשִׁים

כַּדְכֹּד וְשֹׁהַם, חֲרוּזֵי מַרְגָּלִיּוֹת וְאַלְמֻגִּים,

אַבְנֵי-חֵפֶץ בִּידֵי חָרָשִׁים מְמֹרָטִים וּלְטֻשִׁים

עֲדָיֵי דָרְיָוֶשׁ מִדֵּי הִתְעַנֵּג בַּתַּעֲנוּגִים.


אָז יְחַלֵּק אַלֶכְּסַנְדֶּר אֲבָנִים יְקָרוֹת וּמַכְלֻלִים

לִבְחִירֵי גִבּוֹרֵי חֵילוֹ, לִפְאֵר וּלְשַׁעֲשׁוּעִים,

וּבְחַלְּקוֹ מִלֵּא שְׂחוֹק פִּיהוּ וּלְשׁוֹנוֹ הֲתֻלִּים

עַל-גְּבָרִים כְּנָשִׁים, אֹהֲבֵי שְׁלַל צְבָעִים וְצַעֲצֻעִים.


אֶת-טַבַּעַת דָּרְיָוֶשׁ, הִיא חוֹתַם הַמַּמְלָכָה

אֶבֶן שֹׁהַם גְּדוֹלָה, עַל-עָפָר אֵין עֲרוֹךְ אֵלֶיהָ,

שָׁלַח אַרְצָה מֹקְדוֹן, אֶל-אִמּוֹ מִנְחָה וּבְרָכָה

וַתְּהִי לָהּ לִוְיַת-חֵן לְאַחַד עֲנָק מִצַּוְּרֹנֶיהָ.


גַּם לֹא-שָׁכַח אֶת-מוֹרֵהוּ הֶחָכָם כְּקֹהֶלֶת

כִּי אֶבֶן נֹפֶךְ חֲמוּדָה, יְקָרָה עַל-אַחַת שֶׁבַע

שָׁלַח אֶל-אַבִּיר חוֹקְרֵי יָוָן, הוּא אֲרִסְטוֹטֶלֶס

לְשִׂימָהּ בֵּית נְכֹתֹה, בֵּית-אֹסֶף סְגֻלּוֹת הַטֶּבַע.


גַּם-מַחֲרֶזֶת מַרְגָּלִיּוֹת מָצָא תּוֹךְ הָעֲדָייִם

נַחֲלַת שַׁלִּיטֵי פָרַס מִשָּׁנִים קַדְמֹנִיּוֹת,

אוֹתָהּ הִקְרִיב אֶשְׁכָּר, מָנָה אַחַת אַפָּיִם

אֶל-עַלְמָה אֲהוּבָה, מִן-הַמְחֹלְלוֹת הַיְפֵה-פִיּוֹת.


בֵּין מַחְלְפוֹת דַּלַּת רֹאשָׁהּ שָׁתָה אֶת-הַמַּחֲרֶזֶת

תַּיִס הַבַּת הַשּׁוֹבֵבָה, הוֹמִיָה וְסוֹרֵרָה,

בַּעֲלִיצוּת נַפְשָׁהּ עוֹדָהּ מְכַרְכֶּרֶת וּמְפַזֶּזֶת

בְּהֵיכַל פֶּרְסֶפּוֹלִיס הִבְעִירָה בִצְדִיָּה בְּעֵרָה.


נְקַם קוֹרִנְתְּ מוֹלַדְתָּהּ לָתֵת בְּדָרְיָוֶשׁ חָפָצָה

וַתַּשְׁלֵךְ לַפִּיד בֹּעֵר בָּאַרְמוֹן וּבַצְּרִיחַ,

בְּכָל-אַרְבַּע פִּנּוֹתֵיהֶם לַהֶבֶת-אֵשׁ פָּרָצָה

וּמְהֵרָה קַל נֶהֶפְכוּ כֻלָּם לְדֶשֶׁן וָפִיחַ.


אַחֲרֵי מוֹת אַלֶּכְּסַנְדֶּר תַּיִס לְאִישׁ הָיָתָה

אֶת-תַּלְמַי מֶלֶךְ מִצְרַיִם לְבַעַל לָהּ בָּחָרָה,

וּבְמוֹתָהּ בְּעִנְיָן רָע, בַּמַּחֲלָה אֲשֶׁר-לָהּ יָאָתָה

אֶל-אַחַד כֹּהֲנֵי מֹף הַמַּחֲרֶזֶת נִמְכָּרָה.


אַךְ מִידֵי הַכֹּהֲנִים שָׁבָה הַמַּחֲרֶזֶת וּבָאָה

אֶל-בֵּית תַּלְמַי, שַׁלִּיטֵי אַדְמַת-חָם וּמְלָכֶיהָ,

בֵּין עֲדִי קְלֶאֹפַּטְרָה הַמַּלְכָּה גַּם-הִיא נִמְצָאָה

וַתָּלֶן כָּל-הַיּוֹם כִּצְרוֹר הַמֹּר בֵּין שָׁדֶיהָ.


נְצֻרַת-לֵב זֹאת מִבְחַר הַמַּרְגָּלִיּוֹת לָקָחָה

וַתִּטְחְנֵן עַד-דָּק וַתִּזֶר עַל-פְּנֵי יֵין מִשְׁתֶּיהָ,

הוֹסֵף בָּזֹאת עַל-חִין יָפְיָהּ הַמַּלְכָּה בָּטָחָה

לִלְכֹּד לֵב אַנְטוֹנִיּוֹס בְּרִשְׁתָּהּ, בְּעָצְמַת כְּשָׁפֶיהָ.


כַּאֲשֶׁר פָּשְׁטוּ הָעַרְבִיאִים עַל-אֶרֶץ מִצְרָיִם

לָכַד עַמְרוֹ אֶת-אַלֶּכְּסַנְדְּרִיָּה בֶּעֱזוּז מִלְחָמָה,

וַיָּבֹז רַב בִּזָּה וַיִּשְׁלֹל שָׁלָל מְלֹא חָפְנָיִם

וְגַם אֶת-הַמַּרְגָּלִיּוֹת מָצָא בַבִּזָּה שָׁמָּה.


וּבְבוֹא בְנֵי-עֲרָב עַל-אֵירֹפָּא בְּרַעַם גְּבוּרָתָם

בִּסְפָרַד תָּקְעוּ יִתְדָּם, שָׁם מֶמְשַׁלְתָּם נוֹסָדָה,

הֵבִיאוּ אִתָּם שָׁמָּה חָכְמָתָם, דָּתָם וּשְׂפָתָם –

וְגַם מַחֲרֶזֶת הַמַּרְגָּלִיּוֹת הוּבְאָה סְפָרָדָה.


כַּלִּיפֵי קוֹרְדוֹבָה עֲנָדוּהָ עֲטָרוֹת לְמִצְנְפוֹתָם,

וְעֶבֶד-רַחְמָן הַשְּׁלִישִׁי, הַמְהֻלָּל בְּרֹב גְּבוּרָתוֹ

מְהִר צֶדֶק וּמְשׁוֹרֵר, שָׂמָה עַל-לִבּוֹ כַּחוֹתָם

בְּצֵאתוֹ לִכְבָּשׁ-לוֹ לֵב סוּלַיְמָה אֲהוּבָתוֹ.


מִתִּגְרַת יַד-אוֹיְבִים זְרוֹעַ הַכַּלִּיפִים נִשְׁבָּרָה

כִּסְאוֹתָם מֻגְּרוּ לָאָרֶץ, מַמְלַכְתָּם נֶהֱרָסָה

נוֹצְרִים יָרְשׁוּ אַרְצָם – וְגַם הַמַּחֲרֶזֶת הַמְפֹאָרָה

בְּאוֹצַר מַלְכֵי קַסְטִילִיָּה מִנִּי-אָז הִתְנוֹסָסָה.


בְּאֶרֶץ קִנְאַת-דָּת הַלֵּזוּ, מְקוֹם שִׁלְטוֹן הַכְּמָרִים

עַל-צַוְּארֵי הַמְּלָכוֹת הוֹפִיעוּ הַמַּרְגָּלִיּוֹת,

בִּהְיוֹתָן בְּבֵית מִשְׂחָק, בִּרְאוֹתָן מִלְחֶמֶת הַשְּׁוָרִים

וּבְלֶכְתָּן עִם-הַכְּמָרִים הַחֹגְגִים בְּרָאשֵׁי הוֹמִיּוֹת.


גַּם בְּשִׁבְתָּן עַל-בָּמוֹת רָמוֹת וּבְנוּיוֹת לְתַלְפִּיּוֹת

וּמְדוּרוֹת עֵצִים לִפְנֵיהֶן עִם-אֵשׁ מִתְלַקָּחַת,

וְרֵיחַ מִשְׂרְפוֹת יְהוּדִים מְרוּדִים, צְלוּיִים כִּגְדִיּוֹת,

כַּעֲתַר עֲנַן-קְטֹרֶת לְאַפָּן וּכְרֵיחַ מִרְקָחַת.


כְּכֹל אֲשֶׁר הִתְעִיבוּ עֲלִילָה שַׁלִּיטֵי סְפָרָד

כֵּן לְאַרְצָם הִרְבּוּ הָרֵעַ, אַדְמָתָם עָכָרוּ

עַד-כִּי שֶׁמֶשׁ אָשְׁרָם וְעָשְׁרָם עֶשֶׂר מַעֲלוֹת יָרָד

אוֹצְרוֹת הַמְּלוּכָה רֻשֵּׁשׁוּ, כְּמוֹקֵד נִחָרוּ.


וּלְאִישׁ יְהוּדִי קָרְאוּ עֵת אֲחָזוּם חֲבָלִים וְצִירִים

וַיְהִי מֶנְדִּיצַבַּל שַׂר-אוֹצָרוֹת, עָרוֹם וְנָבוֹן…

הוּא הֵרִיק אֶת-בָּתֵּי מִקְלַט הַנְּזִירוֹת וְהַנְּזִירִים

וְגַם אֶת-הַמַּרְגָּלִיּוֹת נָתַן בַּעֲבוֹט וְעֵרָבוֹן.


לְקֵץ הַיָּמִים בְּאַרְמוֹן טוּלִירִי בְּפַרִיס הַבִּירָה

הַמַּרְגָּלִיּוֹת בַּהֲדָרָן עָלוּ מִן-הַנְּשִׁיָּה

עַל-חֶלְקַת צַוְּארֵי הַכְּבוּדָה, עֲדִינָה וּגְבִירָה,

אֵשֶׁת הַשַּׂר שְׁלֹמֹה רוֹטְשִׁילְדְּ – אִשָּׁה עִבְרִיָּה.


אַךְ לֹא-כֵן הָיָה גּוֹרַל הָאַרְגָּז וּמִקְרֵהוּ,

הוּא לֹא הִתְהַלֵּךְ מִיָּד לְיָד בְּהֵיכְלֵי כְמָרִים וְשָׂרִים,

כִּי אַלֶּכְּסַנְדֶּר הַגָּדוֹל בּוֹ חָשַׁק וַיְפַלְּטֵהוּ

וַיַּאַסְפהוּ אֵלָיו, תִּפְאַרְתּוֹ לֹא נָתַן לַזָּרִים.


בּוֹ הִצְפִּין שִׁירֵי הוֹמֵירוֹס, הוֹד עַמּוֹ וּכְבוֹדוֹ,

עִוֵּר גְּלוּי עֵינָיִם, עַל-גִּבּוֹרֵי יָוָן נְגִינָתוֹ,

שִׁירָיו אָהַב אַלֶּכְּסַנְדֶּר בְּכָל-נַפְשׁוֹ וּמְאֹדוֹ

וַיַּצֵּג הָאַרְגָּז עַל-הַשֻּׁלְחָן – לִפְנֵי מִטָּתוֹ.


גִּבּוֹר נַעֲרָץ זֶה מִדֵּי עַל-עַרְשׂוֹ בִּקֵּשׁ מָנוֹחַ

לֹא-נָתַן לְעַפְעַפָּיו תְּנוּמָה וּשְׁנָת לְעֵינָיִם

טֶרֶם שִׁירֵי הוֹמֵירוֹס לְפָנָיו יִשְׁטַח שָׁטוֹחַ

וּבְשׂוֹם לֵב יִקְרָא בָמוֹ דְּלָתוֹת שָׁלֹש אוֹ שְׁתָּיִם.


וּבִתְנוּמוֹת עֲלֵי-מִשְׁכָּב, בִּנְפֹל עָלָיו תַּרְדֵּמָה

מִן-הָאַרְגָּז יַעֲלוּ הָעֲנָקִים, גִּבּוֹרֵי הֶחָיִל

אִישׁ אִישׁ מֵהֶם יַחְשׂף יָדוֹ וּזְרוֹעוֹ רוֹמֵמָה

וְקֶסֶם מִפְעָלָם יִרְאֶה הַמֶּלֶךְ בְּחֶזְיוֹן-לָיִל…


הוֹמֵירוֹס כַּבִּיר רוּחַ! בִּנְעוּרַי גַּם-אָנִי

בִּמְלִיצוֹת שִׁירֶיךָ יוֹמָם וָלַיְלָה הָגִיתִי,

בְּחַבְלֵי-יֹפִי וָנֹעַם אָז אַחֲרֶיךָ מְשַׁכְתָּנִי

וּמֵאַהֲבָתִי אֵלֶיךָ, אָמְנָם רַבּוֹת נָשִׁיתִי.


בְּחֶלְקַת שְׂפָתֶיךָ קָנִיתָ לְבָבִי גַם-כִּלְיּוֹתָי

חָכְמַת חַכְמֵי עַמְּךָ הָיוּ בֵין עֵינַי לְטֹטֶפֶת,

וַתֵּקַל אָז בְּעֵינַי גַּם-חָכְמַת מוֹרַי וַאֲבוֹתָי

נַפְשִׁי וְרוּחִי הִתְמַכְּרוּ אֶל “יְפֵהפִיּוּתוֹ-שֶׁל-יֶפֶת”.


יְמֵי יַלְדוּת נָגוֹזּוּ, עִמָּם גָּז חִישׁ גַּם-חֲלוֹמִי

יָמִים הוֹרוּנִי בִינָה, שָׁנִים אִלְּפוּנִי דָעַת…

חַשְׁתִּי אֶל-אָהֳלֵי-שֵׁם, שַׁבְתִּי יָשַׁבְתִּי עַל-מְקוֹמִי

לִשְׁאֹב חָכְמָה בְּשָׂשׂוֹן מִמְּקוֹר בְּנֵי-עַמִּי נוֹבָעַת.


וּבְכֵן סְלַח-נָא לִי הוֹמֵירוֹס, בְּחִיר מְשׁוֹרְרֵי הַיְוָנִים!

סִלְחוּ-נָא לִי פִילוֹסוֹפִים, אֶת-שִׁמְכֶם אֲכַבֵּדָה,

יִסְלְחוּ-לִי גַּם-חָרָשֵׁי אֶבֶן, יֹצְרֵי-פֶסֶל וָאֳמָנִים

אִם-בְּאָזְנֵי כָל-הַיְּוָנִים אַגִּידָה וְלֹא אֲכַחֵדָה:


לוּ נָפַל הָאַרְגָּז הַלָּז לִי בַּנְּעִמִים חֶבֶל

גַּם-אֲנִי כְאַלֶּכְּסַנְדֶּר לְאִישׁ אַחֵר לֹא אֶתְּנֶנּוּ,

(לוּ רַק הַמַּחְסוֹר לֹא יַכְבִּיד עָלַי הַסֵּבֶל…)

אַךְ בְּשִׁירֵי הוֹמֵירוֹס כָּמֹּהוּ לֹא אֲרַפְּדֶנּוּ;


אֶת-שִׁירֵי רַבִּי יְהוּדָה אֶקְבֹּץ אֶל-חֹבֶרֶת אַחַת

כָּל-שִׁירֵי אֲהָבִים, שִׁירֵי שַׁבָּת מוֹעֵד וָחֹדֶשׁ,

מְאוֹרָה אוֹפָן וּגְאֻלָּה, זוּלַת פִּזְמוֹן וְתוֹכַחַת

גַּם-שִׁירֵי מַסָּעָיו כֻּלָּמוֹ, שִׁירֵי חֹל וָקֹדֶשׁ;


לְסוֹפֵר אָמָן אֲשַׁלֵּם מִמֵּיטַב כַּסְפִּי וּזְהָבִי

לְכָתְבָם כְּתִיבָה תַמָּה וְיָפָה עַל-גְּוִילֵי תִפְאֶרֶת,

וּכְתֻמָּם יַחַד אֶל-תּוֹךְ הָאַרְגָּז אוֹתָם אָבִיא

שָׁם יָנוּחוּ בְכָבוֹד, וְהָיוּ לִפְאֵר וּלְמִשְׁמֶרֶת.


וְאֶת-הָאַרְגָּז גַּם-אָנִי אַצִּיגָה עַל-שֻׁלְחָנִי

הֶעָרוּךְ בַּחֲדַר מַשְׂכִּיתִי, שָׁם עַל-יַד מִטָּתִי,

וּכְבוֹא אֲחֻזַּת מֵרֵעַי לְשַׁחֲרֵנִי בִּנְוֵה מִשְׁכָּנִי

אֲזַי יִשְׁתָּאוּ, אַף-יִשְׁתּוֹמְמוּ עַל-כְּלִי חֶמְדָּתִי:


זֶה יַרְבֶּה לְהַלֵּל זְהָבוֹ הַמְזֻקָּק שִׁבְעָתָיִם

וְזֶה יְפָאֵר אֶת-הֲדַר מְלַאכְתּוֹ, מְלֶאכֶת מַחֲשֶׁבֶת,

זֶה יְשַׁבַּח הַתְּמוּנוֹת הַמְשׁוּחוֹת בְּצֶבַע רִקְמָתָיִם

וְזֶה אֶת-אַבְנֵי הַיְקָר הַנֹּצֲצוֹת כִּבְרַק שַׁלְהֶבֶת;


אַף אֲנִי בִּשְׂחוֹק אֶעֱנֵם: רֵעַי, שִׁמְעוּ-נָא עָדָי!

בַּמֶּה נֶחְשָׁב הוּא הָאַרְגָּז הַזֶּה בִּיקְּרוֹתֵיהוּ

מוּל הָאוֹצָר הַנֶּחְמָד אֶל-קִרְבּוֹ צָפְנוּ יָדָי?

הוּא לֹא-יְסֻלֶּה בַפָּז, פִּטְדַת-כּוּשׁ לֹא תַעַרְכֵהוּ!


עֲדָיֵי הָאַרְגָּז מֵהֶבֶל, נֶחְשְׁבוּ לְנִבְלֵי-חֶרֶשׁ

מוּל הָאֲבָנִים בְּתוֹכוֹ, הוֹד יִפְעַת שָׁמַיִם יָרִיקוּ,

אַבְנֵי אֹדֶם אָדְמוּ מִדַּם לֵב בְּאֵשׁ-דָּת בֹּעֶרֶת

וְאַבְנֵי בָרֶקֶת מִנֹּגַהּ נֶפֶשׁ בְּרוּכָה יַבְרִיקוּ,


גַּם-מַרְגָּלִיּוֹת טוֹבוֹת פֹּה בָאַרְגָּז נִצְמָדוּ

עִם-מַרְגָּלִיּוֹת דָּרְיָוֶשׁ לֹא יִשָּׁקְלוּ בְמֹאזְנָיִם,

יַעַן לֹא כָאֵלֶּה מִשֶּׁרֶץ שִׁקּוּצִים נוֹלָדוּ

מֵחֶלְאַת צָב חֹלֶה בִּתְהוֹמוֹת יָם וְנַחֲלֵי-מָיִם;


הַאֻמְנָם הַמַּרְגָּלִיּוֹת בְּאַרְגָּז זֶה נִמְצָאוּ

יָצְאוּ מִמְּקוֹר נֶפֶשׁ טְהוֹרָה כְּאוֹר שִׁבְעַת הַיָּמִים,

מִמְּקוֹר נֶפֶשׁ עֲמֻקָּה מִנִּי-יָם חוּצָה נָבָעוּ

וּבִיקַר תִּפְאַרְתָּם יַעַמְדוּ עַד-עוֹלְמֵי עוֹלָמִים:


הַמַּרְגָּלִיּוֹת הָהֵן הֵנָּה נְטִפוֹת הַדְּמָעוֹת

אֲשֶׁר נָטְפוּ מֵעֵינֵי הַמְשׁוֹרֵר רַבִּי יְהוּדָה

בִּבְכוֹתוֹ עַל-מַשֻּׁאוֹת עִיר אֱלֹהֵי הַצְּבָאוֹת

וּבְהִשְׁתַּפֵּךְ נַפְשׁוֹ אֶל-חֵיק בַּת-צִיּוֹן הַשְּׁדוּדָה.


פְּנִינֵי הַדְּמָעוֹת הָאֵלֶּה חֹרְזוּ אַחַד אַחַד

עַל-חוּט זָהָב מָשְׁזָר בִּידֵי צֹרֵף חֲכַם חָרָשִׁים,

וַחֲרוּז אֶחָד בְּלֻלָאֹתָיו יְחַבְּרֵן כֻּלָּן יַחַד

וַתִּהְיֶינָה לְשִׁירָה יְקָרָה, שִׁירַת קֹדֶּשׁ הַקֳּדָשִׁים!


זֹאת הַשִּׁירָה קִינָה הִיא, בְּכִי וּמִסְפֵּד תַּמְרוּרִים

בְּכָל-מִשְׁכְּנוֹת יַעֲקֹב עַל-כָּל-אַרְבַּע כַּנְפוֹת הָאָרֶץ,

לֶעָפָר יֵשְׁבוּ וִיקוֹנֲנוּהָ זְקֵנִים עִם-בַּחוּרִים

יוֹם בָּא שֹׁד מִשַּׁדַּי, יוֹם עֶבְרָה כִּלָּיוֹן וָקָרֶץ.


הֲלֹא הוּא יוֹם מַר וְנִמְהָר, יוֹם תּוֹכֵחָה וְצָרָה

יוֹם קֹרָא לְבַת-צִיּוֹן: “הוֹרִידִי עֶדְיֵךְ מֵעָלָיִךְ!”

וּבָנֶיהָ בְנִיהֶם יִשְׂאוּ קִינָה, יִזְעֲקוּ מָרָה

וִיתַנּוּ: “צִיּוֹן הֲלֹא תִשְׁאֲלִי לִשְׁלוֹם אֲסִירָיִךְ!”


כָּל-עַם-יְהוּדָה יְקוֹנֲנוּהָ בִּכְלוֹת כָּל-קִינוֹתֵיהֶם,

כִּי בַעֲזוֹב יְהוּדָה מוֹלַדְתּוֹ וַיִּדֹּר צִיֹּינָה,

לְתַנּוֹת לְמִקְדְּשֵׁי-אֵל לִשְׁפֹּךְ נַפְשׁוֹ עַל-חָרְבוֹתֵיהֶם

"צִיּוֹן" הָיְתָה שִׁירָתוֹ – קִינָתוֹ הָאַחֲרוֹנָה.


רַגְלָיו יְחֵפוֹת, עַל-רֹאשׁוֹ אֵפֶר וְשַׂק עַל-חֲלָצָיִם

לִבּוֹ סֹעֵר כַּהֲמוֹת יַם לְגַלָּיו, רוּחוֹ חֻבָּלָה

כֹּה יִקְרַב אֶל-הַקֹּדֶשׁ, יִקֹּד שֶׁבַע אַפָּיִם

יִכְרַע עַל-שְׂרִיד עַמּוּד הֵיכָל, עַל-מְעִי מַפָּלָה;


שַׂעֲרוֹת רֹאשׁוֹ כְּגֶפֶן שׂרֵק שָׂרִיגֶיהָ הִלְבִּינוּ

עַל-שִׁכְמוֹ וְעַל-חָזֵהוּ יַחַד כִּכְפוֹר נִפְזָרוּ,

כַּמֵּת חָוְרוּ פָנָיו, עֹצֶב וְיָגוֹן בָּם יָלִינוּ

אַךְ עֵינָיו בְּחוֹרֵיהֶן כְּעַפְעַפֵּי-שַׁחַר נָהָרוּ;


מַר צֹרֵחַ הַלֵּוִי, יֶרֶב תַּאֲנִיָּה וַאֲנִיָּה

וְצַלְמוֹ נִדְמָה אֶל-אַחַד הַצּוֹפִים, חוֹזֵי אֱלוֹהַּ,

לַגֶּבֶר רָאָה עֳנִי – הַכֹּהֵן בֶּן-חִלְקִיָּה

רִאשׁוֹן לְצִיּוֹן קָרָא מִסְפֵּד, עֹרֵר בְּכִי וָנֹהַּ.


דִּמְעוֹת יְהוּדָה עַל-עַפְרוֹת צִיּוֹן כְּגַלִּים נִגָּרוּ

כִּרְבִיבִים עֲלֵי-עֵשֶׂב הִרְעִיפוּ עַל-רְגָבֶיהָ;

לְקוֹל תִּתּוֹ הֶהָרִים סָבִיב הֵד-קוֹלָם עָצָרוּ

גַּם-הָרוּחַ מִדֵּי עָבְרָהּ עָמָדָה, רָפוּ כְנָפֶיהָ;


כּוֹס חֳרָבוֹת, קָאַת גַּם-קִפֹּד חֲרָדוֹת לָבָשׁוּ

נֶאֶלְמוּ דוּמִיָּה, לִפְצוֹת פִּיהֶם לֹא הִרְהִיבוּ,

תַּנִּים וִיעֵנִים מִסָּבִיב בִּרְעָדָה נִגָּשׁוּ

וּכְמוֹ בְחֶמְלָה רַבָּה לְקִינַת הַמְשׁוֹרֵר הִקְשִׁיבוּ.


אֶפֶס בְּעוֹדֶנּוּ מִתְפַּלֵּשׁ בָּאֵפֶר וּמִתְיַפֵּחַ

וּבַחֲלוֹם שִׁיבַת צִיּוֹן נַפְשׁוֹ עֹז דֹּרֶכֶת,

עַל-סוּסוֹ עָבַר עֲרָבִי, אַכְזָר וּמְרַצֵּחַ

וַיִּתְקַע בְּכִלְיוֹתָיו לַהַט חֶרֶב מִתְהַפֶּכֶת!


כִּמְעַט כִּלָּה זֵד יָהִיר מַעֲשֵׂה הַשַּׁעֲרוּרִיָּה

חָמַק עָבַר מֵעֵין רֹאִי, עִקְּבוֹתָיו לֹא נוֹדָעוּ,

נֶעֶלְמוּ סוּס וְרֹכְבוֹ כְּמוֹ יָרְדוּ שְׁאוֹל תַּחְתִּיָּה

כְּמוֹ פָצְתָה אֶרֶץ פִּיהָ וּשְׁנֵיהֶם חַיִּים נִבְלָעוּ.


אַךְ הַמְשׁוֹרֵר לֹא קָם וְלֹא זָע, קִינָתוֹ לֹא חָדֵלָה

וַיֵּז נִצְחוֹ עַל-עָפָר, הִתְבּוֹלֵל עִם-דִּמְעָתוֹ,

“בְּשׁוּבֵךְ אֶל-קַדְמוּת נְעוּרָיִךְ” עֵת שְׂפָתוֹ מִלֵּלָה

אֶל-בֵּית אָבִיהָ כִּנְעֻרֶיהָ שָׁבָה גַם-נִשְׁמָתוֹ.


אוֹ אָז נִדְבְּרוּ יִרְאֵי אֲדֹנָי אִישׁ אֶל-רֵעֵהוּ:

לֹא יִשְׁמְעֵאלִי פֶּרֶא אָדָם, שֹׁפֵךְ דָּם כַּמָּיִם

הֵבִיא חַרְבּוֹ בְּלֵב יְהוּדָה, קִפַּד כָּאֹרֵג חַיֵּיהוּ

כִּי אִם-מַלְאָךְ מִמְּרוֹמִים, רֹכֵב עַל-כְּרוּב שָׁמָיִם;


כִּי שָׂרָף מֵעִם-אֲדֹנָי יְרוּשָׁלַיְמָה שָׁלוּחַ

לְהוֹצִיא נֶפֶשׁ יְדִידוֹ מִתּוֹךְ עֵמֶק הַבָּכָא,

לָשֵׂאת אוֹתָהּ חִישׁ קַל, לַהֲבִיאָהּ עַל-כַּנְפֵי-רוּחַ

אֶל-שִׂיא זְבוּלֵי עֲרָב? אֶל-מְעוֹנוֹת הַבְּרָכָה.


שָׁם קִדְּמוּ פְנֵי יְהוּדָה אַלְפֵי שִׁנְאָן רִבֹּאתָיִם

גַּם אֲבוֹת יְשֻׁרוּן וְהַנְּבִיאִים בְּתוֹכָם הוֹפִיעוּ,

הַלְוִיִּם שָׁרֵי קֹדֶשׁ עָרְכוּ מַקְהֵלַת מַחֲנָיִם

וּבְשִׁירָה מַעֲשֵׂה יָדָיו בִּזְמִרוֹת לוֹ הֵרִיעוּ:


"שׁוּבִי, יְחִידָה, אֶל-מְנוּחָתֵךְ שׁוּבִי

תָּמִיד לְכִסֵּא הָאֱלֹהִים שֵׁבִי;

כִּסְאוֹת אֲדָמָה מַאֲסִי, תֵּדְעִי הֲכִי

עָלִית אֱלֵי-מָרוֹם וְשָׁבִיתָ שֶׁבִי;

הִשְׁתַּחֲוִי וּתְנִי לְאֵל כָּבוֹד וָעֹז

וּמְקוֹם בְּנֵי אֵלִים תְּהִלּוֹת הָבִי!"1




ד

בְּאָזְנֵי רַעְיָתִי יָפָתִי קָרָאתִי שִׁירָתִי

וַתַּקְשֵׁב אֵלַי רַב-קֶשֶׁב בְּהַשְׁקֵט וּבִדְמָמָה,

אַךְ עֵת עַל-אַרְגַּז דָּרְיָוֶשׁ הִטַּפְתִּי מִלָּתִי

לָטְשָׁה לִי עֵינֶיהָ יוֹנִים, גַּם-פָּנִים רָעָמָה.


בִּמְרִירוּת-לֵב נֶאֶנְחָה וַתַּעַן לִי אֲמָרֶיהָ:

לוּ אֲהַבְתַּנִי אִישִׁי כַּאֲשֶׁר גַּם-בְּעֵינַי יָקָרְתָּ,

לוּ זָכַרְתָּ לְרַעְיָתְךָ אַהֲבַת כְּלוּלֹתֶיהָ

כִּי עַתָּה אֶת-הָאַרְגָּז בַּחֲדָרֶיךָ לֹא נָטָרְתָּ.


מַה-בֶּצַע אִם-יַעֲמֹד תָּמִיד עַל-שֻׁלְחַן הַפָּנִים

מַה-מּוֹעִיל אִם-בְּיָפְיוֹ רֵעֶיךָ נַפְשָׁם יַשְׂבִּיעוּ,

הֲלֹא טוֹב לָנוּ לְמָכְרוֹ בְּכֶסֶף אַלְפַּיִם מָנִים

וּלְמַלֹּאת דֵּי מַחְסֹרַי, מְנוּחַת לִבִּי יַפְרִיעוּ.


הֲטֶרֶם תֵּדַע כִּי מוֹדַעְתָּנוּ חֵלֶץ הַפְּלוֹנִי

כְּבָר קָנָה לַאֲהוּבָתוֹ שִׂמְלַת-מֶשִׁי צְחוֹרָה

וְלָמָּה-זֶּה אֵפוֹא יִגְרַע גּוֹרָלִי, אֶשָּׂא קְלוֹנִי

הֲלֹא אֲנִי אֶשְׁתְּךָ אֵרַשְׂתָּ לְּךָ כַּדָּת וְכַתּוֹרָה.


גַּם-מֶרְכָּבָה מְרַקֵּדָה נָתַן אֵילוֹן הַזְּבוּלֹנִי

אֶל-רַעְיָתוֹ הָעֲנֻגָּה, הִיא חֲבֶרְתִּי צִפּוֹרָה,

וּמַה-מְּאֹד תִּכְלֶינָה שְׁתֵּי עֵינַי, מַה-יִגְדַּל יְגוֹנִי

בִּרְאוֹתִי אוֹתָהּ רֹכֶבֶת, וַאֲנִי בְּרַגְלַי אֶעֱבֹרָה!


וּלְשִׁירֵי יְהוּדָה מַה-זֶּה חָרַדְתָּ כָּל-הַחֲרָדָה?

הַמְעַט אִם-אֲחַתְּלֵם בְּתַכְרִיךְ-בּוּץ וְתוֹלַעַת שָׁנִי,

עָלָיו אֶקְשֹׁר חוּט תְּכֵלֶת סָבִיב כְּצָמִיד וּצְעָדָה

וַאֲנִיחֵם יַחַד עִם-הַשֵּׁרוֹת, מִנְחָה נְתַתָּנִי;


אוֹ אֶל-בֵּית מְלֶאכֶת אָמָן מַשְׂכִּיל אֶשְׁלְחָה אוֹתָם

לְמַעַן יַלְבִּישֵׁם כָּתְנוֹת עֹרֹת אֵילִם מְאָדָּמִים

וְצִפָּה זָהָב טָהוֹר שְׂפָתָם, מִסְגְּרוֹתָם וְגַבּוֹתָם

וַאֲשִׂימֵם בָּאָרוֹן עִם-יֶתֶר סִפְרֵי הַחֲכָמִים.


גַּם זֹאת פְּלִיאָה דַעַת מִמֶּנִּי, חֵי נַפְשִׁי וְגֵוִי!

כִּי מֵעוֹדִי מְאוּמָה לֹא שָׁמַעְתִּי, לֹא רָאִיתִי

עַל-דִּבְרַת מְשׁוֹרֵר עִבְרִי, רַבִּי יְהוּדָה הַלֵּוֵי,

אַף כִּי-חֹק לִמּוּדֵי בֵית-סֵפֶר גַּם-אֲנִי כִלִּיתִי…


בִּלְשׁוֹן רַכָּה עָנִיתִי לָהּ: שִׁמְעִי-נָא יוֹנָתִי,

אִם עַל-דְּבַר יְהוּדָה בְּבֵית-סִפְרֵךְ לֹא קָנִית דָּעַת

אוֹת נֶאֱמָן הוּא כִּי חִנּוּךְ בְּנוֹת עַמִּי וַעֲדָתִי

עוֹד חֶסְרוֹן רַב בּוֹ, מְעֻוָּת פְּחֶתֶת וּמִגְרָעַת.


אֵיךְ תּוּכַלְנָה בָּנוֹת כָּאֵלֶּה בִּהְיוֹתָן לְנָשִׁים

גַּדֵּל בָּנִים אֹהֲבֵי עַמָּם אַהֲבַת בִּלְתִּי סָרָה,

לֹחֲמִים מִלְחֶמֶת אֲחֵיהֶם הָרְצוּצִים וְהַנִּגָּשִׂים

וּמִתְיַמְּרִים בִּגְאוֹן לְאֻמָּם גַּם לְעִתּוֹת בַּצָּרָה?


הַאֻמְנָם בְּבֵית-סֵפֶר תִּקְנֶינָה שֵׂכֶל וּבִינָה:

לְכָל-חֲנוּטֵי מַלְכֵי מִצְרַיִם לִקְרֹא בְשֵׁם לָמָדוּ

וּלְהוֹצִיא בְמִסְפַּר כָּל-גָּלְמֵי שַׁלִּיטֵי כִינָה

שֶׁבִּן-לַיְלָה הִשְׂתָּרֲרוּ וּבִן-לַיְלָה אָבָדוּ;


לָדַעַת שְׁמוֹת סוֹפְרֵי עָם וָעָם, חֲדָשִׁים גַּם-יְשָׁנִים,

שְׁמוֹת חֹבְרֵי סִפּוּרֵי בַדִּים, יֹצְרֵי עָמָל עֲלֵי-חֹק

וּשְׂפַת צָרְפַת הַיָּפָה תִּלְמַדְנָה רַבּוֹת שָׁנִים

לָבוֹא חֶדֶר בְּחֶדֶר, לַהֲגוֹת בְּסִפְרֵי פּוֹל-דִּי-קוֹק;


לְהָבִין בְּמִסְתְּרֵי עֲצַבֵּי הַגּוֹיִם מָשָׁל וְחִידָה,

לְמִן-הֲדַר כְּבוֹד יֻפִּיטֶר הַגָּדוֹל וְאִשְׁתּוֹ יוּנוֹנָה

עַד-יִצְרֵי מַעַלְלֵי הַקְּטַנִים, אַמּוֹר וְקֻפִּידָה…

אִם-עַל-אֵלֶּה נִשְׁאָלֵן, תְּמַהֵרְנָה עֲנוֹת נְכוֹנָה.


אֶפֶס אִם פִּי הַצְּעִירוֹת הָהֵן נוֹסִיף נִשְׁאָלָה

עַל-דְּבַר מְשֹׁרְרֵי יְשֻׁרוּן, נְשָׁרִים גִּדְלֵי הַכְּנָפָיִם

אֲשֶׁר חָיוּ בְּתֹר-הַזָּהָב, בָּעֵת הַמְהֻלָּלָה

בִּפְרֹחַ חָכְמַת יְהוּדֵי סְפָרַד כְּאֶרֶז עַל-מָיִם;


אִם-נִשְׁאָלֵן עַל-שְׁלֹשֶׁת כּוֹכְבֵי הַשִּׁיר וְהַתְּעוּדָה

פַּאֲתֵי תֵבֵל הֵאִירוּ כִּבְרָקִים בָּאֲפֵלָה:

שְׁלֹמה בֶּן-גְּבִירוֹל, משֶׁה בֶּן-עֶזְרָא וִיהוּדָה

שֶׁבְּחָכְמָתָם וּבְשִׁירָתָם כָּל-יְמֵי עוֹלָם נִתְהַלֵּלָה;


אִם-נִשְׁאָלֵן עַל-שְׁאָר הַחֲכָמִים, גְּדוֹלֵי עַמֵּנוּ

כְּרַבִּי שְׁמוּאֵל הַנָּגִיד וּמַיְמוּן, אַנְשֵׁי-פֶלֶא,

הֲלֹא אָז כִּפְסִילִים אִלְּמִים תַּעֲמֹדְנָה לְפָנֵינוּ

כָּל-חָכְמָתָן תִּתְבַּלָּע, בִּינָתָן תִּיגַע וְתֵלֶא…


לוּ תֹאבִי לְמַלֹּאת הַחֶסְרוֹן, שִׁמְעִי-נָא לַעֲצָתִי,

הֵן עוֹד גֹּאֲלֵךְ חַי, טֶרֶם גָּוַע וַיֵּאָסֶף

מַהֲרִי-נָא לִמְדִי שְׂפַת-עֵבֶר, בִּטְחִי בִי בְּעֶזְרָתִי

אָנֹכִי אָנֹכִי מְלַמְּדֵךְ לְהוֹעִיל, חִנָּם אֵין-כָּסֶף!


אַךְ מִבָּתֵּי מִשְׂחָק וְזִמְרָה מִנְעִי-נָא רַגְלָיִךְ

לִשְׁלוֹם רֵעוֹתַיִךְ דִּרְשִׁי רַק אַחַת בַּחֹדֶשׁ,

גַּם אַל-תַּרְבִּי שִׂיחָה עִמָּהֶן בְּבוֹאָן אֵלָיִךְ

וַאֲנַחְנוּ לְבָדָד יַחַד נִשְׁקֹד עַל-לְשׁוֹן הַקֹּדֶשׁ.


לִתְקוּפַת הַיָּמִים, כִּמְעַט תַּעֲבר שָׁנָה אַחַת

וּשְׂפַת-עֵבֶר תֵּדְעִי הֵיטֵב, בִּי לִבִּי בָטוּחַ,

בִּקְרוֹא שִׁירֵי בֶן-גְּבִירוֹל וּבֶן-עֶזְרָא תִּרְוִי נַחַת

וּמַה-גַּם בְּשִׁירֵי יְהוּדָה אֲהוּבִי, כַּבִּיר הָרוּחַ.


שְׁלֹשֶׁת אֵלֶּה גִבֹּרֵי הַשִּׁיר כְּמוֹ בְרִית כָּרָתוּ

כִּבְרִית-הַשָּׁלִשִׁים בְּשַׁעֲרֵי רוֹמָא וְעַל-שְׂדֵי-לָחֶם,

אוּלָם לֹא לִלְכֹּד אֲרָצוֹת וְאִיִּים אֶת-לִבָּם שָׁת:

כִּי אִם-לִרְכשׁ לָהֶם כִּנּוֹר דָּוִד מִבֵּית-לָחֶם.


אַלְחֲרִיזִי הַסְּפַרְדִּי (גַּם שְׁמוֹ יְהוּדָה לֹא שָׁמַעַתְּ)

חֹבֵר סִפּוּרִים חֲרוּזִים, הוּא סֵפֶר “הַתַּחְכְּמֹנִי”

הַטּוֹבִים גַּם-מִסִּפּוּרֵי הֲרִירִי, אֲשֶׁר יָדַעַתְּ

עַל-פִּי תַרְגּוּם רִקֶּרְט, לֹא בִשְׂפַת-עֲרָב כָּמֹנִי;


וַהֲתֻלָּיו חַדּוּ מֵהֲתֻלֵּי וָלְטֵר הַצָּרְפָתִי

וּבְחֹפֶשׁ-רוּחַ קִדְּמוֹ זֶה כְּבָר שֵׁש מֵאוֹת שָׁנָה

(הֶגְיוֹנֵי וָלְטֵר הֵן מָצְאוּ חֵן בְּעֵינַיִךְ, יָפָתִי),

עַל-אֹדוֹת הַמְשׁוֹרְרִים הָאֵלֶּה כֹּה אָמַר וְעָנָה:2


"שִׁירֵי שְׁלֹמה עָמְקוּ עַד-מְאֹד, רַעְיוֹנָיו נִשְׂגָּבִים

כָּל-חֹקְרֵי חֵקֶר אֱלוֹהַּ עֲלֵיהֶם יִתְמָהוּ;

יְפֵי-תֹאַר שִׁירֵי בֶן-עֶזְרָא, תַּבְנִית הֵיכָל מְחֻטָּבִים

כָּל-חֹמְדֵי מַעֲשֵׂה יְדֵי אָמָן אַחֲרֵיהֶם יִנָּהוּ;


אַךְ בְּשִׁירֵי יְהוּדָה עֹמֶק וָיֹפִי יַחְדָּו חֻבָּרוּ,

כֻּלָּם מְקֻטָּרִים בְּאַבְקַת לְבֹנָה?כָּה וָמֹר,

מִכָּל-הַהֹלְכִים לְפָנָיו עַל-כֵּן אִמְרוֹתָיו יָקָרוּ

כָּל-קֹרְאָיו יִקְרְאוּ: “הֶאָח, חַמּוֹנוּ רָאִינוּ אוֹר!”


וּדְעִי לָךְ כִּי לֹא רַק שִׁירֵיהֶם נִשְׂגְּבוּ וְרָמּוּ,

גַּם דִּבְרֵי יְמֵי חַיֵּיהֶם נִפְלָאִים הַפְלֵא וָפֶלֶא,

אַף-הֵמָּה יְשַׁעַשְׁעוּ נַפְשֵׁךְ, אֶל-חִכֵּךְ יִנְעָמוּ;

אֶעֶרְכָה לְפָנַיִךְ הַפַּעַם אַךְ-מְעַט מֵאֵלֶּה:


חֹטֶר מִגֶּזַע יִשַׁי הָיָה משֶׁה בֶּן-עֶזְרָא

הוּא וּשְׁלֹשֶׁת אֶחָיו, אֶזְרָחֵי קִרְיַת גְּרַנַּדָּה,

אֶל-חָכְמַת יְשֻׁרוּן הָיוּ מִבְטַח-עֹז, מִשְׁעָן וְעֶזְרָה

כֻּלָּם בֵּרַךְ אֵל בָּעֹשֶׁר, גַּם בְּחָכְמָה וּבְמַדָּע.


הָאַחִים שְׁלָשְׁתָּם רָאוּ טוֹבָה, שְׂמָחוֹת שָׂבָעוּ

הִתְעַנְּגוּ עַל-רֹב שָׁלוֹם, נַחֲלָת עֲלֵיהֶם שָׁפָרָה;

אַךְ דַּרְכֵי משֶׁה הִתְעַוְּתוּ וּנְתִיבוֹתָיו לָעוּ

כּוֹס הַתַּרְעֵלָה עָלָיו פִּתְאֹם עָבָרָה.


לֹא אֵשׁ מִן-הַשָּׁמַיִם עַל-בֵּיתוֹ נָפָלָה

לֹא שֹׁדְדִים בָּאוּ עָלָיו, הֵשַׁמּוּ אֶת-נָוֵהוּ

לֹא חַיָּה רָעָה שִׁכְּלָה צֹאנוֹ, טָרְפָה אָכָלָה

לֹא פֶגַע-רָע פְּגָעוֹ אוֹ נֶגַע נָגַע בְּאָהֳלֵהוּ;


הָאַהֲבָה אַבְנֵי אֵשׁ-פְּלָדוֹת עַל-רֹאשׁוֹ קָלָעָה

הָאַהֲבָה כְּשֹׁד מִשַּׁדַּי שָׁדְדָה חָמְסָה מְנוּחָתוֹ,

כְּחַיְתוֹ-טֶרֶף קָרְעָה סְגוֹר לִבּוֹ, דָּמוֹ בָלָעָה

וּבְמַכּוֹת נֶאֱמָנוֹת דִּכְּאָה לָאָרֶץ חַיָּתוֹ!


לֹא בְּבַת אֵל-נֵכָר חָשַׁק מֹשֶה, בְּאַהֲבָתָהּ שָׁגָה

וּמִצְוַת אֱלֹהִים-חַיִּים לְמִפְגָּע לוֹ הָיְתָה,

לֹא אֶל-בַּת גָּדוֹל וָרָם מִמֶּנּוּ נַפְשׁוֹ עָרָגָה

אַף-לֹא אֲהוּבָתוֹ בָחֲלָה בּוֹ, צַלְמוֹ בָזָתָה;


בְּבַת אָחִיו הַבְּכוֹר בָּחַר, גַּם-הִיא אֲהֵבַתּוּ נְדָבָה

וּבְזִקֵּי אַהֲבָתָהּ רֻתַּק כְּמוֹ בַסּוּגַר וְחַחִים,

אַךְ אֶחָיו עָמְדוּ לְשָׂטָן לוֹ, תּוֹחַלְתּוֹ נִכְזָבָה

אֲהוּבָתוֹ הָיְתָה לְאִישׁ-אַחֵר – לְאֶחָד מִן-הָאַחִים!


בְּיוֹם נָכְרוֹ וְאֵידוֹ, עֵת נוֹחֲלָה אָבְדָה תִּקְוָתוֹ

יֵשׁ מֵאַנְשֵׁי שְׁלוֹמוֹ עֲזָבוּהוּ, עָמְדוּ מִנֶּגֶד

וְיֵשׁ גַּם-זָדוּ לִזְרוֹת מֶלַח עַל-מַחַץ מַכָּתוֹ

בֵּין אַחִים שִׁלְּחוּ מְדָנִים וַיִּבְגְּדוּ בוֹ בֶּגֶד.


תְּפִלָּתוֹ בְּמוֹ-פִיהוּ וּבְיָדוֹ מַטֵּה הַהֵלֶךְ

נָטַשׁ בֶּן-עֶזְרָא מוֹלַדְתּוֹ – וַאֲהוּבָתוֹ הָרַכָּה

וַיִּרְחַק נְדוֹד מֵעִיר לְעִיר וּמִפֶּלֶךְ אֶל-פֶּלֶךְ,

אַךְ הָאַהֲבָה לֹא סָרָה וַתְּשִׁיבֵהוּ עַד-דַּכָּא.


אָן יֵלֶךְ אִישׁ מֵרוּחָהּ, אָן יִבְרַח מִפָּנֶיהָ?

אִם-יַעֲלֶה גִּבְעוֹת-עוֹלָם כְּעֹפֶר עַל הָרֵי-בָתֶר,

אִם-יֵרֵד קִצְבֵי-אֶרֶץ עִמְקֵי תַחְתִּיּוֹתֶיהָ

מִיָּדָה לֹא יִבְרָח, מִפָּנֶיהָ לֹא יִסָּתֶר.


עִמּוֹ תִּסַּק שָׁמַיִם, עִמּוֹ גַּם-שְׁאוֹל תַּצִּיעַ

וּבְכָל-מְקוֹמוֹת מַחֲבֹאָיו לֹא תֶחְדַּל עַצָּבֶת,

כָּל-עוֹד לִבּוֹ יָחוּש בְּקִרְבּוֹ, לֹא יִשְׁקֹט יַרְגִּיעַ

כִּי בֵיתָהּ בְּקִיר-לִבּוֹ וּבַחֹמָה הִיא יוֹשָׁבֶת…


בַּהֲלֹךְ מֹשֶׁה בְּאֶרֶץ בְּנֵי-קֵדָר, קֹדֵר בַּלָּחַץ

אַנְשֵׁי חָמָס הָעֹרְקִים צִיָּה עָלָיו הִתְנַפָּלוּ,

אֶל-גַּבֵּי סוּס אַבִּיר קְשָׁרוּהוּ אֵלֶּה בְנֵי-שָׁחַץ

וַיִּסְחָבֻהוּ אֶל-הָעֲרָבָה וּבוֹ הִתְעַלָּלוּ:


עַל-שִׁכְמוֹ נָתְנוּ כָל-מְלָאכָה נְמִבְזָה שִׁפְלַת-עֵרֶךְ

לִשְׁפּוֹת אֶת-הַסִּיר וּלְהַשִּׁיק תַּנּוּר וְכִירָיִם,

אַף-אִלְצוּהוּ בֹּקֶר וָעֶרֶב לִכְרֹעַ עַל-בֶּרֶךְ

לְעַשּׂוֹת דַּדֵּי הַפָּרוֹת וּלְהוֹצִיא חֲלֵב-שָׁדָיִם.


וַיְהִי הַיּוֹם, עֵת בִּמְלַאכְתּוֹ זֹאת יָדוֹ יָגָעָה

רוּחוֹ הֱצִיקַתְהוּ – וַיְשׁוֹרֵר וַיְמָרֵר בַּבֶּכִי

וַיִּשְׁמַע הַכַּהַן, יִלְלַת שִׁירוֹ עַד-לִבּוֹ נָגָעָה,

אָז נִכְמְרוּ רַחֲמָיו וַיִּמַח דִּמְעָתוֹ מִלֶּחִי:


דְּרוֹר קָרָא לוֹ הַשַּׁלִּיט, שִׁכְמוֹ הֵסִיר מִסֵּבֶל

וּלְבוּשֵׁי תִפְאֶרֶת נָתַן לוֹ לִכְסוּת חֶרְפָּתוֹ,

גַּם בַּמִּנְחָה קִדֵּם פָּנֵיהוּ, בְּמִנִּים וָנֵבֶל



אֶפֶס מֹשֶׁה הוֹסִיף נְדוֹד וַיֶּהְגֶּה אַךְ-נְכָאִים

גַּלְמוּד עֲרִירִי הָלַךְ גּוֹלֶה, מִבְּלִי רֵעַ וָעֵזֶר,

וַיּוֹלֶד אֶת “הָעֲנָק”, שִׁירִים בְּתַרְשִׁישׁ מְמֻלָּאִים

וּלְקָדְקֹד בֶּן-עֶזְרָא נְזִיר אֶחָיו יִהְיוּ כַנֵּזֶר.


זֶה חֵלֶק-מְשׁוֹרְרִים, מִיּוֹם בְּרֹא אֵל אֶרֶץ וְשָׁמָיִם

כָּל-יְמֵי חֶלְדָּם יִזְרְעוּ אֶל-קוֹצִים, שָׁמִיר וָשָׁיִת

תַּאֲוָתָם תֹּאבֵד, מַר יִבְכְּיוּן וְיַחַרְקוּ שִׁנָּיִם,

אוּלָם כְּבוֹד מְלִיצִים יַנְחִילוּם – נִזְרֵי עֲלֵי-זָיִת!


כֵּן גַּם-אַפּוֹלוֹ, אֲבִי כָּל-תֹּפֵשׂ כִּנּוֹר וָנֵבֶל

לַשָּׁוְא רָדַף אַחֲרֵי דַפְנָה, לֹא בָהּ רָאָה בְרָכָה

כִּמְעַט תָּפַשׂ אֹתָהּ בַּכָּף, תִּקְוָתוֹ הָיְתָה חֶבֶל,

לְנֵזֶר עֲלֵי-דַפְנָא יַעֲלַת-חֵן בְּכַפּוֹ נֶהְפָּכָה.

*

בְּבֶן-גְּבִירוֹל מִדֵּי אֲדַבֵּר – אָשִׂיחָה וְאֶהֱמָיָה,

גַּם הוּא שָׂבַע מַמְּרוֹרִים בִּטְפָחוֹת יְמֵי חֶלְדֵּהוּ

חֶלְקוֹ לֹא שָׁמַן בָּאָרֶץ, אַף-כּוֹסוֹ לֹא-רְוָיָה

רַק הַנֵּזֶר יָצִיץ עַל-מִצְחוֹ, לֹא יִבּוֹל עָלֵהוּ!


עוֹדוֹ יֶלֶד צָעִיר וָרַךְ אָבִיו וְאִמּוֹ נֶאֱסָפוּ

גַּם אָח וְאָחוֹת לֹא הָיוּ לוֹ, רְפֵה-כֹחַ אֵין-אוֹנִים,

אָכֵן חַסְדֵי אֲבִי יְתוֹמִים עַל-רֹאשׁוֹ חָפָפוּ

רוּחוֹ אָצַל עָלָיו “וְלִבּוֹ בֶּן כְּלֵב בֶּן-הַשְּׁמֹנִים”.


בִּימֵי הַבֵּינַיִם, יְמֵי חֹשֶׁךְ וַאֲפֵלָה נִדַּחַת

עֵת כֹּהֲנִים הֹזִים פֹּחֲזִים בָּאֵימִים הִתְהֹלָלוּ,

עֵת מַשְׂאוֹת שָׁוְא וָרִיק הָיוּ לַדָּת שְׂאֵת וְסַפַּחַת

אֵל-אֵלִים לֹא הִכִּירוּ, מִדַּעַת דְּרָכָיו נוֹאָלוּ;


אָז נֶפֶשׁ בֶּן-גְּבִירוֹל תָּשִׁיר עֱזוּז-יָהּ וּמִפְלְאוֹתָיו

עַל-חוּג שָׁמַיִם תִּתְהַלָּךְ, תַּרְקִיעַ לִשְׁחָקִים;

כָּכָה יְזַמֵּר הַזָּמִיר וִיצַפְצֵף מַנְגִּינוֹתָיו

מִדֵּי יְרַחֵף עַל-אֲהוּבָתוֹ – שׁוֹשַׁנַּת הָעֲמָקִים.


גַּם שִׁירֵי שְׁלֹמה אַבְנֵי-חֵפֶץ הֵם, פְּנִינִים וְאַלְגּוּמִּים

אוֹר-הַגָּנוּז גָּנוּז בָּם, כְּמַרְאֵה הַקֶּשֶׁת יָאִירוּ,

נֹגַהּ לָמוֹ כִּנְגֹהוֹת טוּרֵי הָאוּרִים וְהַתֻּמִּים

וְאַבְנֵי שִׁיר “כֶּתֶר-מַלְכוּת” גַּם כֶּתֶר מַלְכוּת יַחְפִּירוּ.


אַף “מְקוֹר חַיִּים” לַחָכְמָה רִאשֹׁנָה יָדָיו פָּתָחוּ,

וְהַחוֹקְרִים אַנְשֵׁי-שֵׁם בְּרָזֵי עוֹלָם תְבּוֹנָנוּ

מִמְּקוֹר נֶאֱמָן זֶה שָׁאָבוּ, מִלִּקְחוֹ לָקָחוּ

עַד-כִּי שֵׁם “פְּלַטוֹ הָעִבְרִי” בְּצֶדֶק לוֹ קָרָאוּ.


פְּעָמִים שְׁלֹשִׁים רָאָה בְּחַדֵּשׁ פְּנֵי-אֲדָמָה הָאָבִיב

וּכְבָר בְּכָל-הָאָרֶץ יָצָא שֵׁם מְקוֹרוֹ וּמִזְמוֹרוֹ,

כָּבוֹד וָהוֹד עִטְּרוּהוּ כָּל-הַחֲכָמִים מִסָּבִיב

וּמְשׁוֹרְרִים כֻּלָּם צָהֲלוּ לְצִלְצְלֵי קַוֵּי כִנּוֹרוֹ.


אָמְנָם אִם-חִצֵּי אַהֲבָה לֵב בֶּן-עֶזְרָא פִּלֵּחוּ

בְּנֵי אִמּוֹ נִחֲרוּ-בוֹ, מְרָרוּהוּ עַד-זִקְנָה וְשֵׂיבָה;

חִצֵּי חֲמַת הַקִּנְאָה עַצְמוֹת בֶּן-גְּבִירוֹל פִּצֵּחוּ

בִּדְמִי יָמָיו הֲדִמּוֹ זָר, נוֹטֵר שִׂטְנָה וְאֵיבָה.


בְּקֹרְדֻבָּה הָעִיר הַהוֹמִיָּה, מְקוֹם מְגוּרֵי שְׁלֹמה

גָּר גַּם-מְשׁוֹרֵר עֲרָבִי, שָׁכֵן קָרוֹב אֶל-נָוֵהוּ,

אֶפֶס עַל-אֹדוֹתָיו הִתְחָרְשׁוּ כָּל-אַנְשֵׁי מְקוֹמוֹ

כִּי נֹפֵל הוּא מִשְּׁלֹמֹה וּמָךְ מְאֹד מֵעֶרִכֵּהוּ.


וַיִּשְׁמַע וַיִּנְאָץ, קִנְאָתוֹ בָּעֲרָה כַתֹּפֶת,

כְּשָׁמְעוֹ זִמְרַת שְׁלֹמֹה מְרֵרָתוֹ לָאָרֶץ נִשְׁפָּכָה

וְשִׁירָתוֹ הַנְּעִימָה, הַמְּתוּקָה מִדְּבַשׁ וָנֹפֶת

אֶל-חֵךְ הַמְקַנֵּא הַלָּז לְלַעֲנָה וָרֹאשׁ נֶהְפָּכָה,


קִנְאָה תַּצְמִיד מִרְמָה, תֵּלַכְנָה בָאֹפֶל כַּדֶּבֶר

וּלְרַגְלֵיהֶן יֵצְאוּ חֹצֵץ קֶטֶב מְרִירִי וָרֶצַח;

כֹּה הֵבִיא קַיִן עַל-אָחִיו הַשֹּׁד וְהַשֶּׁבֶר

כֹּה יַעֲשֶׂה כָל-מְקַנֵּא הַיּוֹם וְכֹה יוֹסִיף לָנֶצַח.


גַּם-לֵב עֲרָבִי אִישׁ-תְּכָכִים גָּרַשׁ טִיט וָרֶפֶשׁ,

אֶל-בֵּיתוֹ מָשַׁךְ אֶת-שְׁלֹמֹה בַּעֲבֹתוֹת עָרְמָתוֹ,

בַּלַּיְלָה קָם עָלָיו פֶּתַע וַיְרַצְּחֵהוּ נֶפֶשׁ

וּבְיַרְכְּתֵי הַגִּנָּה קָבַר חֶרֶשׁ אֶת-גְּוִיָּתוֹ.


וְהִנֵּה זֶה פֶלֶא, כָּמֹהוּ לֹא הָיָה לְעֹלָמִים:

עַל-גַּל קִבְרַת הַמְשׁוֹרֵר חִישׁ-קַל תְּאֵנָה צָמָחָה

וּבְטֶרֶם עוֹד יַעֲבֹר עָלֶיהָ גַּם-חֹדֶשׁ יָמִים

וַתִּיף בְּגָדְלָהּ וַהֲדָרָהּ, הֵצִיצָה צִיץ גַּם-פָּרָחָה.


פַּגֶּיהָ חָנְטוּ בְלֹא-עֵת, לֵב רוֹאֵיהֶם לָקָחוּ

כָּל-טֹעֲמֵיהֶם שָׂבְעוּ דֶשֶׁן וּמַה-מְּאֹד הִתְפַּלָּאוּ,

כִּי הַתְּאֵנִים הָאֵל רוּחַ-נְעִימוֹת בָּם מָסָכוּ

נַפְשָׁם הִתְהַלְּכָה בִגְדֹלוֹת, גַּם-חֶזְיֹנוֹת רָאוּ.


וַיֵּצֵא בָעִיר שֵׁמַע זֹאת הַתְּאֵנָה הַנִּפְלָאָה

וַיָּשִׂיחוּ בָהּ כָּל-יוֹשְׁבֶיהָ מִזָּקֵן וְעַד-נָעַר

וּבְכֵן גַּם לְאָזְנֵי הַחַאלִיף הַשְּׁמוּעָה בָאָה

בְּשִׁבְתּוֹ עִם-יוֹעֲצָיו וַחֲכָמָיו לִשְׁפֹּט בַּשָּׁעַר.


וַיְצַו לְהָבִיא מִן-הַתְּאֵנִים, וַיִּתְמַהּ עַל-הַפֶּרִי

כִּי אֱמֶת הַשְּׁמוּעָה נוֹכָח בְּאָכְלוֹ אַחַת מֵהֵנָּה,

וַיַּחְלֵט כִּי יָצוּק בַּתְּאֵנָה דְּבַר-קֶסֶם וָמֶרִי

ולִפְנֵי הֲדוֹם כִּסְאוֹ הִבְהִילוּ אֶת-בַּעַל הַתְּאֵנָה.


אַךְ הָעֲרָבִי לֹא חָת, לֹא-יָדַע עַוָּל בּשֶׁת

אֶת-שֵׁם הָאֱלֹהִים לַשֶּׁקֶר נָשָׂא עַל-לְשׁוֹנוֹ,

וַיָּשֶׂם לִבּוֹ כְּאֶבֶן-שַׁיִשׁ וּמִצְחוֹ כִּנְחֹשֶׁת

וַיֹּאמֶר כִּי אֲדֹנָי שָׁלַח הַבְּרָכָה בִּמְעוֹנוֹ.


דִּבְרֵי כַחַשׁ וְתָפֵל לֵב הַחַאלִיף לֹא הִרְגִּיעוּ,

בְּקוֹצִים וּבַרְקָנִים צִוָּה לָדוֹשׁ אֶת-בְּשָׂרוֹ

אָז רַק אָז הִתְוַדָּה וּשְׂפָתָיו פִּשְׁעוֹ הִבִּיעוּ,

עַל-רַגְלָיו שָׂמוּ נְחֻשְׁתַּיִם וּמוֹטָה עַל-צַוָּארוֹ.


רַבֵּי הַמֶּלֶךְ וּזְקֵנָיו וְכָל-הָעָם בְּרַגְלֵיהֶם

בָּאוּ הַגִּנָּה לְמַשְּׁאוֹת הַתְּאֵנָה מִשָּׁרָשֶׁיהָ,

וְהִנֵּה זֶה גּוּפַת רַבִּי שְׁלֹמֹה נִגְלְתָה לְעֵינֵיהֶם

אַךְ הִיא לֹא הִבְאִישָׁה וְרִמָּה לֹא-עָלְתָה עָלֶיהָ.


לְהוֹבִילָהּ לִקְבָרוֹת חָרְדוּ כָל-יְהוּדֵי סְפָרָד

גַּם כָּל-הָעָם מִקָּצֶה נָתְנוּ לָהּ כָּבוֹד אֵין-קֵץ

וּלְרוֹצְחָהּ הוּשַׁב גְּמוּל, חֲמָסוֹ עַל-קָדְקֳדוֹ יָרָד

כִּי בְּעֶצֶם הַיּוֹם הַזֶּה תָּלוּ אֹתוֹ עַל-הָעֵץ!

*

הִנֵּה אִם עַל-קִבְרַת שְׁלֹמֹה הַתְּאֵנָה פָרָחָה

וּמַשְׁחִית נֶפֶשׁ הַצַּדִּיק נִצְמַת בְּיַד-הַפֶּשַׁע,

עַל-גַּל הַמַּפֵּלָה, מְקוֹם נֶפֶשׁ יְהוּדָה נִרְצָחָה

לְבֵית-יְהוּדָה עוֹד יָצִיץ וּפָרַח עֵץ-חַיִּים וָיֶשַׁע.


עֵץ-חַיִּים זֶה יִשָּׂא פְרִי-קֹדֶשׁ לִצְבִי וּלְגָאוֹן

עֵינֵי אֹכְלָיו יִפְקַח לֶאֱהֹב צִיּוֹן וְקָדָשֶׁיהָ;

כָּל תֹּמְכָיו יְאֻשְּׁרוּ בָאָרֶץ, לֹא יִהְיוּ לְדֵרָאוֹן

צִיּוֹן תָּשִׂישׂ בֵּאלֹהֵי יִשְׁעָהּ, וְרַב שְׁלוֹם בָּנֶיהָ!
