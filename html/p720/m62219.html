בַּת הַבּוֹגֵד

2

על פי פרוג

א

עַל-קִרְיַת פְּרָג, בְּיִשְׂרָאֵל רַבָּתִי

פָּרַשׂ הַלַּיִל מֻטּוֹת כְּנָפֵיהוּ,

וּבְבֵית-מִקְלָט כֹּהֲנֵי דַת-נַצָּרֶת

שֶׁשֵּׁם מַרְטִין הַקָּדוֹשׁ עָלָיו קֹרָא,

הִשְׁלִיךְ הָס; שֵׁנָה עֲמֻקָּה שׂרָרֶת.

רוּחַ תַּרְדֵּמָה עַל-כָּל נָסָכָה

עַל-הַגַּנּוֹת וְעַל-הַלְּשָׁכוֹת כֻּלָּנָה,

עַל-נְאוֹת מַחֲשַׁכִּים וּמְעֹנוֹת-אֹפֶל

לִכְמָרִים זֹעֲפִים, עֲצוּבֵי-רוּחַ.


רַק בְּלִשְׁכָּה אַחַת, שָׁם בַּמִּקְצוֹעַ

בְּעַד הַחַלּוֹן אוֹר כֵּהָה הוֹפִיעַ

מִבֵּינוֹת לְעַנְפֵי אַלּוֹנֵי-זֹקֶן

אַךְ לַמִּצְעָר יָפִיץ קַרְנָיִם.

בִּפְאַת מִטָּה, עֶרֶשׂ דְּוָי וָעֹנִי

נָזִיר בּוֹדֵד מִתְהַפֵּךְ בָּחֹלִי,

אֵשׁ הַשַּׁחֶפֶת בְּלִבּוֹ קֹדָחַת.

כַּחֲצִיר גַּגּוֹת מֵחַרְבֹנֵי קַיִץ

נֶהְפַּךְ לְשַׁדּוֹ וַעֲצָמָיו עָשֵׁשׁוּ;

יִקְהַת קְמָטִים כָּל-מִצְחוֹ כִּסָּתָה

וּכְמוֹ עַל-גַּבּוֹ חֹרְשִׁים חָרָשׁוּ

וּבַמּוֹרָג לְמַעֲנִיתָם הֶאֱרִיכוּ,

בֵּין אֵלֶּה הַקְּמָטִים נִטְפֵי יֶזַע

נִצְּבוּ כְמוֹ-נֵד, כַּקֶּרַח נִקְפָּאוּ;

עֵינָיו מְפִיקוֹת יָגוֹן וְדַאֲבוֹן נֶפֶשׁ

מֵחֹרֵיהֶן כְּנִקְרוֹת צוּר עָמָקוּ

מַבִּיטוֹת לִרְגָעִים אֶל-הַפֶּתַח,

בְּתִקְוָה וּבַחֲרָדָה גַּם-יָחַד

לִמְאֹד תִּצְפֶּינָה אֶל-בֹּא אֹרֵחַ.


לִפְנֵי הַחַלּוֹן הָעֵצִים יִרְעָשׁוּ,

וּבְעָנָף רָצוּץ, עוֹדוֹ בַגֶּזַע,

עַל-הַזְּכוּכִית יַךְ כְּפַעַם בְּפַעַם

רוּחַ קַר, רוּחַ סֹעָה וָסָעַר…


נוֹרָא הַחוֹלֶה, אָיוֹם מַרְאֵהוּ!

כְּסָלָיו מָלְאוּ נִקְלֶה, לְחָייָו שֻׁפּוּ

יָדָיו צְנֻמוֹת וּבִרְכָּיו רוֹפָפוּ,

סֵמֶל שִׁמָּמוֹן הוּא, דְּמוּת צַלְמָוֶת!

אֶפֶס לְחָלְיוֹ לֹא יָשִׁית לִבֵּהוּ

וּמַכְאֹבָיו כְּאַיִן לוֹ נֶחְשָׁבוּ,

עַל-דַּל שְׂפָתָיו תְּלוּנָה לֹא עָבָרָה

רַק הִמְלִיט מִלִּין לֹא אַחַת וּשְׁתָּיִם:

"הָהּ, מַדּוּעַ בֹּשֵׁשׁ לָבוֹא הֵנָּה,

מָתַי יָבוֹא, לְעֶזְרָתִי יָחוּשָׁה!"

פִּתְאֹם חָרַד וַיַּקְשֵׁב רַב-קֶשֶׁב…

כִּי מֵאַחֲרֵי הַמְּזוּזָה וְהַדֶּלֶת

נִשְׁמַע קוֹל צַעֲדֵי אִישׁ בְּלֶכְתּוֹ שֶׁפִי;

בַּלָּאט בַּלָּאט הַדֶּלֶת נִפְתָּחָה

וְאִישׁ שֵׂיבָה, גְּבַהּ קוֹמָה וִיפֵה תֹאַר

עֹטֶה מְעִיל שָׁחוֹר – בָּא הַבָּיְתָה.


בְּקוֹל חֹדֵר כְּלָיוֹת וְלֵבָב פֹּלֵחַ:

“הָהּ רַבִּי, יִשְׁעִי וְגֹאֵל חַיָּתִי!”

מִהֵר חִישׁ וַיְקַפֵּץ מִן-הָעֶרֶשׂ;

בִּדְמָעוֹת יוֹרְדוֹת כַּאֲפִיקֵי-מַיִם

לְרַגְלֵי הַזָּקָן הִתְנַפֵּל אָרְצָה;

"הָהּ רַבִּי רַבִּי! לַחֲזוֹת פָּנֶיךָ

שְׁתֵּי עֵינַי כִּמְעַט מִיַּחֵל כָּלוּ;

אֵל שַׁדַּי יִגְמָלְךָ שִׁבְעָתַיִם

עַל-חַסְדְּךָ הַגָּדוֹל לִי הִפְלֵאתָ!

שְׁמַע-נָא עָדַי, אֶל-מִלַּי הַאֲזִינָה:

עוֹד רִגְעֵי מִסְפָּר לְחַיַּי חֻצָּצוּ,

עַד-מְהֵרָה שְׂפָתַי תֵּאָלַמְנָה,

מְאוֹרֵי עֵינַי כַּפִּשְׁתָּה יִדְעָכוּ

וַאֲשֶׁר אֶשְׁכַּב לָקוּם לֹא אוֹסִיפָה…

לָכֵן רַבִּי הוֹאִילָה לִשְׁמוֹעַ:

הֲלֹא עוֹד זָכוֹר תִּזְכֹּר, לֹא שָׁכַחְתָּ

הַקֹּרֹת קָרוּנוּ בְּעִיר בַּסִּילִיָּה?

בִּפְרוֹעַ הֲמוֹן-עָם פְּרָעוֹת קִרְבֵּנוּ,

אָז בְּלֵיל אִישׁוֹן וַאֲפֵלָה נִדָּחַת

פַּעֲמוֹן הַמִּגְדָּל נָתַן קוֹל-רָעַם

וּגְדוּדִים גְּדוּדִים כִּזְאֵבֵי עֶרֶב

לִשְׁלֹל שָׁלָל בַּחוּצוֹת פָּשָׁטוּ…

הֲתִזְכֹּר אֶת-הָאֵשׁ הַמִּתְלַקָּחַת?

זַעֲקַת-שֶׁבֶר כָּל-הָעִיר הִקִּיפָה,

אֲנָחוֹת נוֹרָאוֹת, כַּעַשׂ וָרֹגֶז

אֶנְקַת אֻמְלָלִים לְטֶבַח הוּבָלוּ

יִלְלַת נָשִׁים וּבְתוּלוֹת לְאֵין-קֵצֶה

צִלְצַל נֶשֶׁק, חֶרֶב וּמַאֲכֶלֶת

הֶמְיַת פָּרִיצִים הֲלוּמֵי יַיִן

וּשְׁאוֹן עֲלִיזִים הֵידָד קָרָאוּ –

כָּל-אֵלֶּה הִתְעָרְבוּ הִתְבּוֹלָלוּ

וּלְחֶרְדַת אֱלֹהִים נֶהְפַּךְ הַלָּיְלָה…

הֲתִזְכֹּר אֵיךְ אֶל-הָאֵשׁ הַמְבֹעֶרֶת

בְּשֶׁצֶף קֶצֶף אֶת-אָבִי הִשְׁלִיכוּ,

וּבְרֶגַע זֶה הַנּוֹרָא לְאֵין-חֵקֶר

נֶאֱבַק נִלְחַם עִם-עֹשְׁקָיו כֹּחַ

וַיְנַס לַצִּיל אִמִּי מַיַּד מְחַלֲלֶיהָ –

שִׁכּוֹר פָּרִיץ מֵאַנְשֵׁי הֶחָיִל…

גַּם-כְּבוֹד אֲחוֹתִי בְּקָלוֹן הֵמִירוּ:

יַלְדָּה עֲנֻגָּה עַל-גַּל הָאֵפֶר

פְּרוּעָה לְשִׁמְצָה שָׁם הִתְגַּלְגָּלָה

עֵרֹם וְעֶרְיָה עַל-מַשֻּׁאוֹת בֵּיתֵנוּ,

וּבְחָזְקָה אָחֲזָה בֵין אֶצְבְּעוֹתֶיהָ

קְוֻצַּת שֵׂעָר לָבָן אָז מָרָטָה

מִקָּדְקֹד נָזִיר לֹא-יָדַע בֹּשֶׁת

עֵת בְּעַד תֻּמָּתָהּ לִלְחֹם הִתְאַמֵּצָה…

הָהּ, מַחֲזֶה אָיוֹם מֵאֵין כָּמֹהוּ

כְּמוֹ נִצָּב לְפָנַי, עֵינִי תְשׁוּרֶנּוּ

וְכָל-קְרָבַי יִמָּלְאוּ בַּלְהוֹת-שָׁחַת!

כָּל-אַנְשֵׁי בֵית-אָבִי סָפוּ תַּמּוּ

רַק אָנֹכִי לְבַדִּי חַי נִשְׁאַרְתִּי

לְחֶרְפָּה לִשְׁנִינָה וּלְדִרְאוֹן נֶצַח…

אַהֲבַת הַחַיִּים עָלַי גָּבָרָה,

לְקוֹל מַדִּיחַי הִטֵּיתִי אֹזֶן

וּכְיוֹנָה פוֹתָה אֵין לֵב נִפְתֵּיתִי;

בְּחֶלְקַת שְׂפָתַיִם עָלַי נָסַבּוּ

וּבִלְשׁוֹן רַכָּה עַל-לִבִּי דִבֵּרוּ,

כָּל-הַמֹּצְאוֹת לִשְׁכֹּחַה וֹרוּנוּ

גַּם לִלְבֹּשׁ בֶּגֶד נְזִירִים תִּלְבֹּשֶׁת

וּלְהִסָּגֵר פֹּה בְּלִשְׁכַּת עֲרָפֶל –

לְמַעַן לֹא-יֹאבַד נִצְחִי וְתוֹחַלְתִּי

וּתְשׁוּעַת עוֹלָמִים אִוָּשֵׁעַ…

………………………….

זֶה שְׁלֹשִׁים שָׁנָה חָלְפוּ נָגֹזּוּ

בְּמָעוֹן צַר זֶה בָּדָד אֵשֵׁבָה,

שְׁלֹשִׁים שָׁנָה בִּגְדִי זֶה יְכַסֵּנִי

כָּל-חֻקּוֹת בֵּית-הַמִּקְלָט שָׁמַרְתִּי,

כְּאַחַד הַנְּזִירִים תָּמִיד חָיִיתִי,

אֶפֶס לְאִישׁ אַחֵר לֹא נֶהְפָּכְתִּי…

עָמוֹק עָמוֹק מְאֹד בְּחֶבְיוֹן לִבָּתִי

כְּנִיצוֹץ תַּחַת עֲרֵמַת הָאֵפֶר

גַּחֶלֶת אַהֲבַת עַמִּי בָּעָרָה,

וּמֵרֹב הַיָּמִים לֹא עֻמָּמָה;

זֶה כַּמָּה פְעָמִים חִשְּׁבָה לְהִתְלַקֵּחַ

לְהִתְפָּרֵץ הַחוּצָה בְּלַהַב שַׁלְהֶבֶת,

אַךְ בְּדֵי-עָמָל הִתְאַפֵּק יָכֹלְתִּי

וּבְחֵיקִי כָּל-עַצְּבֹתַי יָגֹרְתִּי,

וּבְכֵן שְׁלֹשִׁים שָׁנָה הֵן עָבָרוּ

לְמִן-הַיּוֹם עָשִׂיתִי בְנַפְשִׁי שָׁקֶר…

עַתָּה בְּחַלּוֹנִי כִּי עָלָה מָוֶת,

וּמֵרוּחַ אַפּוֹ בִּי הֵפִיחַ

זַלְעָפַת הַשַּׁחֶפֶת נִצָּתָה

לֶאֱכֹל כְּקַשׁ יָבֵשׁ כְּלָיוֹת וָבֶטֶן –

גַּם זִיק אַהֲבָתִי לְעַמִּי וּלְאֶחָי

בְּקֶרֶב לִבִּי יִבְעַר כְּאֵשׁ צָרֶבֶת…

אַחַת הִיא שְׁאֵלָתִי הָהּ שְׁמָעֵנִי

בְּקֶבֶר יִשְׂרָאֵל כִּי אִקָּבֵרָה,

בִּמְקוֹם שָׁם עַצְמוֹת אִמִּי הוֹרָתִי

גַּם אֶת-עַצְמֹתַי אָנָּא הַנִּיחוּ!


לוּ יָכֹלְתִּי סְגֹר לִבִּי לִקְרוֹעַ

אֲזַי רָאִיתָ, רַבִּי, וְנוֹכָחְתָּ

כִּי תַחַת מַדֵּי נָזִיר פֹּשֵׁעַ

לֵב עִבְרִי נֶאֱמָן דּוֹפֵק בְּחָזְקָה

וְעַל-מַעֲלוֹ בֵּאלֹהָיו מָלֵא נֹחַם.


לֹא נָזִיר, לֹא מֵפֵר בְּרִית לְפָנֶיךָ

כִּי אִם בֶּן-אֻמְלָל, יָתוֹם גֹּוֵעַ

וּבְתַחֲלֻאָיו עָלָיו דָּמָיו יִֹטֹּשׁוּ…

רַבִּי! הַאִם לֹא תִשְׁמַעְנָה אָזְנֶיךָ

נַחֲרַת הֶחָזֶה וּשְׁאוֹן מֵיתָרֵיהוּ?

שִׁירַת-מָוֶת הִיא, הֶמְיַת נִבְלֵהוּ

בִּנְהִי זֶה לְבֵית עוֹלָמִי יְלַוֵּנִי!

עוֹד מְעַט נַפְשִׁי לֶאֱלוֹהַּ אָשִׁיבָה

וְאֹרַח לֹא-אָשׁוּב מַהֵר אֵלֵכָה…

בֵּאלֹהִים חַיִּים הֵן אַשְׁבִּיעֶךָ

כִּי אֶל-קִבְרוֹת אֲבוֹתַי אוּבָלָה,

עִבְרִי נוֹלַדְתִּי וְעִבְרִי אָשׁוּב שָׁמָּה!

גַּם מֵהִתְפַּלֵּל וּמִשְּׁפָךְ-שִׂיחַ

אַל-תֶּחְדַּל בְּעַד נַפְשִׁי הַחֹטֵאת…

אֶת-אֱלֹהֵי שָׁמַיִם לִי אָעִידָה

כִּי בְיָדְךָ רַבִּי רוּחִי אַפְקִידָה!



ב

מְלֵא סְעִפֵּי יָגוֹן וְרַעְיוֹנֵי עֹצֶב

מִהֵר הָרַב לָשׁוּב אֶל-נָוֵהוּ,

כָּל-זִקְנֵי הָעֵדָה שָׁמָּה נֶאֱסָפוּ

יִוָּעֲצוּ יַחְדָּו, סוֹד יַמְתִּיקוּ.

"הָהּ אֵלִי (הָגָה הָרַב בָּרוּחַ)

לֹא לִי הַמִּשְׁפָּט לַהֲדֹף נָפֶשׁ

נֶפֶשׁ אֻמְלָל תִּתְמוֹגֵג בַּנֹּחַם

וְאֶל-קִבְרוֹת אֲבוֹתֶיהָ שֹׁאָפֶת…

הֵן אַתָּה אֵל רֳאִי שׁוֹפֵט צֶדֶק

בֹּחֵן כְּלָיוֹת וָלֵב חֲקַרְתַּנִי

וַתֵּדַע כִּי לֹא לְנַפְשִׁי אֶדְאָגָה

לֹא חַיַּי וְדָמִי בְּעֵינַי יָקָרוּ…


"הֵן מֵרֵאשִׁית יְמֵי נְעוּרַי עַד-עָתָּה

לִבִּי וּבְשָׂרִי אַךְ לְךָ הִקְדַּשְׁתִּי;

רַבּוֹת פְּעָמִים מַעֲדַנּוֹת הָלַכְתִּי

לִקְרַאת הַמָּוֶת עַל-קִדּוּשׁ שְׁמֶךָ,

לְמַעַנְךָ הִשְׁלַכְתִּי נַפְשִׁי מִנֶּגֶד

מִכְּלִמּוֹת וָרֹק פָּנַי לֹא הִסְתַּרְתִּי

וְכָל-צָרָה וְצוּקָה דּוּמָם נָשָׂאתִי –

אֶפֶס מַה-נּוֹרָא פָעֳלִי הַפָּעַם!"

הַאֻמְנָם לֹא לַשָּׁוְא פָּחַד פָּחַד

לֹא חִנָּם פַּלָּצוּת בִּעֲתַתּוּ

וְלֹא מֵרֹךְ לֵב עֲצָתוֹ נָבָקָה,

אָכֵן כַּאֲשֶׁר נַפְשׁוֹ מְאֹד יָדָעָה

כִּי הַבּוֹגֵד יִתְחַבֵּא בְמוֹ-אָרֶב

וּכְנָמֵר שֹׁקֵד יָשׁוּר עַל-דָּרֶךְ.

בּוֹגֵד זֶה יְהוּדִי מִלֵּדָה מִבֶּטֶן

הָפַךְ לִבּוֹ לִשְׂנֹא עַם יוֹלְדֵהוּ

לְהִתְנַכֵּל בְּאֶחָיו וּלְצוּדָם חֵרֶם;

לַנֵּכָר הִתְמַכֵּר בְּעַד בֶּצַע כָּסֶף

לְצֹרֲרֵי עֲדָתוֹ הָיָה זְרוֹעַ

וּלְהָפִיק זְמָמָם נָתַן יָדָיִם;

בְּעֵינֵי נֶשֶׁר-אֶל-עַמּוֹ הִשְׁגִּיחַ

כָּל-צְעָדֵיהֶם עַפְעַפָּיו בָּחָנוּ

לִמְצֹא עָוֹן וּלְהַלְשִׁין בַּסֵּתֶר;

אֶת לִבּוֹ הָרָע כַּצֹּר הִקְשִׁיחַ

וּבְזַעַם לְשׁוֹנוֹ הֵמִיט שֵׁאת וָשֶׁבֶר,

שִׁחֵת רַחֲמָיו אֱכֹל עַמּוֹ אֱכָל-לֶחֶם

וּבְאַפּוֹ טָרַף זָקֵן וָנַעַר

לֹא רִחַם גַּם-נָשִים וְיוֹנְקֵי שָׁדָיִם.


"לַבּוֹגֵד הַדָּבָר כִּי יִוָּדֵעַ

(חָשַׁב הָרַב בְּלִבּוֹ מַחֲשֶׁבֶת)

הֲלֹא חִישׁ יִגְלֶה אֹזֶן צֹרֲרֵינוּ

אָז אָבַדְנוּ, כֻּלָּנוּ אָבָדְנוּ!

אֵלִי, הַנּוֹתֵן לַיָּעֵף כֹּחַ!

בְּצֵל כְּנָפֶיךָ אָנָּא הַסְתִּירֵנִי

חַזְּקֵנִי וְאַמְּצֵנִי אַךְ הַפָּעַם!


"אֶל-פֹּעַל קֹדֶשׁ הִנְנִי לָגֶשֶׁת

מַעֲשֵׂה הַחֶסֶד יָדַי תְּבַצַּעְנָה –

מַשְׂאַת נֶפֶשׁ-דָּךְ קוֹרְאָה בִשְׁמֶךָ

וְעַל-עַמְּךָ בִּרְעָדָה מִתְרַפֶּקֶת…

וְאִם אַחֶרֶת, הָהּ אֵלִי, יָעַצְתָּ

וּלְמִכְשׁוֹל וּלְפוּקָה פָּעֳלִי יֶהִי,

אִם דְּמֵי עֲבָדֶיךָ לְךָ יֶעֱרָבוּ

עַל-מִזְבַּחֲךָ בְּרָצוֹן חַיַּי אַקְרִיבָה,

אַךְ עַל-נַפְשׁוֹת עֲדָתִי חוּסָה

בַּעֲוֹנִי אָנִי בַּל-יֵרְדוּ שָׁחַת!"


* * * * *

פְּנֵי אֶרֶץ וְחוּצוֹת עוֹד כִּסָּה חֹשֶׁךְ

וּבְחָפְזָה עָבְרוּ אֲנָשִׁים שְׁנַיִם

דֶּרֶךְ גְּדֶרֶת הַגִּינָה בַּתָּוֶךְ,

מִבֵּית-מִקְלַט מַרְטִין הִתְחַמָּקוּ

וְגוּפַת נָזִיר בַּכָּתֵף נָשָׂאוּ…


בַּחֲצַר-הַמָּוֶת דְּמָמָה וָשֶׁקֶט,

וּבִדְמִי הַלַּיִל יָהִים הָרוּחַ

כְּהוֹלֵךְ בֶּחָלִיל עַל-גַּלֵּי קָבֶר,

וּמֵעַל רָאשֵׁי הַבְּכָאִים שָׁמָּה

עָלִים נֹבְלִים יְנַתֵּק יַשְׁלִיךְ אָרְצָה

וִיפִיצֵם אֶל-כָּל-פֵּאָה וָעֵבֶר;

גַּם בְּקֶרֶן פִּנָּה בַּיַּרְכָתַיִם

שְׁנֵי חֹפְרֵי קְבָרִים עֹמְדִים שְׁחוֹחַ

עַל-קֶבֶר פָּתוּחַ יְדֵיהֶם כָּרוּ…


קֻפַּת חֶבְרָה קְדוֹשָׁה לֹא צִלֵּלָה

אֵין קוֹרֵא “צְדָקָה תַּצִּיל מִמָּוֶת”!

גַּם-קִינַת הַסּוֹפְדִים לֹא נִשְׁמָעָה

עֵת גְּוִיַּת מֵת הַבּוֹרָה הוּרָדָה…

רַק לְעִתִּים מִבֵּין עַנְנֵי עֲרָפֶל

זְעֵיר שָׁם זְעֵיר שָׁם בָּרוּחַ יְפֻזָּרוּ

מִשִּׂיא מְרוֹמֵי רְקִיעַ הַשָּׁמַיִם

כּוֹכָבִים מִמְּסִלּוֹתָם נִשְׁקָפוּ,

כְּעֵינֵי יְלָדִים נִדְהָמִים מִפַּחַד

עַל-חֲצַר-מָוֶת בַּחֲרָדָה הֵצִיצוּ;

גַּם-הָרוּחַ עוֹד טֶרֶם יָנוּחַ,

עָלִים נֹבְלִים מִנִשְׁמַת אַפֵּהוּ

מֵעַל-הַבְּכָאִים אַרְצָה יֻשְׁלָכוּ

וְאֶל-כָּל פֵּאָה וְעֵבֶר יִנָּפֵצוּ…

שַׁעֲרֵי חֲצַר-מָוֶת בָּלִים מִזֹּקֶן

בְּלִי הַשְׁמָעוּת קוֹל כִּמְעַט הִגִּיפוּ,

וּמִן-הַמִּצְפָּה הָרַב הִגִּיחַ

וּבִשְׁפַל קוֹלֵהוּ נָתַן אֹמֶר:

אַחַי! פֹּעַל יֶדְכֶם קָדוֹשׁ גָּבוֹהַּ

וִיהִי נֹעַם יְיָ עֲלֵיכֶם סֶלָה!

אַף-גַּם-זֹאת הֲלֹא תֵדְעוּ תַאֲמִינוּ

כִּי אֲנִי לְבַדִּי בְּיוֹם אֵיר וָנֶכֶר…

אַךְ עוֹד טֶרֶם כִּלָּה אִמְרֵי פִיהוּ

הִתְחַלְּחַל פֶּתַע, חָרַד וַיָּנוּעַ,

כִּי קוֹל רַעַשׁ מִצְעֲדֵי-גֶבֶר

מִמּוּל הַגְּדֵרָה אָזְנָיו שָׁמָעוּ,

וּכְהַפְנוֹתוֹ שִׁכְמוֹ אֶל-הַהֵלֶךְ

קַל מְהֵרָה נָדַד צַלְמוֹ בִּן-רֶגַע,

גַּם קוֹל צַעֲדֵי אוֹנוֹ חִישׁ נֶחְבָּאוּ

וּכְצֵל עוֹבֵר עִקְּבוֹתָיו לֹא נוֹדָעוּ…



ג

צְוָחָה וּמִסְפֵּד, אֲנָחוֹת אַלְפַּיִם

רַב-שָׁאוֹן, קִינִים הֶגֶה וָנֶהִי

אֶת-בֵּית-הַכְּנֶסֶת כֻּלָּהּ מָלָאוּ;

בְּכִי זְקֵנִים שְׂבֵעֵי יָגוֹן וָרֹגֶז

יְלֵיל נָשִׁים שֹׁפְכוֹת דִּמְעָה כַּמַּיִם

נַהֲקַת בַּחוּרִים, רוּחָם נָבָקָה,

וְאֶנְקַת יְלָדִים נִבְהָלִים מִפָּחַד –

הִתְבּוֹלְלוּ וְהָיוּ לַאֲנָחָה אַחַת,

אֲנָחָה מְמֻשָּׁכָה עַד לְאֵין-קֵצֶה…

אֲנָחוֹת כָּאֵלֶה לְמַכְבִּיר יָצָאוּ

מִלֵּב עַם-עוֹלָם יֻלַּד לַצֶּלַע,

וְהֵד קוֹלָן עוֹד הַיּוֹם יִשָּׁמֵעַ

מִתּוֹךְ עֲבִי סִבְכֵי-חֹרֶשׁ וָיַעַר

שָׁמָּה מִדֹּלְקָיו בִּקֶּשׁ-לוֹ סֵתֶר,

מִבָּתֵּי כְלָאִים בְּתַחְתִּיּוֹת אֶרֶץ

בָּהֶם הֻכָּה מַכּוֹת חַדְרֵי-בָטֶן;

מֵעַל-סַלְעֵי מָגוֹר וְהָרֵי נֶשֶׁף

רַגְלָיו הַיְחֵפוֹת שָׁם הִתְנַגָּפוּ,

מִכָּל הַמְּקוֹמוֹת עֲלֵיהֶם דָּרָכָה

כַּף רֶגֶל נִדָּח מִדְּחִי אֶל-דֶּחִי

וּבְדִמְעוֹתָיו הִשְׁקָה עֲפַר הָאָרֶץ…


אַךְ מַה-זֶּה וְעַל-מַה-זְֶּה יִבְכּוּ עַתָּה

זַעֲקַת-שֶׁבֶר כָּל-הָעָם יְעֹעֵרוּ?

הַאִם קָרְאוּ לְצוֹם לְמִסְפֵּד וָבֶכִי

לְתַנּוֹת אֶל-יוֹם עִיר קָדְשָׁם חָרֵבָה

וְתִפְאֶרֶת עֻזָּם לְשַׁמָּה הָיָתָה?

אוֹ יוֹם זֶה לָהֶם יוֹם חַג וַעֲצֶרֶת

מוֹעֵד שִׂמְחָה מִיָּמִים יָמִימָה,

בְּהִתְאַסֵּף עָם בֵּית יְיָ בְּרָגֶשׁ.

בְּשִׁירֵי דָוִד נְעִים זְמִרוֹת וָמֶלֶךְ

עַל-חַסְדֵי-אֵל אֶת-הַהַלֵּל יִקְרָאוּ

עֱזוּזוֹ וְנִפְלְאוֹתָיו בַּשִּׁיר יַזְכִּירוּ?

ּאָז לֹא קוֹל עֲנוֹת גְּבוּרָה יָרִימוּ,

אַךְ נְכָאִים יֶהְגּוּ, בִּבְכִי יָרֹנּוּ:

"מִן-הַמֵּצַר קָרָאתִי יָהּ

עָנָנִי בַּמֶּרְחַבְיָה!"

וּכְהָמֵר עַל הבְּכוֹר יוֹדְעֵי אֵבֶל

בִּבְכִי תַמְרוּרִים כֻּלָּם יָרִיעוּ:

"יַסֹּר יִסְּרַנִּי יָהּ

וְלַמָּוֶת לֹא נְתָנָנִי!"

וּלְקוֹל נְגִינָתָם לְבָבוֹת יִמַּסּוּ

כְּהִמֵּס דּוֹנַג מִפְּנֵי הַשַּׁלְהָבֶת…


הֵן לְפָנִים בִּימֵי אוֹרָה כָּאֵלֶּה

הָרַב הַמּוֹרֶה לִצְדָקָה וָקשֶׁט

מִמְּרוֹם הַבָּמָה דְּבָרוֹ הִבִּיעַ

וְעַל-עַם-מַרְעִיתוֹ מַקְשִׁיבֵי לֶקַח

הִטִּיף מִלִּין כִּרְבִיבִים עֲלֵי-עֵשֶׂב;

בְּחָכְמָה וָדַעַת וּאֱמוּנַת אֹמֶן

בְּשֵׂכֶל טוֹב וּבְשִׂפְתֵי חֵן וָנֹעַם

הִזִּיל כַּטַּל אֲמָרוֹת מְשִׁיבוֹת נָפֶשׁ;

פָּנָיו בְּאוֹר הַדְרַת-קֹדֶשׁ הוֹפִיעוּ

עֵינָיו הִבְרִיקוּ כְּסַפִּיר לָטֹהַר

וּזְרוֹעוֹ הֵנִיף בְּשׁוּבָה וָנָחַת.

וְלָמָּה-זֶּה עַתָּה פָּנָיו חֳמַרְמָרוּ

וְכִפְנֵי הָרְפָאִים לִמְאֹד חָוָרוּ,

עֵינָיו מְזָרוֹת דְּאָגָה וָפָחַד

שְׂפָתָיו דֹּלְקִים, יָחוֹגּוּ וְיָנוּעוּ

וְכַפָּיו הַצְּנֻמוֹת פְּרוּשׂוֹת לְמָעְלָה:

פַּעַם כִּמְבַשְּׂרוֹת צָבָא רָב וְחֶרֶב

וּנְטוּיוֹת מוּל צָר בְּעֶבְרָה וָזָעַם,

וּפַעַם כִּנְשֻׂאוֹת לְישְׁבִי שָׁמַיִם

כִּי יִשְׁלַח לְעַמּוֹ עֶזְרוֹ מִקֹּדֶשׁ?

בְּיוֹם זֶה רַב-הָעֵדָה וּנְשִׂיאֶיהָ

קָרְאוּ עַמָּם אֶל-מִסְפֵּד וָאֵבֶל

וְאֶל-מִקְדָּשׁ מְעַט כַּצִּפּוֹר חָרָדוּ

לִרְצוֹת בֵּית-יְיָ, לְחוֹנֵן אֲבָנֵיהוּ

בְּטֶרֶם יִפָּרְדוּ מֵעָלָיו לָנֶצַח

וְדֶרֶךְ לֹא יָשׁוּבוּ-עוֹד יַהֲלֹכוּ…


אוֹיְבָם הַבּוֹגֵד, מְלָשְׁנִי בַסֵּתֶר

גָּלָה סוֹד, הָלַךְ רָכִיל בְּעַמֵּיהוּ,

אֶת-שׁוֹפְטֵי-הַדָּת, לִבָּם לֵב אֶבֶן,

יָבִיא אֶל-הַקֶּבֶר שָׁם בַּחֲצַר-מָוֶת

עַל-יַד הַגְּדֵרָה בַּיַּרְכָתָיִם –

וּמָחָר נָכוֹן לָמוֹ יוֹם חֹשֶׁךְ

בְּכַבְלֵי בַרְזֶל אֶת יְדֵיהֶם יֶאֱסֹרוּ

לִנְחֻשְׁתַּיִם רַגְלֵיהֶם יֻגָּשׁוּ

וְאֶל-בֵּית מִקְלַט מַרְטִין יִסָּחֵבוּ,

בְּחַדְרֵי-אֹפֶל מִּתַּחַת לַבַּיִת

שֹׁפְטֵי הָאֱמוּנָה יַעֲשׂוּ בִקֹּרֶת

לִפְנֵי כִסְאָם שָׁם לְמִשְׁפָּט יַעֲמֹדוּ;

שָׁם עֲוֹנָם עֲלֵיהֶם בְּאַף יָמִיטוּ

בִּכְלֵי-מָוֶת עַל-נַפְשָׁם יָגוֹדּוּ

יְפַצְּחוּ עַצְמֵיהֶם וְעוֹרָם יַפְשִׁיטוּ..




ד

כָּל-הָעָם סָבִיב נֶאֱלַם דּוּמִיָּה;

אַךְ לְעִתִּים הִתְפָּרְצָה אַנְחָתָה

מִלֵּב נוֹאָשִׁים שׂפְקֵי כַפַּיִם

עַל-הֲמוֹן מֵעֵיהֶם לֹא הִבְלִיגוּ

וְסַעֲרַת רוּחָם עֲצוֹר לֹא יָכֹלוּ;

אֶל-הַבָּמָה עֵינֵיהֶם נָשָׂאוּ

וְאֶל-דִּבְרֵי מוֹרָם בִּרְתֵת הִקְשִׁבוּ;

בְּקוֹל אַדִּיר וְחָזָק שְׂפָתָיו צָלָלוּ:

"רַב לָכֶם אַחַי אֲחֻזֵי יִרְאָה וָרָעַד,

כְּמוּגֵי-לֵב הַנּוֹאָשִׁים מֵעֵזֶר

וּכְאֹבְדֵי עֵצוֹת הֶבֶל תִּפְחָדוּ!

אֵל טוֹב וּמֵטִיב חֶסֶד מְשָׁכַנִי

וּבְאַחֲרִית יְמֵי חֶלְדִּי פֹּה חַנַּנִי

עֲשׂוֹת מִפְעָל קָדוֹשׁ, נַעֲלֶה וְרַב-עֶרֶךְ

וּמַה-מְּאֻשָּׁר אָנֹכִי וְצוֹלֵחַ

כִּי לַעֲשׂוֹתוֹ יָדִי מָצְאָה חָיִל.

וְאַתֶּם, אַחַי, אַל-נָא תֵּעָצֵבוּ

אַל-נָא תוֹסִיפוּ מָרֵר בַּבֶּכִי

הֵן מֵרֵאשִׁית יָדַעְתִּי יָדוֹעַ

כִּי חַתְחַתִּים עַל-דֶּרֶךְ אֵלֵכָה

וְכִי נָכוֹן אָנֹכִי לְמוֹעֲדֵי רָגֶל;

יָדַעְתִּי כִּי עֵין בּוֹגֵד צוֹפִיָּה,

הוּא יִשְׁמֹר עֲקֵבַי יוֹמָם וָלָיְלָה

וּכְדֶבֶר הַוּוֹת יַהֲלֹךְ בָּאֹפֶל;

כִּי פְרִי מַעֲלָלִי רַע וָמַר יֶהִי

וְאֶל-גֵּוִי הֶעָיֵף וְהַיָּגֵעַ

מְצָרֵי שְׁאוֹל וְחֶבְלֵי מָוֶת נָכוֹנוּ,

כָּל-אֵלֶּה מִמֶּנִּי לֹא נִכְחָדוּ…

אַךְ הַאִם הַתִּפְאֶרֶת וְהַנֵּצַח,

נַחֲלַת כָּל-מִפְעַל טוֹב וָצֶדֶק,

לֹא יִגְדְּלוּ עַל-אַחַת שֶׁבַע

מִדֵּי אֶת-רֹאשׁ פֹּעֲלוֹ יַכְתִּירוּ

לֹא בַעֲטֶרֶת גֵּאוּת עֲלֵי-זָיִת

כִּי אִם בְּנֵזֶר סַלּוֹנִים וָחוֹחַ,

וִידֵי עֹשֵׂה הַיָּשָׁר וְהַחֶסֶד

לֹא מָר-דְּרוֹר וְקִנְּמָן-בֶּשֶׂם תִּטֹּפְנָה,

אַךְ בִּדְמֵי פִצְעֵי לִבּוֹ תְּגֹאַלְנָה

וּבְדִמְעוֹת עֵינָיו רַחֵץ תְּרֻחַצְנָה?

אִישׁ אֲשֶׁר לִצְדָקָה זָרַע זֶרַע

וּבְקוֹל רִנָּה יִקְצֹר אֲלֻמּוֹתֶיהָ;

אִישׁ גְּדָל-הַחֶסֶד וְרַב הָעֲלִילִיָּה

אֲשֶׁר תְּהִלָּתוֹ יַגִּידוּ בַשָּׁעַר,

בְּמַקְהֵלוֹת-עָם לִקְרָאתוֹ יָרִיעוּ

וּבְפִיּוֹת רַבִּים שֵׁם כְּבוֹדוֹ יְבֹרָךְ,

אִישׁ אֲשֶׁר כָּזֶה לֹא טָעַם רַק שֶׁמֶץ

מִכּוֹס יְיָ, כּוֹס עֵדֶן וָנַחַת

עֲרוּכָה מֵאָז לִקְדוֹשִׁים בָּאָרֶץ,

וְאַךְ בִּשְׂפָתֶיהָ לְשׁוֹנוֹ נָגָעָה;

כִּי רַק בְּתַחְתִּיתָהּ מֶתֶק הָאֹשֶׁר

וּבְקֻבַּעְתָּהּ צְפוּנָה נְעִימַת נֶצַח,

אוֹתָהּ יִשְׁתֶּה יִמְצֶה לְבַד הַגֶּבֶר

שֶׁבְּעַד פֹּעַל קֹדֶשׁ יָדוֹ כּוֹנָנָה

כָּל-יַד עָמָל וּתְלָאָה תְּבוֹאֶנּוּ,

מְחֹלָל וּמְדֻכָּא בְּאַכְזְרִיּוּת רֶצַח

נִשְׁפָּט בַּחֶרֶב וּבְאֵשׁ אֹכֶלֶת

יִשָּׂא אַף-יִסְבֹּל כְּרָחֵל נֶאֱלָמָה;

אַשְׁרֵי הָאִישׁ שֶׁכָּכָה גוֹרָלֵהוּ

מִכּוֹס אֱלֹהָיו הוּא יִשְׂבַּע דֶּשֶׁן

וַעֲדָנִים יִרְוֶה מֵאשֶׁר שָׁמָיִם, –

לוֹ לְבַדּוֹ הַנֵּצַח וְהַתִּפְאָרֶת!


"זִכְרוּ-נָא אַחַי אַלְפֵי קְדֹשֵׁינוּ

לְהֶרֶג וּלְטֶבַח נַפְשָׁם מָסָרוּ,

לְמַרְאֵה מָוֶת פְּנֵיהֶם לֹא נָפָלוּ

וּמִמְּצוּקֵי שַׁחַת לֹא נָמֹגוּ;

הֶאָח, מַה-קָּרַן אָז עוֹר פָּנֵימוֹ

וּבְזִיז יְקָר אֵיךְ הִתְנוֹסֵס הַמֶּצַח

עֵת מְעַנֵּיהֶם בְּיַד-חָמָס וָרֶשַׁע

אֶת-בְּשָׂרָם דָּשׁוּ כְּהִדּוּשׁ הַתֶּבֶן

טֶרֶם תַּעֲבֹר חַיָּתָם בַּשָּׁלַח

אוֹ עַל-מְדוּרוֹת אֵשׁ טֶרֶם יֻשְׁלָכוּ.

הָהּ! מַעַרְכוֹת אֵשׁ וְעֵצִים הָאֵלֶּה

קָדְשׁוּ בְעֵינֵי יְיָ גַּם-יָקָרוּ

מִמִּזְבְּחוֹת הֵיכָל בְּאֶרֶץ הַצֶּבִי,

דְּמֵי הַחֲסִידִים כַּמַּיִם נִגָּרוּ

נִבְחֲרוּ לוֹ מֵרִבְבוֹת נַחֲלֵי-שָׁמֶן

דִּמְעָתָם וּתְחִנָּתָם יִנְעָמוּ

מִשֶּׁלֶם מְרִיאִים, מִנְחָה וָנֶסֶךְ

וּמִקְּטֹרֶת מוֹר, נָטָף וּשְׁחֵלֶת

בִּכְלִי-טָהוֹר אֶל-הַמִּקְדָּשׁ הוּבָאוּ!


"זֶה כַּמֶּה בְּרֹב שַׂרְעַפַּי הָגִיתִי

בַּעֲלִילוֹת-קֹדֶשׁ אֵלֶּה בִּינוֹתִי,

וְעַד-חֵקֶר תַּכְלִיתָם כִּי-בָאתִי

בִּקְדוֹשֵׁי-אֵל מְאֹד קַנֹּא קִנֵּאתִי,

לְיוֹצֵר תֵּבֵל וּמְלֹאָהּ הֶעְתַּרְתִּי

וְאֶל-יוֹשְׁבִי מָרוֹם שָׁפַכְתִּי שִׂיחַ

כִּי יַפִּיל גַּם-לִי בַּנְּעִמִים חֶבֶל

וּמִנֹּפֶת צוּף כּוֹסוֹ יַשְׁקֵנִי…

לְקוֹל תַּחֲנוּנַי הִנֵּה שָׁמַע עָתָּה,

מָחָר אֶמְצָא כּוֹס יְיָ מָלֵא מֶסֶךְ

וְלִשְׁתּוֹתוֹ כֻּלּוֹ נָכוֹן הִנֵּנִי!

אָז תָּבֹא נַפְשִׁי בְּסוֹד אַנְשֵׁי-קֹדֶשׁ

בִּקְהָלָם תֵּחַד כְּבוֹדִי גַם-אָנִי.

כַּדְּרוֹר לָעוּף אֶסַּק שִׂיא-שַׁחַק,

בִּמְעוֹן רוּחוֹת טְהֹרוֹת אֶרְוֶה נָּחַת,

בִּנְוֵה שְׁלוֹם-אֱמֶת וְאַהֲבָה נִצָּחַת!




ה

עֵינֵי הָרַב כִּבְרָקִים הִזְהִירוּ

וּפָנָיו הַנִּזְעָמִים צָהָלוּ,

מֵרֶגַע לְרֶגַע לֵאוֹר הוֹסִיפוּ

עַד-אִם-כִּלָּה דְבָרוֹ וַיֵּרָגַע.

מֵעֵינָיו מְאַזְּרֵי זִיקוֹת וָרֶשֶׁף

רְסִיסֵי טַל דְּמָעוֹת לְמַכְבִּיר זֹרָמוּ,

עַל-קִמְטֵי לְחָיָיו הָרָזוֹת נָפָלוּ,

עַל-קְוֻצּוֹת זְקָנוֹ לָבָן כַּכָּסֶף

יוֹרֵד עַל-פִּי מִדּוֹתָיו בַּהֶדֶר

וְעַל-מְעִילוֹ הַשָּׁחוֹר כְּלִיל-מֶשִׁי;

וּלְנֹגַהּ קַרְנֵי שֶׁמֶשׁ עַרְבָּיִם

כְּּחוּטֵי זָהָב טָהוֹר אָז נָהָרוּ,

מִבַּעַד לְחַלּוֹנֵי בֵּית-הַכְּנֶסֶת,

דְּמָעוֹת אֵלֶה מִפְּנִינִים אָדָמוּ…

רִגְעֵי הַשְׁקֵט מַהֵר הָלְכוּ לָמוֹ

וַיּוֹסֶף הָרַב קוֹלוֹ הַשְׁמִיעַ

וּשְׁתֵּי יָדָיו פְּרוּשׂוֹת הַשָּׁמָיְמָה:

לֹא עַל-יְסוּרֵי מָוֶת חִישׁ יְבֹאֻנִי

תִּדְאַב נַפְשִׁי, עֵינִי יֹרְדָה מָיִם;

אֵלִי! הֲלֹא כֵן מִשְׁפָּטְךָ חָרָצְתָּ!

מַחְשְׁבוֹת לִבִּי כַּסֵּפֶר נָגֹלּוּ,

מִמְּעוֹן קָדְשְׁךָ עֵינֶיךָ תֶחֱזֶינָה

כִּי שָׂשׂ אֲנִי עַל-מִשְׁפְּטֵי צִדְקֶךָ

וּכְשִׂמְחַת צָמֵא עַל-מַבּוּעֵי מַיִם

לִקְרַאת יְסוּרַי אָשִׂישָׂה אָגִילָה,

קַח נַפְשִׁי לְעוֹלָה, הִנְנִי הִנֵּנִי!

אַךְ מַדּוּעַ זֶה עַל-מִזְבַּח קָדְשֶׁךָ

רַק דָּמִי לְבַדִּי לֹא יִשָּׁפֵכוּ?

הֵן אֲנִי לְבַדִּי בְּחֵפֶץ לֵב וָנֶפֶשׁ

הַפְּעֻלָּה הַכְּבֵדָה עֲשׂוֹת אִוִּיתִי

אֶפֶס הֶהָלְכוּא עִמִּי מָה הֵמָּה

כִּי יִשְׂאוּ עֲוֹנִי בְּיוֹם עֶבְרָה וְזָעַם?

הֲלֹא כְגַלֵּי-יָם נִדָּפִים בָּרוּחַ

כָּכָה בְּמִצְוֹתַי עֲלֵיהֶם חָזָקָה

אֶת-חֶפְצִי עָשׂוּ, עֲצָתִי הִשְׁלִימוּ;

אֲנִי זְרוֹעֹתָם חִזַּקְתִּי אִמַּצְתִּי,

וּבִפְקֻדָּתִי הֵן אַחֲרַי הָלָכוּ

כְּלֶכֶת הַצֹּאן בְּיוֹם-סַעַר וָדֶלֶף

אַחֲרֵי רוֹעָם הַמְנַהֲלָם בַּשָּׁבֶט…

וּבְכֵן הַעֲבֶר-נָא אֱלֹהֵי הַחֶסֶד

מִנַּפְשׁוֹת נְקִיִּים אָסוֹן וָפֶגַע,

מִשֹּׁאַת פִּתְאֹם תָּסֵךְ עָלֵימוֹ;

רַק עָלַי לְבַדִּי תִּנְחַת יָדֶךָ,

מְמוֹתֵי תַחֲלֻאִים הָבָה אָמוּתָה,

אָכֵן אֵלֶּה הַצֹּאן מֶה חָטָאוּ

וּבְלֹא-עֵת לָמָּה-זֶּה יֵרְדוּ שְׁאוֹלָה.

בְּיַרְכְּתֵי קִבְרִי לֹא אֶשְׁכֹּן בֶּטַח

תַּחְתַּי אֶרְגָּז, לֹא אֶמְצָא מָנוֹחַ –

מִקּוֹל רְצוּצִים בִּכְלֵי נֶפֶץ יֻכַּתּוּ

מֵאֶנְקַת עֲשׁוּקִים נְתוּנִים לַקָרֶץ,

מִקִּלְלוֹת אַלְמָנוֹת וִיתוֹמִים יַחַד

אֶל-תּוֹךְ מַעֲמַקֵּי בוֹרִי תַּחְדֹּרְנָה,

עָלַי וְעַל-רֹאשִׁי כֻּלָן תָּחוֹלְנָה

אוֹיָה לִי, אוֹיָה, מַת-זֹּאת עָשִׂיתִי?

מָה רַבָּה הָרָעָה אֲשֶׁר הֲסִבֹּתִי

אָנֹכִי זָקֵן שְׂבַע יָמִים וָרֹגֶז

עַל-עֲדָתִי הַנִּלְאָה וְהַנֶּהֱלָאָה…

אָנֹכִי הִרְבֵּיתִי אֶת-חֲלָלֶיהָ,

הָהּ גָּדוֹל כַּיָּם הַשֹּׁד וְהַשֶּׁבֶר!



“הַרְפּוּ נָא!” נִשְׁמַע פִּתְאֹם לְפֶתַע

קוֹל קוֹרֵא בְכֹחַ מִפֶּתַח הַדֶּלֶת

"הַרְפּוּ-נָא! מִנְעוּ קוֹלְכֶם מִבֶּכִי

וּדְמָעוֹת שָׁלִישׁ מְחוּ-נָא מִלֶּחִי,

לָכֶם לְמוֹשִׁיעַ שַׁדַּי שְׁלָחָנִי,

אֶת-כֻּלְּכֶם מִמָּוֶת אַצִּיל אָנִי!"




ו

כְּרֶגַע כָּל-הָעָם פְּנֵיהֶם הָפָכוּ

אֶל-הַפֶּתַח עֵינֵיהֶם נָשָׂאוּ –

וְקוֹל פְּחָדִים נִשְׁמַע מִכָּל-עֵבֶר

גַּם-הֲמֻלָּה גְדוֹלָה סָבִיב נִהְיָתָה:

בֵּין טוּרֵי זְקֵנִים וּזְקֵנוֹת בַּתָּוֶךְ

אֶל-הַבָּמָה נְתִיבָה לָהּ פִּלֵּסָה

עַלְמָה צְעִירָה לְיָמִים וִיפַת-תֹּאַר.

אַךְ אָיוֹם וְנוֹרָא הָיָה מַרְאֶהָ!

פָּנֶיהָ הַיָּפִים כְּשִׂיד חָוָרוּ,

וְשְׁחוֹרִים כָּעוֹרֵב תַּלְתַּלֶּיהָ

בִּבְלִי-סְדָרִים וּפְרוּעִים יָרָדוּ

עַל-מַפְּלֵי בִגְדָּהּ מִּצֶּמֶר צַחַר

שֶׁעַל-גְּוִיָּתָהּ בָּהוֹד חֻטָּבָה;

שְׂפָתֶיהָ פְּתוּחוֹת כְּשֶׁלֶג הִלְבִּינוּ

וּמֵעֵינֶיהָ שְׁחֹרוֹת כְּפֶחָם וָפִיחַ

כִּידוֹדֵי אֵשׁ פְּלָדוֹת הִתְמַלָּטוּ,

בְּיָדֶיהָ לָחֲצָה אֶת-שְׁנֵי שָׁדֶיהָ

וּבַמַּעֲלוֹת עָלְתָה הַבָּמָתָה;

כְּאִישׁ אֲשֶׁר הַקַּדַּחַת אֲחָזַתּוּ

כָּל-יְצֻרֵי גֵוָהּ מְאֹד הִתְחַלְּחָלוּ

וּבִמְרִירוּת-לֵב רַבּוֹת נֶאֱנָחָה…

כְּשׁוּב לִדְמָמָה סַעֲרַת הָרָגֶשׁ

מְעַט מְעַט הִתְאוֹשְׁשָׁה הִתְעוֹדָדָה.

מֵעַל-חָזֶהָ רֹאשָׁהּ הֵרִימָה

וּכְאוֹב מֵאֶרֶץ עָנְתָה אֲמָרֶיהָ:

"הִנֵּה בָאתִי הֲלוֹם, רַבִּי שְׁמָעֵנִי…

לְבַעֲבוּר הַצִּיל אֶתְכֶם הֵנָּה בָאתִי.

בַּת הַבּוֹגֵד לִפְנֵיכֶם נִצָּבֶת!

אָמְנָם כֵּן אָבִי – שְׁמוֹ לֹא אַזְכִּירָה.

הָהּ! אָרוּר הַיּוֹם שִׁבְעִים וָשֶׁבַע

בּוֹ עֹכֵר עַמּוֹ בְּשֵׁם אָב קָרָאתִי…

אֶל-טְפָחוֹת חֶלְדִּי רְאוּ נָא וְהַבִּיטוּ,

אַךְ בַּת שְׁבַע עֶשְׂרֵה שָׁנָה אָנֹכִי

וּכְבָר אֶל-בֵּית כֶּלֶא וּמַהְפֶּכֶת

כָּל-הַתֵּבֵל כֻּלָּהּ לִי נֶהְפָּכָה…

מִלְּפָנִים הַרְחֵק מִזֶּה יָשָׁבְנוּ;

אִמִּי בְּבֹא יוֹמָהּ נַפְשָׁהּ נָפָחָה…

מִתִּגְרַת יַד הָאַכְזָר כָּלָתָה

בַּאֲבִיב יָמֶיהָ מָצְאָה לָּהּ קָבֶר,

אֶת-מְקוֹמָהּ יָרְשָׁה אִשָּׁה אַחֶרֶת –

וְהָאַכְזָר הָיָה לְאִישׁ הַבְּלִיָּעַל…

אֶת-שְׁאֵרִית לִבּוֹ אַךְ הִיא גָזָלָה

וַתָּשֶׂם תַּחְתֶּיהָ פֶּתֶן חֵרֵשׁ

יַפְרִשׂ כָּל-הַיּוֹם רַק רֹאשׁ וָרָעַל…

לְמִן-הָעֵת הַהִיא גַּם-שִׁמְשִׁי בָּאָה,

כְּנָכְרִיָּה אֲנִי וּכְזָרָה נֶחְשָׁבְתִּי,

אֵין דּוֹרֵשׁ לִי, מֵאִיר לִי פָּנֵיהוּ

אֵין חוֹנֵן אֹתִי, מֹשֵׁךְ אֵלַי חָסֶד

וְכִיתוֹמָה עֲזוּבָה טוֹבָה נָשִׁיתִי…

הָהּ! אֵין מִסְפָּר לְמַעֲשֵׂי רֶשַׁע כֶּסֶל

וּנְבָלוֹת גְּדוֹלוֹת בְּעֵינַי רָאִיתִי…

גַּם-הַיּוֹם עַל-שֻׁלְחַן הַצָּהֳרָיִם

שְׁנֵיהֶם יַחַד פִּיהֶם שְׂחוֹק מָלָאוּ

עֲלִיצֻתָם וְשִׂמְחָתָם מַה-גָּדֵלָה…

וּבְשָׁאטָם בְּנֶפֶשׁ מָחְאוּ כַפַּיִם

עַל-הַתָּפְתֶּה מֵאֶתְמוּל עָרָכוּ…

אֶל-קִרְבּוֹ אָז הֵרִיק גְּבִיעַ יַיִן,

כְּגִּבּוֹר מִתְרוֹנֵן שָׁב מִשְּׂדֵה-לָחֶם

יָשִׂישׂ עַל-הַשָּׁלָל וְהַמַּלְקוֹחַ…

רַבּוֹת רַבּוֹת בְּפָנָיו הִתְבּוֹנַנְתִּי

קַו-חֶמְלָה אֶחָד לִמְצֹא נִסִּיתִי,

אַךְ שָׁוְא עָמַלְתִּי, לָרִיק יָגַעְתִּי:

כָּל-אוֹתֹתָם יַגִּידוּ, לֹא יְנַכֵּרוּ

כִּי עֻפְּלָה נַפְשׁוֹ, כָּלִיל נִשְׁחָתָה

וּבְאוֹר אַהֲבָה לֹא תָאִיר עוֹד נֶצַח…

עִם-כִּבְדֵי עָוֹן נִלְאֵיתִי שֶׁבֶת,

אֶבְחַר מַחֲנַק מִדּוּר בְּאֹהֶל רֶשַׁע…

שְׂנֵאתִיו מְאַסְתִּיו, גַּם-חַיַּי מָאַסְתִּי.

אַחַת הִיא לִי… זֶה כְּבָר בְּחַיַּי קַצְתִּי,

עַל-כֵּן חָשַׁבְתִּי, זָמַמְתִּי, חָרָצְתִּי…

גַּשׁ-נָא אֵלַי, רַבִּי, קְרַב-נָא הֵנָּה…

הַטֵּה אָזְנְךָ… הַאֲזִינָה אִמְרָתִי:

שְׁמַעְתִּיו מִתְלַחֵשׁ עִם-הַמִּרְשָׁעַת…

מָחָר בַּבֹּקֶר! כַּעֲלוֹת הַשַּׁחַר

שֹׁפְטִים עָרִיצִים חֹצֵץ יָבֹאוּ

בֵּית מוֹעֵד לְכָל-חַי כֻּלָּם יֵלֵכוּ

שָׁמָּה יַחְפֹּרוּ… אוֹתוֹ יִמְצָאוּ…

הָיוֹ לֹא תִהְיֶה! גָּמַרְתִּי אֹמֶר…

לֹא כֵן, לֹא אוֹתוֹ יִמְצְאוּ בַקָּבֶר!

בִּינוּ-נָא אֵיךְ עַצְמוֹתַי רָחָפוּ…

הֵן רַק בִּכְבֵדוּת רוּחַ אֶשְׁאָפָה…

רַגְלַי הִתְעַוְּתוּ, לִנְפֹּל נִטָּיוּ…

הִנֵּה זֶה בָּא… סָר אֵלַי הַמָּוֶת…

כְּלַבַּת-אֵשׁ יִזַּל בְּעֹרְקַי הָרָעַל,

בְּעֶצֶם יָדַי הֲכִינוֹתִי… שָׁתִיתִי…

מִשְׁלַחְתּוֹ יַעֲשֶׂה, יַשְׁלִים פָּעֳלֵהוּ…

עוֹד מְעַט מִזְעָר אַרְצָה יַפִּילֵנִי…

וּגְוִיָּתִי… נְבֵלָתִי… כָּאֶבֶן

פֹּה תִּשְׁכַּב יוֹמָם, קָרָה כַקֶרַח…

אֲזַי קָחוּנִי-נָא… הַצְפִּינוּנִי…

וּבְהַגִּיעַ תֹּר… כַּחֲצוֹת הַלַּיְלָה

שָׂאוּנִי וַהֲטִלֻנִי… הַבּוֹרָה…

תַּחַת הַנָּזִיר… שָׁמָּה יָנוּחַ…

וּבְבֹא הַשֹּׁפְטִים. רֵיקָם יָשׁוּבוּ…

הָהּ! בֹּעֵר הָרַעַל כְּאֵשׁ יוֹקֶדֶת…

קַרְסֻלַּי יִמְעָדוּ… רַבִּי, תָּמְכֵנִי…

אֵלִי! הִנְנִי מֵתָה… מֵתָה הִנֵּנִי…"

כָּל-הָעוֹמְדִים נִבְהָלוּ נִדְהָמוּ

וְשִׂפְתֵי הַנַּעֲרָה לָעַד נֶאֱלָמוּ…



ז

בְּעִיר פְּרָג, קִרְיָה עַתִּיקָה רַבָּתִי

עַל-חֲצַר-מָוֶת, עָלָיו אָבַד כָּלַח,

לְגַל הַקֶּבֶר הַלָּז עוֹד יֵשׁ זֵכֶר

וְאַנְשֵׁי הַמָּקוֹם מְקוֹמוֹ יַכִּירוּ:

עָלָיו לֹא תָרִים רֹאשׁ מַצְבַת-אָבֶן

אַלּוֹן בָּכוּת שָׁמָּה לֹא יַצְמִיחַ

הַשּׁוֹשַׁנִּים לֹא תִתְּנֶינָה רֵיחַ

גַּם-יְרַק דֶּשֶׁא עֵשֶׂב לֹא יְכַסֶּנּוּ;

עַל-גַּל זֶה עָלָה שָׁמִיר וָשָׁיִת

חֲרֻלִּים וְקִמְשֹׁנִים עָלָיו יְסֻפָּחוּ, –

בָּאָרֶץ – זֹאת נַחֲלַת קְדוֹשֵׁי עַמֵּנוּ

וּפֹה – זֶה חֶלְקֵנוּ וְגוֹרָלֵנוּ…


_________


בְּאָזְנַי סִפֵּר כַּדְּבָרִים הָאֵלֶּה

אִישׁ שֵׂיבָה נִכְבָּד, יָמִים שָׂבֵעַ

כַּאֲשֶׁר גַּם-לוֹ אֲבוֹתָיו סִפֵּרוּ;

אַךְ אִם כָּל-אֵלֶּה נִקְרוּ וַיֶּאֱתָיוּ

אוֹ כֶּחָזוֹן מִלֵּב הֹרוּ וְהֹגוּ,

אוֹ גַּם מַעֲשֶׂה וְחָזוֹן הִתְבּוֹלָלוּ

וּבְשֶׁכְּבָר הַיָּמִים הִתְלַכָּדוּ

יַחְדָּו הָיוּ תַמִּים, לֹא יִתְפָּרָדוּ –

מִי חָכָם יָבִין זֹאת, יַגִּיד פֶּשֶׁר;

אֶפֶס אֶת-זֹאת נַפְשִׁי מְאֹד יוֹדָעַת

כִּי כָל-סִפּוּר בַּדִּים אוֹ חֲזוֹן-רוּחַ

עַל-כּוֹס בַּת-עַמִּי, כּוֹס דִּמְעוֹתֶיהָ

לֹא יוּכְלוּ הוֹסִיף גַּם-נֶטֶף אֶחָד:

עַד-בְּלִי-חֹק הַכּוֹס כְּבָר מָלָאָה,

וּמִדֵּי תְרַחֵף רוּחַ עָלֶיהָ

שֹׁטְפוֹת דְּמָעוֹת מֵעַל לִגְדוֹתֶיהָ!
