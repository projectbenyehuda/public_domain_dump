חֶטְאוֹ שֶׁל צַדִּיק

רבִּי יְשַׁעְיָה הַלֵּוִי הוּרְבִיץ, הוּא הַשְּׁלָ"ה הַקָּדוֹשׁ, מְחַבֵּר הַסֵּפֶר שְׁנֵי לוּחוֹת הַבְּרִית, גָמַר בְּלִבּוֹ לַעֲלוֹת לָאָרֶץ הַקְּדוֹשָׁה, כְּדֵי שֶׁיִּהְיֶה קָרוֹב לַשְּׁכִינָה וְיִלְמַד תּוֹרָה מִתּוֹךְ קְדֻשָּׁה וְטָהֳרָה.

וּבַיָּמִים הָהֵם אֵין עוֹד מְסִלַּת בַּרְזֶל בָּאָרֶץ וְהָיוּ נוֹסְעִים בְּעֶגְלַת סוּסִים עַד סְטַמְבּוּל, הִיא בִּירַת הַתֻּרְכִּים, וּמִסְּטַמְבּוּל הָיוּ הוֹלְכִים בָּאֳנִיָּה עַד יָפוֹ, הִיא שַׁעַר הָאָרֶץ. שָׁבוּעוֹת וִירָחִים אָרְכָה הַנְּסִיעָה וְהַרְבֵּה תְּלָאוֹת וְסַכָּנוֹת הָיוּ כְּרוּכוֹת בָּהּ, וְרַק אֲנָשִׁים גִדְלֵי אֱמוּנָה וְאַמִּיצֵי לֵב קִבְּלוּ עַל עַצְמָם עִנּוּיֵי הַדֶּרֶךְ הָאֵלֶּה כְּדֵי לָבוֹא לְאֶרֶץ־יִשְׂרָאֵל.

בְּמֶשֶׁךְ עֶשֶׂר שָׁנִים וָמַעְלָה קָבַץ הַשְּׁלָ"ה כֶּסֶף לְהוֹצָאוֹת הַדֶּרֶךְ וּלְמִחְיָתוֹ בָּאָרֶץ, שֶׁאָז לֹא הָיְתָה יְכֹלֶת לְהִשְׂתַּכֵּר בְּאֶרֶץ יִשְׂרָאֵל אֲפִלּוּ פְּרוּטָה אַחַת, וְגַם יְשִׁיבָה אָמַר לְיַסֵּד בִּירוּשָׁלַיִם, שֶׁעֲנִיִּים יִלְמְדוּ בָּהּ תּוֹרָה וְקַבָּלָה.

כְּשֶׁהִגִּיעָה שְׁעַת הַנְּסִיעָה קָנָה לוֹ סוּס וַעֲגָלָה קְטַנָּה, וּלְעֶגְלוֹן לָקַח לוֹ אֶת שׁוֹחֵט עִירוֹ, אִישׁ כָּשֵׁר וְיָשָׁר, אַלְמָן מֵאִשְׁתּוֹ וְגַם בָּנִים לֹא הָיוּ לוֹ, וְהָיָה לוֹ גַּם לִשְׁחֹט עוֹף בַּדֶּרֶךְ לִימֵי הַשַּׁבָּתוֹת וְהַחַגִּים. וּכְדֵי שֶׁיְּהֵא בָּטוּחַ מִשּׁוֹדְדִים בַּדֶּרֶךְ, עָשָׂה אֶת מוֹט הָעֲגָלָה נָבוּב, נָתַן בּוֹ אֶת כָּל הַכֶּסֶף וְסָתַם אֶת פִּי הַמּוֹט, שֶׁלֹּא יְהֵא נִכָּר כְּלָל.

הָיוּ שְׁנֵיהֶם נוֹסְעִים מֵעִיר לְעִיר וּמִקְּהִלָּה לִקְהִלָּה, וּבְכָל מָקוֹם שֶׁעָבְרוּ יָצְאוּ חֲשׁוּבֵי הָעֵדָה לְהַקְבִּיל פְּנֵיהֶם בְּכָבוֹד וּבְאַהֲבָה, כִּי שֵׁם הַשְּׁלָ"ה הַקָּדוֹשׁ הוֹלֵךְ לְפָנָיו. וְהָעֲשִׁירִים הָיוּ מְכַבְּדִים אוֹתָם מֵהוֹנָם בְּיָד רְחָבָה, וּבְכֵן הָיוּ מִתְפַּרְנְסִים בַּדֶּרֶךְ בְּשֹׁפִי, וְלֹא נָגְעוּ בַּמַּטְמוֹן הַטָּמוּן בְּמוֹט הָעֲגָלָה. וְאִישׁ זוּלָתָם לֹא יָדַע מַה גָּנוּז בַּמּוֹט.

וְהַשְּׁלָ"ה שָׂמַח שֶׁהַכֶּסֶף נִשְׁמַר בִּשְׁלֵמוּתוֹ, וְכֵן אָמַר לַשׁוֹחֵט בֶּן־לִוְיָתוֹ: אִם יְזַכֵּנוּ הַשֵּׁם וְנָבִיא אֶת כָּל כַּסְפֵּנוּ בְּשָׁלוֹם לָאָרֶץ הַקְּדוֹשָׁה, וְהָיָה לְכַלְכֵּל אוֹתָנוּ כָּל יְמֵי חַיֵּינוּ וְגַם אֶת הַיְשִׁיבָה שֶׁאֲיַסֵּד בְּעֶזְרַת הַשֵּׁם.

כְּשֶׁהִגִּיעוּ לִסְטַמְבּוּל לֹא הָיָה לָהֶם עוֹד צֹרֶךְ בַּסּוּס וּבָעֲגָלָה, שֶׁמִּכָּאן עֲלֵיהֶם לְהַפְלִיג בָּאֳנִיָּה. וְלִפְנֵי שֶׁהוֹצִיאָם לַשּׁוּק לִמְכִירָה פָּתַח הַשְּׁלָ"ה אֶת הַמּוֹט לְהוֹצִיא מִתּוֹכוֹ אֶת הַכֶּסֶף – וְהִנֵּה הַמּוֹט רֵיק, אֵין בּוֹ אֲפִלּוּ פְּרוּטָה אַחַת.

נִבְהַל הַגָּאוֹן מְאֹד, אָחַז בְּשַׁעֲרוֹת רֹאשׁוֹ וְצָוַח: אוֹי לִי, הַכֶּסֶף אֵינֶנּוּ וְאָנָה אֲנִי בָּא? אֵיךְ אַגִּיעַ לָאָרֶץ הַקְּדוֹשָׁה? אֲבָל אָשֵׁם אֲנִי, שֶׁלָּקַחְתִּי בַּדֶּרֶךְ מַתְּנַת בָּשָׂר וָדָם, בְּשָׁעָה שֶׁהָיָה לִי רַב… אוֹי לִי וַאֲבוֹי לִי, וַאֲנִי בְּעִיר נָכְרִיָּה!

כְּשֶׁנָּח מְעַט מִצַּעֲרוֹ הִתְחִיל מְהַרְהֵר: מִי יָכוֹל הָיָה לְהוֹצִיא אֶת הַכֶּסֶף מִתּוֹךְ הַמּוֹט, וְאִישׁ זוּלָתִי וְזוּלַת הַשּׁוֹחֵט לא יָדַע עַל אוֹדוֹת הַמַּטְמוֹן? הָיָה הוֹפֵךְ וּמְהַפֵּךְ בַּדָּבָר עַד שֶׁגָּמַר בְּלִבּוֹ: הַשּׁוֹחֵט הוּא שֶׁגָּנַב אֶת הַכֶּסֶף. אָמְנָם אָדָם כָּשֵׁר וְיָשָׁר הָיָה בְּעִירוֹ, אֲבָל תַּאֲוַת הַמָּמוֹן הֶעֱבִירַתּוּ עַל דַעְתּוֹ וְשָׁלַח יָדוֹ בַּמַּטְמוֹן.

קָרָא אֶת הַשּׁוֹחֵט לְחֶדְרוֹ בְּבֵית הַמָּלוֹן, סָגַר אֶת הַדֶּלֶת וְאָמַר לוֹ: הִתְוַדֵּה. אַתָּה לָקַחְתָּ הַכֶּסֶף מִתּוֹךְ הַמּוֹט. הֲשִׁיבֵהוּ וְלֹא אֲגַלֶּה קְלוֹנְךָ.

נִבְהַל הַשּׁוֹחֵט מִדִּבְרֵי הַגָּאוֹן וְנִשְׁבַּע בְּכָל קָדְשֵׁי יִשְׂרָאֵל, שֶׁאֵינוֹ יוֹדֵעַ מְאוּמָה מִן הַנַּעֲשֶׂה. וְלֹא הֶאֱמִין לוֹ הַשְּׁלָ"ה וְדִבֶּר עַל לִבּוֹ רַכּוֹת וְקָשׁוֹת, שֶׁיּוֹדֶה עַל עֲווֹנוֹ וְיַחֲזִיר הַמָּמוֹן.

וְהַשּׁוֹחֵט עוֹמֵד עַל נַפְשׁוֹ וּמוֹסִיף שְׁבוּעָה עַל שְׁבוּעָה, כִּי יָדָיו נְקִיּוֹת מִן הַחֵטְא הַזֶּה.

נִתְחַמְמוּ הָאֲנָשִׁים בְּרִיבָם, עַד שֶׁהִתְחִילוּ צוֹעֲקִים בְּקוֹל רָם, וְהַקּוֹל נִשְׁמַע בְּבֵית הַמָּלוֹן. נֶאֶסְפוּ אֲנָשִׁים מִבָּאֵי בֵּית הַמָּלוֹן, וּכְשֶׁשָּׁמְעוּ אֶת דִּבְרֵי הַשְּׁלָ"ה, פָּנוּ אֶל הַשּׁוֹחֵט בְּתוֹכְחוֹת מוּסָר וִיעָצוּהוּ שֶׁיִּתְוַדָּה עַל עֲווֹנוֹ. וְהַשּׁוֹחֵט עוֹמֵד עַל דַעְתּוֹ: לֹא יָדַעְתִּי, לֹא לָקַחְתִּי.

אָמְרוּ לְהַשְּׁלָ"ה: יִמְסְרֵהוּ אֲדוֹנֵנוּ לָרָשׁוּת. שָׁם יִתְוַדָּה, וְגַם אֶת הַכֶּסֶף יָשִׁיב.

לֹא חָפֵץ הַשְּׁלָ"ה הַקָּדוֹשׁ לִמְסֹר דִינוֹ שֶׁל יְהוּדִי לִידֵי נָכְרִים, כִּי אִם לְבֵית דִין שֶׁל יִשְׂרָאֵל. אֲבָל בְּאוֹתָהּ שָׁעָה הָיָה בְּבֵית הַמָּלוֹן שׁוֹטֵר תֻּרְכִּי, וְגַם הוּא שָׁמַע אֶת דְבַר הָרִיב. הָלַךְ וּמָסַר אֶת הַדָּבָר לָרָשׁוּת, וּמִיָּד בָּאוּ חַיָּלִים וְכָפְתוּ אֶת הַשּׁוֹחֵט וּנְתָנוּהוּ בַּבּוֹר עַד יוֹם הַמִּשְׁפָּט.

וּמִשְׁפָּט הַגַּנָּבִים בִּסְטַמְבּוּל חָמוּר מְאֹד, וּבְיוֹתֵר לְמִי שֶׁמָּעַל בַּאֲדוֹנָיו. יָצָא דִינוֹ שֶׁל הַשּׁוֹחֵט לְמִיתָה. כְּשֶׁשָּׁמַע הַשְּׁלָ"ה, כִּי בִּגְלָלוֹ יוּמַת יְהוּדִי בְּאֶרֶץ נָכְרִיָּה, מָרָה לוֹ נַפְשׁוֹ מְאֹד. הָלַךְ אֶל עֲשִׁירֵי הָעֵדָה וְהִפִּיל תְּחִנָּתוֹ לִפְנֵיהֶם שֶׁיָּחוּסוּ וְיַצִּילוּ אֶת הַשּׁוֹחֵט מִמָּוֶת.

נָשְׂאוּ הָעֲשִׁירִים אֶת פְּנֵי הַשְּׁלָ"ה הַקָּדוֹשׁ וְהִשְׁתַּדְּלוּ אֵצֶל הָרָשׁוּת וְלֹא חָסוּ עַל מָמוֹנָם, עַד שֶׁהִסְכִּימוּ הַדַּיָּנִים לְהָמִיר לוֹ אֶת מִשְׁפַּט הַמָּוֶת בִּכְרִיתַת אֵבֶר אֶחָד מֵאֶבְרֵי גוּפוֹ. נָתְנוּ הָעֲשִׁירִים סְכוּם כֶּסֶף בִּידֵי הַשְּׁלָ"ה וּמִהֲרוּ לְשַׁלְּחוֹ בָּאֲנִיָּה לָאָרֶץ הַקְּדוֹשָׁה, שֶׁלֹא יִרְאֶה בָּרָעָה אֲשֶׁר מָצְאָה אִישׁ יְהוּדִי בְּעֶטְיוֹ.

נָסַע הַשְּׁלָ"ה לְדַרְכּוֹ מְדֻכָּא מִן הַמְּאֹרָע, וְהַשּׁוֹחֵט נִשְׁאַר כָּלוּא בַּבּוֹר, מְחַכֶּה לְעָנְשׁוֹ. וְהוּא בּוֹכֶה וְנִשְׁבָּע בְּכָל קָדְשֵׁי יִשְׂרָאֵל, שֶׁיָּדוֹ לֹא הָיְתָה בַּמַּעַל וְאֵינוֹ יוֹדֵעַ מְאוּמָה.

לְאַחַר יָמִים אֲחָדִים בָּאוּ עוֹבְרֵי אֹרַח לִסְטַמְבּוּל וְסִפְּרוּ שֶׁמָּצְאוּ מַטְבְּעוֹת כֶּסֶף וְזָהָב אֲחָדִים בְּדֶרֶךְ הַמֶּלֶךְ. הִגִּיעָה הַשְּׁמוּעָה לָרָשׁוּת. בִּקְשׁוּ וּמָצְאוּ, כִּי בְּמֶרְחַק כַּמָּה מִילִים מְפֻזָּרִים בַּדֶּרֶךְ מַטְבְּעוֹת כֶּסֶף וְזָהָב, הִיא הַדֶּרֶךְ שֶׁעָבַר בָּהּ הַשְּׁלָ"ה בְּעֶגְלָתוֹ.

בָּדְקוּ יָפֶה אֶת מוֹט הָעֲגָלָה וּמָצְאוּ שֶׁהַמּוֹט נִסְדַק מִן הַחֹם וּמִטִּלְטוּל הָעֲגָלָה, וּבְנָסְעָהּ נָפְלוּ הַמַּטְבְּעוֹת מִתּוֹךְ הַמּוֹט, עַד כִּי לֹא נוֹתַר אֶחָד. יָצָא הַשּׁוֹחֵט לַחָפְשִׁי, רַק תְּנוּךְ אָזְנוֹ הַשְּׂמָאלִית כָּרְתוּ לְאוֹת הָאַשְׁמָה שֶׁהוּטְלָה עָלָיו. בִּקְשׁוּ לְהוֹדִיעַ לְהַשְּׁלָ"ה, אֲבָל הוּא הָיָה בְּלֵב יָם וְאֵין שָׁלִיחַ שֶׁיּוּכַל לְהַגִּיעַ אֵלָיו.

נִשְׁאַר הַשּׁוֹחֵט בִּסְטַמְבּוּל בְּעֵירֹם וּבְחֹסֶר כֹּל וְלֹא יָדַע מַה יַּעֲשֶׂה לְהַחֲיוֹת אֶת נַפְשׁוֹ.

הָיָה מַחֲזִיר בִּרְחוֹבוֹת הָעִיר, אֶפְשָׁר תִּזְדַמֵּן לוֹ עֲבוֹדָה כָּל שֶׁהִיא וְלֹא מָצָא. יוֹם אֶחָד עָבַר עַל חֲנוּת שֶׁל מַעֲשֵׂי אֶרֶג. עָמַד וְהִסְתַּכֵּל בַּחֲנוּת הַמְפֹאֶרֶת, וּבְעָמְדוֹ שָׁמַע אֶת הַחֶנְוָנִית עוֹשָׂה חֶשְׁבּוֹן עִם סוֹחֵר מֵאֶרֶץ אַחֶרֶת. וּבָרוּר הָיָה לוֹ, שֶׁהִיא טוֹעָה בְּחֶשְׁבּוֹנָהּ בְּאֶלֶף זְהוּבִים.

נִכְנַס לַחֲנוּת וְאָמַר לַחֶנְוָנִית: תִּמְחֹל לִי מַעֲלָתָהּ, בָּרִי לִי שֶׁטָּעֲתָה בְּאֶלֶף זְהוּבִים.

חִשְּׁבָה הָאִשָּׁה שֵׁנִית אֶת חֶשְׁבּוֹנָהּ יַחַד עִם הַסּוֹחֵר הַזֶּה, וְהִנֵּה אֱמֶת נָכוֹן הַדָּבָר: הִיא עָמְדָה לְקַבֵּל מִן הַסּוֹחֵר אֶלֶף זְהוּבִים פָּחוֹת מִן הַמַּגִּיעַ לָהּ. הוֹדְתָה הַחֶנְוָנִית לַשּׁוֹחֵט וְצִוְּתָה עַל הַמְשָׁרֵת, שֶׁיַּגִּישׁ לוֹ כִּסֵּא לָשֶׁבֶת עָלָיו. וְהַסּוֹחֵר הַזָּר הִתְנַצֵּל, שֶׁגַּם הוּא טָעָה בְּחֶשְׁבּוֹנוֹ וְלֹא הִתְכַּוֵּן חָלִילָה לְרַמּוֹת אֶת הַחֶנְוָנִית. פָּרַע אֶת כָּל חוֹבוֹ וְהָלַךְ לוֹ.

שָׁאֲלָה הַחֶנְוָנִית אֶת הַשּׁוֹחֵט מִי הוּא וּמֵאַיִן בָּא (שֶׁהִכִּירָה בּוֹ, כִּי מִקָּרוֹב בָּא מֵאֶרֶץ נָכְרִיָּה). סִפֶּר לָהּ אֶת כָּל הַמּוֹצְאוֹת אוֹתוֹ מֵאָלֶף עַד תָּיו.

רִחֲמָה עָלָיו הָאִשָּׁה וְהִצִּיעָה לְפָנָיו שֶׁיַּעֲבֹד בַּחֲנוּתָהּ וְתִתֶּן לוֹ שָׂכָר מוּעָט וְגַם מְזוֹנוֹת עַל שֻׁלְחָנָהּ, וְאִם יַצְלִיחַ בִּמְלַאכְתּוֹ, תַּעֲשֶׂה אֶת עֲבוֹדָתוֹ קֶבַע.

הוֹאִיל הַשּׁוֹחֵט לָשֶׁבֶת עִם הָאִשָּׁה וּלְשָׁרֵת בַּחֲנוּתָהּ. עַד מְהֵרָה לָמַד אֶת דַרְכֵי הַמִּסְחָר וְהַלָּקוֹחוֹת אֲהֵבוּהוּ עַל זְרִיזוּתוֹ וְיָשְׁרוֹ וְהָיוּ מַרְבִּים לָבוֹא לִקְנוֹת בִּגְלָלוֹ. שְׂכָרַתּוּ הָאִשָּׁה לְשָׁנָה תְּמִימָה וְקָרְאָה לוֹ “עוֹזֵר” וְנָתְנָה שְׂכָרוֹ בְּעַיִן יָפָה.

וְהָ"עוֹזֵר" הָיָה עוֹסֵק בֶּאֱמוּנָה וְהָיָה מַעֲשִׁיר אֶת גְבִרְתּוֹ. גַם הָיָה שׁוֹמֵר עִתּוֹת הַתְּפִלָּה וּבַלֵּילוֹת הָיָה הוֹגֶה בַּתּוֹרָה. וּמִסְחַר הָאִשָּׁה פָּרַץ מְאֹד, וּכְטוֹב לְמִסְחָרָהּ הֵיטִיבָה מַשְׂכֻּרְתּוֹ, וְהוּא הִתְעַצֵּם בִּשְׁקִידָתוֹ וְגָמַל לָהּ כֵּפֶל כִּפְלַיִם.

וְהָאִשָּׁה אַלְמָנָה מֵאִישָׁהּ, וּכְשֶׁרָאֲתָה אֶת הָאִישׁ, כִּי מַשְׂכִּיל הוּא בְּכָל דְּרָכָיו וְהוּא גַם תַּלְמִיד חָכָם וְשׁוֹמֵר מִצְוָה, אָמְרָה לַחֲסוֹת בְּצֵל כְּנָפָיו. שָׁלְחָה לִקְרֹא לְאֶחָד מִקְּרוֹבֶיהָ וְגִלְּתָה לוֹ אֶת מַחֲשַׁבְתָּהּ לְהִנָּשֵׂא לָאִישׁ הָעוֹבֵד בַּחֲנוּתָהּ. הָלַךְ הָאִישׁ אֶל הָ"עוֹזֵר" וְהִגִּיד לוֹ אֶת דִּבְרֵי גְבִרְתּוֹ. שָׂמַח הָאִישׁ שִׂמְחָה גְּדוֹלָה, שֶׁהִכִּיר אֶת הָאִשָּׁה כִּי טוֹבַת שֵׂכֶל הִיא וְיִשְׁרַת לֵב וּמִנְהָגָהּ כְּבַת יִשְׂרָאֵל כְּשֵׁרָה.

נִשְּׂאָה לוֹ הַחֶנְוָנִית, וּבְיוֹם חֲתֻנָּתָם חָלְקוּ מַתָּנוֹת רַבּוֹת לָעֲנִיִּים, וְרַבִּים הָיוּ הַקְּרוּאִים מִנְּשִׂיאֵי הָעֵדָה וְהָיוּ שָׂשִׂים וּשְׂמֵחִים וּמְסִיחִים בְּדִבְרֵי תוֹרָה עַד עֲלוֹת הַשַּׁחַר.

לַאֲדוֹנֵי הַחֲנוּת הַגְּדוֹלָה. וּכְשֶׁעָלָה לִגְדֻלָּה לֹא שָׁכַח, כִּי אֱלֹהִים הֱרִימוֹ מֵאַשְׁפּוֹת, וְהָיְתָה יָדוֹ פְתוּחָה לְכָל עָנִי וּלְכָל אִישׁ מָצוֹק, וּבְיוֹתֵר לָעֲנִיִּים שֶׁבָּאוּ מֵאֶרֶץ רְחוֹקָה, כִּי יָדַע אֶת נֶפֶשׁ הַגֵּר בְּאֶרֶץ נָכְרִיָּה. וּבְרֹב הַיָּמִים קָרְאוּ לוֹ “הַגְּבִיר הַנָּדִיב” וְהָיוּ מִתְבָּרְכִים בּוֹ קְרוֹבִים וּרְחוֹקִים.

וְהַשְּׁלָ"ה הַקָּדוֹשׁ הִגִּיעַ לִירוּשָׁלַיִם וְשָׁם קִבְּלוּהוּ בְּכָבוֹד גָדוֹל. הָיָה יוֹשֵׁב וְעוֹסֵק בְּתוֹרָה וּבְמַעֲשִׂים, וְגַם יְשִׁיבָה יָסַד וְהָיוּ לוֹמְדִים שָׁם בַּחוּרִים עֲנִיִּים וְעוֹשִׂים חַיִל בַּתּוֹרָה. וְהַשּׁוֹחֵט שֶׁנִּשְׁאַר בִּסְטַמְבּוּל נִשְׁתַּכַּח מִלִּבּוֹ וְלֹא זְכָרוֹ.

לֹא עָבְרוּ שָׁלֹשׁ שָׁנִים עַד שֶׁהוֹצִיא הַשְּׁלָ"ה אֶת הַכֶּסֶף שֶׁבְּיָדוֹ, וְגַם הַיְשִׁיבָה עָמְדָה לְהִסָּגֵר מֵאֵין כֶּסֶף לְכַלְכֵּל אֶת לוֹמְדֶיהָ. וּבִירוּשָׁלַיִם גָבְרָה אָז הַמְּצוּקָה, וְלֹא יָכֹל לְקַיֵּם אֶת הַתּוֹרָה אֲפִלּוּ מֵעֹנִי. וְגַם שְׁנַת בַּצֹּרֶת הָיְתָה בָּאָרֶץ וְהַיֹּקֶר הֶאֱמִיר, וְלֹא הָיָה בִּידֵי הַיְּהוּדִים הַיּוֹשְׁבִים בִּירוּשָׁלַיִם לִקְנוֹת לֶחֶם וָמַיִם, וְהָיוּ רַבִּים מִתְעַטְּפִים בְּרָעָב בִּרְחוֹבוֹת הָעִיר.

וְלֹא יָכֹל עוֹד הַשְּׁלָ"ה לִרְאוֹת בְּצָרַת אֶחָיו, אָסַף אֶת הָאֲמַרְכְּלִים וּמְנַהֲלֵי הַצְּדָקָה בָּאָרֶץ לְטַכֵּס עֵצָה מַה לַּעֲשׂוֹת. נִמְנוּ וְגָמְרוּ לִשְׁלֹחַ אִישׁ בַּעַל שֵׁם לְאַרְצוֹת הַגּוֹלָה לְעוֹרֵר רַחֲמִים בְּלֵב אַחֵינוּ בְּנֵי יִשְׂרָאֵל, שֶׁיָּבוֹאוּ לְעֶזְרַת לוֹמְדֵי הַתּוֹרָה בָּאָרֶץ הַקְּדוֹשָׁה, וּבָחֲרוּ בַּשְּׁלָ"ה, שֶׁיְּהֵא שָׁלִיחַ לִדְבַר מִצְוָה זֶה, בְּדַעְתָּם שֶׁשְּׁמוֹ הַטּוֹב יִפְעַל עַל לִבּוֹת נְדִיבֵי עַם יִשְׂרָאֵל לְהַרְבּוֹת נִדְבוֹתָם.

יָצָא הַשְּׁלָ"ה מִירוּשָׁלַיִם לִהְיוֹת שְׁדָ"ר (שְׁלִיחַ דְרַבָּנָן) בְּאַרְצוֹת הַגּוֹלָה, וּבְדַרְכּוֹ הִגִּיעַ בָּאֳנִיָּה לִסְטַמְבּוּל יוֹמַיִם לִפְנֵי הַפֶּסַח. כְּשֶׁבָּא לְבֵית הַכְּנֶסֶת רָאָהוּ “הַגְּבִיר הַנָּדִיב” וְהִכִּירוֹ, וְהַשְּׁלָ"ה לֹא הִכִּירוֹ, כִּי הָעֹשֶׁר שִׁנָּה אֶת פְּנֵי הָאִישׁ וּלְבוּשׁוֹ הַנָּאֶה עָשָׂהוּ כְּאַחַד הַשּׁוֹעִים.

שָׁלַח הַגְּבִיר אֶת הַשַּׁמָּשׁ לְהַזְמִין אֶת הַשְּׁדָ"ר לְבֵיתוֹ שֶׁיְּסַדֵּר עִמּוֹ בְּלֵיל הַפֶּסַח. וּבְלֵיל הַסֵּדֶר צִוָּה הַגְּבִיר לַעֲרֹךְ בְּבֵיתוֹ שְׁנֵי שֻׁלְחָנוֹת: אֶחָד לוֹ וְלִבְנֵי בֵּיתוֹ וְאֶחָד לַשְּׁדָ"ר. וְהַשֻּׁלְחָנוֹת עֲרוּכִים בִּכְלֵי כֶסֶף וּכְלֵי זָהָב וּבְדֹלַח וְיַיִן מְשֻׁבָּח לְאַרְבַּע כּוֹסוֹת וּפֵרוֹת וּמִגְדָנוֹת, הַכֹּל כְּיַד הַמֶּלֶךְ. גַם שְׁתֵּי מִטּוֹת הֲסִבָּה הֵכִינוּ עִם מַצָּעוֹת יְקָרִים מְצֻפִּים מֶשִׁי לָבָן.

אָמַר הַשְּׁלָ"ה בְּלִבּוֹ: “הַגְּבִיר חוֹשֵׁשׁ שֶׁמָּא אֲנִי מִתְבַּיֵּשׁ לֶאֱכֹל עִם בְּנֵי הַבַּיִת כְּאוֹרֵחַ עָנִי, עַל כֵּן שָׂם לִי שֻׁלְחָן לְבַד, שֶׁאֶהְיֶה בְּעֵינַי אָדוֹן לְעַצְמִי, וַהֲרֵיהוּ נוֹהֵג כְּאֶחָד מֵאֲצִילֵי בְּנֵי יִשְׂרָאֵל”. וְהוּא לֹא יָדַע, שֶׁבַּעַל הַבַּיִת עָשָׂה כָּךְ, שֶׁלֹּא יִסְתַּכֵּל בּוֹ הַגָּאוֹן מִקָּרוֹב וְיַכִּירֶנּוּ.

וּשְׁנֵיהֶם נִרְאִים שְׂמֵחִים וְטוֹבֵי לֵב: הַשְּׁלָ"ה עַל שֶׁהִקְרָה לוֹ אֱלֹהִים לֶאֱכֹל בְּחַג הַפֶּסַח בְּבֵית אִישׁ נִכְבָּד וְשׁוֹמֵר חֻקֵּי הַפֶּסַח לְכָל פִּרְטֵיהֶם, וְגַם קִוָּה בְּלִבּוֹ, כִּי לְאַחַר הַחַג יַעֲנִיק לוֹ הַגְּבִיר נְדָבָה הֲגוּנָה בִּשְׁבִיל עֲנִיֵּי יְרוּשָׁלַיִם; וּבַעַל הַבַּיִת – עַל הַכָּבוֹד שֶׁנָּפַל בְּחֶלְקוֹ לְאָרַח בְּבֵיתוֹ אִישׁ נוֹדָע וְקָדוֹשׁ כְּהַשְּׁלָ"ה. אַחֲרֵי תֹּם הַסֵּדֶר כְּהִלְכָתוֹ הָלְכוּ בְּנֵי הַבַּיִת לִישׁן וְאוֹר הַנֵּרוֹת כָּהָה כִּי מִעֵטוּ. בָּא הַגְּבִיר וְיָשַׁב לְיַד הָאוֹרֵחַ וְדִבֶּר עִמּוֹ בְּדִבְרֵי הֲלָכָה. שָׂמַח הַשְּׁלָ"ה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא זִמְּנוֹ לְבֵיתוֹ שֶׁל תַּלְמִיד חָכָם.

כְּשֶׁהִגִּיעַ חֲצִי הַלַּיְלָה אָמַר הַגְּבִיר: אֵחַרְנוּ לָשֶׁבֶת. עַתָּה יוֹאִיל כְּבוֹדוֹ לָלֶכֶת אַחֲרַי וַאֲבִיאֶנּוּ אֶל הַחֶדֶר שֶׁפִּנִּינוּ לוֹ.

הוֹלִיךְ בַּעַל הַבַּיִת אֶת אוֹרְחוֹ דֶרֶךְ פְּרוֹזְדוֹרִים וּמִסְדְרוֹנוֹת מְפֹאָרִים, עַד שֶׁהֱבִיאוֹ אֶל חֶדֶר מֻפְרָשׁ בְּיַרְכְּתֵי הַבַּיִת.

זֶה הַחֶדֶר. – אָמַר הַגְּבִיר וּפָתַח אֶת הַדֶּלֶת.

נִכְנַס הַשְּׁלָ"ה לַחֶדֶר וְגַם הַגְּבִיר נִכְנַס עִמּוֹ. תָּמַהּ הַשְּׁלָ"ה עַל כָּךְ, אֲבָל חָשַׁב: “אֶפְשָׁר רוֹצֶה הוּא לִרְאוֹת אִם הַכֹּל עַל מְקוֹמוֹ בַּחֶדֶר”. וּמַה גָּדַל תִּמְהוֹנוֹ, כְּשֶׁרָאָה אֶת בַּעַל הַבַּיִת נוֹעֵל אֶת הַדֶּלֶת וְנוֹתֵן אֶת הַמַּפְתֵּחַ בְּכִיס בִּגְדוֹ.

עָמַד לְפָנָיו הַגְּבִיר וְקָרָא בְּקוֹל: אֱמֹר וִדּוּי בְּטֶרֶם תָּמוּת!

נִבְהַל הַגָּאוֹן וְשָׁאַל רְתֵת: מַה זֹּאת? מָה רָעָה עָשִׂיתִי לוֹ?

אָמַר בַּעַל הַבַּיִת: אֲנִי הוּא הַשּׁוֹחֵט שֶׁעָלָה עִמְּךְ לָאָרֶץ הַקְּדוֹשָׁה וְהוֹצֵאתָ עָלַי שֵׁם רָע שֶׁגָּנַבְתִּי אֶת מָמוֹנְךָ. רְאֵה אֶת תְּנוּךְ אָזְנִי שֶׁנִּכְרַת בְּמִצְוַת הַשּׁוֹפְטִים. וַאֲנִי לֹא נָגַעְתִּי בְּכָל אֲשֶׁר לְךָ, וְאַתָּה הִתְעַלַּלְתָּ בִּי וְהֵבֵאתָ עָלַי אֵימַת מָוֶת וְיִסּוּרִים, שֶׁאֵין בֶּן אָדָם יָכוֹל לַעֲמֹד בָּהֶם.

הִתְבּוֹנֵן הַשְּׁלָ"ה בְּפָנָיו וְהִכִּירוֹ. נָפְלָה עָלָיו אֵימָה גְדוֹלָה, פָּרַשׂ כַּפָּיו לָאִישׁ וְקָרָא: חוּסָה עָלַי וְאַל תְּמִיתֵנִי. בְּצָרָה גְדוֹלָה הָיִיתִי וְהַשֵּׁם נָתַן אֶת הַחֲשָׁד הַנּוֹרָא בְּלִבִּי. אַף לֹא אֲנִי הִלְשַׁנְתִּי עָלֶיךָ, הַדָּבָר נוֹדַע לָרָשׁוּת בְּמִקְרֶה.

לְחִנָּם תִּצְטַדֵּק, קָרָא הָאִישׁ בְּזַעַם, עֵינֶיךָ הָרוֹאוֹת, כִּי בֵּרְכַנִי אֱלֹהִים בְּעֹשֶׁר וּבְכָבוֹד, כִּגְמוּל עַל כָּל הַיִּסּוּרִים שֶׁהֵבֵאתָ עָלַי עַל לֹא חָמָס בְּכַפִּי. וְאִלּוּ הָיִיתָ בֶּאֱמֶת צַדִּיק וְקָדוֹשׁ, כְּמוֹ שֶׁיַּחְשְׁבוּךָ רַבִּים, לֹא נִכְשַׁלְתָּ בְּחֵטְא זֶה לְהָבִיא עַל אִישׁ נָקִי מִשְׁפָּט מָוֶת, כִּי רַק בְּנֵס וּבְחֶמְלַת אֱלֹהִים עָלַי נִצַּלְתִּי מִמָּוֶת. אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא מֵבִיא תַּקָּלָה עַל יְדֵי צַדִּיקִים וּמְגַלְגְּלִין חוֹבָה עַל יְדֵי חַיָּב.

נִסָּה הַשְּׁלָ"ה לְעוֹרֵר רַחֲמֵי הָאִישׁ עָלָיו וְלֹא עָלְתָה בְּיָדוֹ, וְכַעֲסוֹ שֶׁל הַגְּבִיר הוֹלֵךְ וְסוֹעֵר: כַּלֵּה דְבָרֶיךָ וְהִתְוַדֵּה! אִם לֹא תְּמַהֵר לִקְרֹא אֶת הַוִּדּוּי, אֶשְׁחָטְךָ בְּלֹא וִדּוּי.

וּבְדַבְּרוֹ אֶת הַדְּבָרִים הָאֵלֶּה, הֵסִיר מֵעַל הָאָרוֹן נַרְתִּיק יָשָׁן, וּבוֹ הָיָה הַחַלָּף שֶׁלָּקַח עִמּוֹ אָז בְּצֵאתוֹ לַדֶּרֶךְ.

כְּשֶׁרָאָה הַשְּׁלָ"ה כִּי כָּלְתָה אֵלָיו הָרָעָה וְאֵין פֶּתַח הַצָּלָה, עָמַד בְּקֶרֶן זָוִית וְהִתְפַּלֵּל בְּלֵב קָרוּעַ וּמֻרְתָּח, וְקָרָא אֶת הַוִּדּוּי עַד תֻּמּוֹ. אַחַר הִתְפַּלֵּל לִשְׁלוֹם כָּל יִשְׂרָאֵל, וְגַם בְּעַד הָאִישׁ הָעוֹמֵד לָקַחַת אֶת נַפְשׁוֹ הִתְפַּלֵּל, וְכֵן אָמַר: רִבּוֹנוֹ שֶׁל עוֹלָם! יְהִי רָצוֹן מִלְּפָנֶיךָ שֶׁתְּהֵא מִיתָתִי כַּפָּרָה עַל כָּל עֲווֹנוֹתַי וְעַל כָּל עֲווֹנוֹת עַמְּךָ יִשְׂרָאֵל, וְגַם לְלוֹקֵחַ נַפְשִׁי תִּסְלַח כְּרֹב חַסְדְךָ.

וּכְשֶׁכִּלָּה הַשְּׁלָ"ה אֶת וִדּוּיוֹ וּתְפִלָּתוֹ עֲקָדוֹ בַּעַל הַבַּיִת וְהִנִּיחוֹ עַל הַמִּטָּה וְלָקַח אֶת הַחַלָּף לְשָׁחֳטוֹ. עָצַם הַשְּׁלָ"ה אֶת עֵינָיו וְקָרָא בְּכַוָּנָה עֲצוּמָה: “שְׁמַע יִשְׂרָאֵל ה' אֱלֹהֵינוּ, ה' אֶחָד”. וְהַגְּבִיר הֶעָבִיר אֶת הַחַלָּף עַל צַוָּארוֹ אַחַת וּשְׁתַּיִם. אֲבָל לֹא אֶת חֹד הַחַלָּף הֶעֱבִיר עַל הַצַּוָּאר, אֶלָּא אֶת גַב הַחַלָּף, וְקָרָא בְּקוֹל: הִנֵּה סָר עֲווֹנְךָ וְחַטָּאתְךָ תְּכֻפָּר!

מִהֵר וְהִתִּיר אֶת אֲסוּרָיו, אָחַז בְּיָדוֹ, הֶעֱמִידוֹ עַל רַגְלָיו וְאָמַר: עַתָּה קוּם, רַבִּי וּקְדוֹשִׁי. הֵבֵאתִי עָלֶיךָ אֵימַת מַר הַמָּוֶת. כְּמוֹ שֶׁהִרְגַשְׁתִּי אֲנִי אָז אַחֲרֵי הַמִּשְׁפָּט, כְּדֵי שֶׁתִּנָּקֶה מֵחֶטְאֲךְ הָאֶחָד אֲשֶׁר נִכְשַׁלְתָּ בּוֹ. אָמַרְתִּי: אֵיךְ יָבוֹא צַדִּיק וְקָדוֹשׁ כָּמוֹךָ בְּיוֹם הַדִּין לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים וּבְיָדוֹ עָווֹן כָּבֵד אֲשֶׁר כָּזֶה? וְעַתָּה סְלַח לִי, רַבִּי, נֵר יִשְׂרָאֵל וּקְדוֹשׁוֹ, כְּגֹדֶל חַסְדְךָ. קָרָא הַשְּׁלָ"ה בִּדְמָעוֹת מִתְרַגְּשׁוֹת: לֹא אַתָּה חָטָאתָ, יְדִידִי, כִּי אֲנִי חָטָאתִי לְךָ וְהֵבֵאתִי עָלֶיךָ רָעָה רַבָּה וֵאלֹהִים פָּדָה נַפְשְׁךָ. וְאַתָּה סְלַח לַעֲווֹנִי, כִּי רַב הוּא.

נָפְלוּ אִישׁ עַל צַוּארֵי רֵעֵהוּ וּבָכוּ שְׁנֵיהֶם.

כָּל יְמֵי הַחַג יָשַׁב הַשְּׁלָ"ה בְּבֵית הַגְּבִיר בִּכְבוֹדוֹ שֶׁל עוֹלָם. וּבְאִסְרוּ חַג נָתַן עַל יָדוֹ סְכוּם גָדוֹל בִּשְׁבִיל עֲנִיֵּי אֶרֶץ יִשְׂרָאֵל וְהַיְשִׁיבָה וְשִׁלְּחוֹ מִבֵּיתוֹ בְּכָבוֹד גָדוֹל.

וּכְשֶׁחָזַר הַשְּׁלָ"ה הַקָּדוֹשׁ מִמַּסָּעָיו לִירוּשָׁלַיִם הִרְבָּה לַעֲסֹק בְּתוֹרָה וּבְמִצְווֹת, וְשָׁם כָּתַב אֶת סִפְרוֹ “שְׁנֵי לוּחוֹת הַבְּרִית”, וְקָבַע אֶת הַוִּדּוּי לְיַד הִלְכוֹת פֶּסַח, לְזֵכֶר הַמְּאֹרָע. וְאַהֲבָה רַבָּה הָיְתָה בֵּינוֹ וּבֵין הַגְּבִיר הַנָּדִיב מִסְּטַמְבּוּל, וְאִגְּרוֹת הֵרִיצוּ אִישׁ לְרֵעֵהוּ, וְהַנָּדִיב הִרְבָּה לַעֲשׂוֹת צְדָקָה בָּאָרֶץ וְהָיָה מַחֲזִיק בַּעֲנִיֵּי אֶרֶץ יִשְׂרָאֵל וּבַחֲכָמֶיהָ כָּל יְמֵי חַיָּיו.

את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
