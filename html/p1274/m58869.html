סְעַדְיָה תּוֹפֵר הַקְּדֵרוֹת

היְתָה שְׁנַת בַּצֹּרֶת בְּאֶרֶץ תֵּימָן. כָּל הָאָרֶץ צָמְאָה לְמַיִם וְסַכָּנַת רָעָב הָיְתָה תְּלוּיָה עַל רֹאשׁ כָּל הַיִּשּׁוּב. צִוּוּ הַחֲכָמִים לַקָּהָל לְהִתְפַּלֵּל בְּבָתֵּי כְּנֵסִיּוֹת עַל הַגֶּשֶׁם, וְלֹא יָרְדוּ גְּשָׁמִים. גָּזְרוּ תַּעֲנִית בְּשֵׁנִי וַחֲמִישִׁי וְשֵׁנִי – וְלֹא יָרְדוּ גְּשָׁמִים. עָשׂוּ כָּל חַכְמֵי תֵּימָן קְהִלָּה גְּדוֹלָה בְּצַנְעָה הָעִיר, יָשְׁבוּ וְדָנוּ שְׁלשָׁה יָמִים, וְלַסּוֹף נִמְנוּ וְגָמְרוּ לַעֲשׂוֹת יוֹם כִּפּוּר קָטָן בְּשַׁבָּת, אֶפְשָׁר תִּזָּקֵק לָהֶם הַשְּׁכִינָה וְיֵרְדוּ גְּשָׁמִים לְהַעֲבִיר אֶת הַבַּצֹּרֶת. וּכְרוּז הוֹצִיאוּ הַחֲכָמִים לְכָל הָעֵדָה לְהִתְאַסֵּף בְּיוֹם הַשַּׁבָּת לְבָתֵּי הַכְּנֵסִיּוֹת לְקַיֵּם אֶת הַצּוֹם וּלְהַרְבּוֹת בִּתְפִלָּה וּבְתַחֲנוּנִים, אוּלַי יְרַחֵם.

הָיָה שָׁם יְהוּדִי פָּשׁוּט אֶחָד, אִישׁ תָּם וְיָשָׁר וּמְלַאכְתּוֹ תִּקּוּן כְּלֵי חֶרֶס. וְהָיוּ קוֹרְאִים לוֹ סְעַדְיָה תּוֹפֵר הַקְּדֵרוֹת. הָיוּ הַבְּרִיּוֹת מְבִיאִים אֶל בֵּיתוֹ אֶת שִׁבְרֵי כְּלֵיהֶם וְהוּא מְחַבְּרָם אֵבֶר אֶל אֵבֶר, קוֹדֵחַ בַּשּׁוּלַיִם נְקָבִים קְטַנִּים, תּוֹפְרָם וּמְהַדְּקָם בְּחוּטֵי בַּרְזֶל וּנְחשֶׁת וְעוֹשֶׂה עַל גַבָּם מַעֲשֵׂה רֶשֶׁת, וְהָיָה מַפְלִיא לַעֲשׂוֹת בִּמְלַאכְתּוֹ. אֲבָל שְׁכַר מְלָאכָה זוֹ פְּרוּטָה וַחֲצִי פְּרוּטָה, וְעַל כֵּן הָיְתָה פַּרְנָסָתוֹ וּפַרְנָסַת בְּנֵי בֵּיתוֹ בְּצַעַר גָדוֹל. אֲבָל אֶת הַשַּׁבָּת הָיָה שׁוֹמֵר מִכָּל מִשְׁמָר וּמִשְׁתַּדֵּל לַעֲשׂוֹתָהּ בְּתַעֲנוּגִים, כְּכֹל שֶׁיָּדוֹ מַגַּעַת.

בְּאוֹתוֹ יוֹם הַשִּׁשִּׁי הָלְכוּ כָּל הַגְּבָרִים לְגַלַּח אֶת רָאשֵׁיהֶם וְלִטְבֹּל בְּמַיִם טְהוֹרִים, הִתְפַּלְלוּ מִנְחָה מִבְּעוֹד יוֹם וְאָכְלוּ הַסְּעֻדָּה הַמַּפְסֶקֶת כְּמוֹ בְּעֶרֶב יוֹם כִּפּוּר, שֶׁכֵּן גָזְרוּ הַחֲכָמִים. אֲבָל סְעַדְיָה לֹא נָתַן דַעְתּוֹ עַל גְּזֵרַת הַחֲכָמִים, מִפְּנֵי שֶׁמְּקֻבָּל הָיָה בְּיָדוֹ מֵאָבִיו, עָלָיו הַשָׁלוֹם וְהַתְּפִלָּה: אָסוּר לְהִתְעַנּוֹת בְּשַׁבָּת, וְעֹנֶג שַׁבָּת מִצְוָה מִן הַתּוֹרָה.

הָלְכוּ כָּל הָעָם אַחֲרֵי הַסְּעֻדָּה הַמַּפְסֶקֶת לְבָתֵּי הַכְּנֵסִיּוֹת, וְגַם סְעַדְיָה הָלַךְ לְבֵית הַכְּנֶסֶת שֶׁהָיָה רָגִיל לְהִתְפַּלֵּל שָׁם, הִתְפַּלֵּל עַרְבִית, וְאַחַר הַתְּפִלָּה הָלַךְ לְבֵיתוֹ לְקַדֵּשׁ עַל הַיַּיִן וְלַעֲרֹךְ סְעֻדַּת הַשַּׁבָּת כְּהִלְכָתָהּ. כָּל הָעָם מִקָּצֶה הָיוּ מְכֻנָּסִים בְּבָתֵּי כְּנֵסִיּוֹת וְלֹא יָצְאוּ מִשָּׁם כָּל הַלַּיְלָה, עִנּוּ נַפְשׁוֹתֵיהֶם וְקָרְאוּ לֵאלֹהִים בְּחָזְקָה בְּקוֹלוֹת זְעָקָה וְזִמְרָה. וּסְעַדְיָה יָשַׁב עִם בְּנֵי בֵּיתוֹ, הֵרִיחוּ אֶת הַצֶּמַח “רֵיחָן” וְאָכְלוּ מַטְעַמֵּי שַׁבָּת וּפִצְּחוּ זֵרְעוֹנִים וְזִמְּרוּ זְמִירוֹת שֶׁל שַׁבָּת, כְּמוֹ בְּכָל שַׁבְּתוֹת הַשָּׁנָה.

וְאִשְׁתּוֹ, שֶׁהָיְתָה בַּת חָכָם, רָאֲתָה אֶת מַעֲשֵׂי בַּעְלָהּ וְשָׁתְקָה. אֲבָל כְּתֹם הַסְּעֻדָּה לֹא יָכְלָה לְהִתְאַפֵּק עוֹד וְאָמְרָה לוֹ: מָה אַתָּה עוֹשֶׂה, סְעַדְיָה? כָּל הָעָם מִקָּצֶה מִתְעַנִּים וּמִתְפַּלְלִים וּמְשַׁבְּרִים אֶת הַשָּׁמַיִם בְּזַעֲקָתָם, וְאַתָּה יוֹשֵׁב בַּבַּיִת, אוֹכֵל וְשׁוֹתֶה וּמִתְעַנֵּג. וְכִי אֵין עָלֶיךָ מוֹרָא הַחֲכָמִים? וְכִי אֵין צַעַר הַצִּבּוּר נוֹגֵעַ לְלִבְּךָ? אָמַר לָהּ: הַשֵּׁם יִתְבָּרַךְ צִוָּנוּ: “אָז תִּתְעַנַּג עַל הַשֵּׁם וְקָרָאתָ לַשַּׁבָּת עֹנֶג”, וְלֹא צִוָּה אוֹתָנוּ: “אָז תִּתְעַנֵּה עַל הַשֵּׁם וְקָרָאתָ לַשַׁבָּת נֶגַע”. לֹא עַל יְדֵי בִּטּוּל הַשַּׁבָּת אֶפְשָׁר לְהָבִיא אֶת הַגְּשָׁמִים אֶלָּא עַל יְדֵי קִיּוּמָהּ.

שָׁמְעָה הָאִשָּׁה וְשָׁתְקָה.

לְמָחָר בַּבֹּקֶר הָלַךְ לְבֵית הַכְּנֶסֶת לְהִתְפַּלֵּל עִם הַקָּהָל, וְאַחֲרֵי תְּפִלַּת מוּסָף חָזַר לְבֵיתוֹ לְקַדֵּשׁ וְלִסְעֹד עִם בְּנֵי בֵּיתוֹ.

הֵם יוֹשְׁבִים וְאוֹכְלִים, וְהִנֵּה גָבְרוּ מְאֹד הַקּוֹלוֹת וְהַצְּעָקוֹת בְּבֵית הַכְּנֶסֶת, עַד שֶׁנִּשְׁמְעוּ לָהֶם הֵיטֵב, אַף עַל פִּי שֶׁהַבַּיִת הָיָה רָחוֹק מִשָּׁם. הִנִּיחָה הָאִשָּׁה אֶת הַכַּף מִיָּדָהּ, וְהַיְלָדִים נָשְׂאוּ עֵינֵיהֶם אֶל אֲבִיהֶם וּפְנֵיהֶם הִלְבִּינוּ מִפַּחַד, עוֹד מְעַט וְיִגְעוּ בִּבְכִי.

אָמַר סְעַדְיָה: מַה לָּהֶם שֶׁהֵם בּוֹכִים וּמִצְטָעֲרִים כָּל כָּךְ? אֲנִי אֵלֵךְ וְאָבִיא אֶת הַגֶּשֶׁם.

הָלַךְ לְפִנַּת הַחֶדֶר וְנָטַל אֶת תַּרְמִיל הָעוֹר, נָתַן בּוֹ שִׁבְרֵי כֵּלִים מִמָּה שֶּׁהֵבִיאוּ לוֹ לְתִקּוּן, לָקַח כְּלֵי אֻמָּנוּתוֹ: צְבָת וּמַקְדֵחַ וְחוּטֵי בַּרְזֶל וְהִנִּיחָם בְּתוֹךְ הַתַּרְמִיל. נִתְעַטֵּף בְּטַלִּיתוֹ וְאֶת הַתַּרְמִיל כִּתֵּף עַל שִׁכְמוֹ, יָצָא וְהָלַךְ לְבֵית הָעָלְמִין וְיָשַׁב בֵּין הַקְּבָרִים. לְאַחַר שֶׁיָּשַׁב פָּתַח אֶת הַתַּרְמִיל וְהוֹצִיא אֶת כְּלֵי הָאֻמָּנוּת וְאֶת שִׁבְרֵי הַכֵּלִים וַעֲרָכָם לְפָנָיו עַל אֶבֶן מַצֵּבָה, מוּכָן לַמְּלָאכָה.

אַחַר נָשָׂא עֵינָיו לַשָּׁמַיִם וְדִבֶּר לִפְנֵי הַשֵּׁם יִתְבָּרַךְ: רִבּוֹנוֹ שֶׁל עוֹלָם! רְאֵה יֵשׁ מְלָאכָה בְּיָדִי וְהָאֲנָשִׁים הֵבִיאוּ לִי עֲבוֹדָה וַאֲנִי רוֹצֶה לְהִשְׂתַּכֵּר לְמִחְיַת בְּנֵי בֵּיתִי, אֲבָל אֲנִי מְבַטֵּל אֶת רְצוֹנִי וְשׁוֹבֵת מִמְּלָאכָה מִפְּנֵי רְצוֹנְךָ שֶׁאֶשְׁמֹר אֶת הַשַׁבָּת לְכַבְּדָהּ וּלְהִתְעַנֵּג בָּהּ. לָמָּה אֲנִי מְבַטֵּל רְצוֹנִי מִפְּנֵי רְצוֹנְךָ וְשׁוֹבֵת מִמְּלַאכְתִּי וְאַתָּה אֵינְךָ מְבַטֵּל רְצוֹנְךָ מִפְּנֵי רְצוֹנִי לְהוֹרִיד גְּשָׁמֶיךָ לָאָרֶץ? כְּלוּם יֵשׁ בָּזֶה מִן הַיּשֶׁר, שֶׁשְּׁנַיִם שֶׁכָּרְתוּ אֲמָנָה בֵּינֵיהֶם, הָאֶחָד הִתְחַיֵּב עַל “וּשְׁמַרְתֶּם אֶת הַשַּׁבָּת” וְהַשֵּׁנִי הִתְחַיֵּב עַל “וְנָתַתִּי גִשְׁמֵיכֶם בְּעִתָּם”, וּבִזְמָן שֶׁהָאֶחָד מְקַיֵּם אֶת הָאֲמָנָה, הַשֵּׁנִי מֵפִיר אוֹתָהּ?

יָשַׁב כַּמָּה רְגָעִים מַחֲשֶׁה לָדַעַת מַה יַעֲנוּהוּ מִן הַשָּׁמַיִם. וּכְשֶׁלֹּא קִבֵּל תְּשׁוּבָה אָחַז בְּיָדוֹ אֶת הַצְּבָת וְהַמַּקְדֵחַ, נָשָׂא עֵינָיו לַמָּרוֹם וְקָרָא: עַתָּה אֱמֹר, מַה טּוֹב בְּעֵינֶיךָ – שֶׁיֵּרְדוּ גְשָׁמִים אוֹ שֶׁאֲחַלֵּל אֶת הַשַּׁבָּת?

וְחָזַר עַל דְבָרָיו שֶׁבַע פְּעָמִים. אַחַר הוֹרִיד רֹאשׁוֹ וְשָׁתַק כַּמָּה רְגָעִים.

כְּשֶׁנָּשָׂא אַחַר כָּךְ עֵינָיו לַשָּׁמַיִם רָאָה עַל קְצוֹתָם עָב קְטַנָּה כְּכַף אִישׁ עוֹלָה וּמִתְקָרֶבֶת. וְלֹא עָבְרוּ עֲשָׂרָה רְגָעִים וְהַשָּׁמַיִם הִתְקַדְּרוּ עָבִים וְגֶשֶׁם כָּבֵד נִתַּךְ אַרְצָה, וּסְעַדְיָה רָץ לְבֵיתוֹ כֻּלּוֹ נוֹזֵל מַיִם.

רָאוּהוּ בְּנֵי בֵּיתוֹ נִכְנָס וְהִבִּיטוּ אֵלָיו כְּאֶל אֶחָד קָדוֹשׁ, שֶׁהָלַךְ וְהֵבִיא עִמּוֹ אֶת הַגֶּשֶׁם בְּתַרְמִילוֹ שֶׁעַל כְּתֵפָיו.

נוֹדַע הַדָּבָר בָּרַבִּים וַיְהִי לְנֵס בְּכָל אֶרֶץ תֵּימָן.

את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
