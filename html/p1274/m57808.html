הֶחָסִיד וְהָאֲרִי

רבִּי שְׁמוּאֵל הֶחָסִיד, אָבִיו שֶׁל רַבִּי יְהוּדָה הֶחָסִיד, הָיָה גוֹלֶה מִמָּקוֹם לְמָקוֹם לִלְמֹד תּוֹרָה, שֶׁכָּל מַה שֶׁהָיָה מַרְוֶה אֶת עַצְמוֹ בְּדִבְרֵי תוֹרָה, הָיָה צִמְאוֹנוֹ גוֹבֵר וְהוֹלֵךְ. לֹא מָצְאָה עוֹד יָדָם שֶׁל חֲכָמִים בְּעִיר פְּלוֹנִית לְלַמְּדוֹ דָבָר חָדָשׁ, הָלַךְ לְעִיר אַחֶרֶת, שֶׁשָּׁם יוֹשְׁבִים חֲכָמִים גְדוֹלִים מֵהֶם.

פַּעַם אַחַת בָּא לְעִיר שֶׁל חוֹף. יָשַׁב בָּהּ כִּשְׁלֹשָה חֳדָשִׁים וְשָׁמַע כָּל מַה שֶׁהַחֲכָמִים יָכְלוּ לְהַשְׁמִיעוֹ. יָרַד אֶל הַנָּמֵל וְרָאָה סְפִינָה עוֹמֶדֶת לְפָרֵשׂ בַּיָּם, נָתַן שְׂכָרָהּ וְיָרַד לָבוֹא לִמְדִינוֹת הַיָּם, שֶׁיּוֹשְׁבִים שָׁם מְאוֹרֵי הַתּוֹרָה. הִפְלִיגָה הַסְּפִינָה כְּשָׁעָה בַּיָּם. פִּתְאֹם שָׁמַע מִן הַיַּבָּשָׁה קוֹל יְלֵל מַחֲרִיד לֵב. בִּקֵּשׁ רַבִּי שְׁמוּאֵל הֶחָסִיד מֵאֵת הַקַּבַּרְנִיט שֶׁיְּקָרֵב אֶת הַסְּפִינָה לַיַּבָּשָׁה. וְהַקַּבַּרְנִיט, שֶׁרָאָה כִּי אִישׁ קָדוֹשׁ הוּא, עָשָׂה אֶת רְצוֹנוֹ וְעָמַד סָמוּךְ לַיַבָּשָׁה. יָרַד רַבִּי שְׁמוּאֵל בַּסֻּלָּם, וּכְשֶׁעָלָה עַל הַיַּבָּשָׁה הָלַךְ אַחֲרֵי הַקּוֹל עַד שֶׁהִגִּיעַ לְעַרְבַת חוֹל גְּדוֹלָה וְלוֹהֶטֶת בַּחַמָּה. רָאָה אֲרִי רָץ בַּחוֹל וְהוּא בּוֹרֵחַ עַל נַפְשׁוֹ מִפְּנֵי פַּנְתֵּר גָּדוֹל וְנוֹרָא. הַפַּנְתֵּר רָדַף אַחֲרֵי הָאֲרִי, וְהָאֲרִי, שֶׁאֵימַת הָרוֹדֵף עָלָיו, רָץ וְיִלֵּל וְקוֹלוֹ הִתְפַּשֵּׁטּ לְאַרְבַּע רוּחוֹת הַשָּׁמַיִם, מִפְּנֵי שֶׁהַפַּנְתֵּר הָיָה יוֹרֵק מִפִּיו לַהֲבוֹת אֵשׁ, וְכָל מָקוֹם שֶׁהֵָאֵשׁ פּוֹגַעַת הִיא שׂוֹרֶפֶת, אֲפִלּוּ אֶת הֶעָפָר וַאֲפִלּוּ אֶת הַחוֹל.

כְּשֶׁרָאָה הֶחָסִיד, נִתְמַלֵּא לִבּוֹ רַחֲמִים וְרָץ לְהַצִּיל אֶת הָאֲרִי מִן הַסַּכָּנָה. עָמַד בֵּין הָאֲרִי וּבֵין הַפַּנְתֵּר וְנָשָׂא אֶת שְׁתֵּי יָדָיו כְּלַפֵּי הָרוֹדֵף, כְּדֶרֶךְ שֶׁהַכֹּהֲנִים נוֹשְׂאִים יְדֵיהֶם בְּדוּכָנָם. מִיַּד פָּסְקָה הָאֵשׁ מִפִּי הַפַּנְתֵּר, הָפַךְ אֶת פָּנָיו וּבָרַח בִּצְרִיחָה נוֹקֶבֶת אֶל מֶרְחַקֵּי הָעֲרָבָה.

נִצַּל הָאֲרִי מֵרוֹדְפוֹ וְעָמַד כָּל עוֹד נִשְׁמָתוֹ בּוֹ.

כְּשֶׁרָאָה הָאֲרִי אֶת הֶחָסִיד שֶׁהִצִּילוֹ מֵאִשּׁוֹ שֶׁל הַפַּנְתֵּר הַנּוֹרָא, בָּא אֶצְלוֹ וְהִלְחִית לְרַגְלָיו כְּכֶלֶב, לִקְקוֹ בִּלְשׁוֹנוֹ וְלֹא רָצָה לָזוּז מִשָּׁם. בִּקֵּשׁ הֶחָסִיד לֵילֵךְ, חִבֵּק הָאֲרִי אֶת רַגְלָיו בִּשְׁתֵּי כַּפּוֹתָיו וְלֹא הִנִּיחוֹ לֵילֵךְ.

שְׁאֵלוֹ רַבִּי שְׁמוּאֵל: מָה רְצוֹנְךָ, בְּנִי?

כָּרַע הָאֲרִי לִפְנֵי הֶחָסִיד וְזָקַר אֶת גַבּוֹ שֶׁיִרְכַּב עָלָיו. יָשַׁב רַבִּי שְׁמוּאֵל עַל הָאֲרִי וְרָכַב עָלָיו, עַד שֶׁהֱבִיאוֹ אֶל הַסְּפִינָה שֶׁעָמְדָה עַל עָגְנָהּ וְהִמְתִּינָה לוֹ.

נִכְנַס רַבִּי שְׁמוּאֵל לְתוֹךְ הַסְּפִינָה וְעָמַד עַל סִפּוּנָהּ מַשְׁקִיף עַל הַיַּבָּשָׁה. וְהָאֲרִי עָמַד כָּל אוֹתָהּ שָׁעָה עַל שְׂפַת הַיָּם וְהָיָה מִסְתַּכֵּל אַחֲרֵי מַצִּילוֹ עַד שֶׁנֶּעְלְמָה הַסְּפִינָה מִן הָעַיִן, כְּאָדָם הַמְלַוֶּה אֶת חֲבֵרוֹ וְהוּא מִסְתַּכֵּל אַחֲרָיו כָּל זְמַן שֶׁהוּא יָכוֹל לִרְאוֹתוֹ.


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
