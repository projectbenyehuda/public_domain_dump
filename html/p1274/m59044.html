שְׁלִיחוּת הַנָּחָשׁ

רַבִּי אֶלְעָזָר בֶּן שִׁמְעוֹן בֶּן יוֹחַאי הָיָה מְהַלֵּךְ כְּחֹם הַיּוֹם עִם חֲבֵרָיו בִּסְבִיבוֹת אוּשָׁא וְעוֹסֵק עִמָּהֶם בְּתוֹרָה.

הַחַמָּה עָמְדָה בְּלֵב הָרָקִיעַ וְהָיְתָה מְלַהֶטֶת כְּכִבְשָׁן, עַד שֶׁצִּלָּם לֹא הָיָה נוֹפֵל וְהָיוּ מִצְטַעֲרִים מֵרֹב חֲמִימוּת. נָשְׂאוּ עֵינֵיהֶם וְרָאוּ לִפְנֵיהֶם שְׂדֵה דֶשֶׁא רָטֹב וְרַעֲנָן וּבוֹ כַּמָּה אִילָנוֹת נָאִים וְטוֹבִים לְהִסְתּוֹפֵף בְּצִלָּם, וּבֵין הָעֲשָׂבִים עֵינוֹת מַיִם קָרִים לְהָשִׁיב נֶפֶשׁ עֲיֵפָה וְרוּחַ חֲרִישִׁית מַשִּׁיבָה רֵיחוֹת עֲרֵבִים. פָּנוּ וְעָלוּ אֶל הַמָּקוֹם הַהוּא וְיָשְׁבוּ תַּחַת אַחַד הָעֵצִים לַהֲגוֹת בְּתוֹרָה בְּנַחַת וּבִנְעִימִים.

אָמַר רַבִּי אֶלְעָזָר: כַּמָּה נָאֶה מָקוֹם זֶה לַעֲסֹק בְּתוֹרָה. הֲרֵי זֶה כְּגַן־עֵדֶן הַתַּחְתּוֹן.

עָנוּ חֲבֵרָיו וְאָמְרוּ: בָּרוּךְ שֶׁכָּכָה לוֹ בְּעוֹלָמוֹ.

וְהָיוּ שׁוֹאֲפִים הָרֵיחוֹת הַטּוֹבִים וּמְכַסִּים רַגְלֵיהֶם הַיְחֵפוֹת בְּאִצְטְלוֹתֵיהֶם מִפְּנֵי כְּבוֹד הַתּוֹרָה וּמַפְלִיגִים בַּהֲלָכָה. עוֹדָם יוֹשְׁבִים וְשׁוֹנִים וְהִנֵּה רָאוּ נָחָשׁ גָּדוֹל בָּא וְעוֹבֵר לִפְנֵיהֶם בְּחִפָּזוֹן. נָתַן רַבִּי אֶלְעָזָר עֵינָיו בַּנָּחָשׁ וְקָרָא: נָחָשׁ, נָחָשׁ, חֲזוֹר לַאֲחוֹרֶיךָ, שֶׁהָאִישׁ שֶׁאַתָּה הוֹלֵךְ לְהָרְגוֹ כְּבָר חָזַר בִּתְשׁוּבָה וּשְׁלִיחוּתְךָ בָּטְלָה.

כְּשֶׁשָּׁמְעוּ הַחֲבֵרִים אֶת דִבְרֵי רַבִּי אֶלְעָזָר הָיוּ בְּתִמָּהוֹן גָדוֹל.

אָמַר הָאֶחָד: בְּבַקָשָׁה מִמְּךָ, לַמְּדֵנוּ מָה עִנְיָנוֹ שֶׁל נָחָשׁ זֶה? אָמַר לָהֶם: אַחַר כָּךְ אַגִּיד לָכֶם אֶת הָעִנְיָן.

וְהַנָּחָשׁ אֵינוֹ מַפְנֶה רֹאשׁוֹ וְגוֹרֵר עַצְמוֹ הָלְאָה בְּחִפָּזוֹן גָדוֹל.

קָרָא רַבִּי אֶלְעָזָר שׁוּב: נָחָשׁ, נָחָשׁ, אַחַר שֶׁצִוּוּ אוֹתְךָ מִן הַשָּׁמַיִם וְנָתְנוּ לְךָ רְשׁוּת לֵילֵךְ לַהֲרֹג אוֹתוֹ הָאִישׁ, נִחַם הָאִישׁ עַל רָעָתוֹ וְקִבֵּל עַל עַצְמוֹ שֶׁלֹּא יָשׁוּב וְיֶחֱטָא לְעוֹלָם וַהֲרֵיהוּ כְּצַדִּיק גָמוּר. כְּלוּם תֵּלֵךְ וְתַהֲרֹג צַדִּיק?

לַדְּבָרִים הָאֵלֶּה עָמַד הַנָּחָשׁ וְהָיָה מוּטָל כְּמַקֵּל וְלֹא זָז לֹא לְפָנִים וְלֹא לְאָחוֹר. אֲבָל לְאַחַר שֶׁעָמַד כָּךְ רְגָעִים אֲחָדִים חָזַר הַנָּחָשׁ לֵילֵךְ לְדַרְכּוֹ כְּבָרִאשׁוֹנָה.

דִבֶּר אֵלָיו רַבִּי אֶלְעָזָר דִבְרֵי כִּבּוּשִׁים: נָחָשׁ בְּנִי, יוֹדֵעַ אֲנִי מָה רְצוֹנְךָ. חֲזוֹר מִדַּרְכְּךָ וְתִמְצָא עַכּוּ"ם אֶחָד, שֶׁעָשָׂה רָעָה לְזַרְעוֹ שֶׁל אַבְרָהָם אָבִינוּ, וְהוּא שׁוֹכֵב נִרְדָם בִּמְעָרָתְךָ – לֵךְ וְהָרְגֵהוּ וּשְׁפֹךְ חֲמָתְךָ וְיִרְוַח לְךָ. אַבָּא זִכְרוֹ לִבְרָכָה הָיָה אוֹמֵר: “טוֹב שֶׁבְּעַכּוּ”ם הֲרֹג" – עַכּוּ"ם רָשָׁע לֹא כֹּל שֶׁכֵּן.

מִיָּד הָפַךְ הַנָּחָשׁ אֶת פָּנָיו וְחָזַר בִּמְרוּצָה אֶל מְעָרָתוֹ.

אָמַר רַבִּי אֶלְעָזָר לַחֲבֵרָיו: תֵּדְעוּ שֶׁיְּהוּדִי אֶחָד עָבַר עֲבֵרָה גְדוֹלָה וְנִתְחַיֵּב מִיתָה וְשָׁלְחוּ מִן הַשָּׁמַיִם לְאוֹתוֹ נָחָשׁ לְהָרְגוֹ. וּבְשָׁעָה שֶׁהַנָּחָשׁ הָלַךְ בִּשְׁלִיחוּתוֹ חָזַר אוֹתוֹ אִישׁ לְמוּטָב, וּבִזְכוּת תְּשׁוּבָתוֹ בָּטְלָה הַגְּזֵרָה.

אָמְרוּ לוֹ חֲבֵרָיו: כָּל זֶה מִנַּיִן לְךָ?

אָמַר לָהֶם: רְאִיתֶם, בְּשָׁעָה שֶׁהָיָה הַנָּחָשׁ הוֹלֵךְ בִּשְׁלִיחוּתוֹ הָיוּ הַקַּשְׂקַשִּׂים שֶׁעַל גוּפוֹ זְקוּפוֹת וְעוֹמְדוֹת וּזְנָבוֹ נָטוּי לְמַעְלָה, מִפְּנֵי שֶׁבְּדֶרֶךְ רִיצָתוֹ בָּא כְּנֶגְדוֹ רוּחַ וְלָחַשׁ לוֹ: חֲזֹר לַאֲחוֹרֶיךְ, שֶׁבָּטְלָה הַגְּזֵרָה. וְהַנָּחָשׁ אָטָם אָזְנָיו לְדִבְרֵי הָרוּחַ, מִפְּנֵי שֶׁרָצָה שֶׁיִּתְּנוּ לוֹ אָדָם אַחֵר בִּמְקוֹם הָרִאשׁוֹן. שֶׁבְּרִית כְּרוּתָה לוֹ לַנָּחָשׁ, כְּשֶׁנּוֹתְנִים לוֹ רְשׁוּת מִן הַשָּׁמַיִם לַהֲרֹג אֵיזֶה אָדָם אֵינוֹ זָז מִשְּׁלִיחוּתוֹ עַד שֶׁיַּהֲרֹג לָאִישׁ שֶׁנִּמְסַר לִרְשׁוּתוֹ, אוֹ שֶׁיִּתְּנוּ לוֹ אַחֵר תְּמוּרָתוֹ. וְזֶה דֶרֶךְ רֹב הַמַּשְׁחִיתִים.

כְּשֶׁשָּׁמְעוּ הַחֲבֵרִים כָּךְ, תָּמְהוּ מְאֹד עַל גֹדֶל חָכְמָתוֹ שֶׁל רַבִּי אֶלְעָזָר, וּמֵרֹב תִּמָּהוֹן הָיוּ יוֹשְׁבִים וְשׁוֹתְקִים.

אָמַר לָהֶם רַבִּי אֶלְעָזָר: בֹּאוּ וְנֵלֵךְ לִרְאוֹת אֶת מַעֲשֶׂה הַנָּחָשׁ.

קָמוּ וְהָלְכוּ אֶל הָהָר שֶׁשָּׁם הָיְתָה הַמְּעָרָה. מָצְאוּ שָׁם אֶת הָעַכּוּ"ם מֻשְׁלָךְ מֵת וְהַנָּחָשׁ כָּרוּךְ בַּעֲקֵבוֹ וְאֵינוֹ פּוֹרֵשׁ מִמֶּנּוּ. אַחַר כָּךְ נִבְדַל מֵעֲקֵבוֹ וְעָלָה עַל צַוָּארוֹ וְנִכְרַךְ בּוֹ. אַחַר הִנִּיחַ אֶת הַצַּוָּאר וְחָזַר אֶל הֶעָקֵב. וְעַל זֶה הַדֶּרֶךְ פְּעָמִים רַבּוֹת, וּמִן הָעַכּוּ"ם לא פָּרַשׁ אֲפִלּוּ לְרֶגַע אֶחָד.

הָלַךְ רַבִּי אֶלְעָזָר אֵצֶל הָעַכּוּ"ם וּפִשְׁפֵּשׁ בִּלְבוּשׁוֹ וּמָצָא שָׁם כִּיס מָלֵא זְהוּבִים. לָקַח רַבִּי אֶלְעָזָר אֶת הַכִּיס, נְתָנוֹ בְּחֵיקוֹ וְאָמַר: בָּרוּךְ הַמָּקוֹם, שֶׁעוֹשֶׂה שְׁלִיחוּתוֹ עַל־יְדֵי כָּל בִּרְיוֹתָיו לְרָעָה וּלְטוֹבָה.

וְסִפֵּר לָהֶם, שֶׁאוֹתוֹ הַכִּיס הוּא שֶׁל הַיְּהוּדִי שֶׁפָּגַע בּוֹ הָעַכּוּ"ם בַּדֶּרֶךְ וְהִכָּהוּ וּפְצָעוֹ וְגָזַל מִמֶּנּוּ אֶת כָּל מָמוֹנוֹ.

יָצְאוּ מִן הַמְּעָרָה וְיָשְׁבוּ בַּשָּׂדֶה וְהִפְלִיגוּ שׁוּב בַּהֲלָכָה.

עוֹדָם יוֹשְׁבִים וְשׁוֹנִים, רָאוּ יְהוּדִי אֶחָד בָּא עָיֵף וְיָגֵעַ מִן הַדֶּרֶךְ וּמֵחֹם הַשֶּׁמֶשׁ וְהוּא מֻכָּה וּפָצוּעַ כֻּלּוֹ. יָשַׁב הַיְּהוּדִי כְּנֶגְדָם עַל שֹׁרֶשׁ אִילָן לָנוּחַ, הָיָה גוֹנֵחַ וּבוֹכֶה: צַדִּיק הוּא הַשֵּׁם וְצַדִּיק דִינוֹ. רִבּוֹנוֹ שֶׁל עוֹלָם, לא אִכְפַּת לִי צַעַר גוּפִי וְאֵינִי חָס עַל מָמוֹנִי, שֶׁוַּדַּאי יָשָׁר מִשְׁפָּטְךָ, וְאִם לֹא הָיִיתִי חַיָּב לְפָנֶיךָ לֹא אֵרַע לִי כָּל זֶה. אֲבָל מִצְטַעֵר אֲנִי עַל אָבִי וְאִמִּי, שֶׁהֵם זְקֵנִים וַחֲלוּשִׁים כִּקְטַנִים, וְלֹא נִשְׁאָר בְּיָדִי כְּלוּם לְפַרְנְסָם אֲפִלּוּ כָּעוֹרֵב וְכַכֶּלֶב. עַל זֶה יִדְוֶה לִבִּי וְיֶהֱמוּ מֵעַי. וַי לִי וַי, טוֹב מוֹתִי מֵחַיַּי!

קָמוּ רַבִּי אֶלְעָזָר וַחֲבֵרָיו וְנִתְקָרְבוּ אֶצְלוֹ לְפַיְּסוֹ. וְרַבִּי אֶלְעָזָר תָּפַס יָדוֹ וְאָמַר לוֹ: אָחִי יְרֵא הַשֵּׁם, אַל תִּתְיָרֵא וְאַל תִּצְטַעֵר, שֶׁוַּדַּאי אָדָם כָּשֵׁר אַתָּה, אַחֲרֵי שֶׁרָאִינוּ אֶת הַנֵּס הַגָּדוֹל שֶׁעָשָׂה עִמְךָ בַּעַל הַנִּסִּים. הֲרֵי לְךָ כִּיס הַזְּהוּבִים שֶׁנִּגְזַל מִמְּךָ, קָחֶנּוּ וּבוֹא עִמָּנוּ לִרְאוֹת מַה שֶׁנַּעֲשָׂה בִּגְּלָלְךָ.

הָלַךְ עִמָּהֶם עַד שֶׁבָּאוּ שׁוּב לַמְּעָרָה שֶׁהָיָה שָׁם הָעַכּוּ"ם מֻשְׁלָךְ מֵת וְרָאָה אוֹתוֹ וְהִכִּירוֹ, וְהַנָּחָשׁ עֲדַיִן כָּרוּךְ עָלָיו בִּרְתִיחָה וּבִרְעִידָה וְעוֹשֶׂה בּוֹ נְקָמוֹת גְדוֹלוֹת.

נָפַל הַיְּהוּדִי עַל פָּנָיו וְנָתַן הוֹדָאוֹת וְתִשְׁבָּחוֹת לַמָּקוֹם בָּרוּךְ הוּא עַל הַנֵּס שֶׁעָשָׂה לוֹ. אַחַר כָּךְ קָם וְנָשַׁק יְדֵי רַבִּי אֶלְעָזָר וַחֲבֵרָיו וְאָמַר לָהֶם בַּדְמָעוֹת: בְּוַדַּאי שֶׁלֹא נַעֲשָׂה לִי הַנֵּס הַזֶה אֶלָּא בִּזְכוּתְכֶם וּבִזְכוּת תּוֹרַתְכֶם הַקְּדוֹשָׁה, שֶׁאֲנִי מַכִּיר בְּעַצְמִי שֶׁאֵינִי כְּדַאי וְהָגוּן לַעֲשׂוֹת לִי נִסִים וְנִפְלָאוֹת כָּאֵלֶּה.

וְהֶרְאָה לָהֶם אֶת סִמָּנֵי הַהַכָּאוֹת וְהַפְּצִיעוֹת שֶׁעָשָׂה לוֹ הָרָשָׁע כְּדֵי לְהוֹצִיא מִיָּדוֹ אֶת הַמָּמוֹן.

חָזַר רַבִּי אֶלְעָזָר וְנִתְקָרֵב אֵצֶל הַמֵּת וְרָאָה שֶׁהַנָּחָשׁ עוֹדֶנוּ מַחֲזִיק בּוֹ וְעוֹשֶׂה נְקָמוֹת בְּגוּפוֹ. אָמַר לוֹ רַבִּי אֶלְעָזָר לַנָּחָשׁ: נָחָשׁ, נָחָשׁ, כָּל מַה שֶּׁעָשִׂיתָ יָפֶה עָשִׂיתָ, הֶרְאֵיתָ גְבוּרָתְךָ וְכֹחֲךָ הַגָּדוֹל, וְעַכְשָׁיו דַי לְךָ, הִסְתַּלֵּק וַחֲזֹר לִמְעָרָתְךָ. וְגוֹזֵרְנִי עָלֶיךָ שֶׁמִּכָּאן וְאִילֵךְ לֹא תַזִּיק עוֹד לַבְּרִיוֹת וּתְקַיֵּם אֶת הַצִּוּוּי “וְעָפָר תֹּאכַל”. מִיָּד נָפַל הַנָּחָשׁ מִגּוּפוֹ שֶׁל הָעַכּוּ"ם, הִרְכִּין רֹאשׁוֹ לָאָרֶץ, חָפַז זְנָבוֹ וְרָץ לִמְקוֹמוֹ.

לָקַח הַיְהוּדִי אֶת הַכִּיס, הִשְׁתַּחֲוָה לִפְנֵי הַחֲכָמִים וְהָלַךְ לְדַרְכּוֹ שָׂמֵחַ וְטוֹב לֵב.

הָלְכוּ רַבִּי אֶלְעָזָר וַחֲבֵרָיו לַחֲזֹר לְאוּשָׁא.

כְּשֶׁעָבְרוּ עַל תֵּל אֶחָד רָאוּ אִישׁ עוֹמֵד עַל הַתֵּל, פָּנָיו כְּלַפֵּי יְרוּשָׁלַיִם הַחֲרֵבָה וְהוּא כּוֹרֵעַ וּמִשְׁתַּחֲוֶה וּמִתְפַּלֵּל מִנְחָה וְאֵינוֹ מַרְגִישׁ בְּעוֹבְרִים וְשָׁבִים.

לָחַשׁ רַבִּי אֶלְעָזָר לַחֲבֵרָיו: אִלְמָלֵא חָזַר זֶה הַיּוֹם בִּתְשׁוּבָה שְׁלֵמָה הָיָה הַנָּחָשׁ עוֹשֶׂה בּוֹ שְׁלִיחוּתוֹ, וּשְׂפָתָיו הַנָּעוֹת עַכְשָׁו בִּתְפִלָּה הָיוּ מְלַחֲכוֹת עָפָר.

את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
