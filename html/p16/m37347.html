בַּמוֹלֶדֶת

אֵלַיִךְ בָּאתִי שׁוּב, מוֹלֶדֶת יְקָרָה;

בְּרֹאשׁ מוּרָד אֵלַיִךְ שַׁבְתִּי וְיָגֵעַ,

אַךְ עֲלִיצוּת אִלֶּמֶת נִשְׁמָתִי מָלֵאָה,

זוֹ עֲלִיצוּת שְׁקֵטָה, זוֹ דְמָמָה בְּהִירָה,

הָרוֹעֲפוֹת בַּדִּמְדּוּמִים הַחַכְלִילִים

מִמַּעֲלוֹת חוֹפֵי הַדְּנֶפְּר וְהַשִּׁפּוּלִים,

מְקוֹם שָׁם בְּלִבִּי שְׁנֵי צִלְצוּלִים נוֹלָדוּ

בָּרִאשׁוֹנָה, קְרוֹבִים וְתוֹאֲמִים רָעָדוּ,

כִּשְׁנַיִם מַעְיָנוֹת בִּמְעֵי הָאֲדָמָה, –

וְשִׁירָתִי הָרִאשׁוֹנָה צֶחָה, חַמָּה,

הֻצְּתָה בְּפִי וְתַחַת אֶצְבְּעוֹתַי רָתָחָה,

בְּשֶׁטֶף צִלְצוּלִים מֵעַל נִימִים נִשְׁפָּכָה…

*

הָהּ, לֹא לַשָּׁוְא, מוֹלַדְתִּי, לִבִּי נִמְשַׁךְ

אֵלָיִךְ…עֶרֶב זַךְ זָכוֹר אֶזְכֹּר עֲדַיִן,

עֵת פַּעַם רִאשׁוֹנָה הָלַכְתִּי מֵעִמָּךְ.

כְּמוֹ מַרְבַד נִפְלָא, פּוֹרֶה וּרְחַב־יָדַיִם

נִיר הַמּוֹלֶדֶת מִסָּבִיב לִי הִשְׂתָּרַע

וְהַדְּנֶפְּר הָרָחָב, זַךְ־מַיִם כְּמַרְאָה,

מוּאָר בַּמֶּרְחַקִּים בְּאֹדֶם בֵּין־עַרְבָּיִם,

הָמָה מִתַּחַת הָר, וּכְרֶשֶׁת שֶׁל קַרְנַיִם

נָח נֹגַהּ אוֹר־חוֹזֵר זָהֹב

מִזִּיז גַלִּים קַלִּים עֲלֵי מִדְרוֹן הַחוֹף.

לִבְלֵב לִילָךְ, הִבְרִיק טַל־עֶרֶב כְּבָרֶקֶת.

הַכֹּל הָיָה סָפוּג שָׁלוֹם, עֶדְנָה וָשֶׁקֶט.

*

הֲלַחַשׁ שֶׁל גַּלִּים הָיָה זֶה אִם חָזוֹן?

זָכוּרְנִי, בִּי הַכֹּל פִּתְאֹם הִזְדַּעֲזֵעַ,

נָגְעוּ בְּנִשְׁמָתִי צְלִילִים קְצוּבִים כִּשְׁאוֹן

כְּנַף נֶשֶׁר עוּל־יָמִים בְּקֶצֶב תִּתְנוֹעֵעַ.


פִּנַּת־מוֹלַדְתִּי אָמְרָה לִי בִדְמָמָה:

"לֵךְ לְשָׁלוֹם, יַלְדִּי! הַרְבֵּה, הַרְבֵּה עָצְמָה

טִפַּחְתִּי בְּלִבְּךָ…לְבַשׁ גְּבוּרָה וָלֵכָה

לִמְקוֹם שָׁם שְׁאִיפָה נִשְׂגֶּבֶת תִּשָּׂאֶךָ,

לוֹחֵם הַדְּרוֹר וְהָאֱמֶת;

חֲיֵה, לְחַם, אֱהַב! וּכְבוֹא הָעֵת

וִיגַע־קְרָבוֹת קָשִׁים יָשֹׁחַ

רֹאשְׁךָ הַדַּךְ וְהֶעָיֵף מֵאֶפֶס־כֹּחַ –

תָּשׁוּב אֵלָי"…הָהּ, אֶרֶץ־מוֹלַדְתִּי,

קַבְּלִי יַלְדֵּךְ חוֹלֶה, קַבְּלִינִי־נָא! אַתְּ שֶׁפַע

כֹּחוֹת לִי הֶעֱנַקְתְּ – שִׁבְרִי וְהַוָּתִי

מָצוּ אוֹתָם טִפָּה טִפָּה כְּצֶפַע…

נִלְחַמְתִּי לְנַפְשִׁי, סָבַלְתִּי סֵבֶל רָב,

סָבַלְתִּי וָאֹהַב וָאֶעֱמֹד בַּקְּרָב,

אוּלָם הַוָּה לֹא־יְדוּעָה, לֹא־מְשֹׁעֶרֶת,

רְחָבָה מִמְּלֹא־עוֹלָם, מִתְּהוֹם־הַיָּם סוֹעֶרֶת,

מִתּוֹךְ אַלְפֵי לִבּוֹת דָּוִים חָדְרָה

אֶל תּוֹךְ לִבִּי. בְּנִשְׁמָתִי שְׁבוּרָה

עָלְתָה שַׁוְעַת יֵאוּשׁ נוֹרָא כֹה וּמְשַׁגֵּעַ,

וְשָׁוְא אַךְ שָׁוְא עָמַלְתִּי הַחֲרִישֵׁהוּ…

*

אַךְ שַׁבְתִּי עָדֵיכֶן, שַׁדְמוֹת מוֹלַדְתִּי,

כְּגֶשֶׁם־נְדָבוֹת יוֹרְדָה בִּרְכַּת־אֱלֹהַּ

עַל לְבָבִי…מְלֵאָה בִי נִשְׁמָתִי

זוֹ עֲלִיצוּת שְׁקֵטָה, זוֹ בְהִירוּת מַרְגּוֹעַ,

שֶׁבְּעֶרֶב רֵיחָנִי עִם חֲשֵׁכָה תִּרְעַף

מִמַּעֲלוֹת חוֹפֵי הַדְּנֶפְּר וְשִׁפּוּלָיו…

*

הַרְעֵשׁ גַּלֶּיךָ, דְּנֶפְּר, עַל לֵב דַּוָּי הָפִיחָה

קְרִירוּת גַּלִּים זַכִּים, בְּרֹאשׁ מַשְׁמִים הַשְׁבִּיחָה

אֶת סַעַר הָגוּתִי הָעֲצוּבָה,

הָקִיצָה, שַׁחַר, בְּחֶשְׁכַת נַפְשִׁי דָוָה

וְשׁוּב אֶקְרָא לַצָּר, וְשׁוּב בִּקְרָב אָחֵלָּה,

אֶסְבֹּל, אֹהַב, אֶקֹּב וְאֶתְפַּלֵּלָה!…


את הטקסט[ים] לעיל הפיקו מתנדבי פרויקט בן־יהודה באינטרנט.
